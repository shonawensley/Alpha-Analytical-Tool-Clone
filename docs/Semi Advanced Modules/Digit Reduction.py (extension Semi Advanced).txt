"""
digit_reduction.py

This module implements a comprehensive digit reduction system designed for Pick-3 analysis.
It performs multiple functions:
  1. Eliminates “noise” digits from a given string (e.g., digits from the most recent draw).
  2. Applies digit shifting to reveal hidden 3-value patterns.
  3. Extracts and scores 3-value patterns from longer (4-digit or 5-digit) clusters.
  4. Aggregates and scores patterns extracted from R2/R4/R6/R8 structures both vertically and horizontally.
  
Key Features:
  • Basic reduction by removal of specified digits.
  • Digit shifting transformation.
  • Extraction of all 3-value substrings from longer clusters.
  • Scoring for stability (vertical and horizontal) and permutation persistence.
  • Consensus detection for final-column patterns.
  
Configuration:
  - The module can be configured to use a list of “noise” digits.
  - Shift amount for digit shifting can be set.
  - Bonus scores for exact order (vertical multi-straight) and consensus are parameterized.
  
Usage:
  Import the module and call the main function `reduce_and_score_patterns(...)`
  with your R2/R4/R6/R8 structured data.
"""

import itertools
from collections import defaultdict

# --- Configuration Constants ---
# Digits to eliminate (for example, from the most recent draw)
DEFAULT_NOISE_DIGITS = "597"  # (this can be configured dynamically)

# Mirror mapping for V-Trac (if needed in a transformation)
MIRROR_MAP = {
    '0': '5', '1': '6', '2': '7', '3': '8', '4': '9',
    '5': '0', '6': '1', '7': '2', '8': '3', '9': '4'
}

# Bonus scores (tweak these to your preferences)
VERTICAL_STABILITY_BONUS = 3
HORIZONTAL_STRAIGHT_BONUS = 2
BOXED_PERMUTATION_BONUS = 1
CONSENSUS_BONUS = 4

# Digit shift amount (e.g., shift each digit by +1, with wrap-around)
DIGIT_SHIFT_AMOUNT = 1

# --- Utility Functions ---

def mirror_digit(d: str) -> str:
    """Return the mirror of digit d using MIRROR_MAP."""
    return MIRROR_MAP.get(d, d)

def mirror_string(s: str) -> str:
    """Return the mirrored string of s."""
    return "".join(mirror_digit(ch) for ch in s)

def shift_digits(s: str, shift: int = DIGIT_SHIFT_AMOUNT) -> str:
    """
    Shifts each digit in s by the specified shift (wrap-around 9->0).
    For example, "459" with shift=1 becomes "560".
    """
    shifted = []
    for ch in s:
        if ch.isdigit():
            new_digit = (int(ch) + shift) % 10
            shifted.append(str(new_digit))
        else:
            shifted.append(ch)
    return "".join(shifted)

def eliminate_noise(s: str, noise_digits: str = DEFAULT_NOISE_DIGITS) -> str:
    """
    Removes all characters in noise_digits from string s.
    """
    return "".join(ch for ch in s if ch not in noise_digits)

def extract_substrings(s: str, min_len: int = 3, max_len: int = 5) -> list:
    """
    Extracts all substrings of s with lengths between min_len and max_len.
    Returns a list of substrings.
    """
    substrings = set()
    n = len(s)
    for i in range(n):
        for j in range(i + min_len, min(i + max_len + 1, n + 1)):
            substrings.add(s[i:j])
    return sorted(substrings)

def is_valid_3value(cluster: str) -> bool:
    """
    Checks if cluster qualifies as a 3-value pattern.
    A valid pattern has a length between 3 and 5 and at most 3 unique digits.
    """
    if not (3 <= len(cluster) <= 5):
        return False
    return len(set(cluster)) <= 3

def extract_3value_patterns(s: str) -> list:
    """
    Extracts all substrings from s that qualify as 3-value patterns.
    """
    all_subs = extract_substrings(s, 3, 5)
    return [sub for sub in all_subs if is_valid_3value(sub)]

# --- Core Digit Reduction Functions ---

def basic_digit_reduction(s: str, noise_digits: str = DEFAULT_NOISE_DIGITS) -> str:
    """
    Eliminates noise digits from s.
    """
    return eliminate_noise(s, noise_digits)

def dynamic_digit_shift(s: str, shift: int = DIGIT_SHIFT_AMOUNT) -> str:
    """
    Applies digit shifting to s.
    """
    return shift_digits(s, shift)

def reduce_to_3value(s: str) -> list:
    """
    If s is longer than 3 digits (e.g. 4 or 5 digits) and qualifies as a 3-value cluster,
    generate all possible 3-digit substrings.
    This is intended to capture additional key digits.
    """
    if len(s) <= 3:
        return [s] if is_valid_3value(s) else []
    subs = []
    for i in range(len(s) - 2):
        candidate = s[i:i+3]
        if is_valid_3value(candidate):
            subs.append(candidate)
    return sorted(set(subs))

def aggregate_r_values(r2: str, r4: str, r6: str, r8: str) -> dict:
    """
    Given strings from R2, R4, R6, R8, compute:
      - vertical_stability: a score based on how similar their ending portions are.
      - vertical_patterns: a list of extracted 3-value patterns from the final parts.
      - If the final columns (e.g., last 3 digits) are identical (consensus), flag that.
    """
    # For each row, take the last 3 (or 4 if available) digits
    def tail(s, n=3):
        return s[-n:] if len(s) >= n else s
    
    tails = [tail(r) for r in [r2, r4, r6, r8] if r]
    common = set(tails[0])
    for t in tails[1:]:
        common &= set(t)
    vertical_stability = len(common)
    
    # Consensus: if all tails are identical (or if all digits in last tail are the same)
    consensus = (len(set(tails)) == 1) if tails else False
    
    # Extract 3-value patterns from the tails
    vertical_patterns = []
    for t in tails:
        vertical_patterns.extend(extract_3value_patterns(t))
    vertical_patterns = sorted(set(vertical_patterns))
    
    return {
        "vertical_stability": vertical_stability,
        "consensus": consensus,
        "vertical_patterns": vertical_patterns
    }

def horizontal_repeat_bonus(column_strings: list) -> int:
    """
    For a given row type across columns (e.g. all columns of R2),
    check if the last 3 digits in one column appear exactly in the next.
    Add a bonus score for each such repetition.
    """
    bonus = 0
    # For each adjacent pair of columns, check the last 3 digits
    for i in range(len(column_strings) - 1):
        s1 = column_strings[i]
        s2 = column_strings[i+1]
        if len(s1) >= 3 and len(s2) >= 3:
            tail1 = s1[-3:]
            tail2 = s2[-3:]
            if tail1 == tail2:
                bonus += HORIZONTAL_STRAIGHT_BONUS
            elif set(tail1) == set(tail2):
                bonus += BOXED_PERMUTATION_BONUS
    return bonus

# --- Comprehensive Reduction and Scoring Function ---

def reduce_and_score_patterns(r_data: dict, noise_digits: str = DEFAULT_NOISE_DIGITS, shift: int = DIGIT_SHIFT_AMOUNT) -> dict:
    """
    Given a dictionary with keys "R2", "R4", "R6", "R8" representing strings from each row,
    this function performs digit reduction and extracts 3-value patterns.
    
    It applies:
      1. Basic reduction (eliminates noise digits).
      2. Optional digit shifting to reveal hidden patterns.
      3. Extraction of 3-value patterns from each reduced string.
      4. Aggregation of vertical structure (using the last 3-4 digits) to score stability.
      5. Horizontal bonus: checks for repeats in adjacent columns.
      6. Computes a final "reduction_score" for each extracted pattern.
      
    Returns:
        A dictionary containing:
          - "extracted_patterns": List of dicts with pattern and its score.
          - "vertical_aggregate": Output of aggregate_r_values.
          - "horizontal_bonus": Bonus score from horizontal repeats.
    """
    # Assume r_data is a dictionary with keys "R2", "R4", "R6", "R8", each a list of strings (columns)
    # We perform reduction on each string in each row.
    reduced_rows = {}
    for key in ["R2", "R4", "R6", "R8"]:
        row = r_data.get(key, [])
        reduced = []
        for s in row:
            if s:
                # Apply basic reduction
                s_reduced = basic_digit_reduction(s, noise_digits)
                # Optionally apply digit shift
                s_shifted = dynamic_digit_shift(s_reduced, shift)
                # We can choose to keep both versions; here we concatenate for additional clues
                combined = s_reduced + "|" + s_shifted
                reduced.append(combined)
            else:
                reduced.append("")
        reduced_rows[key] = reduced

    # Aggregate vertical information using the rightmost (last) column from each row.
    final_r2 = reduced_rows["R2"][-1] if reduced_rows["R2"] else ""
    final_r4 = reduced_rows["R4"][-1] if reduced_rows["R4"] else ""
    final_r6 = reduced_rows["R6"][-1] if reduced_rows["R6"] else ""
    final_r8 = reduced_rows["R8"][-1] if reduced_rows["R8"] else ""
    vertical_info = aggregate_r_values(final_r2.split("|")[0], final_r4.split("|")[0],
                                         final_r6.split("|")[0], final_r8.split("|")[0])
    
    # Horizontal bonus across each row type (for each R-level, bonus for adjacent columns repeating tail)
    horizontal_bonus_total = 0
    for key in ["R2", "R4", "R6", "R8"]:
        horizontal_bonus_total += horizontal_repeat_bonus(reduced_rows.get(key, []))
    
    # Extract patterns from each reduced string (for each row, each column)
    pattern_scores = defaultdict(int)
    pattern_details = defaultdict(list)
    for row_key in ["R2", "R4", "R6", "R8"]:
        for s in reduced_rows.get(row_key, []):
            # We consider both parts if combined by "|"
            parts = s.split("|")
            for part in parts:
                # Extract all 3-value patterns from this part
                patterns = extract_3value_patterns(part)
                for p in patterns:
                    # Base score: length bonus (3-digit =2, 4-digit =3, 5-digit =4)
                    base = 2 if len(p) == 3 else (3 if len(p) == 4 else 4)
                    # Bonus for fewer unique digits (doubles/triples)
                    unique_bonus = 2 if len(set(p)) == 1 else (1 if len(set(p)) == 2 else 0)
                    # Add the bonus for each occurrence in this cell (simulate frequency)
                    pattern_scores[p] += base + unique_bonus
                    pattern_details[p].append(row_key)
    
    # Now add vertical stability bonus: for each pattern that appears in vertical aggregate patterns, add bonus
    for p in vertical_info.get("vertical_patterns", []):
        if p in pattern_scores:
            pattern_scores[p] += vertical_info.get("vertical_stability", 0) * VERTICAL_STABILITY_BONUS

    # Also add consensus bonus if consensus is detected
    if vertical_info.get("consensus", False):
        for p in pattern_scores:
            pattern_scores[p] += CONSENSUS_BONUS

    # Incorporate horizontal bonus uniformly to all patterns (or could be weighted by frequency)
    for p in pattern_scores:
        pattern_scores[p] += horizontal_bonus_total

    # Prepare final list of patterns with scores
    extracted_patterns = []
    for p, score in pattern_scores.items():
        extracted_patterns.append({
            "pattern": p,
            "final_reduction_score": score,
            "occurs_in_rows": pattern_details[p]
        })
    
    # Sort extracted patterns by final score descending
    extracted_patterns = sorted(extracted_patterns, key=lambda x: x["final_reduction_score"], reverse=True)

    return {
        "vertical_aggregate": vertical_info,
        "horizontal_bonus": horizontal_bonus_total,
        "extracted_patterns": extracted_patterns
    }
