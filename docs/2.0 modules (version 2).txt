#!/usr/bin/env python3
"""
Stable Pattern Progression Module
---------------------------------

This module scans the R2/R4/R6/R8 structures (columns 7..1) for 3-value
digit patterns ("stable patterns") in each cell, then scores them
based on:

 - Vertical coverage (repeats across R2/R4/R6/R8 in same column),
 - Horizontal adjacency (repeats across adjacent columns in same row),
 - V-Trac equivalence (mirror-like offsets),
 - Consensus columns (where all R2/R4/R6/R8 cells are identical),
 - Optional detection of star/hot-zone cells (but no advanced hot-zone logic),
 - Mirror elimination check (pattern's digits aren't in the prior row or its mirrors),
 - Extended length bonus,
 - Persistence (placeholder, if you pass 'persistence' in the data).

No digit-reduction or specialized hot-zone iteration is done here.
That will be handled in separate modules.

The final output is a dictionary of patterns keyed by their canonical
(actual-digit) rotation, each with:
 - "vtrac" for the V-Trac canonical,
 - "locations": list of (row, col),
 - "score": numeric total,
 - "score_breakdown": detail of how that score was computed.

You'll typically feed this output into your aggregator or synergy module.
"""

from collections import defaultdict

#########################################
# 1. V-TRAC MAPPING & HELPER FUNCTIONS
#########################################

# Mapping digits 0..9 to v-trac classes (1..5), matching the 0↔5,1↔6,2↔7,3↔8,4↔9 rules.
VTRAC_MAP = {
    str(d): str(((d % 10) % 5) + 1)
    for d in range(10)
}
# Example: '0'->'1','5'->'1','1'->'2','6'->'2','2'->'3','7'->'3','3'->'4','8'->'4','4'->'5','9'->'5'


def vtrac_transform(num_str: str) -> str:
    """Convert an actual digit string into its V-Trac representation, digits '1'..'5'."""
    return "".join(VTRAC_MAP[ch] for ch in num_str if ch in VTRAC_MAP)


def canonical_rotation(s: str) -> str:
    """
    Return the lexicographically smallest rotation of the string s.
    E.g. '312' => '123', '6688' => '6688' vs '8668' => pick whichever is lexicographically minimal.
    """
    if not s:
        return s
    rotations = [s[i:] + s[:i] for i in range(len(s))]
    return min(rotations)


def parse_cell(cell_value: str):
    """
    Separate numeric digits from trailing '*' (or '**').
    Returns (digits_str, stars):
      digits_str = string of digits only,
      stars = 0 if no star, 1 if '*', 2 if '**'.
    """
    if not cell_value:
        return "", 0

    s = cell_value.strip()
    if not s:
        return "", 0

    stars = 0
    while s.endswith('*'):
        stars += 1
        s = s[:-1]
    s = s.strip()

    digits_str = "".join(ch for ch in s if ch.isdigit())
    return digits_str, stars


#########################################
# 2. EXTRACT 3-VALUE PATTERNS FROM A CELL
#########################################

def extract_cell_clusters(digits_str: str, min_len=3):
    """
    Identify all 3-value clusters (length >= min_len) in 'digits_str'.
    A cluster is 3-value if it has <=3 unique digits in actual form OR <=3 unique digits in V-Trac form.

    Returns a list of dicts:
      {
        "pattern": <actual substring>,
        "vtrac": <vtrac of substring>,
        "canonical": <canonical rotation of actual>,
        "canonical_vtrac": <canonical rotation of vtrac>,
        "digit_pattern": bool,
        "vtrac_pattern": bool
      }
    """
    clusters = []
    length = len(digits_str)
    if length < min_len:
        return clusters

    for size in range(min_len, length+1):
        for start in range(0, length - size + 1):
            sub = digits_str[start:start+size]
            unique_d = set(sub)
            vt_sub = vtrac_transform(sub)
            unique_v = set(vt_sub)

            digit_ok = (len(unique_d) <= 3)
            vtrac_ok = (len(unique_v) <= 3)
            if not (digit_ok or vtrac_ok):
                continue

            cano_sub = canonical_rotation(sub)
            cano_vt = canonical_rotation(vt_sub)
            clusters.append({
                "pattern": sub,
                "vtrac": vt_sub,
                "canonical": cano_sub,
                "canonical_vtrac": cano_vt,
                "digit_pattern": digit_ok,
                "vtrac_pattern": vtrac_ok
            })
    return clusters


#########################################
# 3. FINDING VERTICAL/HORIZONTAL PATTERNS
#########################################

def find_vertical_patterns(clusters_by_col):
    """
    Group clusters by their canonical actual rotation in each column.
    If a pattern appears in 2+ rows in that column => it's a vertical repeat.
    We note if it's an exact repeat or a cyclic rotation.
    Returns { col: { canonical_str: {...}}}
    """
    vertical_patterns = {}
    for col, cluster_list in clusters_by_col.items():
        group_by_canon = defaultdict(list)
        for cl in cluster_list:
            group_by_canon[cl["canonical"]].append(cl)

        col_pattern_info = {}
        for canon, clist in group_by_canon.items():
            if len(clist) < 2:
                continue
            # We have a vertical repeat
            exact_repeat = False
            cyc_repeat = False

            actual_strings = {}
            for citem in clist:
                patt = citem["pattern"]
                if patt in actual_strings:
                    exact_repeat = True
                else:
                    actual_strings[patt] = True
            if len(actual_strings) > 1:
                cyc_repeat = True

            col_pattern_info[canon] = {
                "canonical": canon,
                "vtrac": clist[0]["canonical_vtrac"],  # assume same vTrac if only rotation difference
                "locations": [(itm["row"], itm["col"]) for itm in clist],
                "exact_vertical_repeat": exact_repeat,
                "cyclic_vertical_repeat": cyc_repeat,
                "count_vertical": len(clist)
            }
        if col_pattern_info:
            vertical_patterns[col] = col_pattern_info
    return vertical_patterns


def find_vtrac_vertical_matches(clusters_by_col):
    """
    Similar approach but grouping by vtrac canonical. If multiple different actual
    canons share the same vtrac => vtrac-only vertical match.
    Returns { col: { vtrac_canon: { "canonical_vtrac":..., "locations":..., ...}}}
    """
    vtrac_results = {}
    for col, cluster_list in clusters_by_col.items():
        group_by_vcanon = defaultdict(list)
        for cl in cluster_list:
            group_by_vcanon[cl["canonical_vtrac"]].append(cl)

        col_vtrac_info = {}
        for vcanon, c_list in group_by_vcanon.items():
            if len(c_list) < 2:
                continue
            actual_canons = {item["canonical"] for item in c_list}
            if len(actual_canons) < 2:
                continue  # all same actual => normal vertical
            col_vtrac_info[vcanon] = {
                "canonical_vtrac": vcanon,
                "locations": [(itm["row"], itm["col"]) for itm in c_list],
                "count_vertical": len(c_list)
            }
        if col_vtrac_info:
            vtrac_results[col] = col_vtrac_info
    return vtrac_results


def find_horizontal_patterns(clusters_by_row):
    """
    Check each row for patterns repeated in adjacent columns.
    We unify by actual canonical or vtrac canonical if they match.
    Returns { row: { pattern_key: {...} } }
    """
    horizontal_results = {}
    for row, c_list in clusters_by_row.items():
        # cluster each column for that row
        row_cols = defaultdict(list)
        for cl in c_list:
            row_cols[cl["col"]].append(cl)

        row_pattern_info = {}
        for col in range(1, 7):
            if col not in row_cols or (col+1) not in row_cols:
                continue
            left_clusters = row_cols[col]
            right_clusters = row_cols[col+1]

            for lc in left_clusters:
                for rc in right_clusters:
                    same_actual = (lc["canonical"] == rc["canonical"])
                    same_vtrac = (lc["canonical_vtrac"] == rc["canonical_vtrac"])
                    if same_actual or same_vtrac:
                        key = lc["canonical"] if same_actual else lc["canonical_vtrac"]
                        if key not in row_pattern_info:
                            row_pattern_info[key] = {
                                "canonical": lc["canonical"],
                                "vtrac": lc["canonical_vtrac"],
                                "locations": set(),
                                "adjacent_cols": []
                            }
                        row_pattern_info[key]["locations"].add((lc["row"], lc["col"]))
                        row_pattern_info[key]["locations"].add((rc["row"], rc["col"]))
                        row_pattern_info[key]["adjacent_cols"].append((col, col+1))

        # Convert to lists
        for k, info in row_pattern_info.items():
            info["locations"] = list(info["locations"])
        if row_pattern_info:
            horizontal_results[row] = row_pattern_info

    return horizontal_results


#########################################
# 4. SCORING FUNCTION
#########################################

def apply_scoring(patterns_dict,
                  vertical_patterns,
                  vtrac_vertical_patterns,
                  horizontal_patterns,
                  consensus_cols,
                  hot_cells,
                  prev_draw_digits):
    """
    Assign numeric scores to each pattern, storing pattern["score"].

    Key factors:
      - Vertical coverage (# of rows in same col)
      - exact/cyclic repeats
      - vtrac-only vertical synergy
      - horizontal adjacency
      - consensus col match
      - hot zone presence (optional)
      - extended length
      - mirror elimination
      - optional persistence
    """
    for pkey, pat in patterns_dict.items():
        breakdown = {}
        total = 0

        # Gather columns usage
        col_map = defaultdict(set)
        for (r,c) in pat["locations"]:
            col_map[c].add(r)

        # (1) vertical coverage
        max_vertical = 0
        for c, rowset in col_map.items():
            if len(rowset) > 1:
                coverage = len(rowset)
                score_c = coverage * 2  # e.g. 2 points per row beyond the first
                breakdown[f"vertical_coverage_col{c}"] = score_c
                total += score_c
                if coverage > max_vertical:
                    max_vertical = coverage

                # exact/cyclic
                if c in vertical_patterns and pkey in vertical_patterns[c]:
                    vinfo = vertical_patterns[c][pkey]
                    if vinfo["exact_vertical_repeat"]:
                        breakdown[f"vertical_exact_col{c}"] = breakdown.get(f"vertical_exact_col{c}", 0) + 3
                        total += 3
                    if vinfo["cyclic_vertical_repeat"]:
                        breakdown[f"vertical_cyclic_col{c}"] = breakdown.get(f"vertical_cyclic_col{c}", 0) + 2
                        total += 2

                # vtrac-only
                if c in vtrac_vertical_patterns:
                    vtcan = pat["vtrac"]
                    if vtcan in vtrac_vertical_patterns[c]:
                        # indicates multiple actual canons share same vtrac => synergy
                        breakdown[f"vertical_vtrac_col{c}"] = breakdown.get(f"vertical_vtrac_col{c}", 0) + 2
                        total += 2

        # full 4-row coverage
        if max_vertical == 4:
            breakdown["vertical_full_coverage"] = 5
            total += 5

        # (2) horizontal adjacency
        row_map = defaultdict(set)
        for (r,c) in pat["locations"]:
            row_map[r].add(c)

        for r, cset in row_map.items():
            if r in horizontal_patterns:
                # prefer actual canonical in the row-based dictionary, else vtrac
                ac = pat["canonical"]
                vc = pat["vtrac"]
                row_h = horizontal_patterns[r]
                key = None
                if ac in row_h:
                    key = ac
                elif vc in row_h:
                    key = vc
                if key and "adjacent_cols" in row_h[key]:
                    adj_pairs = row_h[key]["adjacent_cols"]
                    breakdown[f"horizontal_{r}"] = len(adj_pairs)
                    total += len(adj_pairs)
                    if len(adj_pairs) >= 2:
                        breakdown[f"horizontal_span_{r}"] = 2
                        total += 2

        # (3) consensus columns
        for c, cons_val in consensus_cols.items():
            # if pat's canonical = canonical_rotation of cons_val => synergy
            ccons = canonical_rotation(cons_val)
            if pat["canonical"] == ccons:
                breakdown[f"consensus_col{c}"] = 5
                total += 5
                if len(cons_val) <= 3:
                    breakdown[f"consensus_short_col{c}"] = 2
                    total += 2

        # (4) star/hot zone presence
        # We'll keep a small local bonus if found in star cells,
        # but you can set it to 0 or pass to aggregator if you prefer
        star_bonus = 0
        for (r,c) in pat["locations"]:
            if (r,c) in hot_cells:
                star_count = hot_cells[(r,c)]
                if star_count == 1:
                    star_bonus = max(star_bonus, 2)
                elif star_count == 2:
                    star_bonus = max(star_bonus, 3)
        if star_bonus:
            breakdown["hot_zone_presence"] = star_bonus
            total += star_bonus

        # (5) extended length
        plen = len(pat["canonical"])
        if plen > 3:
            ext_points = (plen - 3)
            breakdown["extended_length"] = ext_points
            total += ext_points

        # (6) mirror elimination
        # if none of the pattern's digits are in prior row's digits or their mirrors => bonus
        if prev_draw_digits:
            pat_digits = set(pat["canonical"])
            eliminated = False
            for (r,c) in pat["locations"]:
                # e.g. R2 => R1, R4 => R3, ...
                try:
                    prior_num = int(r[1:]) - 1
                    prior_row = f"R{prior_num}"
                except:
                    prior_row = None
                if prior_row and prior_row in prev_draw_digits:
                    excluded = set(prev_draw_digits[prior_row])
                    # add mirror digits
                    for d in list(excluded):
                        if d.isdigit():
                            mirror_d = str((int(d) + 5) % 10)
                            excluded.add(mirror_d)
                    if pat_digits.isdisjoint(excluded):
                        eliminated = True
                        break
            if eliminated:
                breakdown["mirror_elimination"] = 2
                total += 2

        # (7) optional persistence
        if "persistence" in pat:
            pval = pat["persistence"]
            if pval > 1:
                addp = (pval - 1)*3
                breakdown["persistence"] = addp
                total += addp

        pat["score_breakdown"] = breakdown
        pat["score"] = total

def extract_stable_patterns(data):
    """
    Primary function to find stable (3-value) patterns in R2,R4,R6,R8 columns 7..1
    and score them based on vertical/horizontal coverage, vTrac equivalence,
    consensus columns, optional star presence, mirror elimination, etc.

    'data' is a dict like:
      data["R2"][7] = "6688*"
      data["R2"][6] = "1688"
      ...
    where row "R2" etc. has col->cellValue.

    Returns a dict of patterns:
      {
         <canonical_pattern_str>: {
            "canonical": "...",
            "vtrac": "...",
            "locations": [(row, col), ...],
            "score": <int>,
            "score_breakdown": {...}
         },
         ...
      }
    """
    # 1) detect consensus in columns
    consensus_cols = {}
    for col in range(1, 8):  # col1..col7
        vals = []
        for rr in [2,4,6,8]:
            rkey = f"R{rr}"
            if rkey not in data or col not in data[rkey]:
                vals = []
                break
            cell_str = str(data[rkey][col]).strip()
            vals.append(cell_str)
        if len(vals) == 4 and vals[0] and all(v == vals[0] for v in vals):
            consensus_cols[col] = vals[0]

    # 2) gather digits from R1,R3,R5,R7 for mirror elimination
    prev_draw_digits = {}
    for rr in [1,3,5,7]:
        rkey = f"R{rr}"
        if rkey in data:
            ddset = set()
            for col_val in data[rkey].values():
                if col_val is None: 
                    continue
                s = str(col_val).strip()
                digs = "".join(ch for ch in s if ch.isdigit())
                ddset.update(digs)
            if ddset:
                prev_draw_digits[rkey] = ddset

    # 3) parse each cell in R2,R4,R6,R8 => extract clusters
    from collections import defaultdict
    clusters_by_col = defaultdict(list)
    clusters_by_row = defaultdict(list)
    hot_cells = {}

    for rr in [2,4,6,8]:
        rkey = f"R{rr}"
        if rkey not in data:
            continue
        for col, cell_val in data[rkey].items():
            dig_str, star_cnt = parse_cell(str(cell_val) if cell_val else "")
            if not dig_str:
                continue
            if star_cnt > 0:
                hot_cells[(rkey,col)] = star_cnt

            # extract clusters
            found_clusters = extract_cell_clusters(dig_str, min_len=3)
            for fc in found_clusters:
                fc["row"] = rkey
                fc["col"] = col
                clusters_by_col[col].append(fc)
                clusters_by_row[rkey].append(fc)

    # 4) identify vertical/horizontal patterns
    vertical_pts = find_vertical_patterns(clusters_by_col)
    vtrac_vertical_pts = find_vtrac_vertical_matches(clusters_by_col)
    horizontal_pts = find_horizontal_patterns(clusters_by_row)

    # 5) aggregate patterns by canonical
    patterns_dict = {}
    # first from vertical pts
    for col, cinfo in vertical_pts.items():
        for canon, info in cinfo.items():
            if canon not in patterns_dict:
                patterns_dict[canon] = {
                    "canonical": canon,
                    "vtrac": info["vtrac"],
                    "locations": []
                }
            for loc in info["locations"]:
                if loc not in patterns_dict[canon]["locations"]:
                    patterns_dict[canon]["locations"].append(loc)
    # also from row-based clusters
    for rr, c_list in clusters_by_row.items():
        for citem in c_list:
            cano = citem["canonical"]
            if cano not in patterns_dict:
                patterns_dict[cano] = {
                    "canonical": cano,
                    "vtrac": citem["canonical_vtrac"],
                    "locations": []
                }
            loc = (citem["row"], citem["col"])
            if loc not in patterns_dict[cano]["locations"]:
                patterns_dict[cano]["locations"].append(loc)

    # 6) apply scoring
    apply_scoring(patterns_dict,
                  vertical_pts,
                  vtrac_vertical_pts,
                  horizontal_pts,
                  consensus_cols,
                  hot_cells,
                  prev_draw_digits)

    return patterns_dict

) What Is the Mirror Elimination Stuff?
1a) The Concept of “Mirror Elimination”
•	In Pick-3 (and similar) lottery systems, mirror digits are those which differ by 5 modulo 10:
o	0 ↔ 5, 1 ↔ 6, 2 ↔ 7, 3 ↔ 8, 4 ↔ 9.
•	The code in apply_scoring tries to detect if a newly discovered pattern does not contain any digit from the immediately preceding draw (the previous row) or the mirror of those digits.
o	If the pattern’s digits are “emergent” (i.e. they do not appear in the prior draw or its mirror), it assigns a small “mirror elimination bonus” (like +2).
1b) Why Is It Useful?
•	This is meant to capture the idea that “new digits not recently seen (nor their mirrors) might be strong ‘fresh hits.’”
•	In many lottery strategies, if digits and their mirrors have been heavily used, you might “eliminate” them from predictions. The module awards a bonus to patterns whose digits appear to be “outside” that prior set + mirror set.
1c) In Practice:
1.	It checks which digits were used in the prior row—for example R2 checks R1, R4 checks R3, etc.
2.	It also adds each digit’s mirror to that “excluded set.”
3.	If your pattern’s digits are disjoint from that excluded set, you get a “mirror_elimination” bonus in your score_breakdown.
NOTE: If you do not want to handle mirror elimination in the stable‐pattern stage (maybe you prefer a separate “Mirror Analysis” module), you can remove or comment out that check.

o	Exact vertical repeat means the same substring appears in multiple rows of the same column. This is effectively a “straight match” in vertical direction.
o	Horizontal adjacency means the pattern (or its V‐Trac variant) appears in adjacent columns of the same row.

#!/usr/bin/env python3
"""
digit_reduction_module.py

Digit Reduction & Hidden Long‐String Patterns
============================================
This module complements the main stable pattern extractor by focusing on "longer" columns
or certain specialized areas, applying up to four approaches:
  - Method A: remove digits that appear in "past_draw_digits" exactly.
  - Method B: remove each digit or, if absent, its mirror, once.
  - Method C: remove each digit and its mirror entirely.
  - Method D: single-transit removal (iteratively remove each digit + mirror).

We detect newly revealed 3-value (<=3 unique digits or <=3 unique VTrac classes) patterns
after each reduction step, assigning scores. The aggregator can then merge these results
with the main stable-pattern module, hotzone, etc.

Covers:

  2) Long String Area #2 -> simpler columns for Set1 only (e.g., Draw4 col3 & Draw6 col1), rows=R2/R4/R6/R8

Each area has a separate function returning a structured "analysis_results" array.

#############################################
# 1. Mirror & VTrac Helpers
#############################################

MIRROR_MAP = {
    '0':'5','1':'6','2':'7','3':'8','4':'9',
    '5':'0','6':'1','7':'2','8':'3','9':'4'
}

def vtrac_transform(digits: str) -> str:
    """
    We apply a simple transform so that each digit maps to its "mirror" group.
    This is not the same as the standard 5-class VTrac,
    but it suffices to see if the string's digits are within 3 distinct mirror-classes.
    If you want the standard 5-group logic, you can adapt it.
    """
    return "".join(MIRROR_MAP[ch] if ch in MIRROR_MAP else ch for ch in digits)

def is_3value_pattern(digits_str: str) -> bool:
    """
    Returns True if 'digits_str' has <=3 unique actual digits
    OR <=3 unique 'mirror/vtrac' digits.
    """
    if not digits_str:
        return False
    unique_actual = set(digits_str)
    if len(unique_actual) <= 3:
        return True
    # else check vtrac
    vt = vtrac_transform(digits_str)
    unique_vt = set(vt)
    return (len(unique_vt) <= 3)

#############################################
# 2. Digit Reduction Methods (A, B, C)
#############################################

def method_a_remove(original: str, past_draw_digits: str) -> str:
    """
    Method A: remove each digit in past_draw_digits from the string,
              if present, as many times as it appears.
    """
    remove_set = set(past_draw_digits)
    return "".join(ch for ch in original if ch not in remove_set)

def method_b_remove(original: str, past_draw_digits: str) -> str:
    """
    Method B: For each digit d in past_draw_digits:
       - remove d once if present
       - else remove its mirror once if present
    """
    s_list = list(original)
    used_indices = set()
    for d in past_draw_digits:
        found_it = False
        # try removing d
        for i, ch in enumerate(s_list):
            if i in used_indices:
                continue
            if ch == d:
                used_indices.add(i)
                found_it = True
                break
        # if not found, remove mirror
        if (not found_it) and d in MIRROR_MAP:
            mirror_d = MIRROR_MAP[d]
            for i, ch in enumerate(s_list):
                if i in used_indices:
                    continue
                if ch == mirror_d:
                    used_indices.add(i)
                    found_it = True
                    break
        # else do nothing
    # rebuild
    final_str = "".join(
        ch for idx, ch in enumerate(s_list) if idx not in used_indices
    )
    return final_str

def method_c_remove(original: str, past_draw_digits: str) -> str:
    """
    Method C: For each digit d in past_draw_digits,
       remove *all occurrences* of d and d's mirror from 'original'
    """
    to_remove = set()
    for d in past_draw_digits:
        to_remove.add(d)
        if d in MIRROR_MAP:
            to_remove.add(MIRROR_MAP[d])
    return "".join(ch for ch in original if ch not in to_remove)

#############################################
# 3. Single-Digit Transit Approach (Method D)
#############################################

def transit_digit_removal(original: str, past_draw_digits: str):
    """
    For each digit d in 'past_draw_digits' (in order),
    remove d (and possibly its mirror).
    At each step, check if the new string forms a stable 3-value pattern.
    Return a list describing each step's outcome.
    """
    steps_data = []
    current = original
    for idx, d in enumerate(past_draw_digits, start=1):
        new_str = current.replace(d, '')
        if d in MIRROR_MAP:
            new_str = new_str.replace(MIRROR_MAP[d], '')
        stable_after = []
        if is_3value_pattern(new_str):
            stable_after.append(new_str)
        steps_data.append({
            "step_index": idx,
            "removed_digit": d,
            "result_str": new_str,
            "stable_3value": stable_after
        })
        current = new_str
    return steps_data

#############################################
# 4. Finding 3-value Substrings Within A String
#############################################

def find_3value_substrings(digits_str: str, min_len=3):
    """
    Return all substrings of length >=3 
    that are 3-value in actual or mirror sense.
    """
    res = set()
    n = len(digits_str)
    for size in range(min_len, n+1):
        for start in range(0, n-size+1):
            sub = digits_str[start:start+size]
            if is_3value_pattern(sub):
                res.add(sub)
    return list(res)

# 5. The Main Class

class LongStringDigitReduction:
    """
    This class covers the "long string" analysis 
    for both area #1 (Set3/Set2/Set1->Draw1, columns 7..5) 
    and area #2 (Set1 only, special columns).
    
    We:
     - parse the relevant cells
     - do pre-reduction stable substring detection
     - apply A/B/C methods for 'own' digits, 'combined' digits
     - apply single-digit approach (Method D)
     - score patterns
     - produce structured output for aggregator
    """

    def __init__(self, big_data: dict, config: dict=None):
        """
        big_data: your entire JSON-based data structure
        config: optional, containing e.g.:
          {
             "past_draw_digits_own": "341",
             "past_draw_digits_combined": "632",
             "scoring": {
                "initial_stable": 3,
                "method_a": 2,
                ...
             }
          }
        """
        self.data = big_data
        self.config = config or {}
        # scoring
        self.scoring = {
            "initial_stable": 3,
            "method_a": 2,
            "method_b": 2,
            "method_c": 2,
            "method_d": 3,
            "multi_method_bonus": 1
        }
        if "scoring" in self.config:
            self.scoring.update(self.config["scoring"])

    #######################################
    # 5A. Gathering logic for Area #1
    #######################################
    def gather_area1_cells(self):
        """
        area #1 => columns 7,6,5 in 
          Set3->Draw1, Set2->Draw1, Set1->Draw1
          for Midday, Evening, Combined
          rows=R2,R4,R6,R8
        Returns a list of (cell_id, original_str, meta).
        """
        results = []
        sections = ["Midday","Evening","Combined"]
        sets_ = ["Set3","Set2","Set1"]
        needed_columns = [7,6,5]
        needed_rows = ["R2","R4","R6","R8"]

        for sec in sections:
            if sec not in self.data["sections"]:
                continue
            sets_dict = self.data["sections"][sec]["sets"]
            for st in sets_:
                if st not in sets_dict:
                    continue
                ddict = sets_dict[st]["draws"]
                if "Draw1" not in ddict:
                    continue
                dd = ddict["Draw1"]
                if "pattern_variations" not in dd:
                    continue
                patvars = dd["pattern_variations"]
                for row_k in needed_rows:
                    if row_k not in patvars:
                        continue
                    row_list = patvars[row_k]
                    # row_list is from col7..col1 => indexes [0..6]
                    for i, col in enumerate(needed_columns):
                        if i >= len(row_list):
                            continue
                        original_str = row_list[i]
                        cell_id = f"{sec}|{st}|Draw1|{row_k}|col{col}"
                        meta = {
                            "section": sec,
                            "set": st,
                            "draw": "Draw1",
                            "row": row_k,
                            "col": col
                        }
                        results.append((cell_id, original_str, meta))
        return results

    #######################################
    # 5B. Gathering logic for Area #2
    #######################################
    def gather_area2_cells(self):
        """
        area #2 => for Set1 only,
          and columns => (Draw4 col3), (Draw6 col1),
          sections => [Midday,Evening,Combined],
          rows => R2,R4,R6,R8
        """
        results = []
        sections = ["Midday","Evening","Combined"]
        st = "Set1"
        # We want Draw4 col3 => index for col3 is (7-3)=4
        # We want Draw6 col1 => index for col1 is (7-1)=6
        # We'll map col-> index
        col_idx_map = {3:4, 1:6}
        draws_cols = [("Draw4", 3), ("Draw6", 1)]
        rows_of_interest = ["R2","R4","R6","R8"]

        for sec in sections:
            if sec not in self.data["sections"]:
                continue
            sets_dict = self.data["sections"][sec]["sets"]
            if st not in sets_dict:
                continue
            draws_dict = sets_dict[st]["draws"]
            for (dw, col) in draws_cols:
                if dw not in draws_dict:
                    continue
                ddata = draws_dict[dw]
                if "pattern_variations" not in ddata:
                    continue
                patvars = ddata["pattern_variations"]
                if col not in col_idx_map:
                    continue
                arr_index = col_idx_map[col]
                for row_k in rows_of_interest:
                    if row_k not in patvars:
                        continue
                    row_list = patvars[row_k]
                    if arr_index>=len(row_list):
                        continue
                    original_str = row_list[arr_index]
                    cell_id = f"{sec}|Set1|{dw}|{row_k}|col{col}"
                    meta = {
                        "section": sec,
                        "set": "Set1",
                        "draw": dw,
                        "row": row_k,
                        "col": col
                    }
                    results.append((cell_id, original_str, meta))
        return results

    #######################################
    # 5C. Internal function: process a single cell
    #######################################
    def process_cell(self, original_str: str, meta: dict) -> dict:
        """
        1) Find stable 3-value substrings prior to reduction
        2) For each method (A,B,C) with own & combined digits
        3) Single-digit approach (Method D) with own & combined
        4) Score patterns
        5) Return structured cell-level data
        """
        # We'll store everything in a result dict
        out = {
            "metadata": meta,
            "original_string": original_str,
            "initial_stable": [],
            "method_a": {},
            "method_b": {},
            "method_c": {},
            "method_d": {},
            "final_score": 0,
            "notable_patterns": []
        }
        score_map = defaultdict(int)

        # Step1: pre-reduction stable
        stable_subs = find_3value_substrings(original_str)
        for sb in stable_subs:
            score_map[sb] += self.scoring["initial_stable"]
        out["initial_stable"] = list(stable_subs)

        # Past draw digits
        p_own = self.config.get("past_draw_digits_own", "")
        p_comb = self.config.get("past_draw_digits_combined", "")

        def apply_method(method_func, method_key, raw_string, digits):
            new_str = method_func(raw_string, digits)
            found = find_3value_substrings(new_str)
            for fz in found:
                score_map[fz] += self.scoring[method_key]
            return {
                "past_draw_digits": digits,
                "reduced_string": new_str,
                "found_3value_substrings": found
            }

        # Method A
        out["method_a"]["own"] = apply_method(method_a_remove, "method_a", original_str, p_own)
        out["method_a"]["combined"] = apply_method(method_a_remove, "method_a", original_str, p_comb)

        # Method B
        out["method_b"]["own"] = apply_method(method_b_remove, "method_b", original_str, p_own)
        out["method_b"]["combined"] = apply_method(method_b_remove, "method_b", original_str, p_comb)

        # Method C
        out["method_c"]["own"] = apply_method(method_c_remove, "method_c", original_str, p_own)
        out["method_c"]["combined"] = apply_method(method_c_remove, "method_c", original_str, p_comb)

        # Single digit approach (D)
        def do_transit_d(raw_string, digits):
            steps = transit_digit_removal(raw_string, digits)
            discovered = []
            for stp in steps:
                for pat in stp["stable_3value"]:
                    discovered.append(pat)
                    score_map[pat] += self.scoring["method_d"]
            return {
                "past_draw_digits": digits,
                "transit_steps": steps,
                "all_discovered": list(set(discovered))
            }
        out["method_d"]["own"] = do_transit_d(original_str, p_own)
        out["method_d"]["combined"] = do_transit_d(original_str, p_comb)

        # multi-method bonus if a pattern was found in 2+ ways
        method_presence = defaultdict(set)
        # from initial_stable
        for sb in stable_subs:
            method_presence[sb].add("initial")
        # from method A/B/C
        for mk in ["method_a","method_b","method_c"]:
            for subcat in ["own","combined"]:
                fl = out[mk][subcat]["found_3value_substrings"]
                for fz in fl:
                    method_presence[fz].add(mk)
        # from method_d
        for subcat in ["own","combined"]:
            dd = out["method_d"][subcat]
            for fz in dd["all_discovered"]:
                method_presence[fz].add("method_d")

        for pat, ways in method_presence.items():
            if len(ways) >= 2:
                score_map[pat] += self.scoring["multi_method_bonus"]

        # finalize patterns
        final_list = []
        for pat, scv in score_map.items():
            if scv>0:
                final_list.append((pat, scv))
        final_list.sort(key=lambda x: x[1], reverse=True)
        out["notable_patterns"] = [
            {"pattern": fp[0], "score": fp[1]} for fp in final_list
        ]
        out["final_score"] = final_list[0][1] if final_list else 0

        return out

    #######################################
    # 5D. Main calls
    #######################################
    def analyze_area1_long_strings(self):
        """
        Gathers 'Area #1' cells, processes them, returns a structured dict.
        """
        cell_list = self.gather_area1_cells()
        analysis_output = []
        for (cid, orig_str, meta) in cell_list:
            cell_res = self.process_cell(orig_str, meta)
            cell_res["cell_id"] = cid
            analysis_output.append(cell_res)
        return {
            "module_name": "LongStringReduction_Area1",
            "timestamp": datetime.datetime.now().isoformat(),
            "analysis_results": analysis_output
        }

    def analyze_area2_long_strings(self):
        """
        Gathers 'Area #2' cells, processes them, returns a structured dict.
        """
        cell_list = self.gather_area2_cells()
        analysis_output = []
        for (cid, orig_str, meta) in cell_list:
            cell_res = self.process_cell(orig_str, meta)
            cell_res["cell_id"] = cid
            analysis_output.append(cell_res)
        return {
            "module_name": "LongStringReduction_Area2",
            "timestamp": datetime.datetime.now().isoformat(),
            "analysis_results": analysis_output
        }

o	In your system, they are basically the same initial digits (since R2, R4, R6, R8 in that box typically match).
o	But it’s simpler to do the elimination once per row in code, then move on.
o	The result is the same stable patterns for R2 col7 vs R4 col7 if the strings match exactly. Or you might discover an edge case if they differ by a star or some small difference.
3.	After we have done that for each row, the final aggregator or synergy step can see that “the same stable pattern was discovered in R2 col7, R4 col7, R6 col7, R8 col7,” giving it a big “vertical coverage or vertical stability” bonus.
In short, the code is not ignoring or losing the fact that “R2/R4/R6/R8 are the same.” It’s just that the code is structured so each row is processed individually for the digit removal step. The final synergy is a separate step. This approach is typical in many multi‐pass designs: do transformations row by row, then unify across rows.
________________________________________
3. “But If R2/R4 Have the Same Digits, Why Bother Reducing Both?”
3.1 Real‐World “Star” or “Hot Indications” Differences
In your data, sometimes R2 might have a star (6688*) while R4 does not (6688)—or vice versa. That might lead to a different weighting or a difference in how you want to interpret that row’s “hotness.” So you might end up with slightly different code branches or notes.
3.2 Future or Past Ties
In the full “box” approach, you might store which row had something “excluded from a prior draw,” or which row belongs to midday vs. evening if you plan to do cross‐midday synergy. That can sometimes lead to small differences in the digit removal approach or the final scoring. So even though the initial strings might appear the same, you keep them separated so you can track the “where did we find it?” location.
3.3 It's Usually Zero Extra Complexity
In practice, if R2 col7 and R4 col7 truly have the same string, they get the same digit‐reduced result anyway. The code path is the same. Then your aggregator sees “the same stable pattern discovered in R2 col7 and R4 col7,” awarding synergy for that. If they are truly identical, you effectively get the same outcome, but it is simpler for the code to handle each row systematically than to do “if row is identical to row above, skip.”
________________________________________
4. So Does the Code “Follow the PSAT Logic”?
Yes, it does in the sense that:
1.	We gather columns 7,6,5 from sets 3,2,1 (the “long string area #1”), or we gather those special columns in area #2.
2.	For each row (R2, R4, R6, R8) in that box, we do digit removal with methods A/B/C + single transit. We find newly stable 3‐value patterns.
3.	We record which patterns were discovered, in which column, row, set, etc.
4.	You then do a synergy step (or aggregator step) to see how that pattern persists across boxes or how it lines up across R2→R4→R6→R8.
Hence if the same pattern is revealed for R2 col7, R4 col7, R6 col7, and R8 col7, you see that in the aggregator as “vertical coverage” or “R2=R4=R6=R8 → stable.”
If it also reappears from col7 → col6 → col5, you might note “horizontal coverage or column progression.”
If it also appears in Set2 or Set3, you note “persistence.”
All of that is part of the final synergy—not in the raw digit‐reduction code.
________________________________________
5. Summary of Where “Vertical Stability” & “Set3→Set2→Set1 Progression” Appear
•	Digit‐Reduction is purely about removing digits from each big string (like “668811059324”) and discovering newly stable patterns. It does that for each row in each relevant box.
•	Stability across the same column or across multiple sets (Set3→Set2→Set1) is not done in that same function. Instead:
1.	Either you do a second pass in the same module that checks “did we see the same pattern in R2 col7 and R4 col7?” or “did col7 in Set3’s result match col7 in Set2’s result?”
2.	Or (as is typical) you do that synergy in the final aggregator, which sees all discovered patterns from all modules and merges them.
________________________________________
6. Are We “Wasting Time” Because R2=R4=R6=R8?
No. Even if R2=R4=R6=R8 are the same string in a single box, we still do the same digit reduction code for each row. The outcome is usually identical stable patterns (unless row differences exist). Then the synergy step merges them. This is normal in code design; a row‐by‐row approach is simpler to maintain than special‐casing “the same row string.”

2.	Digit Reduction:

#!/usr/bin/env python3
"""
hotzones_module.py

Comprehensive Hot Zone Analysis
===============================
This module scans your 'data' structure for R2/R4/R6/R8 cells flagged
with hot (*) or super-hot (**) indicators. It then:

1) Collects these cells from Midday/Evening/Combined, sets, draws.
2) Performs single-digit transit removal (Method D) to reveal hidden
   3-value or 3-digit patterns.
3) Checks for local synergy:
   - Patterns repeated across R2/R4/R6/R8 in the same column
   - Minimal cross-section synergy (Midday vs. Evening vs. Combined)
   - Potential "consensus" if R2=R4=R6=R8 are identical
4) Scores patterns with local weighting for:
   - hot vs. super-hot
   - stable after transit removal
   - discovered in multiple rows (or multiple sections)
   - minimal V-Trac synergy
   - "consensus" detection if the entire column has the same short string
5) Outputs a structured dictionary, letting the aggregator unify
   these findings with the rest (Pattern Extractor, LongStrings, etc.).

Author: [Your Name / Team]
Date: [Date]
"""

from collections import defaultdict

#####################################################
# 1) Mirror & V-Trac Helpers
#####################################################
MIRROR_MAP = {
    '0': '5','1': '6','2': '7','3': '8','4': '9',
    '5': '0','6': '1','7': '2','8': '3','9': '4'
}

def vtrac_transform(digits_str: str) -> str:
    """
    Transforms each digit to its 'mirror' to represent V-Trac classes.
    If you prefer the strict classic V-Trac mapping (0/5->1,1/6->2,...),
    you can adapt this function. Here we just do a mirror-based approach
    that lumps digits in pairs, effectively capturing a 2-group pattern.
    We'll produce a *sorted* version so we can unify patterns that differ
    only in order.
    """
    mapped = [MIRROR_MAP.get(ch, ch) for ch in digits_str]
    return ''.join(sorted(mapped))


#####################################################
# 2) Single-Digit Transit Reduction
#####################################################
def remove_one_digit_and_mirror(string: str, digit: str, remove_mirror=True) -> str:
    """
    Removes the given 'digit' from 'string', and if remove_mirror=True,
    also removes its mirror partner (e.g. if digit='3', remove '8' too).
    Returns the resulting string after removal.
    """
    result = list(string)
    # remove 'digit'
    result = [ch for ch in result if ch != digit]

    if remove_mirror and digit in MIRROR_MAP:
        mir = MIRROR_MAP[digit]
        result = [ch for ch in result if ch != mir]

    return "".join(result)

def single_transit_iterative_reduction(original: str, past_digits: list, remove_mirror=True):
    """
    Iteratively removes each digit in 'past_digits' from 'original',
    also removing its mirror if remove_mirror=True.
    Returns a list of intermediate strings (including the final).
    Example: step by step transformation that might reveal stable patterns.
    """
    results = [original]
    current = original
    for d in past_digits:
        new_str = remove_one_digit_and_mirror(current, d, remove_mirror)
        if new_str != current:
            results.append(new_str)
            current = new_str
        # else no change if digit not found
    return results


#####################################################
# 3) 3-Value/3-Digit Pattern Detection for Short Strings
#####################################################
def is_3value_pattern(digits_str: str) -> bool:
    """
    Returns True if the string has <=3 unique digits (actual or VTrac),
    i.e. a '3-value' pattern in your methodology.
    Here we check actual digits first; if it fails, we check mirror-based VTrac.
    """
    if not digits_str:
        return False
    unique_actual = set(digits_str)
    if len(unique_actual) <= 3:
        return True
    # else check mirror-based
    vt = vtrac_transform(digits_str)
    if len(set(vt)) <= 3:
        return True
    return False

def find_all_3value_substrings(digits_str: str, min_len=3):
    """
    For a short string (typically <=6..7 digits in hot zone),
    find all substrings of length >=3 that are '3-value' in actual or V-Trac sense.
    Return them as a set of substring patterns (no duplicates).
    """
    found = set()
    n = len(digits_str)
    for length in range(min_len, n+1):
        for start in range(0, n-length+1):
            sub = digits_str[start:start+length]
            if is_3value_pattern(sub):
                found.add(sub)
    return found

def is_straight_equiv(p1: str, p2: str) -> bool:
    """Check if two patterns are exactly the same => 'straight' match."""
    return p1 == p2

def is_box_equiv(p1: str, p2: str) -> bool:
    """
    Check if p1 and p2 are 'box' equivalents => same digits in different order.
    Sort them and compare.
    """
    return ''.join(sorted(p1)) == ''.join(sorted(p2))


#####################################################
# 4) Scoring Function
#####################################################
def score_hotzone_pattern(
    pattern: str,
    hot_level: int,
    stable_after_reduction: bool,
    found_in_multiple_rows: bool,
    found_in_multiple_sections: bool,
    is_consensus_column: bool,
    is_box: bool,
    is_straight: bool,
    vtrac_repeated: bool
) -> int:
    """
    Example scoring for a discovered pattern in a hot-zone cell.
    Feel free to tweak the numeric weights to your preference.
    """
    score = 0

    # Base: found in a hot zone => +5
    score += 5

    # hot_level => * or **
    if hot_level == 1:   # '*'
        score += 2
    elif hot_level == 2: # '**'
        score += 4

    # if stable after single-digit transit => +2
    if stable_after_reduction:
        score += 2

    # synergy for multiple rows (R2/R4/R6/R8) => +3
    if found_in_multiple_rows:
        score += 3

    # synergy for multiple sections (Midday/Evening/Combined) => +2
    if found_in_multiple_sections:
        score += 2

    # if the entire column was a "consensus" (R2=R4=R6=R8 identical) => +3
    if is_consensus_column:
        score += 3

    # extra if it is a "straight" 3-digit pattern => +1
    # or if it's a "box" match => +1
    # (some people prefer to weigh straights more heavily, etc.)
    if is_straight:
        score += 1
    elif is_box:
        score += 1

    # vtrac_repeated => +2 if it matched the same vtrac pattern in another row
    if vtrac_repeated:
        score += 2

    return score

class HotZonesAnalyzer:
    """
    Encapsulates the scanning of R2/R4/R6/R8 cells to find hot/super-hot boxes,
    apply single-digit transit, detect stable 3-value patterns,
    do minimal synergy checks (rows, sections), detect consensus columns, etc.
    """

    def __init__(self, data, config=None):

        data: your entire JSON-like structure (OntarioCanada4 ...).

            "past_own_digits_midday": [...],
            "past_own_digits_evening": [...],
            "past_combined_digits": [...],
            ...
        }
        The config can store the lists of digits you want removed
        in single-digit transit approach for each section, or
        store any scoring overrides. We'll show a simple approach.
        """
        self.data = data
        self.config = config or {}
        # Example of how you might store different "past_draw_digits"
        # for midday vs. evening vs. combined:
        self.past_draw_digits_midday = self.config.get("past_draw_digits_midday", [])
        self.past_draw_digits_evening = self.config.get("past_draw_digits_evening", [])
        self.past_draw_digits_combined = self.config.get("past_draw_digits_combined", [])
        # If you want the same for all, just set them all to the same list.

    def _get_past_draw_digits_for_section(self, section_name):
        """
        Return the appropriate 'past draw digits' for single-digit transit,
        depending on if the current section is 'Midday', 'Evening', or 'Combined'.
        If you have a special approach for 'Combined', do it here.
        """
        if section_name == "Midday":
            return self.past_draw_digits_midday
        elif section_name == "Evening":
            return self.past_draw_digits_evening
        else:
            # combined
            return self.past_draw_digits_combined

    def analyze_hotzones(self):
        """
        Main entry: scans the entire data structure for hot zone or super-hot zone
        cells in R2/R4/R6/R8. Does single-digit transit to find stable patterns.
        Does minimal synergy checks: row synergy, cross-section synergy,
        consensus columns. Returns a structured result.
        """
        # We'll create a data structure for all discovered patterns:
        # "results" => list of patterns
        all_discovered = []

        # Also track discovered patterns in a local dictionary so we can see
        # if the same pattern appears in multiple rows or multiple sections.
        # Key might be (section, set, draw, col_index) -> row-> set_of_patterns
        # Then we do a second pass to label synergy.
        local_tracker = defaultdict(lambda: defaultdict(set))

        # We'll also store "consensus check" if R2/R4/R6/R8 are identical strings
        # in the same column. We'll do that in one pass as we read them.

        sections = self.data.get("sections", {})
        for section_name, section_data in sections.items():
            sets_dict = section_data.get("sets", {})
            for set_key, set_val in sets_dict.items():
                draws_dict = set_val.get("draws", {})
                for draw_key, draw_val in draws_dict.items():
                    metadata = draw_val.get("metadata", {})
                    hot_zone_indicators = metadata.get("hot_zone_indicators", {})
                    # We also can detect if the entire draw is flagged is_hot_zone => or partial
                    # Typically you said we rely on the star indicator in each cell.

                    pattern_variations = draw_val.get("pattern_variations", {})

                    # We'll gather column_count from any row's pattern_variations to do a consensus check
                    # We'll do a small approach: columns = length of the array in R2, etc.
                    row_arrays = {r: pattern_variations[r] for r in ["R2","R4","R6","R8"] if r in pattern_variations}

                    # 1) Determine consensus columns (R2=R4=R6=R8 identical raw string):
                    # We'll do a dictionary consensus_cols = { col_index : True/False } per row. 
                    # If we find that col_index is identical across all 4, we mark it True.
                    col_count = max((len(arr) for arr in row_arrays.values()), default=0)
                    consensus_flags = [False]*col_count
                    # only if we have all 4 rows:
                    if len(row_arrays) == 4:
                        for c_idx in range(col_count):
                            # we gather each row's raw string if it exists
                            strings = []
                            for rtype in ["R2","R4","R6","R8"]:
                                rowarr = row_arrays[rtype]
                                val = rowarr[c_idx] if (c_idx < len(rowarr)) else ""
                                strings.append(val)
                            # check if all are identical and non-empty
                            if strings[0] and all(s == strings[0] for s in strings):
                                consensus_flags[c_idx] = True

                    # 2) Now for each row R2,R4,R6,R8 => for each column => if star indicator => gather patterns
                    for rtype in ["R2","R4","R6","R8"]:
                        if rtype not in pattern_variations:
                            continue
                        row_str_array = pattern_variations[rtype]
                        # hot_zone_indicators[rtype] => an array of "" or "*" or "**"
                        row_indicators_array = hot_zone_indicators.get(rtype, [])

                        for col_idx, raw_string in enumerate(row_str_array):
                            if not raw_string:
                                continue
                            # check star indicator
                            star_val = ""
                            if col_idx < len(row_indicators_array):
                                star_val = row_indicators_array[col_idx]

                            if star_val not in ("*", "**"):
                                # not a hot zone cell, skip
                                continue

                            hot_level = 2 if (star_val == "**") else 1
                            is_consensus_column = bool(consensus_flags[col_idx])

                            # 3) Single-digit transit approach
                            # pick the relevant "past_draw_digits" for this section
                            own_digits = self._get_past_draw_digits_for_section(section_name)

                            # e.g. single transit iterative
                            intermediate_strings = single_transit_iterative_reduction(raw_string, own_digits, remove_mirror=True)

                            # 4) find stable patterns in the original raw + all intermediates
                            discovered_patterns = set()
                            # check raw
                            raw_subs = find_all_3value_substrings(raw_string)
                            for s in raw_subs:
                                discovered_patterns.add( (s, False) )  # (pattern, stable_after_reduction=False)
                            # check intermediate
                            for idx_str in intermediate_strings[1:]: # skip index 0 => raw
                                subs = find_all_3value_substrings(idx_str)
                                for sc in subs:
                                    discovered_patterns.add( (sc, True) )

                            # Store them in local_tracker so we can do synergy pass
                            # local_tracker[(section, set, draw, col_idx)][rtype] => set_of_patterns
                            for (patt, stable_after_red) in discovered_patterns:
                                local_tracker[(section_name, set_key, draw_key, col_idx)][rtype].add(patt)

                            # We'll store a preliminary record now (some synergy we can do later).
                            for (patt, stable_after_red) in discovered_patterns:
                                all_discovered.append({
                                    "section": section_name,
                                    "set": set_key,
                                    "draw": draw_key,
                                    "rtype": rtype,
                                    "column_index": col_idx,
                                    "raw_string": raw_string,
                                    "hot_level": hot_level,
                                    "pattern": patt,
                                    "pattern_len": len(patt),
                                    "stable_after_reduction": stable_after_red,
                                    "is_consensus_column": is_consensus_column
                                ])

        # 6) Now we do a second pass: synergy annotation
        # We'll build a final list with synergy-based scores.
        final_results = []
        # We'll store a helper so we can see if pattern repeats in multiple rows or multiple sections.
        # We'll also store (section, set, draw, col_idx)-> list of sections found => to detect cross-section synergy

        # For cross-section synergy (Midday/Evening/Combined):
        # we can check if the same pattern in the same (set, draw, col_idx) but different sections.
        # We'll do a small local approach. Full synergy is aggregator's job.

        # Convert all_discovered to a better structure for synergy lookups
        # Key: (set, draw, col_idx, pattern), then store all (section, rtype, stable_after_reduction, is_consensus_col, hot_level)
        synergy_map = defaultdict(list)
        for item in all_discovered:
            synergy_key = (
                item["set"],
                item["draw"],
                item["column_index"],
                item["pattern"]
            )
            synergy_map[synergy_key].append(item)

        # Build the final
        for synergy_key, item_list in synergy_map.items():
            # item_list are all the occurrences of the same pattern in the same set/draw/column
            # but possibly in different sections or different rows in that column
            # We'll unify them.
            set_k, draw_k, col_i, pattern_k = synergy_key

            # gather synergy info
            rtypes_found = set()
            sections_found = set()
            stable_after_red_any = False
            consensus_col_any = False
            hot_level_max = 0

            # check if it's "straight" or "box" => we can store that by comparing
            # the pattern to its sorted self
            is_straight = False
            is_box = False
            sorted_patt = ''.join(sorted(pattern_k))
            if pattern_k == sorted_patt:
                # That might mean it's "straight" in ascending order, but not necessarily "straight"
                # Typically "straight" means 'exact same digits in same order' across multiple occurrences.
                # We'll do a simpler approach: "straight" means pattern length <= 3 and not rearranged.
                # Or just set is_straight = True => if we want to consider the discovered pattern "straight" by itself.
                # We'll do a simple approach: if the pattern is 3 digits and we didn't reorder => is_straight = True
                if len(pattern_k) == 3:
                    is_straight = True
            else:
                # if the pattern has the same sorted form as itself => trivially. Let's define "box" = pattern length == 3
                if len(pattern_k) == 3:
                    is_box = True

            # For a more accurate approach: typically "straight" vs. "box" is about duplicates. We'll keep it simple.

            # We'll do V-Trac approach
            vt_form = vtrac_transform(pattern_k)
            # We'll see if the same vt_form appears in multiple rows. For that, we need to check item_list's rows:
            # Actually let's do a simpler approach: if multiple occurrences in item_list => vtrac repeated = True
            vtrac_repeated = False  # aggregator might do bigger synergy. We'll do minimal local check if there's >1 row
            # We'll gather the vtrac of pattern_k across item_list => it's the same pattern, so same vt_form.
            # If it appears in multiple rtypes, we can call that a local vtrac synergy.

            for it in item_list:
                rtypes_found.add(it["rtype"])
                sections_found.add(it["section"])
                hot_level_max = max(hot_level_max, it["hot_level"])
                if it["stable_after_reduction"]:
                    stable_after_red_any = True
                if it["is_consensus_column"]:
                    consensus_col_any = True

            # if we found multiple rtypes => that means pattern repeated across rows
            found_in_multiple_rows = (len(rtypes_found) > 1)
            found_in_multiple_sections = (len(sections_found) > 1)
            # if found_in_multiple_rows => we can consider that vtrac_repeated or a local synergy
            # We'll define vtrac_repeated => True if found_in_multiple_rows for now
            if found_in_multiple_rows:

######################################################
# 1) UTILITY: Canonicalization, V-Trac, Rotation
######################################################

MIRROR_MAP = {
    '0': '5', '5': '0',
    '1': '6', '6': '1',
    '2': '7', '7': '2',
    '3': '8', '8': '3',
    '4': '9', '9': '4'
}


def vtrac_digit(d: str) -> str:
    """
    Convert a single digit to its VTrac digit:
      0,5 -> 1
      1,6 -> 2
      2,7 -> 3
      3,8 -> 4
      4,9 -> 5
    """
    # We can interpret 'mirror' as 'vtrac' but more standard is:
    # VTrac(0)=1, VTrac(5)=1, VTrac(1)=2, ...
    # Let's do the direct approach:
    val = int(d)
    # group = val%5 + 1
    group = (val % 5) + 1
    return str(group)


def vtrac_transform(num_str: str) -> str:
    """Return the VTrac string for num_str, e.g. '509' -> '115'."""
    return ''.join(vtrac_digit(ch) for ch in num_str if ch.isdigit())


def canonical_rotation(s: str) -> str:
    """
    Return the lexicographically smallest rotation of string s.
    E.g., '312' => '123'.
    This helps group cyc-shift patterns together.
    """
    if not s:
        return s
    rotations = [s[i:] + s[:i] for i in range(len(s))]
    return min(rotations)


def canonical_pattern(num_str: str) -> str:
    """
    A multi-step approach to get a canonical representative for
    a multi-digit pattern:
      1) Possibly sort or do rotation-min to unify cyc-shifts
      2) Possibly keep the pattern as 'rotation-min' if you consider cyc
         the same pattern, but preserve exact digit order if you want 'straight' vs. 'box'.

    In many systems, you might do a rotation-min *or* a sorted form.
    But let's do rotation-min to unify cyc-shifts.
    """
    return canonical_rotation(num_str)


def same_pattern_check(p1: str, p2: str, check_rotation=True, check_sorted=False) -> bool:
    """
    Check if p1 and p2 represent the same pattern under rotation or box equivalence.
    This can be customized.
    - If check_rotation=True, we unify cyc-shifts.
    - If check_sorted=True, we unify sorted forms => '123' ~ '321'.
    Return True if they unify, else False.
    """
    if p1 == p2:
        return True
    if check_rotation:
        # compare canonical rotations
        return canonical_rotation(p1) == canonical_rotation(p2)
    if check_sorted:
        return ''.join(sorted(p1)) == ''.join(sorted(p2))
    return False

def collect_stable_patterns(stable_output) -> list:
    """
    stable_output: the dictionary returned by stable_extractor, e.g.:
    {
      "patternA": {
         "canonical": ...,
         "vtrac": ...,
         "locations": [...],
         "score": ...,
         ...
      },
      ...
    }
    Return a list of dicts, each representing a discovered pattern
    with enough info to unify later.
    """
    # The stable extractor used "patterns_dict" keyed by canonical.
    results = []
    for patt_key, info in stable_output.items():
        # We assume "canonical" is info["canonical"]
        # vtrac might be info["vtrac"]
        # total score is info["score"]
        # locations is info["locations"]
        results.append({
            "module_name": "stable_extractor",
            "pattern_str": patt_key,  # or info["canonical"]
            "canonical_str": info["canonical"],  # rotation-min of actual digits
            "vtrac_str": info["vtrac"],          # rotation-min of vtrac
            "locations": info.get("locations", []),
            "module_score": info.get("score", 0),
            "score_breakdown": info.get("score_breakdown", {}),
        })
    return results


def collect_longstring_patterns(long_output) -> list:
    """
    long_output typically is:
    {
      "module_name": "LongStringReduction_Area1",
      "analysis_results": [
         {
           "cell_id": "...",
           "metadata": {...},
           "original_string": "...",
           "initial_stable_patterns": [...],
           "method_a": {...},
           "method_b": {...},
           "method_c": {...},
           "method_d": {...},
           "notable_patterns": [
             {"pattern":"...", "score": 12}, ...
           ],
           "final_score": ... 
         },
         ...
      ]
    }
    We want to unify these "notable_patterns" as discovered patterns.
    """
    if not long_output:
        return []

    results = []
    analysis_results = long_output.get("analysis_results", [])
    for cell_info in analysis_results:
        cell_id = cell_info.get("cell_id", "")
        final_score = cell_info.get("final_score", 0)
        # We go through "notable_patterns"
        for p in cell_info.get("notable_patterns", []):
            patt = p["pattern"]
            sc = p["score"]
            # We'll define vtrac from the aggregator approach or
            # we can do it from a subfunction. But let's do it now:
            vstr = vtrac_transform(patt)
            # canonical rotation for aggregator
            cstr = canonical_pattern(patt)

            results.append({
                "module_name": long_output.get("module_name", "longstring_reduction"),
                "cell_id": cell_id,
                "pattern_str": patt,
                "canonical_str": cstr,
                "vtrac_str": canonical_pattern(vstr),  # or just vstr
                "module_score": sc,
                "locations": [cell_id],  # we only know it from that cell
                "score_breakdown": {},
            })
    return results


def collect_hotzone_patterns(hotzone_output) -> list:
    """
    hotzone_output is typically:
    {
      "hotzone_patterns": [
         {
           "section": "Midday",
           "set": "Set1",
           "draw": "Draw3",
           "rtype":"R2",
           "column_index": 3,
           "raw_digits":"9138",
           "pattern":"241",
           "local_hotzone_score":12,
           "vtrac_form":"679",
           "stable_after_reduction": True,
           "hot_level": 2,
         },
         ...
      ]
    }
    We'll unify them into a list of aggregator-friendly dicts.
    """
    if not hotzone_output or "hotzone_patterns" not in hotzone_output:
        return []

    results = []
    for item in hotzone_output["hotzone_patterns"]:
        patt = item.get("pattern", "")
        sc = item.get("local_hotzone_score", 0)
        vstr = item.get("vtrac_form", "")
        # aggregator's canonical
        cstr = canonical_pattern(patt)
        # or we can unify cstr with item.get(...) if they've used rotation-min
        # but let's ensure we do it consistently here
        results.append({
            "module_name": "hotzones",
            "pattern_str": patt,
            "canonical_str": cstr,
            "vtrac_str": canonical_pattern(vstr),
            "module_score": sc,
            "locations": [(
                item.get("section"),
                item.get("set"),
                item.get("draw"),
                item.get("rtype"),
                item.get("column_index")
            )],
            "score_breakdown": {},
            "hot_level": item.get("hot_level", 0),
            "stable_after_reduction": item.get("stable_after_reduction", False),
        })

    return results

def aggregate_all_modules(
    data,
    stable_out,
    longstring_out,
    hotzone_out,
    config=None
):
    """
    Merges the pattern lists from stable extractor, digit reduction, hot zones.
    Applies synergy scoring across modules, sets, sections, vtrac synergy, etc.

    data: original big data if needed for referencing sets, or for additional
          cross-check
    stable_out: dict from stable_extractor
    longstring_out: dict (or list) from digit_reduction
    hotzone_out: dict (or list) from hot_zones
    config: synergy weighting rules, e.g.:
      {
        "w_found_in_multiple_modules": 5,
        "w_cross_set": 3,
        "w_cross_section": 4,
        "w_length_bonus": 2,
        "w_if_3digit": 3,
        "w_vtrac_consistency": 2,
        ...
      }
    """

    if config is None:
        config = {}
    # define some default synergy weights
    synergy_weights = {
        "found_in_multiple_modules": 5,
        "found_in_2_modules": 3,
        "found_in_3_modules": 5,
        "cross_set_persistence": 3,
        "cross_section_synergy": 3,
        "length_bonus_per_digit_over_3": 1,
        "bonus_if_exact_3digit": 2,
        "vtrac_straight_bonus": 2,
        "mirror_elimination_bonus": 2,
        "hot_level_mul": 2,   # for super-hot we might multiply or add
        # etc.
    }
    synergy_weights.update(config)

    # 1) unify all discovered patterns from each module
    stable_list = collect_stable_patterns(stable_out)  # from stable extractor
    long_list   = []
    if isinstance(longstring_out, dict):
        # single area or multiple area outputs
        if "analysis_results" in longstring_out:
            # single area
            long_list = collect_longstring_patterns(longstring_out)
        else:
            # maybe we have multiple area outputs => combine
            # e.g. { "area1":..., "area2":... }
            # you can adapt this as needed
            for k, v in longstring_out.items():
                if isinstance(v, dict) and "analysis_results" in v:
                    long_list.extend(collect_longstring_patterns(v))
    elif isinstance(longstring_out, list):
        # might be a list of area outputs
        for area_item in longstring_out:
            if "analysis_results" in area_item:
                long_list.extend(collect_longstring_patterns(area_item))

    hotz_list = []
    if isinstance(hotzone_out, dict) and "hotzone_patterns" in hotzone_out:
        hotz_list = collect_hotzone_patterns(hotzone_out)
    elif isinstance(hotzone_out, list):
        # or a list of multiple hotzone results
        for h in hotzone_out:
            if isinstance(h, dict) and "hotzone_patterns" in h:
                hotz_list.extend(collect_hotzone_patterns(h))

    all_patterns = stable_list + long_list + hotz_list

    # 2) group them by a "super-canonical key" to unify duplicates
    # define a function that merges (canonical_str, length, vtrac_str)
    # Possibly we also consider sorted or rotation-min or box/straight.
    # For demonstration, let's unify by (canonical_rotation, len, sorted VTrac).
    grouped = defaultdict(list)
    for item in all_patterns:
        cstr = item["canonical_str"]
        length = len(cstr)
        vt = item["vtrac_str"]

        # define group key
        group_key = (cstr, length, vt)
        grouped[group_key].append(item)

    # 3) For each group, compute synergy
    results = []
    for (group_key, items) in grouped.items():
        # items are patterns from possibly multiple modules
        # let’s unify them into a single "super pattern" entry
        cstr, length, vt = group_key
        # gather info
        modules_found = set()
        total_locations = []
        total_module_score = 0.0

        # track hot levels, stable_after_reduction, other features
        max_hot_level = 0
        found_stable_after_reduction = False

        # keep track if found in different sets or sections
        sets_found = set()
        sections_found = set()

        # also track total pattern_str for debug
        # or we store them all
        pattern_str_examples = set()

        for it in items:
            modules_found.add(it["module_name"])
            pattern_str_examples.add(it["pattern_str"])
            total_module_score += it["module_score"]

            # accumulate location info
            locs = it.get("locations", [])
            for lc in locs:
                total_locations.append(lc)

            # hot-level
            hl = it.get("hot_level", 0)
            if hl > max_hot_level:
                max_hot_level = hl

            if it.get("stable_after_reduction", False):
                found_stable_after_reduction = True

            # if we have set/draw in location we parse them:
            # e.g. (section, set, draw, rtype, col) or a cell_id
            # We attempt to extract set or section if possible
            # This depends on each item’s structure
            # For "stable_extractor" we might have row/col, but not set.
            # For "hotzones" we have (section, set, draw, rtype, col).
            # We'll do a best-effort approach
            if isinstance(lc, tuple) and len(lc) >= 2:
                # e.g. (section, set, draw, rtype, col)
                ssec = lc[0]
                sset = lc[1]
                if isinstance(ssec, str):
                    sections_found.add(ssec)
                if isinstance(sset, str):
                    sets_found.add(sset)

        synergy_score = 0
        synergy_breakdown = {}

        # 3a) base module synergy
        # if pattern found in multiple modules => synergy
        module_count = len(modules_found)
        if module_count >= 2:
            if module_count == 2:
                synergy_score += synergy_weights["found_in_2_modules"]
                synergy_breakdown["found_in_2_modules"] = synergy_weights["found_in_2_modules"]
            elif module_count >= 3:
                synergy_score += synergy_weights["found_in_3_modules"]
                synergy_breakdown["found_in_3_modules"] = synergy_weights["found_in_3_modules"]

        # 3b) cross-set synergy (e.g., if pattern found in multiple sets: Set1,Set2,Set3)
        if len(sets_found) >= 2:
            # If it appears in 2 or 3 sets => add synergy
            val = synergy_weights["cross_set_persistence"] * (len(sets_found) - 1)
            synergy_score += val
            synergy_breakdown["cross_set_persistence"] = val

        # 3c) cross-section synergy (midday, evening, combined)
        if len(sections_found) >= 2:
            val = synergy_weights["cross_section_synergy"] * (len(sections_found) - 1)
            synergy_score += val
            synergy_breakdown["cross_section_synergy"] = val

        # 3d) pattern length considerations
        # if exactly 3-digit => +some
        # if >3 => maybe add a small bonus per digit beyond 3
        if length == 3:
            val = synergy_weights.get("bonus_if_exact_3digit", 2)
            synergy_score += val
            synergy_breakdown["bonus_if_exact_3digit"] = val
        elif length > 3:
            # e.g. +1 per digit beyond 3
            extra = length - 3
            val = extra * synergy_weights.get("length_bonus_per_digit_over_3", 1)
            synergy_score += val
            synergy_breakdown["length_bonus"] = val

        # 3e) vtrac synergy - if the pattern’s actual digits match vtrac in a 'straight' sense
        # or if we want to see if vtrac is an all-same-digit or something
        # For now, let’s just do a small bonus for having a valid vtrac transform
        if vt and len(set(vt)) <= 3:
            # treat that as a VTrac synergy
            val = synergy_weights["vtrac_straight_bonus"]
            synergy_score += val
            synergy_breakdown["vtrac_straight_bonus"] = val

        # 3f) hot-level synergy
        # if max_hot_level==2 => super-hot => maybe +N or multiply
        if max_hot_level == 2:
            # e.g. +4 synergy
            addv = synergy_weights["hot_level_mul"] * 2
            synergy_score += addv
            synergy_breakdown["super_hot_bonus"] = addv
        elif max_hot_level == 1:
            # +2 synergy
            addv = synergy_weights["hot_level_mul"]
            synergy_score += addv
            synergy_breakdown["hot_bonus"] = addv

        # 3g) stable_after_reduction synergy
        if found_stable_after_reduction:
            synergy_breakdown["stable_after_reduction"] = 1
            synergy_score += 1

        # 3h) sum up the module’s own total scores (the aggregator can partially incorporate them)
        synergy_score += total_module_score
        synergy_breakdown["sum_of_module_scores"] = total_module_score

        # Possibly, we do additional logic for repeated permutations,
        # mirror elimination, or advanced “straight vs. box” checks. 
        # For example, if any pattern_str is exactly 3 digits in same order => 
        # synergy_breakdown["straight_box_bonus"] = ...
        # if a pattern doesn't contain any digit from the last draw => synergy_breakdown["mirror_elimination_bonus"] ...
        # etc. We'll skip a full example, but here's a placeholder:
        # synergy_score += synergy_weights["mirror_elimination_bonus"] (if something is found)
        # synergy_breakdown["mirror_elimination_bonus"] = synergy_weights["mirror_elimination_bonus"]

        # finalize
        sp = {
            "canonical_str": cstr,
            "pattern_length": length,
            "vtrac_str": vt,
            "all_pattern_variants": list(pattern_str_examples),  # e.g. '662', '626', '266'
            "modules_found": list(modules_found),
            "sets_found": list(sets_found),
            "sections_found": list(sections_found),
            "locations_aggregate": total_locations,
            "final_synergy_score": synergy_score,
            "synergy_breakdown": synergy_breakdown
        }
        results.append(sp)

    # 4) sort results by synergy desc
    results.sort(key=lambda x: x["final_synergy_score"], reverse=True)

    # wrap up
    return {
        "aggregator_module": "Advanced Synergy Aggregator",
        "timestamp": datetime.datetime.now().isoformat(),
        "pattern_count": len(results),
        "patterns": results
    }

ggregator & Synergy Analysis Module – Part 1 (Design & Implementation)
Overview
Part 1 of the Aggregator & Synergy Analysis Module focuses on integrating and scoring pattern outputs from all previous modules: Pattern Extractor, Digit Reduction (Long String Stability), Hot Zones Analysis, and V-Trac transformations. It consolidates patterns from Midday, Evening, and Combined analyses, then groups equivalent patterns and evaluates them using various scoring criteria. The goal is to identify patterns that are recurrent, stable, cross-confirmed by multiple analyses, “pending” in recent draws, or part of synergistic clusters. This forms a foundation for Part 2 (where advanced prediction and final number selection will occur) by providing a structured list of patterns with rich metadata and preliminary scores.
Pattern Aggregation & Normalization
In this stage, the module accepts inputs from all sources (Midday, Evening, Combined) and organizes them into a unified view. Patterns that are essentially the same (regardless of order or minor variations) are grouped by a canonical form. The canonical form chosen here is the sorted digits of the pattern (i.e. the “boxed” form of a Pick-3 number). For example, patterns "321" and "123" are grouped together under the canonical key "123". This ensures that permutations of the same combination are treated as one underlying pattern group.
Each pattern group preserves source metadata about where it came from. This includes:
•	Section (e.g. Midday, Evening, Combined)
•	Set (e.g. Set1, Set2, Set3 indicating different draw segments or time frames)
•	Draw reference (specific draw identifiers or indices)
•	Result type (rtype) such as "straight" or "boxed" if applicable (e.g. from the Pattern Extractor or stability analysis)
•	Hotness level (e.g. no mark, * for Hot, ** for Super Hot from Hot Zones analysis)
Additionally, each pattern is mapped to its V-Trac representation for further grouping. V-Trac is a transformation that maps actual digits 0–9 into 5 abstract groups (mirror pairs) to reveal deeper relationships. Based on the provided V-Trac table, digits are paired as: 0↔5, 1↔6, 2↔7, 3↔8, 4↔9. Using this, a Pick-3 pattern can be converted into a V-Trac code (each digit replaced by its group number). For instance, pattern "579" would map to V-Trac groups 1-5-5 (since 5→1, 7→3, 9→5) which we denote as v135 when sorted. The image below (excerpt from the supplied V-Trac table) illustrates how “Boxed V-Trac” combinations correspond to actual Pick 3 patterns, highlighting the grouping of singles, doubles, and triples. The aggregator uses this mapping to group patterns by V-Trac category as well, which is crucial for detecting cross-pattern synergies (explained later).
Each unique pattern group (identified by its sorted digits) is stored in a structure with its aggregated occurrences and metadata from all modules. This normalized grouping avoids duplication and allows holistic scoring across all evidences of that pattern.
Scoring Methodology (Part 1 Scope)
Once patterns are aggregated, the module computes a comprehensive score for each pattern. The scoring in Part 1 is based on a combination of factors that reflect the pattern’s stability, frequency across analyses, recency, and synergy with other patterns. Key components of the scoring include:
•	Stable Pattern Base Score: If a pattern is identified as stable by the Digit Reduction (Long String Stability) module, it receives a base score bonus. These patterns survived the reduction process, indicating a consistent presence. (Marked as stable_after_reduction = True in output.)
•	Stable Boxed/Straight Bonus: Patterns that were found to repeat in exact form (straight) or in any order (boxed) across draws get additional points. For example, if the Pattern Extractor noted a number repeating straight in consecutive draws, or repeating as a combination in different order, it indicates higher stability. (The module checks the rtype metadata: e.g. rtype="straight" or "boxed".)
•	V-Trac Grouping Bonus: Simply belonging to a particular V-Trac group can be informative. In this foundational stage, we account for V-Trac groupings by enabling cluster synergy (discussed later) and by mapping each pattern to its V-Trac code for reference. (Each output entry shows vtrac_group, e.g. v123.) While we do not give a flat score to every V-Trac mapping (since all patterns have one), the V-Trac concept is used in synergy scoring to identify patterns that share a group.
•	Pattern Length Bonus: Patterns containing more digits than usual get a small bonus. The typical pattern length is 3 (for Pick-3 combos); if a longer digit sequence emerges (say a 4-digit sequence from the long string analysis that might encompass a Pick-3), it’s slightly rewarded for carrying more information. For instance, a 4-digit stable sequence would score a bit higher than a 3-digit sequence. (In practice, each digit beyond 2 adds a small increment – so a 3-digit pattern gets a +1, a 4-digit gets +2, etc.)
•	Cross-Module Confirmation: If a pattern appears in multiple different modules’ outputs, it’s considered more robust. We increase the score when more than one analysis method flags the same pattern. For example, a pattern found in Long String and also highlighted in Hot Zones (and/or Pattern Extractor) will earn a higher score than one found by only a single module. Each additional distinct module that confirms the pattern contributes a bonus. (We consider the module type, ignoring whether it was Midday/Evening, since this is about different analytical techniques.)
•	Repeats Across Draw Sets: Patterns that persist across time – appearing in analyses of sequential draw sets – get a score increase. For instance, if a pattern showed up in older data (Set3), then again in more recent data (Set2 and Set1), it indicates a continuing trend. The more consecutive sets a pattern spans, the higher this cross-draw repetition bonus. This captures temporal stability: a pattern that keeps resurfacing as draws progress. (In implementation, for each section (Midday/Evening/Combined), we check how many distinct sets the pattern was present in. Each additional set beyond the first in the same section yields a bonus. e.g. Pattern in Set3 and Set2 of Midday → bonus, if also in Set1 → larger bonus.)
•	“Super Hot” Indicator: If any occurrence of the pattern was marked with ** (Super Hot) in the Hot Zones analysis, it signifies an extremely active pattern. Such patterns get an extra boost in the score. (They are likely among the top-tier frequently occurring or highly weighted patterns in Hot Zones.)
All these components contribute to a score breakdown that is calculated for each pattern. Importantly, these base and bonus scores are additive – a pattern might earn multiple bonuses if it satisfies multiple criteria. For example, a stable pattern that appears in two modules and was marked super-hot will accumulate several bonuses, reflecting its strong overall significance.
Pending Pattern Logic (Recency Check)
A critical part of this module is identifying “pending” patterns, which are patterns that have not shown up in recent actual draws, and might be “due” to appear. We compare each pattern against the last 5 draws (using the Combined recent draws data). If a pattern’s digits have NOT appeared in any form in the last 5 draws, we mark it as pending (hasnt_hit_last5 = True) and award a pending bonus. This check is done in a thorough way: we consider the pattern in any permutation (boxed form) and also its V-Trac equivalent. For instance, if the pattern is “123”, we look for any permutation of 1-2-3 (like 312 or 231) in the last five draws, and also any number that shares the same V-Trac group as 1-2-3. If none are found, the pattern is truly absent recently. Such a pattern gets a score increase because, in many lottery pattern strategies, something that hasn’t hit in a while may be more likely upcoming (due to regression to mean or rotation of hot/cold periods).
Crucially, no pattern is penalized for having appeared recently – they simply won’t get the pending bonus if they showed up. All other base scores still apply. This ensures that we don’t ignore patterns just because they hit recently (since they might hit again), but we give a slight edge to those that haven’t hit in the last few draws.
Pattern Clustering & Synergy Detection
Beyond evaluating patterns individually, the module begins to identify clusters of patterns that reinforce each other. Specifically, we look for cases where multiple different patterns point to the same underlying V-Trac group or share a permutation group. If several patterns from different modules all map into the same V-Trac combination (and that V-Trac group itself hasn’t appeared in recent draws), this is a strong sign of a synergistic clue. In such cases, each pattern in this cluster gets an additional cluster synergy bonus.
For example, suppose Long String analysis yielded pattern “128” and Hot Zones yielded “367”. These are different numbers, but if we convert them to V-Trac we might find both map to, say, v234 (hypothetically). If neither any permutation of 128/367 nor any pattern from V-Trac group 234 has shown in recent draws, this cluster of patterns all hinting at group 234 would be significant. The module would tag that V-Trac group as a cluster and boost the score of all patterns in that group. This encourages patterns that are inter-related (by the mirror logic of V-Trac) to surface to the top of the list together. (In our implementation, we collect all pattern groups by their V-Trac code and check for groups that have at least two distinct patterns coming from at least two different module sources. If so, and if that V-Trac group hasn’t hit recently, we add a cluster bonus to each member pattern.)
This clustering logic in Part 1 is kept basic (“begin detecting clusters”) – we identify the condition and reward those patterns. In Part 2, this could be expanded to form actual cluster predictions or combined patterns for number picking. But for now, it gives an extra nudge to patterns that are individually good and also part of a broader convergence signal.
Output Structure
The output of Part 1 is a structured dictionary containing an entry for each unique pattern (group). Each entry includes the pattern’s canonical form and various metadata and scoring details. Specifically, for each pattern we provide:
•	pattern: The pattern’s canonical digit string (usually the sorted digits representing the combination in boxed form).
•	source_modules: A list of sources where the pattern was found, combining section and module name (e.g. "Midday LongString", "Evening HotZones"). This gives a quick view of how widespread the pattern is across analyses.
•	vtrac_group: The V-Trac group code for the pattern, denoted with a "v" prefix (e.g. v123, v355). This corresponds to the mirrored group of digits the pattern falls into, as per the V-Trac table.
•	stable_after_reduction: True/False indicating if the pattern was identified as stable in the digit reduction (long string) process.
•	hasnt_hit_last5: True/False indicating if the pattern (in any permutation or V-Trac equivalent) did not appear in the last 5 actual draws. (True means it’s “pending” or due; False means it has hit recently.)
•	score_breakdown: A dictionary breaking down the scoring components for transparency. Each factor that contributed to the score is listed with the points it added (e.g. stable_pattern_base, cross_module_bonus, pending_bonus, etc.), so one can see why the pattern got its total score.
•	score: The final aggregated score computed for the pattern (sum of all applicable components).
The dictionary keys themselves are the canonical pattern strings. The module is designed for scalability and modularity – new scoring factors can be added easily, and the weight of each factor is configurable. This ensures the system can be tuned or expanded (for example, if we integrate new modules or adjust the scoring logic in Part 2, it can plug into this structure without major refactoring).

def aggregate_synergy_analysis_part1(module_outputs, last5_draws):
    """
    Integrates outputs from Pattern Extractor, Long String Stability, Hot Zones, etc. 
    Groups patterns and computes scores based on cross-module presence, stability, recency, and synergy.
    
    Parameters:
        module_outputs (dict): Nested dict of patterns from each section & module.
            Format example:
            {
              "Midday": {
                "LongString": [ { "pattern": "123", "set": "Set3", "draw": ..., "rtype": "straight", "hot": None }, ... ],
                "HotZones":   [ { "pattern": "321", "set": "Set2", "draw": ..., "rtype": None, "hot": "*" }, ... ],
                ... 
              },
              "Evening": { ... },
              "Combined": { ... }
            }
        last5_draws (list of str): Last 5 actual draw results (e.g. ["045","672","128","093","556"]) for recency checks.
    
    Returns:
        dict: Aggregated patterns with metadata and scoring breakdown.
    """
    # V-Trac digit mapping (mirror pairs as per the V-Trac system).
    vtrac_map = {'0': '1', '5': '1',
                 '1': '2', '6': '2',
                 '2': '3', '7': '3',
                 '3': '4', '8': '4',
                 '4': '5', '9': '5'}
    
    # Scoring weights (tunable parameters for modularity)
    BASE_STABLE_REDUCTION = 5     # base score if stable from long string reduction
    BASE_STABLE_BOX = 3          # base score if identified as stable (boxed)
    BASE_STABLE_STRAIGHT = 3     # base score if identified as stable (straight)
    LENGTH_BONUS_PER_DIGIT = 1   # bonus per digit beyond length 2
    CROSS_MODULE_BONUS = 2       # bonus for each additional module that confirms the pattern
    CROSS_SET_BONUS = 2          # bonus for each additional draw set (within the same section) the pattern appears in
    SUPER_HOT_BONUS = 2          # bonus if pattern is marked "**" super hot in Hot Zones
    PENDING_BONUS = 3            # bonus if pattern hasn't appeared in last 5 draws
    CLUSTER_BONUS = 3            # bonus if pattern is part of a V-Trac cluster synergy
    
    # Helper function to get sorted canonical form of a pattern (for grouping permutations)
    def canonical_form(pattern_str):
        return ''.join(sorted(pattern_str))
    
    # Helper function to compute the V-Trac code of a pattern (sorted to canonicalize the group)
    def get_vtrac_code(pattern_str):
        # Map each digit to its V-Trac value and sort the resulting code
        return ''.join(sorted(vtrac_map[d] for d in pattern_str))
    
    # 1. Aggregate patterns from all modules into a dictionary grouped by canonical form
    aggregated = {}
    for section, modules in module_outputs.items():
        for module_name, patterns in modules.items():
            for entry in patterns:
                pat = entry["pattern"]
                canon = canonical_form(pat)  # e.g. "321" -> "123"
                if canon not in aggregated:
                    aggregated[canon] = {
                        "pattern": canon,
                        "occurrences": []  # list to accumulate metadata of all instances
                    }
                # Preserve source metadata for each occurrence of the pattern
                aggregated[canon]["occurrences"].append({
                    "section": section,
                    "module": module_name,
                    "set": entry.get("set"),
                    "draw": entry.get("draw"),
                    "rtype": entry.get("rtype"),
                    "hot": entry.get("hot")
                })
    
    # Precompute sets for last5 draws in sorted and V-Trac form (for quick lookup)
    last5_sorted_set = { canonical_form(x) for x in last5_draws }
    last5_vtrac_set = { get_vtrac_code(x) for x in last5_draws }
    
    # 2. Identify V-Trac clusters: find V-Trac groups with multiple patterns from different modules
    vtrac_clusters = set()  # set of V-Trac codes that qualify as synergy clusters
    # Map each V-Trac code to all pattern groups that have that code
    vtrac_to_patterns = {}
    for canon, data in aggregated.items():
        v_code = get_vtrac_code(canon)
        vtrac_to_patterns.setdefault(v_code, []).append(canon)
    # Check each V-Trac group for cluster conditions
    for v_code, pattern_list in vtrac_to_patterns.items():
        if len(pattern_list) < 2:
            continue  # need at least 2 different patterns to form a cluster
        # Gather distinct module names (ignoring section) from all patterns in this group
        module_names = set()
        for pat in pattern_list:
            for occ in aggregated[pat]["occurrences"]:
                module_names.add(occ["module"])
        # If patterns originate from at least 2 different analysis modules and group hasn’t hit recently, mark cluster
        if len(module_names) >= 2 and v_code not in last5_vtrac_set:
            vtrac_clusters.add(v_code)
    
    # 3. Compute scores for each pattern group
    results = {}
    for canon, data in aggregated.items():
        occ_list = data["occurrences"]
        # Initialize aggregated attributes for this pattern
        source_modules = set()
        module_names = set()
        section_set_pairs = set()
        stable_flag = False
        stable_box_flag = False
        stable_straight_flag = False
        super_hot_flag = False
        
        # Consolidate info from all occurrences
        for occ in occ_list:
            section = occ["section"]
            module = occ["module"]
            set_id = occ.get("set")
            rtype = occ.get("rtype")
            hot_level = occ.get("hot")
            
            source_modules.add(f"{section} {module}")
            module_names.add(module)
            if section and set_id:
                section_set_pairs.add((section, set_id))
            # Check for stability flags
            if module.lower() in ["longstring", "digit reduction", "digit_reduction", "digitreduction"]:
                stable_flag = True  # came from Long String Stability analysis
            if rtype:
                if rtype.lower() in ["boxed", "box"]:
                    stable_box_flag = True
                if rtype.lower() == "straight":
                    stable_straight_flag = True
            # Check hot level for super hot
            if hot_level and str(hot_level).strip() == "**":
                super_hot_flag = True
        
        # Determine if pattern is "pending" (not seen in last 5 draws in any form)
        v_code = get_vtrac_code(canon)
        if canon not in last5_sorted_set and v_code not in last5_vtrac_set:
            pending_flag = True
        else:
            pending_flag = False
        
        # Score breakdown calculation
        score_breakdown = {}
        total_score = 0
        # Base scores for stability
        if stable_flag:
            score_breakdown["stable_pattern_base"] = BASE_STABLE_REDUCTION
            total_score += BASE_STABLE_REDUCTION
        if stable_box_flag:
            score_breakdown["stable_box_base"] = BASE_STABLE_BOX
            total_score += BASE_STABLE_BOX
        if stable_straight_flag:
            score_breakdown["stable_straight_base"] = BASE_STABLE_STRAIGHT
            total_score += BASE_STABLE_STRAIGHT
        # Bonus for pattern length beyond 2 digits
        pattern_len = len(canon)
        if pattern_len > 2:
            length_bonus = (pattern_len - 2) * LENGTH_BONUS_PER_DIGIT
            score_breakdown["length_bonus"] = length_bonus
            total_score += length_bonus
        # Cross-module confirmation bonus
        if len(module_names) > 1:
            # Each additional module beyond the first adds bonus
            cross_mod_bonus = (len(module_names) - 1) * CROSS_MODULE_BONUS
            score_breakdown["cross_module_bonus"] = cross_mod_bonus
            total_score += cross_mod_bonus
        # Cross-draw (across sets) repetition bonus
        # Calculate per section how many sets, then give bonus per extra set in each section
        cross_draw_bonus = 0
        sets_by_section = {}
        for (sec, set_id) in section_set_pairs:
            sets_by_section.setdefault(sec, set()).add(set_id)
        for sec, sets in sets_by_section.items():
            if len(sets) > 1:
                cross_draw_bonus += (len(sets) - 1) * CROSS_SET_BONUS
        if cross_draw_bonus > 0:
            score_breakdown["cross_draws_bonus"] = cross_draw_bonus
            total_score += cross_draw_bonus
        # Super Hot bonus
        if super_hot_flag:
            score_breakdown["super_hot_bonus"] = SUPER_HOT_BONUS
            total_score += SUPER_HOT_BONUS
        # Pending pattern bonus
        if pending_flag:
            score_breakdown["pending_bonus"] = PENDING_BONUS
            total_score += PENDING_BONUS
        # Cluster synergy bonus
        if v_code in vtrac_clusters:
            score_breakdown["cluster_synergy_bonus"] = CLUSTER_BONUS
            total_score += CLUSTER_BONUS
        
        # Compile final result entry
        results[canon] = {
            "pattern": canon,
            "source_modules": sorted(source_modules),
            "vtrac_group": f"v{v_code}",
            "stable_after_reduction": stable_flag,
            "hasnt_hit_last5": pending_flag,
            "score_breakdown": score_breakdown,
            "score": total_score
        }
    return results

# Mirror-based or V-Trac mapping:
VTRAC_MAP = {
    '0':'1','5':'1',
    '1':'2','6':'2',
    '2':'3','7':'3',
    '3':'4','8':'4',
    '4':'5','9':'5'
}

def canonical_sort_digits(digit_str):
    """ Sort digits to unify permutations (like a 'box' form). """
    return ''.join(sorted(digit_str))

def vtrac_transform(digit_str):
    """
    Convert each digit to its V-Trac group,
    then sort the result for consistent grouping.
    """
    # Filter out any non-digit if present
    mapped = []
    for ch in digit_str:
        if ch.isdigit() and ch in VTRAC_MAP:
            mapped.append(VTRAC_MAP[ch])
    return ''.join(sorted(mapped))

def is_3value_pattern(digit_str):
    """
    Check if up to 3 unique digits. 
    This is how we confirm it's a "3-value" cluster.
    """
    return len(set(digit_str)) <= 3

def in_last_n_draws(digit_str, last_ndraws, check_vtrac=True):
    """
    See if 'digit_str' or any box/permutation or V-Trac code 
    appears in `last_ndraws`. Each draw in last_ndraws is 
    presumably a 3-digit string, or maybe 4-digit. 
    We'll assume 3-digit for simplicity.
    Return True if it appears, else False.
    """
    # We'll gather all permutations (boxed) + vtrac if desired
    # But let's keep it simple:
    cform = canonical_sort_digits(digit_str)   # box
    vt = vtrac_transform(digit_str)

    # Build sets of last draws in canonical-sorted form, plus vtrac form
    last_canon = {canonical_sort_digits(d) for d in last_ndraws}
    last_vtrac = {vtrac_transform(d) for d in last_ndraws}

    # If cform or vt is in those sets => we consider it "recently appeared"
    if cform in last_canon:
        return True
    if check_vtrac and vt in last_vtrac:
        return True
    return False

def collect_stable_patterns(stable_out):
    """

        ...
      }
    We'll transform them into a standard list of items.
    """
    results = []
    for k,info in stable_out.items():
        # k might be the pattern key or canonical
        pat_str = k  
        # The code might store "canonical" as info["canonical"] 
        # but let's trust 'k' is the actual pattern. Adapt as needed.
        item = {
            "module_source": "StableExtractor",
            "pattern_str": pat_str,
            "score_in_module": info.get("score",0),
            # We can store if it was stable_after_reduction if that module sets a flag
            "stable_after_reduction": False,
            "is_3value": (len(set(pat_str)) <= 3),  # guess
            "locations": info.get("locations", []),
            # We can store a "lingers_across_boxes" if the module indicated so, or we check length of locations
            "lingers_across_boxes": False,  
            "hot_level": 0,  # stable_extractor typically doesn't do stars
        }
        results.append(item)
    return results

def collect_longstring_patterns(long_out):
    """
    If the long_out is structured similarly to your previous code,
    we parse 'analysis_results' => 'notable_patterns'.
    Each 'notable_pattern' has "pattern", "score", maybe a "stable_after_reduction" 
    or we can infer it from method outputs.
    """
    # Could be single or multiple area outputs
    results = []
    if not long_out:
        return results

    def handle_area(one_area):
        out = []
        arr = one_area.get("analysis_results", [])
        for cellinfo in arr:
            # we can do "notable_patterns" 
            nps = cellinfo.get("notable_patterns", [])
            for np in nps:
                pat = np["pattern"]
                sc = np["score"]
                # Checking if cellinfo indicated stable after reduction? 
                # Possibly we check cellinfo["method_a"] or method_d 
                # for stable insights.
                # We'll do a simplified approach:
                stable_flag = False
                if cellinfo.get("final_score",0) >= 10:  # random threshold
                    stable_flag = True
                item = {
                    "module_source":"LongStringReduction",
                    "pattern_str": pat,
                    "score_in_module": sc,
                    "stable_after_reduction": stable_flag,
                    "is_3value": (len(set(pat)) <= 3),
                    "locations": [cellinfo.get("cell_id","")],
                    "lingers_across_boxes": False, # or True if the data indicates
                    "hot_level": 0,
                }
                out.append(item)
        return out

    # If it's dict with "analysis_results": handle directly
    if isinstance(long_out, dict) and "analysis_results" in long_out:
        results = handle_area(long_out)
    elif isinstance(long_out, dict):
        # might be multiple area keys
        for area_name, area_val in long_out.items():
            if isinstance(area_val, dict) and "analysis_results" in area_val:
                results.extend(handle_area(area_val))
    elif isinstance(long_out, list):
        # might be a list of area outputs
        for area in long_out:
            if isinstance(area, dict) and "analysis_results" in area:
                results.extend(handle_area(area))
    return results

### 3) Master Aggregation & Synergy Scoring ###

def aggregator_synergy_all(
    stable_output, 
    longstring_output,
    hotzone_output,
    last6_draws,
    synergy_config=None
):
    """
    Merges stable, longstring, hotzone patterns,
    then calculates synergy-based final scores.
    last6_draws: e.g. ["045","672","128","093","556","241"] 
        used to determine pending logic (not in these draws).
    synergy_config: optional dict to override scoring weights.
    """
    if synergy_config is None:
        synergy_config = {}
    # Default synergy weights
    weights = {
        "base_module_score_factor": 1.0,       # how much we trust each module's internal score
        "cross_module_bonus_per_extra": 3,     # synergy if found in multiple modules
        "cross_set_bonus_per_extra": 2,        # synergy if found in multiple sets
        "cross_section_bonus_per_extra": 2,    # synergy if found in multiple sections (Midday,Evening,Combined)
        "hot_star_bonus": 2,                   # if hot_level=1
        "hot_super_bonus": 4,                  # if hot_level=2
        "stable_after_reduction_bonus": 3,
        "linger_across_boxes_bonus": 2,
        "three_value_bonus": 2,                # if is_3value
        "pending_bonus": 3,                    # if not in last6_draws
        "vtrac_cluster_bonus": 2,              # if multiple patterns share same vtrac, or if repeated in multiple modules
        "length_bonus_per_digit_over_3":1      # each digit beyond 3 => +1
    }
    weights.update(synergy_config)

    ### 3a) Collect from each module
    stable_list = collect_stable_patterns(stable_output)
    long_list   = collect_longstring_patterns(longstring_output)
    hotz_list   = collect_hotzone_patterns(hotzone_output)
    # Potentially other modules...

    all_items = stable_list + long_list + hotz_list

    ### 3b) Group them by a "super key"
    # We'll group by (canonical_sort_digits, is_3value, pattern_length) 
    # and also store the vtrac code for synergy. 
    grouped = defaultdict(list)

    for it in all_items:
        pat_str = it["pattern_str"]
        cform   = canonical_sort_digits(pat_str)
        plen    = len(pat_str)
        is3v    = it["is_3value"]
        grouped[(cform,plen,is3v)].append(it)

    ### (Optional) We'll also track the vtrac code for each group so we can find clusters
    # We'll do a pass to map vtrac_code => list of group keys
    vtrac_code_map = defaultdict(set)  # vtrac_code => set of group_keys
    for gkey, recs in grouped.items():
        # pick the first rec to get a pattern, or check them all 
        # since cform is a box, let's pick any item
        # We'll just do the cform for vtrac since cform is sorted digits
        (cform, plen, is3v) = gkey
        vt_code = vtrac_transform(cform)  # cform is sorted digits
        vtrac_code_map[vt_code].add(gkey)

    ### Identify vtrac clusters: if a vt_code maps to multiple group_keys from different modules
    # We'll define "vtrac synergy" if a code has 2+ group_keys from 2+ distinct modules. 
    vtrac_clusters = set()
    for vt_code, gkeys in vtrac_code_map.items():
        if len(gkeys) < 2:
            continue
        # check if there's at least 2 distinct module_source across these group keys
        module_sources = set()
        for gk in gkeys:
            for rec in grouped[gk]:
                module_sources.add(rec["module_source"])
        if len(module_sources) >= 2:
            # define this vt_code as cluster
            vtrac_clusters.add(vt_code)

    ### 3c) Score each group with synergy
    final_results = []

    for gkey, recs in grouped.items():
        cform, plen, is3v = gkey
        # gather everything
        module_names = set()
        sections_found = set()  # "Midday","Evening","Combined"
        sets_found = set()      # "Set1","Set2","Set3" or whatever
        hot_level_max = 0
        stable_after_red_flag = False
        linger_flag = False
        sum_module_scores = 0.0

        # We might have multiple pattern_str among these recs 
        # (like "213","321","123"), all with the same cform.
        pattern_variants = set()
        
        for r in recs:
            pattern_variants.add(r["pattern_str"])
            module_names.add(r["module_source"])
            # accumulate sum of module's own scores (optionally scaled)
            sum_module_scores += (r["score_in_module"] * weights["base_module_score_factor"])

            # stable after reduction
            if r.get("stable_after_reduction",False):
                stable_after_red_flag = True
            # linger across boxes
            if r.get("lingers_across_boxes",False):
                linger_flag = True
            # hot level
            hl = r.get("hot_level",0)
            if hl>hot_level_max:
                hot_level_max=hl

            # parse location for set/section if we have it
            locs = r.get("locations",[])
            for loc in locs:
                # loc could be e.g. ("Midday","Set1","Draw3","R2",3) or a string
                if isinstance(loc, (list,tuple)) and len(loc)>=2:
                    ssec = loc[0]
                    sset = loc[1]
                    if isinstance(ssec,str):
                        sections_found.add(ssec)
                    if isinstance(sset,str):
                        sets_found.add(sset)

        # build synergy
        synergy_score = 0
        synergy_breakdown = {}

        # 1) sum of module scores
        synergy_score += sum_module_scores
        synergy_breakdown["sum_module_scores"] = sum_module_scores

        # 2) cross-module synergy
        if len(module_names)>1:
            # for each extra module beyond the first, add synergy
            extra_mod_count = len(module_names)-1
            val = extra_mod_count * weights["cross_module_bonus_per_extra"]
            synergy_score += val
            synergy_breakdown["cross_module_synergy"] = val

        # 3) cross-set synergy
        # We can do an approach: if pattern found in multiple sets, synergy
        # We'll do the same with sections
        if len(sets_found)>1:
            val = (len(sets_found)-1)*weights["cross_set_bonus_per_extra"]
            synergy_score += val
            synergy_breakdown["cross_set_synergy"] = val

        # 4) cross-section synergy
        if len(sections_found)>1:
            val = (len(sections_found)-1)*weights["cross_section_bonus_per_extra"]
            synergy_score += val
            synergy_breakdown["cross_section_synergy"] = val

        # 5) hot-level synergy
        if hot_level_max==1:
            synergy_score += weights["hot_star_bonus"]
            synergy_breakdown["hot_star_bonus"] = weights["hot_star_bonus"]
        elif hot_level_max==2:
            synergy_score += weights["hot_super_bonus"]
            synergy_breakdown["hot_super_bonus"] = weights["hot_super_bonus"]

        # 6) stable_after_reduction
        if stable_after_red_flag:
            synergy_score += weights["stable_after_reduction_bonus"]
            synergy_breakdown["stable_after_reduction_bonus"] = weights["stable_after_reduction_bonus"]

        # 7) linger_across_boxes
        if linger_flag:
            synergy_score += weights["linger_across_boxes_bonus"]
            synergy_breakdown["linger_across_boxes_bonus"] = weights["linger_across_boxes_bonus"]

        # 8) three_value
        if is3v:
            synergy_score += weights["three_value_bonus"]
            synergy_breakdown["three_value_bonus"] = weights["three_value_bonus"]

        # 9) pattern length bonus
        if plen>3:
            extra = (plen-3)*weights["length_bonus_per_digit_over_3"]
            synergy_score+=extra
            synergy_breakdown["length_bonus"] = extra

        # 10) pending logic => if not in last6_draws
        # We'll pick any pattern_variant to check or check them all
        # Let's define "recently appeared" if ANY variant is in last6. 
        # So only if none appear do we do pending bonus
        recently_appeared = False
        for varp in pattern_variants:
            if in_last_n_draws(varp,last6_draws,check_vtrac=True):
                recently_appeared = True
                break
        if not recently_appeared:
            synergy_score += weights["pending_bonus"]
            synergy_breakdown["pending_bonus"] = weights["pending_bonus"]

        # 11) vtrac synergy => if this vtrac_code belongs to a cluster
        vt_code = vtrac_transform(cform)  # cform is sorted digits
        if vt_code in vtrac_clusters:
            synergy_score += weights["vtrac_cluster_bonus"]
            synergy_breakdown["vtrac_cluster_bonus"] = weights["vtrac_cluster_bonus"]

        # store final result
        final_entry = {
            "canonical_form": cform,
            "pattern_length": plen,
            "pattern_variants": list(pattern_variants),
            "modules_found": list(module_names),
            "sections_found": list(sections_found),
            "sets_found": list(sets_found),
            "is_3value": is3v,
            "max_hot_level": hot_level_max,
            "final_synergy_score": synergy_score,
            "synergy_breakdown": synergy_breakdown,
        }
        final_results.append(final_entry)

    # sort descending
    final_results.sort(key=lambda x: x["final_synergy_score"], reverse=True)

    # wrap in a dict for convenience
    return {
        "aggregator_name": "Advanced Synergy Aggregator",
        "timestamp": datetime.datetime.now().isoformat(),
        "pattern_count": len(final_results),
        "patterns": final_results
    }

How This Addresses Your Key Concerns
1.	Repeating / Lingering 3-Value Patterns
o	Each record can set is_3value=True if it’s up to 3 unique digits.
o	If your earlier modules specifically flagged “this pattern repeated across R2/R4 columns,” you can set lingers_across_boxes=True in that module’s output for that pattern. The aggregator picks it up and adds linger_across_boxes_bonus.
o	The aggregator also does a “cross-set” synergy if it found the pattern in older sets too.
2.	V-Trac Permutation & Clusters
o	We create a vtrac_transform(...) code that yields a sorted vtrac code.
o	We map each aggregator group to that vtrac code.
o	If multiple pattern groups share that code from different modules → that vtrac code is declared a “cluster.”
o	Then each pattern in that cluster gets a synergy bonus (vtrac_cluster_bonus).
o	This effectively encourages patterns that show up as “different digit strings but same V-Trac group,” or show up across multiple modules.
3.	Pending Logic vs. Last 6 Draws
o	The aggregator checks each pattern’s “pattern_variants” (the actual permutations discovered: e.g. “81324,” “13248,” etc.).
o	If none of them appear in the last 6 draws (in any canonical or vtrac sense), the aggregator flags it as “pending,” awarding pending_bonus.
o	If it appeared, we do not penalize it. We simply don’t add the pending bonus.
o	You can make last6_draws a bigger list if you want to check the last 10 draws.
4.	Cross-Midday / Evening / Combined
o	The aggregator sees from which “section” each record came. If a pattern is found in Midday + Evening → synergy. If it’s also in Combined → even more synergy.
o	You see it in the code: sections_found.add(section). Then if len(sections_found)>1, we do a synergy bonus.
5.	Extended Patterns (≥ 4 digits)
o	The aggregator awards length_bonus_per_digit_over_3. For example, if you have a 6-digit 3-value stable pattern (like “777111”), that’s 3 digits beyond 3, so we add (6-3)*1=3 points. You can tweak that to make large stable patterns even more valuable.
6.	Combining Each Module’s Own Score
o	By default, each item has a score_in_module. The aggregator sums them up. Then synergy is layered on top.
o	So if a pattern had big local “module scores” from stable or hot zones, that contributes.
7.	Multi-Module Repeats
o	The aggregator lumps all occurrences of the same “canonical form” together. If they come from 2+ modules, that triggers cross_module_bonus_per_extra.
8.	Straight vs. Box
o	If you want to specifically reward “straight repeating patterns,” you can store a flag like found_straight=True in the module output. Then aggregator can add synergy for “straight.” Similarly for “box.” (We show in the code how to handle “rtype.”)
9.	3-Value Repeats or Single 3-Value
o	Because is_3value is recognized, we add three_value_bonus. If you want an additional synergy for multiple 3-value occurrences, you can expand logic to check “did we see multiple 3-value columns for the same pattern?”
In short, we aren’t ignoring your repeated stable patterns or the synergy of 3-value repeats – you just need to ensure your earlier modules annotate them. For example, if your Long String module sees a pattern crossing from column7 to column6 to column5, set lingers_across_boxes=True. Then the aggregator sees that field and awards synergy.
If you have a separate concept of “3-value pattern repeated across columns,” just create a boolean or integer in the item’s dictionary (e.g. repeat_across_columns_count=3). In the aggregator, if repeat_across_columns_count >= 2, we give synergy.

Below is an updated aggregator design that explicitly integrates permutation analysis right in the synergy-scoring step. The goal is to show how each pattern’s permutations (and/or V Trac permutations) can factor into synergy, cross-checking with other patterns or with the last X draws, ensuring we don’t lose the nuance you’ve repeated many times:
1.	A pattern’s permutations (straight vs. box forms, or partial expansions) need to be recognized to see if they reappear in the same module or across modules.
2.	Cross-permutation synergy: If Pattern A is “695” and Pattern B is “140” but they share a V-Trac “straight” alignment or if they have some key relationship via permutations, we want to score that.
3.	Storing “lingering across columns” or “3-value repeats” can be integrated.
4.	Checking last draws against all permutations (and their V Trac permutations) to decide if a pattern is “pending.”
5.	Flow: We unify patterns from your prior modules (Stable, Long String, Hot Zones, etc.), but now each item specifically includes (or can generate) its relevant permutations for synergy checks.

Summary of New/Extra Steps
1) Each module’s output now can (optionally) store all_permutations or all_vtrac_permutations.
•	E.g. if a pattern is “695,” the module (or aggregator) generates [695, 956, 569, ...]. For a 3-digit pattern, that’s up to 6 permutations if all digits differ.
•	For V Trac permutations, we transform the pattern to V Trac digits, then gather permutations.
2) When the aggregator merges patterns, it can also unify them if any of their permutations match.
•	That is, if we see that pattern “695” is the same group as “956” because they are permutations.
•	Or if we want “695” grouped with “140” because in V Trac they form the same “straight.” (For instance, if the V Trac forms match exactly in the same order—some advanced logic is possible.)
3) A special “Permutation Synergy” pass:
•	The aggregator loops over each pattern group and checks if any permutations or V Trac permutations appear in other patterns discovered across modules. If yes, we add synergy.
•	Or we do a single function to check if the aggregator’s grouping lumps them.
4) Pending check across permutations:
•	Instead of just checking “canonical sort” or “vtrac sort” in the last N draws, we check all permutations + all V Trac permutations to see if any was drawn. If none appear, we mark the pattern “pending.”
Below is an illustrative aggregator that explicitly does:
•	Step A: Collect all patterns from stable, long string, hot zones.
•	Step B: For each pattern, generate or store all_permutations and all_vtrac_permutations.
•	Step C: Group them in a synergy approach that lumps patterns that share any permutation or vtrac permutation.
•	Step D: Score synergy (including cross-module synergy, hot zone, stable-lingering, etc.).
•	Step E: Check last 6 draws using permutations. If none of the permutations appear, we add a “pending” bonus.
•	Step F: Also do a pass to see if pattern A’s permutations align with pattern B’s permutations, awarding a “permutation synergy” if they do.
You can adapt the exact logic to match your system’s definitions (like 3-value or how you interpret “695 vs. 140 is a V-Trac straight”).

#!/usr/bin/env python3
"""
Aggregator & Synergy Analysis with Explicit Permutation Checking
--------------------------------------------------------------
Collects patterns from:
  - stable_extractor
  - long_string_reduction
  - hotzones_analysis

Then unifies them, focusing on:
  - Generating permutations for each pattern (and V-Trac permutations).
  - Grouping patterns if they share permutations in any form.
  - Cross-module synergy, hot-level, stable-lingering, etc.
  - "Pending" logic if none of a pattern's permutations (nor V-Trac perms)
    appear in the last X draws.

This addresses the repeated request to incorporate explicit permutation 
analysis, ensuring that if pattern A's permutations or V-Trac permutations 
match pattern B or the actual draws, we handle it properly.
"""

import datetime
from collections import defaultdict
import itertools

########################
# 1) Utility Functions #
########################

# Mirror / VTrac map
VTRAC_MAP = {
    '0':'1','5':'1',
    '1':'2','6':'2',
    '2':'3','7':'3',
    '3':'4','8':'4',
    '4':'5','9':'5'
}

def generate_permutations(digit_str):
    """
    Return all unique permutations of digit_str. 
    E.g. '695' => {'695','659','569','596','965','956'}
    """
    return {''.join(p) for p in itertools.permutations(digit_str)}

def vtrac_transform(digit_str):
    """Return the V-Trac mapped string (same order) for digit_str."""
    return ''.join(VTRAC_MAP.get(ch,ch) for ch in digit_str)

def generate_vtrac_permutations(digit_str):
    """
    Transform digit_str to V-Trac digits (in same order), 
    then generate permutations of that V-Trac string.
    If the user wants "straight" V-Trac only, you might skip permutations,
    but let's do permutations to unify all forms.
    """
    vt_str = vtrac_transform(digit_str)
    # permutations of the V-Trac string
    perms = {''.join(p) for p in itertools.permutations(vt_str)}
    return perms

def any_permutation_in_draws(perms_set, draws):
    """
    Check if ANY of the permutations in `perms_set` appear in the given `draws`.
    `draws` is a list of strings (like '695', '140', '267', etc.).
    If any one is found, return True, else False.
    """
    # We can store the draws in a set for quick membership
    drawset = set(draws)
    return any((p in drawset) for p in perms_set)

def is_3value(digit_str):
    """Check if up to 3 distinct digits => '3-value' pattern."""
    return len(set(digit_str)) <= 3

########################
# 2) Collect from Modules
########################

def collect_stable_patterns(stable_out):
    """
    Suppose stable_out is your stable_extractor's result:
      { pattern_key: {
          "canonical":..., 
          "vtrac":..., 
          "score":..., 
          "locations":..., 
          "score_breakdown":..., 
          ...
        },
        ...
      }
    We'll produce a list of dict with minimal consistent fields:
      {
        "module_source": "Stable",
        "pattern_str": <the actual digits>,
        "score_in_module": <float or int>,
        "hot_level": 0 or 1 or 2,
        "stable_lingering": bool,
        ...
      }
    Then later aggregator merges them.
    """
    results = []
    if not stable_out:
        return results
    
    for patk, info in stable_out.items():
        # 'patk' might be the canonical pattern or actual
        pat_str = patk
        # or if you prefer info["canonical"] as the main string
        sc = info.get("score",0)
        # If you had something indicating "lingers across columns," store it:
        lingers = info.get("lingers_across_boxes",False)
        # if there's a boolean "stable_after_reduction" or not
        stable_red = info.get("stable_after_reduction",False)
        # hot level doesn't usually come from stable, so 0
        hot_lvl = 0

        results.append({
            "module_source":"Stable",
            "pattern_str": pat_str,
            "score_in_module": sc,
            "hot_level": hot_lvl,
            "lingers_across_boxes": lingers,
            "stable_after_reduction": stable_red,
            "locations": info.get("locations",[]),
        })
    return results


def collect_longstring_patterns(long_out):
    """
    If your long_string_reduction.py returns something like:
    {
      "module_name":"LongStringReduction_Area1",
      "analysis_results": [
        {
          "cell_id":"Midday|Set3|Draw1|R2|col7",
          "notable_patterns":[{"pattern":"695","score":12}, ...],
          "final_score": 14,
          ...
        },
        ...
      ]
    }
    We'll parse that into a consistent aggregator format.
    """
    results = []
    if not long_out:
        return results
    
    def parse_area(area_dict):
        out = []
        arr = area_dict.get("analysis_results", [])
        for cell in arr:
            cell_id = cell.get("cell_id","")
            nps = cell.get("notable_patterns", [])
            for np in nps:
                pat = np["pattern"]
                sc = np["score"]
                out.append({
                    "module_source": area_dict.get("module_name","LongString"),
                    "pattern_str": pat,
                    "score_in_module": sc,
                    "hot_level": 0,
                    "lingers_across_boxes": False,
                    "stable_after_reduction": False, 
                    "locations": [cell_id]
                })
        return out
    
    # if single dict with 'analysis_results'
    if isinstance(long_out, dict) and "analysis_results" in long_out:
        results = parse_area(long_out)
    elif isinstance(long_out, dict):
        # might have multiple "areas"
        for k,v in long_out.items():
            if isinstance(v, dict) and "analysis_results" in v:
                results.extend(parse_area(v))
    elif isinstance(long_out, list):
        # each item might be an area
        for area_item in long_out:
            if isinstance(area_item, dict) and "analysis_results" in area_item:
                results.extend(parse_area(area_item))
    return results


def collect_hotzone_patterns(hotz_out):
    """
    If hotz_out has "hotzone_patterns":[...],
    each item might be:
      {
        "pattern":"695",
        "local_hotzone_score":12,
        "hot_level":2,
        "stable_after_reduction":True,
        "locations":[("Midday","Set1","Draw2","R2",1)],
        "lingers_across_boxes":False,
        ...
      }
    We'll unify them similarly.
    """
    results = []
    if not hotz_out:
        return results
    hotz_list = []
    if isinstance(hotz_out, dict) and "hotzone_patterns" in hotz_out:
        hotz_list = hotz_out["hotzone_patterns"]
    elif isinstance(hotz_out, list):
        for hx in hotz_out:
            if isinstance(hx, dict) and "hotzone_patterns" in hx:
                hotz_list.extend(hx["hotzone_patterns"])
    
    for item in hotz_list:
        pat = item.get("pattern","")
        sc = item.get("local_hotzone_score",0)
        hl = item.get("hot_level",0)
        stable_red = item.get("stable_after_reduction",False)
        ling = item.get("lingers_across_boxes",False)
        locs = item.get("locations",[])
        results.append({
            "module_source":"HotZones",
            "pattern_str": pat,
            "score_in_module": sc,
            "hot_level": hl,
            "stable_after_reduction": stable_red,
            "lingers_across_boxes": ling,
            "locations": locs
        })
    return results

def aggregator_synergy_with_permutation(
    stable_data,
    longstring_data,
    hotzone_data,
    last6_draws,
    synergy_config=None
):
    """
    Merges stable, longstring, hotzone patterns, 
    then explicitly uses permutations to unify patterns 
    and to check "pending" logic or cross-synergy.

    last6_draws: list of strings (like ['695','140','267','718','332','556'])
                 used to see if a pattern or any permutation 
                 has appeared recently => if not => pending bonus
    synergy_config: optional dict to override synergy weights
    """
    if synergy_config is None:
        synergy_config = {}
    # default synergy weighting
    weights = {
        "base_module_score_factor": 1.0,
        "cross_module_bonus": 3,  
        "lingers_across_boxes_bonus": 2,
        "stable_after_reduction_bonus": 3,
        "hot_level_1_bonus": 2,
        "hot_level_2_bonus": 4,
        "three_value_bonus":2,
        "pending_bonus":3,
        "permutation_cluster_bonus": 2,  # synergy if different patterns share permutations
        # etc.
    }
    weights.update(synergy_config)

    # 1) Collect items
    stable_list   = collect_stable_patterns(stable_data)
    long_list     = collect_longstring_patterns(longstring_data)
    hotz_list     = collect_hotzone_patterns(hotzone_data)

    all_items = stable_list + long_list + hotz_list

    # 2) For each item, generate permutations & vtrac permutations
    # We'll store them in the record for aggregator grouping
    for it in all_items:
        pstr = it["pattern_str"]
        # all permutations of that pattern
        perm_set = generate_permutations(pstr)
        # also vtrac permutations
        vtp_set  = set()
        # either transform pstr then permute or permute then transform each perm
        # let's do transform first => then permute
        vt_str = vtrac_transform(pstr)
        vtp_set = generate_permutations(vt_str)
        
        it["all_permutations"] = perm_set
        it["all_vtrac_permutations"] = vtp_set
        it["is_3value"] = is_3value(pstr)

    # 3) Group items if they share any permutation in common
    # We'll build an index that maps each "perm" => a list of item indices
    perm_index = defaultdict(list)
    for idx, rec in enumerate(all_items):
        for pm in rec["all_permutations"]:
            perm_index[pm].append(idx)
        # We could also store vtrac perms in the same index or a separate one
        # but let's unify them so that if item A's vtrac perm = item B's normal perm => synergy
        for vpm in rec["all_vtrac_permutations"]:
            perm_index[vpm].append(idx)

    # Now we define "groups" by connected components:
    # If item A shares any permutation with item B, they belong to the same group.
    visited = set()
    groups = []
    def dfs(start):
        stack = [start]
        cluster = set()
        while stack:
            node = stack.pop()
            if node in cluster: 
                continue
            cluster.add(node)
            # find neighbors => any item that shares a permutation
            # we look at all perms in item[node]
            allp = all_items[node]["all_permutations"] | all_items[node]["all_vtrac_permutations"]
            neighbors = set()
            for p in allp:
                for nxt in perm_index[p]:
                    if nxt not in cluster:
                        neighbors.add(nxt)
            for nb in neighbors:
                if nb not in cluster:
                    stack.append(nb)
        return cluster

    for i in range(len(all_items)):
        if i not in visited:
            comp = dfs(i)
            visited |= comp
            groups.append(comp)
    # groups is a list of sets, each set are item indices that share permutations or vtrac perms
    
    # 4) For each group, compute synergy
    results = []
    for comp_indices in groups:
        comp_indices = list(comp_indices)
        # gather union of module_source, max hot level, sum of module scores, etc.
        modules_found = set()
        max_hot = 0
        stable_after_red_flag = False
        linger_flag = False
        sum_module_score = 0
        patterns_str = set()
        is3val_any = False

        for idx in comp_indices:
            it = all_items[idx]
            modules_found.add(it["module_source"])
            if it["hot_level"]>max_hot:
                max_hot=it["hot_level"]
            if it.get("stable_after_reduction",False):
                stable_after_red_flag=True
            if it.get("lingers_across_boxes",False):
                linger_flag=True
            sum_module_score += (it["score_in_module"]*weights["base_module_score_factor"])
            patterns_str.add(it["pattern_str"])
            if it["is_3value"]:
                is3val_any=True

        # synergy
        synergy_score = sum_module_score
        synergy_breakdown = {
            "sum_module_scores": sum_module_score
        }

        # cross module synergy if multiple distinct modules
        if len(modules_found)>1:
            cmod = (len(modules_found)-1)*weights["cross_module_bonus"]
            synergy_score += cmod
            synergy_breakdown["cross_module_bonus"] = cmod

        if max_hot==1:
            synergy_score += weights["hot_level_1_bonus"]
            synergy_breakdown["hot_star_bonus"] = weights["hot_level_1_bonus"]
        elif max_hot==2:
            synergy_score += weights["hot_level_2_bonus"]
            synergy_breakdown["hot_super_bonus"] = weights["hot_level_2_bonus"]

        if stable_after_red_flag:
            synergy_score += weights["stable_after_reduction_bonus"]
            synergy_breakdown["stable_after_reduction_bonus"] = weights["stable_after_reduction_bonus"]

        if linger_flag:
            synergy_score += weights["lingers_across_boxes_bonus"]
            synergy_breakdown["lingers_across_boxes_bonus"] = weights["lingers_across_boxes_bonus"]

        if is3val_any:
            synergy_score += weights["three_value_bonus"]
            synergy_breakdown["three_value_bonus"] = weights["three_value_bonus"]

        # pending check => if none of the permutations in any item appear in last6
        # build union of permutations for the entire group
        union_all_perms = set()
        for idx in comp_indices:
            union_all_perms |= all_items[idx]["all_permutations"]
            union_all_perms |= all_items[idx]["all_vtrac_permutations"]
        # if none appear in last6 => pending
        if not any_permutation_in_draws(union_all_perms, last6_draws):
            synergy_score += weights["pending_bonus"]
            synergy_breakdown["pending_bonus"] = weights["pending_bonus"]

        # Because we formed groups by permutation connectivity, each group
        # is inherently a "Permutation Cluster." 
        # If a group has more than 1 item => we can add a "permutation cluster bonus"
        if len(comp_indices)>1:
            synergy_score += weights["permutation_cluster_bonus"]
            synergy_breakdown["permutation_cluster_bonus"] = weights["permutation_cluster_bonus"]

        # finalize
        results.append({
            "patterns_in_group": list(patterns_str),
            "modules_found": list(modules_found),
            "group_size": len(comp_indices),
            "max_hot_level": max_hot,
            "is_3value_any": is3val_any,
            "final_synergy_score": synergy_score,
            "synergy_breakdown": synergy_breakdown
        })

    # sort descending
    results.sort(key=lambda x: x["final_synergy_score"], reverse=True)

    return {
        "aggregator_module":"AggregatorWithPermutation",
        "timestamp": datetime.datetime.now().isoformat(),
        "group_count": len(results),
        "groups": results
    }

Explanation of the Key Permutation Steps
1.	generate_permutations(digit_str)
o	Returns all permutations (straight forms) for that digit string. For a 3-digit pattern, that’s up to 6 permutations if all digits are distinct.
2.	generate_vtrac_permutations(digit_str)
o	First transforms the pattern into its V Trac digits (keeping the same order). Then enumerates permutations.
o	If you interpret “V Trac straight” in a special way (e.g., not permuting the order), you can skip the permutations. Or if you want every possible V Trac reordering, do it as shown.
3.	Storing those sets in each item:
o	it["all_permutations"]
o	it["all_vtrac_permutations"]
So each item in the aggregator has a union of “normal permutations” + “vtrac permutations.”
4.	Building a “perm_index”:
o	We map each permutation string (like "695", "596", "269", etc.) to all item indices that contain that permutation.
o	If two items share at least one permutation, we consider them connected.
5.	Grouping by connected components:
o	We do a DFS so that if item A is connected to item B (they share permutations), and item B is connected to C, then A/B/C form a single group.
o	This effectively lumps together all patterns that are “the same or permutation-related in any way.”
o	In your example: “695” might connect with “140” if, say, some V Trac permutation of 695 is “something” that matches a normal perm of 140. Possibly that’s the scenario you wanted to highlight.
6.	Scoring:
o	Each group is assigned synergy:
	Summation of the items’ module scores
	Cross-module bonus if 2+ modules contributed items in that group
	Hot-level synergy if any item was hot
	“Permutation cluster bonus” if group size > 1
	“Pending bonus” if none of the group’s permutations are in the last N draws
	etc.
Result: We end up with fewer “super-groups,” each capturing all patterns that are permutations (or V-Trac permutations) of each other. Inside that group, you see multiple module sources and patterns that all revolve around the same underlying “permutation cluster.”
________________________________________
Why This May Solve the “695 vs 140” V Trac Straight Issue
If “695” in V Trac is, for example, “2 5 2” (just hypothetical) and “140” in V Trac is “2 5 2” as well, the aggregator code sees that they share a vtrac permutation. They get grouped → synergy is awarded.
Hence we explicitly incorporate permutations in the aggregator’s grouping and synergy.
Additional Tweaks
•	If you want your aggregator to unify patterns only if they share “the exact same V Trac straight,” you can skip permutations in generate_vtrac_permutations. Then they’ll only connect if the V Trac transformation is identical.
•	If you want “box or rotation-min” approach, you can adapt the aggregator so that it does rotation-min on each string and lumps them if they match. The example code uses permutations because you specifically stressed the importance of checking all permutations.
•	For 2-digit or 4-digit patterns, this is all general. It will just create permutations as needed.
•	You can incorporate “3-value repeats across columns” more deeply by setting a field col_repeats_count in each item. Then give synergy if col_repeats_count≥2.
