#!/usr/bin/env python3
"""
long_string_reduction.py

Module: Long String Digit Reduction & Hidden Patterns
-----------------------------------------------------
This module handles specialized analysis for "long-string" columns (7, 6, 5) 
in Set3, Set2, and Set1 Draw1 across Midday/Evening/Combined. It:

1) Identifies any stable 3-value patterns before digit reduction.
2) Applies digit reduction methods (A, B, C) + single-digit transit removal (Method D).
3) Discovers or reinforces hidden stable patterns.
4) Scores patterns for final aggregator to combine with the main pattern extractor's output.

Integration:
-----------
- This module is designed to be called AFTER the main data is loaded into a 
  structured dict (matching your JSON format).
- The final output is a structured dict/JSON that the overall Analysis Module 
  can merge with other module outputs (Pattern Extractor, Hot Zone Analysis, etc.).
"""

import copy
from collections import defaultdict
import datetime

###########################################
# 1. Mirror & V-Trac Helpers
###########################################
MIRROR_MAP = {
    '0':'5','1':'6','2':'7','3':'8','4':'9',
    '5':'0','6':'1','7':'2','8':'3','9':'4'
}

def vtrac_transform(digits: str) -> str:
    """
    Simple VTrac transform if you want to check if 
    string has <=3 unique vtrac digits.
    (Optionally, you can refine if you want the 
     0->5 / 5->0 approach, etc.)
    """
    return "".join(MIRROR_MAP[ch] if ch in MIRROR_MAP else ch for ch in digits)

def is_3value_cluster(digits_str: str) -> bool:
    """
    Checks if 'digits_str' has <=3 unique actual digits 
    OR <=3 unique VTrac digits.
    """
    if not digits_str:
        return False
    unique_actual = set(digits_str)
    if len(unique_actual) <= 3:
        return True
    
    # If not <=3 in actual, check VTrac
    vt = vtrac_transform(digits_str)
    unique_vt = set(vt)
    return (len(unique_vt) <= 3)

###########################################
# 2. Digit Reduction Methods (A, B, C)
###########################################
def method_a(original: str, past_draw_digits: str) -> str:
    """
    Removes EXACT digits from 'original' if they appear in 'past_draw_digits'.
    E.g., if 'past_draw_digits' = '341', we remove '3','4','1' from the string.
    """
    to_remove = set(past_draw_digits)
    return "".join(ch for ch in original if ch not in to_remove)

def method_b(original: str, past_draw_digits: str) -> str:
    """
    Eliminates actual digit if present; else eliminates its mirror if present.
    
    For each digit d in 'past_draw_digits':
      - If d is in 'original', remove d.
      - Else if MIRROR_MAP[d] is in 'original', remove that.
      - Otherwise remove d anyway by default. 
        (In practice, if 'd' not in original but we want 
         to show some partial fallback, your logic might differ.)
    """
    result = list(original)
    result_indices_removed = set()  # track which positions are removed

    # We do an iterative approach so we remove each digit once if it exists.
    # (But your logic might interpret differently. 
    #  This is a simplified version.)
    for d in past_draw_digits:
        # attempt to remove d first
        removed_any = False
        for i, ch in enumerate(result):
            if i in result_indices_removed:
                continue
            if ch == d:
                result_indices_removed.add(i)
                removed_any = True
                break
        # if not found d, try mirror
        if (not removed_any) and d in MIRROR_MAP:
            m = MIRROR_MAP[d]
            for i, ch in enumerate(result):
                if i in result_indices_removed:
                    continue
                if ch == m:
                    result_indices_removed.add(i)
                    removed_any = True
                    break
        # else if not found, do nothing or remove d from somewhere
        # We'll do nothing if we can't find either digit or its mirror 
        # in the string.

    # build final
    final_str = "".join(
        ch for idx, ch in enumerate(result) if idx not in result_indices_removed
    )
    return final_str

def method_c(original: str, past_draw_digits: str) -> str:
    """
    Eliminates BOTH the actual digit and its mirror 
    from 'original', if either are present.
    E.g. if 'past_draw_digits' includes '4', we remove 
    '4' and '9' from 'original' entirely.
    """
    to_remove = set()
    for d in past_draw_digits:
        to_remove.add(d)
        if d in MIRROR_MAP:
            to_remove.add(MIRROR_MAP[d])
    return "".join(ch for ch in original if ch not in to_remove)

###########################################
# 3. Single-Digit Transit Removal (Method D)
###########################################
def iterative_transit_reduction(original: str, past_draw_digits: str):
    """
    Removes each digit in 'past_draw_digits' one at a time, 
    checking for stable patterns at each step.
    Returns a list of dicts with intermediate results & discovered patterns.
    
    Example:
      step_0: original
      step_1: remove digit1 -> new_str1
               stable patterns found
      step_2: remove digit2 -> new_str2
               stable patterns found
      ...
    """
    steps_info = []
    current_str = original
    for idx, d in enumerate(past_draw_digits, start=1):
        new_str = current_str.replace(d, '')
        # also remove mirror if present
        if d in MIRROR_MAP:
            new_str = new_str.replace(MIRROR_MAP[d], '')
        stable_3val = []
        if is_3value_cluster(new_str):
            stable_3val.append(new_str)
        steps_info.append({
            "step_idx": idx,
            "removed_digit": d,
            "result_str": new_str,
            "stable_3value_patterns": stable_3val
        })
        current_str = new_str
    return steps_info

###########################################
# 4. Utility: Finding 3-Value Substrings
###########################################
def find_all_3value_substrings(digits_str: str, min_len=3):
    """
    Return all substrings of length >=3 that are '3-value' in either actual or VTrac.
    Simple approach: O(n^2) scanning. 
    """
    results = []
    n = len(digits_str)
    for length in range(min_len, n+1):
        for start in range(0, n-length+1):
            sub = digits_str[start:start+length]
            if is_3value_cluster(sub):
                results.append(sub)
    return list(set(results))  # remove duplicates

# 5. The Main Class or Main Functions
###########################################
class LongStringReductionModule:
    """
    Encapsulates the logic to process 
    'long string' columns 7/6/5 in 
    Set3, Set2, Set1 (Draw1) for Midday/Evening/Combined,
    applying digit reduction methods (A,B,C,D).
    """

    def __init__(self, data_dict, reduction_config=None):
        """
        data_dict: The full structured data (like your JSON).
        reduction_config: 
          - e.g. {"past_draw_digits_own":"341", 
                  "past_draw_digits_combined":"632", 
                  "scoring":{...} }
        You can store scoring weights or references to 
        whichever draws are 'previous' here.
        """
        self.data = data_dict
        self.config = reduction_config or {}
        # define some default scoring weights
        self.scoring_weights = {
            "initial_stable": 3,
            "method_a": 2,
            "method_b": 2,
            "method_c": 2,
            "method_d": 3,  # single-transit approach
            "multi_method_bonus": 1,
            "cross_set_persistence": 2
            # you can add more 
        }
        if "scoring" in self.config:
            self.scoring_weights.update(self.config["scoring"])

    def gather_long_string_cells(self):
        """
        Return a list of (cell_id, original_string, meta)
        specifically from columns 7,6,5 
        in Set3.Draw1, Set2.Draw1, Set1.Draw1 
        for Midday/Evening/Combined 
        across R2,R4,R6,R8.
        
        We'll parse self.data accordingly.
        
        cell_id example: "Midday|Set3|Draw1|R2|col7"
        """
        results = []
        # define the "sections" (Midday,Evening,Combined)
        # define the sets: (Set3->Draw1, Set2->Draw1, Set1->Draw1)
        # define the columns: 7,6,5
        # define the rows: R2,R4,R6,R8
        sections = ["Midday","Evening","Combined"]
        sets_ = ["Set3","Set2","Set1"]
        columns_of_interest = [7,6,5]
        rows_of_interest = ["R2","R4","R6","R8"]

        for sec in sections:
            if sec not in self.data["sections"]:
                continue
            sets_dict = self.data["sections"][sec]["sets"]
            for st in sets_:
                if st not in sets_dict:
                    continue
                draws_dict = sets_dict[st]["draws"]
                # we only want Draw1 for Set3,Set2 
                # but for Set1 we also want 'Draw1' specifically 
                # (the module only focuses on that anyway).
                # So let's unify that logic: we'll attempt 'Draw1'.
                if "Draw1" not in draws_dict:
                    continue
                draw_data = draws_dict["Draw1"]
                # pattern_variations => R2,R4,R6,R8
                if "pattern_variations" not in draw_data:
                    continue
                pattern_variations = draw_data["pattern_variations"]
                # we have pattern_variations["R2"] = list of strings 
                # for columns 7..1 in that order
                # we want indexes 0..2 for col7, col6, col5
                for rowkey in rows_of_interest:
                    if rowkey not in pattern_variations:
                        continue
                    row_strings = pattern_variations[rowkey]
                    # row_strings is a list from col7..col1 => indexes [0..6]
                    # we want col7-> index0, col6->index1, col5->index2
                    for i, col in enumerate(columns_of_interest):
                        if i >= len(row_strings):
                            continue
                        original_str = row_strings[i]  # i=0->col7, i=1->col6 ...
                        cell_id = f"{sec}|{st}|Draw1|{rowkey}|col{col}"
                        results.append((cell_id, original_str, {
                            "section": sec,
                            "set": st,
                            "draw": "Draw1",
                            "row": rowkey,
                            "col": col
                        }))

        return results

    def analyze_long_strings(self):
        """
        The main routine: 
         1) gather the relevant cells 
         2) find stable patterns pre-reduction 
         3) apply digit reductions 
         4) produce final scoring
        Returns a structured dict ready for the aggregator.
        """
        cell_list = self.gather_long_string_cells()

        analysis_output = []
        for (cell_id, orig_str, meta) in cell_list:
            # we'll store a dict with all detail
            cell_result = {
                "cell_id": cell_id,
                "metadata": meta,
                "original_string": orig_str,
                "initial_stable_patterns": [],
                "method_a": {},
                "method_b": {},
                "method_c": {},
                "method_d": {},
                "final_score": 0,
                "notable_patterns": []
            }

            # Step 1: Pre-Reduction stable 3-value
            pre_red_subs = find_all_3value_substrings(orig_str)
            stable_pre = []
            for sub in pre_red_subs:
                stable_pre.append({
                    "pattern": sub,
                    "score_contrib": self.scoring_weights.get("initial_stable",3)
                })
            cell_result["initial_stable_patterns"] = stable_pre

            # We can accumulate pattern->score
            pattern_score_map = defaultdict(int)
            for item in stable_pre:
                pattern_score_map[item["pattern"]] += item["score_contrib"]

            # Step 2: We define 'past_draw_digits_own' + 'past_draw_digits_combined'
            # For now, we'll do a simplified approach:
            p_own = self.config.get("past_draw_digits_own","")
            p_comb = self.config.get("past_draw_digits_combined","")
            # We can try each:
            # Method A/B/C for "own draw" and for "combined draw" => total 6 tries
            # or you might do fewer. Up to your logic.

            # We'll store them in cell_result["method_a"]["own"], etc.
            # Each variant returns a reduced_string, plus discovered patterns.

            def apply_and_record(method_func, method_key, reduce_label, raw_string, past_digits):
                reduced = method_func(raw_string, past_digits)
                found_subs = find_all_3value_substrings(reduced)
                # scoring
                pat_list = []
                for fs in found_subs:
                    # if new pattern => add method_x points
                    # if already seen => add method_x points
                    pat_list.append(fs)
                    # basic scoring
                    pattern_score_map[fs] += self.scoring_weights.get(method_key, 2)
                return {
                    "past_draw_digits": past_digits,
                    "reduced_string": reduced,
                    "found_3value_substrings": pat_list
                }

            # We'll define a small helper to unify a/b/c:
            # For demonstration, let's just do "own draw" for each method. 
            # In reality, you'd do "own" + "combined" for each method 
            # if you want the full approach.

            # ---------- Method A -----------
            cell_result["method_a"]["own"] = apply_and_record(method_a, "method_a", "own", orig_str, p_own)
            cell_result["method_a"]["combined"] = apply_and_record(method_a, "method_a", "combined", orig_str, p_comb)

            # ---------- Method B -----------
            cell_result["method_b"]["own"] = apply_and_record(method_b, "method_b", "own", orig_str, p_own)
            cell_result["method_b"]["combined"] = apply_and_record(method_b, "method_b", "combined", orig_str, p_comb)

            # ---------- Method C -----------
            cell_result["method_c"]["own"] = apply_and_record(method_c, "method_c", "own", orig_str, p_own)
            cell_result["method_c"]["combined"] = apply_and_record(method_c, "method_c", "combined", orig_str, p_comb)

            # Step 3: Single-digit transit approach (Method D)
            # We'll do it for the 'own draw' digits. Possibly also for combined.
            def do_transit_d(raw_string, dkey, label):
                steps_info = iterative_transit_reduction(raw_string, dkey)
                # gather all discovered stable patterns 
                # along the way
                discovered = []
                for stp in steps_info:
                    for pat in stp["stable_3value_patterns"]:
                        discovered.append(pat)
                        pattern_score_map[pat] += self.scoring_weights.get("method_d",3)
                return {
                    "past_draw_digits": dkey,
                    "transit_steps": steps_info,
                    "all_discovered_patterns": list(set(discovered))
                }

            cell_result["method_d"]["own"] = do_transit_d(orig_str, p_own, "own")
            cell_result["method_d"]["combined"] = do_transit_d(orig_str, p_comb, "combined")

            # Optionally, if a pattern was discovered in multiple methods, 
            # we can add a "multi_method_bonus"
            # We'll do a naive approach: if a pattern is found in 2 or more methods => +1

            method_presence_map = defaultdict(set) 
            # pattern -> set of methods that revealed it
            # let's gather
            for meth in ["method_a","method_b","method_c","method_d"]:
                subdict = cell_result.get(meth,{})
                for subk, subres in subdict.items():
                    if isinstance(subres, dict):
                        found = subres.get("found_3value_substrings", [])
                        # or "all_discovered_patterns" for method_d
                        if not found:
                            found = subres.get("all_discovered_patterns", [])
                        for pat in found:
                            method_presence_map[pat].add(meth)
            
            # also include "initial_stable_patterns"
            for item in stable_pre:
                pat = item["pattern"]
                method_presence_map[pat].add("pre_reduction")

            for pat, mset in method_presence_map.items():
                if len(mset) >= 2:
                    # add multi_method_bonus
                    pattern_score_map[pat] += self.scoring_weights.get("multi_method_bonus",1)

            # cross-set persistence 
            # (only relevant if we see the same pattern in older sets vs new sets),
            # that might require a bigger global approach. 
            # We'll skip for now, or do a placeholder:
            # pattern_score_map[xyz] += cross_set_persistence etc.

            # Final tallied patterns
            # we can pick out patterns that have total score > 0
            final_patterns = []
            for patt, sc in pattern_score_map.items():
                if sc > 0:
                    final_patterns.append((patt, sc))
            # sort them descending
            final_patterns.sort(key=lambda x: x[1], reverse=True)
            cell_result["notable_patterns"] = [
                {"pattern": fp[0], "score": fp[1]} for fp in final_patterns
            ]
            top_score = final_patterns[0][1] if final_patterns else 0
            cell_result["final_score"] = top_score

            analysis_output.append(cell_result)

        # We can wrap up in a top-level dict
        return {
            "module_name": "LongStringReduction",
            "timestamp": datetime.datetime.now().isoformat(),
            "analysis_results": analysis_output
        }

"""
long_string_reduction.py

Module: Long String Digit Reduction & Hidden Patterns
=====================================================
This module handles specialized analysis for "long-string" areas. In particular:

1) "Long String Area #1":
   - Columns 7, 6, 5 in Set3/Set2/Set1 (Draw1) for Midday/Evening/Combined
   - R2/R4/R6/R8 boxes
2) "Long String Area #2":
   - A simplified area focusing only on Set1 (Midday/Evening/Combined),
     but specifically: 
       * Draw4, Column3
       * Draw6, Column1
     (Again, R2/R4/R6/R8 rows if applicable.)
   
We:

    - Identify stable 3-value patterns BEFORE digit reduction.
    - Apply digit reduction methods (A,B,C) and single-digit transit (Method D).
    - Reveal or reinforce hidden 3-value patterns that appear after reduction.
    - Score each pattern for final aggregator usage.

Integration:
-----------
After main data is loaded into a structured dict (matching your JSON format),
you can call:
   1) analyze_area1_long_strings(...) 
      -> processes the first "long string area #1"
   2) analyze_area2_long_strings(...)
      -> processes the second "long string area #2"

Both return structured JSON/dict, which your aggregator merges with
other modules' outputs (pattern_extractor, hotzones_analysis, etc.).

import copy
from collections import defaultdict
import datetime

###########################################
# 1. Mirror & V-Trac Helpers
###########################################
MIRROR_MAP = {
    '0':'5','1':'6','2':'7','3':'8','4':'9',
    '5':'0','6':'1','7':'2','8':'3','9':'4'
}

def vtrac_transform(digits: str) -> str:
    """
    Simple VTrac transform: each digit replaced by mirror_map for quick check 
    if string has <=3 unique 'vtrac' digits.
    """
    return "".join(MIRROR_MAP[ch] if ch in MIRROR_MAP else ch for ch in digits)

def is_3value_cluster(digits_str: str) -> bool:
    """
    Checks if 'digits_str' has <=3 unique actual digits 
    OR <=3 unique VTrac digits.
    """
    if not digits_str:
        return False
    unique_actual = set(digits_str)
    if len(unique_actual) <= 3:
        return True
    # If not <=3 in actual, check VTrac
    vt = vtrac_transform(digits_str)
    unique_vt = set(vt)
    return (len(unique_vt) <= 3)

# 2. Digit Reduction Methods (A, B, C)
###########################################
def method_a(original: str, past_draw_digits: str) -> str:
    """
    Removes EXACT digits from 'original' if they appear in 'past_draw_digits'.
    E.g., if 'past_draw_digits' = '341', we remove '3','4','1' from the string.
    """
    to_remove = set(past_draw_digits)
    return "".join(ch for ch in original if ch not in to_remove)

def method_b(original: str, past_draw_digits: str) -> str:
    """
    Eliminates actual digit if present; else eliminates its mirror if present.
    We do a simplistic pass: 
      For each digit in 'past_draw_digits', 
        we remove it once if found, else remove its mirror once if found.
    """
    result = list(original)
    result_indices_removed = set()

    for d in past_draw_digits:
        removed_any = False
        # try removing d once
        for i, ch in enumerate(result):
            if i in result_indices_removed:
                continue
            if ch == d:
                result_indices_removed.add(i)
                removed_any = True
                break
        # if not found, try mirror
        if (not removed_any) and d in MIRROR_MAP:
            m = MIRROR_MAP[d]
            for i, ch in enumerate(result):
                if i in result_indices_removed:
                    continue
                if ch == m:
                    result_indices_removed.add(i)
                    removed_any = True
                    break
        # else do nothing 
        # (some advanced logic might forcibly remove something 
        #  even if not found, but let's keep it simple)
    final_str = "".join(
        ch for idx, ch in enumerate(result) 
        if idx not in result_indices_removed
    )
    return final_str

def method_c(original: str, past_draw_digits: str) -> str:
    """
    Eliminates BOTH the actual digit and its mirror from 'original'.
    E.g., if digit=4 => remove '4' and '9'.
    We remove them entirely from the string (all occurrences).
    """
    to_remove = set()
    for d in past_draw_digits:
        to_remove.add(d)
        if d in MIRROR_MAP:
            to_remove.add(MIRROR_MAP[d])
    return "".join(ch for ch in original if ch not in to_remove)

###########################################
# 3. Single-Digit Transit Removal (Method D)
###########################################
def iterative_transit_reduction(original: str, past_draw_digits: str):
    """
    Removes each digit in 'past_draw_digits' one at a time, 
    also removing mirror, checking for stable patterns at each step.
    Returns a list describing each step.
    """
    steps_info = []
    current_str = original
    for idx, d in enumerate(past_draw_digits, start=1):
        new_str = current_str.replace(d, '')
        # remove mirror too
        if d in MIRROR_MAP:
            mir = MIRROR_MAP[d]
            new_str = new_str.replace(mir, '')
        stable_3val = []
        if is_3value_cluster(new_str):
            stable_3val.append(new_str)
        steps_info.append({
            "step_idx": idx,
            "removed_digit": d,
            "result_str": new_str,
            "stable_3value_patterns": stable_3val
        })
        current_str = new_str
    return steps_info

###########################################
# 4. Utility: Finding 3-Value Substrings
###########################################
def find_all_3value_substrings(digits_str: str, min_len=3):
    """
    Return all substrings of length >=3 that are '3-value' 
    in either actual or VTrac sense.
    """
    results = []
    n = len(digits_str)
    for length in range(min_len, n+1):
        for start in range(0, n-length+1):
            sub = digits_str[start:start+length]
            if is_3value_cluster(sub):
                results.append(sub)
    return list(set(results))  # deduplicate

# 5. Main Class

class LongStringReductionModule:
    """
    Encapsulates logic to process "long string" columns for:
     1) area #1: (Set3,Set2,Set1->Draw1)(col7..col5, row=R2/R4/R6/R8) 
     2) area #2: (Set1 ONLY, but Draw4 col3, Draw6 col1, row=R2/R4/R6/R8)
    """

    def __init__(self, data_dict, reduction_config=None):
        self.data = data_dict
        self.config = reduction_config or {}
        # default scoring
        self.scoring_weights = {
            "initial_stable": 3,
            "method_a": 2,
            "method_b": 2,
            "method_c": 2,
            "method_d": 3,  # single-transit approach
            "multi_method_bonus": 1
        }
        if "scoring" in self.config:
            self.scoring_weights.update(self.config["scoring"])

    ###########################################
    # 5A. Gathering "Long String Area #1"
    ###########################################
    def gather_long_string_area1_cells(self):
        """
        area #1 => columns (7,6,5) in 
          Set3->Draw1, Set2->Draw1, Set1->Draw1 
        for sections [Midday, Evening, Combined], rows=R2,R4,R6,R8
        """
        results = []
        sections = ["Midday","Evening","Combined"]
        sets_ = ["Set3","Set2","Set1"]
        columns_of_interest = [7,6,5]
        rows_of_interest = ["R2","R4","R6","R8"]

        for sec in sections:
            if sec not in self.data["sections"]:
                continue
            sets_dict = self.data["sections"][sec]["sets"]
            for st in sets_:
                if st not in sets_dict:
                    continue
                draws_dict = sets_dict[st]["draws"]
                if "Draw1" not in draws_dict:
                    continue
                draw_data = draws_dict["Draw1"]
                if "pattern_variations" not in draw_data:
                    continue
                pattern_variations = draw_data["pattern_variations"]
                for rowkey in rows_of_interest:
                    if rowkey not in pattern_variations:
                        continue
                    row_strings = pattern_variations[rowkey]
                    # col7-> index0, col6->index1, col5->index2
                    for i, col in enumerate(columns_of_interest):
                        if i >= len(row_strings):
                            continue
                        original_str = row_strings[i]
                        cell_id = f"{sec}|{st}|Draw1|{rowkey}|col{col}"
                        results.append((cell_id, original_str, {
                            "section": sec,
                            "set": st,
                            "draw": "Draw1",
                            "row": rowkey,
                            "col": col
                        }))
        return results

    ###########################################
    # 5B. Gathering "Long String Area #2"
    ###########################################
    def gather_long_string_area2_cells(self):
        """
        area #2 => 
          * Set1 ONLY
          * for Midday/Evening/Combined
          * specifically Draw4 col3, Draw6 col1
          * rows=R2,R4,R6,R8
        """
        results = []
        sections = ["Midday","Evening","Combined"]
        # we only want set1
        st = "Set1"
        # specific draws => Draw4 col3, Draw6 col1
        # we parse them individually
        # rows=R2,R4,R6,R8
        rows_of_interest = ["R2","R4","R6","R8"]

        for sec in sections:
            if sec not in self.data["sections"]:
                continue
            sets_dict = self.data["sections"][sec]["sets"]
            if st not in sets_dict:
                continue
            draws_dict = sets_dict[st]["draws"]

            # we want "Draw4" col3, "Draw6" col1
            for d,cidx in [("Draw4",3), ("Draw6",1)]:
                if d not in draws_dict:
                    continue
                ddata = draws_dict[d]
                if "pattern_variations" not in ddata:
                    continue
                pattern_variations = ddata["pattern_variations"]
                for rowkey in rows_of_interest:
                    if rowkey not in pattern_variations:
                        continue
                    row_strings = pattern_variations[rowkey]
                    # cidx might be outside the row_strings length
                    # row_strings is a list for col7..col1 => index0..6
                    # col7->idx0 ... col1->idx6
                    # So to find col3 => that is idx(7-3)=4 ? 
                    # Actually if the array is col7->0, col6->1, col5->2, col4->3, col3->4, col2->5, col1->6
                    # So col3 => index=4, col1 => index=6
                    # We'll do a small map
                    col_to_index = {7:0,6:1,5:2,4:3,3:4,2:5,1:6}
                    if cidx not in col_to_index:
                        continue
                    arr_index = col_to_index[cidx]
                    if arr_index >= len(row_strings):
                        continue
                    original_str = row_strings[arr_index]
                    cell_id = f"{sec}|{st}|{d}|{rowkey}|col{cidx}"
                    results.append((cell_id, original_str, {
                        "section": sec,
                        "set": st,
                        "draw": d,
                        "row": rowkey,
                        "col": cidx
                    }))
        return results

    ###########################################
    # 5C. Generic Processing for Each Cell
    ###########################################
    def process_cell_string(self, original_str: str, meta: dict) -> dict:
        """
        Applies the 3 steps:
         1) Pre-reduction stable patterns
         2) Digit Reduction methods A/B/C for both own & combined
         3) Single-digit approach D for both own & combined
        Returns a dictionary with all details and a final notable patterns list.
        """
        cell_result = {
            "metadata": meta,
            "original_string": original_str,
            "initial_stable_patterns": [],
            "method_a": {},
            "method_b": {},
            "method_c": {},
            "method_d": {},
            "final_score": 0,
            "notable_patterns": []
        }

        # Step1: pre-reduction stable
        pre_subs = find_all_3value_substrings(original_str)
        stable_pre = []
        pattern_score_map = defaultdict(int)

        for sb in pre_subs:
            stable_pre.append({"pattern": sb, "score_contrib": self.scoring_weights["initial_stable"]})
            pattern_score_map[sb] += self.scoring_weights["initial_stable"]
        cell_result["initial_stable_patterns"] = stable_pre

        # define helper to apply method
        def apply_and_record(method_func, method_key, raw_string, past_digits):
            reduced = method_func(raw_string, past_digits)
            found_subs = find_all_3value_substrings(reduced)
            for fs in found_subs:
                pattern_score_map[fs] += self.scoring_weights[method_key]
            return {
                "past_draw_digits": past_digits,
                "reduced_string": reduced,
                "found_3value_substrings": found_subs
            }

        p_own = self.config.get("past_draw_digits_own","")
        p_comb = self.config.get("past_draw_digits_combined","")

        # method A
        cell_result["method_a"]["own"] = apply_and_record(method_a, "method_a", original_str, p_own)
        cell_result["method_a"]["combined"] = apply_and_record(method_a, "method_a", original_str, p_comb)

        # method B
        cell_result["method_b"]["own"] = apply_and_record(method_b, "method_b", original_str, p_own)
        cell_result["method_b"]["combined"] = apply_and_record(method_b, "method_b", original_str, p_comb)

        # method C
        cell_result["method_c"]["own"] = apply_and_record(method_c, "method_c", original_str, p_own)
        cell_result["method_c"]["combined"] = apply_and_record(method_c, "method_c", original_str, p_comb)

        # single-digit approach (D)
        def do_transit_d(raw_string, digits):
            steps_info = iterative_transit_reduction(raw_string, digits)
            discovered = []
            for stp in steps_info:
                for pat in stp["stable_3value_patterns"]:
                    discovered.append(pat)
                    pattern_score_map[pat] += self.scoring_weights["method_d"]
            return {
                "past_draw_digits": digits,
                "transit_steps": steps_info,
                "all_discovered_patterns": list(set(discovered))
            }

        cell_result["method_d"]["own"] = do_transit_d(original_str, p_own)
        cell_result["method_d"]["combined"] = do_transit_d(original_str, p_comb)

        # multi-method bonus
        method_presence_map = defaultdict(set)
        # gather from initial stable
        for item in stable_pre:
            method_presence_map[item["pattern"]].add("pre_reduction")
        # gather from method A/B/C
        for meth in ["method_a","method_b","method_c"]:
            for subcat in ["own","combined"]:
                foundlist = cell_result[meth][subcat]["found_3value_substrings"]
                for fz in foundlist:
                    method_presence_map[fz].add(meth)
        # gather from method_d
        for subcat in ["own","combined"]:
            dd = cell_result["method_d"][subcat]
            fl = dd["all_discovered_patterns"]
            for fz in fl:
                method_presence_map[fz].add("method_d")

        for patt, mmset in method_presence_map.items():
            if len(mmset) >= 2:
                # add multi_method_bonus
                pattern_score_map[patt] += self.scoring_weights["multi_method_bonus"]

        # compile final patterns
        final_patterns = [(patt, sc) for patt, sc in pattern_score_map.items() if sc>0]
        final_patterns.sort(key=lambda x: x[1], reverse=True)
        cell_result["notable_patterns"] = [
            {"pattern": fp[0], "score": fp[1]} for fp in final_patterns
        ]
        cell_result["final_score"] = final_patterns[0][1] if final_patterns else 0

        return cell_result

    ###########################################
    # 5D. MAIN RUN for "Long String Area #1"
    ###########################################
    def analyze_area1_long_strings(self):
        """
        Processes the 7,6,5 columns in (Set3,Set2,Set1)->Draw1 
        across Midday/Evening/Combined, returning structured output.
        """
        cell_list = self.gather_long_string_area1_cells()
        analysis_output = []
        for (cell_id, orig_str, meta) in cell_list:
            cell_result = self.process_cell_string(orig_str, meta)
            cell_result["cell_id"] = cell_id
            analysis_output.append(cell_result)
        return {
            "module_name": "LongStringReduction_Area1",
            "timestamp": datetime.datetime.now().isoformat(),
            "analysis_results": analysis_output
        }

    ###########################################
    # 5E. MAIN RUN for "Long String Area #2"
    ###########################################
    def analyze_area2_long_strings(self):
        """
        Processes the second simplified area:
         - Set1 ONLY
         - (Draw4 col3) and (Draw6 col1)
         - Midday/Evening/Combined
         - R2/R4/R6/R8
        """
        cell_list = self.gather_long_string_area2_cells()
        analysis_output = []
        for (cell_id, orig_str, meta) in cell_list:
            cell_result = self.process_cell_string(orig_str, meta)
            cell_result["cell_id"] = cell_id
            analysis_output.append(cell_result)
        return {
            "module_name": "LongStringReduction_Area2",
            "timestamp": datetime.datetime.now().isoformat(),
            "analysis_results": analysis_output
        }

Below is a concise yet detailed set of suggestions for optimally incorporating these advanced features (digit shifting, lingering sets, minimal box/straight scoring) into the Long String and Hot Zones modules, based on everything we’ve discussed and your own guidelines.

1) Handling “Lingering Patterns” Across Sets (Set3 / Set2 / Set1)
Primary Logic Resides in the Analysis (Aggregator) Module
Rationale
The aggregator (or final ML stage) is designed to combine patterns from all sets and evaluate synergy, persistence, and re-emergence. That’s the best place to do in-depth scoring or awarding big synergy bonuses if a pattern discovered in Set2 reappears in Set1.

Minimal Implementation in Long String / Hot Zones
Store an origin-set label: If your Long String or Hot Zone module extracts a pattern from Set2 or Set3, you can tag it in the output, e.g. "origin_set": "Set2".

Flag if the pattern also appears in Set1: If you have quick access to the Set1 data, you could do a local mini-check. For instance:

Why:

This approach helps you pass relevant info “(Set2 origin, reappears in Set1 = True)” up to the aggregator.

The aggregator will combine that with all other modules’ data to produce the final synergy weighting.

2) “Digit Shifting” or “Removing One Extra Digit” to Uncover Hidden 3-Value Patterns
Where to Implement
Within the Long String or Hot Zones modules**—the specialized modules that do advanced manipulations or “helper” checks to reveal hidden patterns.

Optimal Implementation Details
Identify Potential 4-Digit Substrings: If a substring is 4 digits but “almost” a 3-value cluster (like it has 4 unique digits or 4 total digits with 3 unique + 1 mismatch).

Try Removing Each Digit:

For substring S of length 4, remove the i-th digit, forming a 3-digit substring.

Check if that 3-digit substring is a valid 3-value pattern.

Record the “digit-shifted” pattern in your local output with an attribute like "shifted_pattern": True or "derived_from": "2254 → remove '4' → 225".

Optionally apply a small local score: e.g. +1 for “digit shifting success.”

Alternatively, just store the derived pattern. Let the aggregator see it as a stable 3-digit pattern and do the main scoring synergy.

Why This Matters
You specifically asked for a way to find stable 3-value clusters that are overshadowed by a single outlier digit. This “digit shifting” approach systematically accomplishes that.

3) Minimal Boxed/Straight Scoring Inside the Long String & Hot Zones Modules
Why Minimal
The main Pattern Extractor or aggregator might already do big row/column synergy for “straight vs. box,” so we want to avoid duplicating it fully.

How to Implement Minimally
When a new 3-digit pattern is found (whether from digit reduction or from shifting):

Check if it’s a permutation of an existing pattern found in the same step (or same cell).

def detect_hidden_patterns_in_box(box_string, method_label):
    """
    Example function that tries digit shifting and then
    checks if the resulting 3-digit pattern is new or a box-dup of an existing pattern.
    """
    found_patterns = []

    # 1) If length=4, attempt digit shifting
    if len(box_string) == 4:
        for i in range(4):
            candidate = box_string[:i] + box_string[i+1:]
            if is_3_value_pattern(candidate):
                found_patterns.append({
                    "pattern": candidate,
                    "shifted_pattern": True,
                    "removed_digit": box_string[i],
                    "origin_substring": box_string,
                    "method": method_label
                    # we can add local minor scoring or flags
                })

def unify_box_groups(pattern_list):
    """
    Minimal approach:
    - assign each new pattern to a box_group_id based on sorted digits
    - check if it's an exact string match with an existing pattern for 'is_straight_dup'
    """
    box_groups = {}
    for pinfo in pattern_list:
        cano = ''.join(sorted(pinfo["pattern"]))
        if cano not in box_groups:
            box_groups[cano] = []
        # check if exact
        is_exact_dup = any(x["pattern"] == pinfo["pattern"] for x in box_groups[cano])
        pinfo["is_straight_duplicate"] = is_exact_dup
        pinfo["box_group"] = cano
        box_groups[cano].append(pinfo)

#!/usr/bin/env python
"""
hotzones_analysis.py

Comprehensive Hot Zone & Super-Hot Zone Analysis Module

Core Steps:
1. Identify cells flagged as hot (*) or super-hot (**).
2. Apply single transit-digit reduction (Method D).
3. Extract stable 3-value patterns (3-digit or fewer unique digits).
4. Check minimal local repeats (box/straight, 3-digit repeats across columns).
5. Record local synergy, V-Trac checks if desired.
6. Produce structured output with local scores for each pattern.
"""

import copy
from collections import defaultdict

#####################################################
# 1) Single Transit Digit Reduction Helpers
#####################################################

MIRROR_MAP = {
    '0':'5','1':'6','2':'7','3':'8','4':'9',
    '5':'0','6':'1','7':'2','8':'3','9':'4'
}

def single_transit_digit_reduction(string: str, transit_digit: str, remove_mirror: bool = True):
    """
    Removes one transit_digit from string, and if remove_mirror=True,
    also removes the digit's mirror if present.
    Returns: (new_string, digit_removed_flag)
    """
    digit_removed = False
    s = list(string)  # manipulate as list for clarity

    # Remove the transit_digit if present
    if transit_digit in s:
        s = [ch for ch in s if ch != transit_digit]
        digit_removed = True

    if remove_mirror:
        mirror_d = MIRROR_MAP.get(transit_digit)
        if mirror_d and mirror_d in s:
            s = [ch for ch in s if ch != mirror_d]
            digit_removed = True

    return "".join(s), digit_removed


def iterative_transit_reduction(string: str, past_draw_digits: list, remove_mirror: bool = True):
    """
    Applies single_transit_digit_reduction *one digit at a time* 
    for each digit in 'past_draw_digits'.
    
    Returns a list of intermediate strings (including final).
    Example usage: to track the progression of patterns as each digit is removed.
    """
    current_string = string
    reductions = [current_string]  # store initial as well
    for d in past_draw_digits:
        new_str, removed = single_transit_digit_reduction(current_string, d, remove_mirror=remove_mirror)
        if removed and new_str != current_string:
            # we have changed the string, record it
            current_string = new_str
            reductions.append(current_string)
        else:
            # no real change or digit not present
            continue
    return reductions


#####################################################
# 2) 3-Value (3-digit) Pattern Extractors for short strings
#####################################################

def find_3value_clusters(digits_str: str, min_len=3, max_len=6):
    """
    Return all sub-clusters in digits_str that have length >=3 and
    contain <=3 unique digits. 
    Because hot zone strings are short, we can do a quick substring check.
    max_len is typically ~6 because hot zone rarely >6 digits.
    """
    found_clusters = set()
    length = len(digits_str)
    if length < min_len:
        return found_clusters

    upper_bound = min(length, max_len)
    for size in range(min_len, upper_bound + 1):
        for start in range(length - size + 1):
            sub = digits_str[start:start+size]
            unique_d = set(sub)
            if len(unique_d) <= 3:
                found_clusters.add(sub)
    return found_clusters


def pattern_is_straight(p1: str, p2: str) -> bool:
    """Checks if p1 == p2 exactly => 'straight match'."""
    return p1 == p2

def pattern_is_boxed_equiv(p1: str, p2: str) -> bool:
    """Checks if sorted digits are the same => 'boxed' equivalent."""
    return ''.join(sorted(p1)) == ''.join(sorted(p2))


#####################################################
# 3) (Optional) Basic V-Trac Transform
#####################################################

def vtrac_transform(digits_str: str):
    """
    Transforms a short string into V-Trac representation (sorted for simpler matching).
    '0'↔'5','1'↔'6','2'↔'7','3'↔'8','4'↔'9'.
    We'll unify by sorted, so '241' => transform => e.g. '2->7,4->9,1->6' => '796', 
    then sorted => '679'.
    """
    mapped = []
    for ch in digits_str:
        mapped.append(MIRROR_MAP[ch])
    return ''.join(sorted(mapped))


#####################################################
# 4) Local Scoring for Hotzone Patterns
#####################################################

def score_hotzone_pattern(
    pattern: str,
    hot_level: int,
    repeated_across_boxes: bool = False,
    repeated_across_structures: bool = False,
    vtrac_match: bool = False,
    stable_after_reduction: bool = False
):
    """
    Basic local scoring formula for hot-zone patterns. 
    This is a simple example; adjust weighting as desired.
    """
    base = 0

    # If pattern is stable in a hot zone, give base points
    base += 5  

    # Add points if 'hot_level' is 2 => super-hot
    if hot_level == 2:
        base += 3
    elif hot_level == 1:
        base += 1

    # +2 if found repeated across multiple R-boxes or columns
    if repeated_across_boxes:
        base += 2

    # +3 if repeated across sets or across midday/evening/combined structures
    if repeated_across_structures:
        base += 3

    # +2 if it has a known V-Trac synergy
    if vtrac_match:
        base += 2

    # +1 if it remains stable after single-transit-digit reduction
    if stable_after_reduction:
        base += 1

    return base

# 5) Main Function: analyze_hotzones
#####################################################

def analyze_hotzones(
    data: dict,
    prev_draw_digits_own: list,
    prev_draw_digits_combined: list = None,
    remove_mirror: bool = True
):
    """
    - data: your entire state/sections data structure (OntarioCanada4...).
    - prev_draw_digits_own: digits from the previous draw of the same type (midday->midday).
    - prev_draw_digits_combined: digits from the 'combined' previous draw if you want that logic. 
      If not needed, can pass None or [].
    - remove_mirror: if True, we remove mirror digit as well in single transit-digit approach.
    
    Returns: A dictionary with all discovered hot-zone patterns and local scoring info.
    """

    if prev_draw_digits_combined is None:
        prev_draw_digits_combined = []

    hotzone_results = []
    # We'll parse data["sections"] -> each "Midday"/"Evening"/"Combined"
    # -> "sets" -> each Set -> "draws" -> each Draw => see "metadata" for hotzone indicators.

    sections = data.get("sections", {})

    for section_name, section_data in sections.items():
        sets_data = section_data.get("sets", {})
        for set_key, set_info in sets_data.items():
            draws_info = set_info.get("draws", {})
            for draw_key, draw_data in draws_info.items():

                # 1) Check if the entire draw_data is flagged is_hot_zone in 'metadata'
                metadata = draw_data.get("metadata", {})
                if not metadata.get("is_hot_zone", False):
                    # skip if it's not a hot zone draw
                    continue

                # If it's a hot zone draw, we gather the pattern_variations R2/R4/R6/R8
                pattern_variations = draw_data.get("pattern_variations", {})
                hot_zone_indicators = metadata.get("hot_zone_indicators", {})
                # e.g., hot_zone_indicators["R2"] => array of strings ["", "", "*", ...], etc.

                # For each R-type in R2, R4, R6, R8
                for rtype in ["R2","R4","R6","R8"]:
                    if rtype not in pattern_variations:
                        continue
                    # a list of digit strings in columns 7->1
                    col_strings = pattern_variations[rtype]

                    # check the indicators array for that row
                    rtype_indicators = hot_zone_indicators.get(rtype, [])

                    for col_idx, cell_str in enumerate(col_strings):
                        if not cell_str:
                            continue
                        indicator = ""
                        if col_idx < len(rtype_indicators):
                            indicator = rtype_indicators[col_idx]  # "", "*", or "**"

                        # only proceed if indicator is "*" or "**"
                        if indicator not in ("*","**"):
                            continue

                        hot_level = 2 if (indicator=="**") else 1

                        # 2) We have a hot zone cell. Let's parse the digits:
                        raw_digits = cell_str

                        # 3) Single-transit-digit reductions (own + combined)
                        #    We'll unify them if you do both. Or do them separately.
                        #    For example, let's do a single pass with own digits => gather patterns
                        #    then with combined => gather patterns.
                        stable_patterns_in_raw = find_3value_clusters(raw_digits)

                        # do iterative approach for own
                        own_reductions = iterative_transit_reduction(raw_digits, prev_draw_digits_own, remove_mirror=remove_mirror)
                        combined_reductions = []
                        if prev_draw_digits_combined:
                            combined_reductions = iterative_transit_reduction(raw_digits, prev_draw_digits_combined, remove_mirror=remove_mirror)

                        # 4) For each intermediate reduced string, find stable clusters
                        #    We unify them in a set so we don't duplicate
                        final_clusters = set()

                        # raw first
                        for pat in stable_patterns_in_raw:
                            final_clusters.add( (pat, False) )  
                            # (pattern, stable_after_reduction=False)

                        for red_str in own_reductions[1:]:  # skip the first which is raw
                            sub_clusters = find_3value_clusters(red_str)
                            for sc in sub_clusters:
                                final_clusters.add( (sc, True) )

                        for red_str in combined_reductions[1:]:
                            sub_clusters = find_3value_clusters(red_str)
                            for sc in sub_clusters:
                                final_clusters.add( (sc, True) )

                        # 5) minimal local synergy checks:
                        #    check 3-digit repeats with other columns in the same draw
                        #    (We won't do a big cross-set check here, aggregator can do that.)
                        #    We'll do a quick approach: if the same pattern appears in multiple columns of the same (rtype)...

                        # We'll store them in a structure to see if there's repeated across columns
                        # We'll do a small local approach only.
                        # In reality aggregator might do more thorough synergy.
                        # But let's do a dictionary to see how many times each pattern appears in this entire row.
                        # We can parse pattern_variations[rtype] fully, ignoring star or not, just to detect any repeats?
                        # Actually for "hot zone only" we might just see if final_clusters themselves have duplicates. 
                        # It's simpler to skip big synergy or do a small approach:

                        # We'll build a local dictionary below. For now let's just produce the final info for each pattern.

                        for (pat, stable_after_red) in final_clusters:
                            # check if 3-digit length => if len(pat)==3 => we might consider it a direct "3-digit pattern"
                            repeated_across_boxes = False
                            repeated_across_structures = False  # aggregator territory, but we can do local placeholder

                            # minimal V-Trac approach
                            vt_p = vtrac_transform(pat)
                            # We won't do cross-check here, just record we have a vtrac version

                            # let's do a local score
                            local_score = score_hotzone_pattern(
                                pattern=pat,
                                hot_level=hot_level,
                                repeated_across_boxes=repeated_across_boxes,
                                repeated_across_structures=repeated_across_structures,
                                vtrac_match=False,  # we skip deeper synergy, aggregator can do advanced
                                stable_after_reduction=stable_after_red
                            )

                            # store final result
                            hotzone_results.append({
                                "section": section_name,   # e.g. "Midday", "Evening", "Combined"
                                "set": set_key,            # e.g. "Set1"
                                "draw": draw_key,          # e.g. "Draw3"
                                "rtype": rtype,            # "R2"/"R4"/"R6"/"R8"
                                "column_index": col_idx,   # index in the string array
                                "raw_digits": raw_digits,
                                "pattern": pat,
                                "pattern_length": len(pat),
                                "stable_after_reduction": stable_after_red,
                                "hot_level": hot_level,    # 1 for '*', 2 for '**'
                                "local_hotzone_score": local_score,
                                "vtrac_form": vt_p
                            })

    # Return a structured object for aggregator usage
    return {
        "hotzone_patterns": hotzone_results
    }

import datetime
from collections import defaultdict

def unify_pattern_key(actual_digits: str) -> str:
    """
    Returns a canonical form used for grouping across modules.
    E.g. the lexicographically smallest rotation or just sorted.
    We'll do sorted for simplicity here.
    """
    return ''.join(sorted(actual_digits))

def aggregator_main(
    pattern_extractor_results: dict,
    longstring_area1: dict,
    longstring_area2: dict,
    hotzones_results: dict,
    ml_model=None
):
    """
    Aggregates results from all modules, applies synergy & advanced scoring,
    optionally calls an ML model to re-rank or finalize predictions.

    :param pattern_extractor_results: dict from pattern_extractor.py
    :param longstring_area1: dict from long_string_reduction.py (area #1)
    :param longstring_area2: dict from long_string_reduction.py (area #2)
    :param hotzones_results: dict from hotzones_analysis.py
    :param ml_model: optional ML model for advanced ranking
    :return: final aggregated dictionary with synergy scores
    """

    # 1) Convert pattern_extractor results into a universal list
    # Example structure:
    # [
    #   {
    #     "pattern_key": "123",
    #     "actual_digits": "123",
    #     "vtrac": "678",
    #     "source_module": "PatternExtractor",
    #     "module_score": <score from pattern extractor>,
    #     "locations": [...]
    #   },
    #   ...
    # ]
    extractor_list = convert_pattern_extractor_to_universal(pattern_extractor_results)

    # 2) Convert long string area #1 & area #2
    ls_area1_list = convert_longstring_results_to_universal(longstring_area1, "LongStringArea1")
    ls_area2_list = convert_longstring_results_to_universal(longstring_area2, "LongStringArea2")

    # 3) Convert hotzones
    hz_list = convert_hotzones_to_universal(hotzones_results)

    # combine them
    combined_raw = extractor_list + ls_area1_list + ls_area2_list + hz_list

    # 4) Group by pattern_key
    aggregator = defaultdict(lambda: {
        "patterns": [],
        "final_aggregated_score": 0,
        "sources_encountered": set(),
        "sets_encountered": set(),       # e.g. {Set1, Set2, Set3}
        "sections_encountered": set(),   # e.g. {Midday, Evening, Combined}
        "vtrac_forms": set(),
        "digit_lens": set(),
    })

    for entry in combined_raw:
        pk = entry["pattern_key"]
        aggregator[pk]["patterns"].append(entry)
        aggregator[pk]["sources_encountered"].add(entry["source_module"])
        if "set" in entry:
            aggregator[pk]["sets_encountered"].add(entry["set"])
        if "section" in entry:
            aggregator[pk]["sections_encountered"].add(entry["section"])
        aggregator[pk]["vtrac_forms"].add(entry.get("vtrac",""))
        aggregator[pk]["digit_lens"].add(len(entry["actual_digits"]))

    # 5) synergy scoring
    for pk, data_block in aggregator.items():
        base_score = 0
        patterns_in_block = data_block["patterns"]

        # sum up module scores (for example)
        base_score += sum(p["module_score"] for p in patterns_in_block if p["module_score"] > 0)

        # synergy: multiple modules
        if len(data_block["sources_encountered"]) > 1:
            base_score += 5  # e.g. synergy

        # synergy: multiple sets
        if len(data_block["sets_encountered"]) > 1:
            base_score += 3

        # synergy: multiple sections (midday/evening/combined)
        if len(data_block["sections_encountered"]) > 1:
            base_score += 3

        # synergy: if any pattern was >3 digits => extended
        max_len = max(data_block["digit_lens"]) if data_block["digit_lens"] else 3
        if max_len > 3:
            base_score += (max_len - 3)  # example

        # store final
        data_block["final_aggregated_score"] = base_score

    # 6) optional ML re-ranking
    if ml_model:
        # we would build a features DataFrame for each pk,
        # predict using ml_model, maybe do data_block["ml_score"] = ...
        # then combine with final_aggregated_score
        pass

    # 7) produce final output
    # Let’s transform aggregator dict into a sorted list
    final_list = []
    for pk, block in aggregator.items():
        final_list.append({
            "pattern_key": pk,
            "final_aggregated_score": block["final_aggregated_score"],
            "sources": list(block["sources_encountered"]),
            "sets": list(block["sets_encountered"]),
            "sections": list(block["sections_encountered"]),

def convert_pattern_extractor_to_universal(pattern_extractor_results):
    """
    pattern_extractor_results is a dict: { "patternA": {...}, ... }
    We'll produce a list of dicts with [pattern_key, actual_digits, vtrac, source_module, module_score].
    """
    out = []
    for cano, info in pattern_extractor_results.items():
        out.append({
            "pattern_key": cano,
            "actual_digits": cano,  # or info["canonical"] if you want
            "vtrac": info.get("vtrac",""),
            "source_module": "PatternExtractor",
            "module_score": info.get("score", 0),
            "locations": info.get("locations", [])
        })
    return out


def convert_longstring_results_to_universal(longstring_dict, source_label):
    """
    We expect something like:
    {
      "module_name": "LongStringReduction_Area1",
      "analysis_results": [
        {
          "cell_id": "...",
          "notable_patterns": [ {"pattern": "5594", "score": 7}, ...]
        },
        ...
      ]
    }
    We'll map each "pattern" to a universal entry.
    """
    out = []
    analysis_list = longstring_dict.get("analysis_results", [])
    for cell_data in analysis_list:
        cell_meta = cell_data.get("metadata", {})
        cell_id = cell_data.get("cell_id","")
        for patinfo in cell_data.get("notable_patterns", []):
            pat = patinfo["pattern"]
            sc = patinfo["score"]
            pk = unify_pattern_key(pat)
            entry = {
                "pattern_key": pk,
                "actual_digits": pat,
                "vtrac": "",  # you could compute a vtrac if you want
                "source_module": source_label,
                "module_score": sc,
                "cell_id": cell_id,
                "section": cell_meta.get("section"),
                "set": cell_meta.get("set"),
                "draw": cell_meta.get("draw"),
            }
            out.append(entry)
    return out


def convert_hotzones_to_universal(hotzones_dict):
    """
    hotzones_dict e.g. { "hotzone_patterns": [ { "section":"Midday", "pattern":"6411", ...}, ... ] }
    We'll produce a universal structure similarly.
    """
    out = []
    for pinfo in hotzones_dict.get("hotzone_patterns", []):
        pat = pinfo["pattern"]
        sc = pinfo.get("local_hotzone_score", 0)
        pk = unify_pattern_key(pat)
        entry = {
            "pattern_key": pk,
            "actual_digits": pat,
            "vtrac": pinfo.get("vtrac_form",""),
            "source_module": "HotZones",
            "module_score": sc,
            "section": pinfo.get("section"),
            "set": pinfo.get("set"),
            "draw": pinfo.get("draw"),
            "rtype": pinfo.get("rtype"),
        }
        out.append(entry)
    return out

def form_consensus_doubles(consensus_digit, key_digits):
    """
    Returns up to 3 x 3 = 9 combos of double + key digit.
    e.g. consensus_digit='7', key_digits=['1','0','5'] => 9 combos.
    """
    result = []
    for kd in key_digits:
        # 3 permutations of double
        result.append(consensus_digit+consensus_digit+kd)
        result.append(consensus_digit+kd+consensus_digit)
        result.append(kd+consensus_digit+consensus_digit)
    return sorted(set(result))

def keep_pair_mirror_3rd(d3):
    """
    If the 3-digit is 'abc', produce mirrored combos of the third 
    (and optionally the first or second).
    Simplest approach: 
      1) Mirror the 3rd:  ab(MIRROR(c))
      2) Mirror the 1st:  (MIRROR(a))bc
      3) Mirror the 2nd:  a(MIRROR(b))c
      Then permute each 3-digit
    """
    import itertools
    
    # a quick mirror map
    mirror_map = {'0':'5','1':'6','2':'7','3':'8','4':'9',
                  '5':'0','6':'1','7':'2','8':'3','9':'4'}
    
    a,b,c = d3[0], d3[1], d3[2]
    combos = set()
    # build the mirrored variants
    triple_set = [
      a+b+mirror_map.get(c,c),
      mirror_map.get(a,a)+b+c,
      a+mirror_map.get(b,b)+c
    ]
    # now permute each
    for tvar in triple_set:
        for p in itertools.permutations(tvar, 3):
            combos.add(''.join(p))
    return combos

        actual_dig = cano_form  # or info["canonical"] if that’s how you store it
        out_item = {
            "pattern_key": cano_form,
            "actual_digits": actual_dig,
            "vtrac_form": info.get("vtrac",""),  # if stored
            "source_module": "PatternExtractor",
            "module_score": info.get("score", 0),
            # You might have other booleans like "consensus_flag", "double_trigger"
            # stored in the pattern_extractor data. For now, we just do minimal.
        }
        output_list.append(out_item)
    return output_list

def convert_longstring_results_to_universal(longstring_dict: dict, source_label: str):
    """
    longstring_dict often has structure:
      {
         "module_name": "LongStringReduction_Area1",
         "analysis_results": [
           {
             "cell_id": "...",
             "metadata": { "section": "Midday", "set": "Set1", ... },
             "notable_patterns": [
               {"pattern": "5594", "score": 7},
               ...
             ],
             ...
           },
           ...
         ]
      }
    We’ll parse out the "notable_patterns" and unify them.
    """
    out_list = []
    analysis_arr = longstring_dict.get("analysis_results", [])
    for cell_data in analysis_arr:
        meta = cell_data.get("metadata", {})
        cell_id = cell_data.get("cell_id","")
        patterns_list = cell_data.get("notable_patterns", [])
        for patinfo in patterns_list:
            pat_str = patinfo["pattern"]
            sc = patinfo.get("score", 0)
            pk = canonical_key(pat_str)  # unify key
            item = {
                "pattern_key": pk,
                "actual_digits": pat_str,
                "vtrac_form": "",  # optionally compute from pat_str if needed
                "source_module": source_label,
                "module_score": sc,
                "cell_id": cell_id,
            }
            # if meta has section/set/draw, store them
            if "section" in meta:
                item["section"] = meta["section"]
            if "set" in meta:
                item["set"] = meta["set"]
            if "draw" in meta:
                item["draw"] = meta["draw"]
            out_list.append(item)
    return out_list

def convert_hotzones_to_universal(hotzones_dict: dict):
    """
    hotzones_dict is typically:
      {
        "hotzone_patterns": [
          {
            "pattern": "6411",
            "local_hotzone_score": 12,
            "section": "Midday",
            "set": "Set1",
            "draw": "Draw3",
            "hot_level": 2,  # or "hot_zone_level"
            ...
          },
          ...
        ]
      }
    We'll produce the universal format.
    """
    out_list = []
    arr = hotzones_dict.get("hotzone_patterns", [])
    for pinfo in arr:
        pat_str = pinfo["pattern"]
        sc = pinfo.get("local_hotzone_score", 0)
        pk = canonical_key(pat_str)
        # also see if there's a "hot_zone_level" or "hot_level"
        hz_lvl = pinfo.get("hot_zone_level", 0) or pinfo.get("hot_level", 0)
        item = {
            "pattern_key": pk,
            "actual_digits": pat_str,
            "vtrac_form": pinfo.get("vtrac_form",""),
            "source_module": "HotZones",
            "module_score": sc,
            "section": pinfo.get("section"),
            "set": pinfo.get("set"),
            "draw": pinfo.get("draw"),
            "hot_zone_level": hz_lvl
        }
        out_list.append(item)
    return out_list

def enhanced_analysis_aggregator(
    pattern_extractor_results: dict,
    longstring_area1: dict,
    longstring_area2: dict,
    hotzones_results: dict,
    synergy_weights: dict = None,
    ml_model = None
):
    """
    Main aggregator function:
      1) Convert outputs from modules -> universal format
      2) Merge them into a single list
      3) Group by pattern_key
      4) Compute synergy-based final 'aggregated_score'
      5) Sort & return final results

    synergy_weights: optional custom scoring for synergy factors
    ml_model: optional machine-learning model to re-rank or add an ML-based score
    """
    if synergy_weights is None:
        # default synergy weights
        synergy_weights = {
            "consensus_bonus": 5,
            "double_trigger_bonus": 3,
            "double_consensus_combo": 2,
            "hotzone_level_1": 2,
            "hotzone_level_2": 3,
            "repeating_hotzone_bonus": 2,
            "cross_section_bonus": 3,
            "vtrac_match_bonus": 2,
            # perhaps synergy for multi-module or multi-set
            "multi_module_bonus": 4,
            "multi_set_bonus": 3
        }

    # 1) Convert
    extractor_list = convert_pattern_extractor_to_universal(pattern_extractor_results)
    ls1_list = convert_longstring_results_to_universal(longstring_area1, "LongStringArea1")
    ls2_list = convert_longstring_results_to_universal(longstring_area2, "LongStringArea2")
    hz_list = convert_hotzones_to_universal(hotzones_results)

    combined_raw = extractor_list + ls1_list + ls2_list + hz_list

    # 2) Group by pattern_key
    aggregator = defaultdict(lambda: {
        "patterns": [],
        "aggregated_score": 0,
        "sections_encountered": set(),
        "modules_encountered": set(),
        "sets_encountered": set()
    })

    for item in combined_raw:
        pk = item["pattern_key"]
        aggregator[pk]["patterns"].append(item)
        # track cross-section synergy
        sec = item.get("section")
        if sec:
            aggregator[pk]["sections_encountered"].add(sec)
        # track multi-module synergy
        aggregator[pk]["modules_encountered"].add(item["source_module"])
        # track multi-set synergy
        st = item.get("set")
        if st:
            aggregator[pk]["sets_encountered"].add(st)

    # 3) Apply synergy scoring
    for pk, block in aggregator.items():
        patterns_list = block["patterns"]

        base_score = 0
        # sum local scores from all items
        base_score += sum(x.get("module_score",0) for x in patterns_list)

        # check if any pattern is flagged consensus/double/hotzone/vtrac
        # (a) consensus_flag
        if any(x.get("consensus_flag", False) for x in patterns_list):
            base_score += synergy_weights["consensus_bonus"]

        # (b) double_trigger
        if any(x.get("double_trigger", False) for x in patterns_list):
            base_score += synergy_weights["double_trigger_bonus"]

            # double + consensus synergy
            if any(x.get("consensus_double", False) for x in patterns_list):
                base_score += synergy_weights["double_consensus_combo"]

        # (c) hotzone synergy: find max hot_zone_level
        max_hz = 0
        for x in patterns_list:
            lvl = x.get("hot_zone_level", 0)
            if lvl>max_hz:
                max_hz = lvl
        if max_hz == 1:
            base_score += synergy_weights["hotzone_level_1"]
        elif max_hz >= 2:
            base_score += synergy_weights["hotzone_level_2"]

        # repeating hot zone check
        hot_count = sum(1 for x in patterns_list if x.get("hot_zone_level",0)>0)
        if hot_count > 1:
            base_score += synergy_weights["repeating_hotzone_bonus"]

        # (d) cross section synergy
        if len(block["sections_encountered"])>1:
            base_score += synergy_weights["cross_section_bonus"]

        # (e) vtrac synergy
        if any(x.get("vtrac_strong_match", False) for x in patterns_list):
            base_score += synergy_weights["vtrac_match_bonus"]

        # (f) multi-module synergy
        if len(block["modules_encountered"])>1:
            base_score += synergy_weights["multi_module_bonus"]

        # (g) multi-set synergy
        if len(block["sets_encountered"])>1:
            base_score += synergy_weights["multi_set_bonus"]

        block["aggregated_score"] = base_score

    # 4) (Optional) If we have an ML model, we can build a feature row for each pattern
    #    and do ml_model.predict or predict_proba, then combine or re-rank.
    if ml_model:
        # placeholder for advanced usage
        pass

    # 5) produce final sorted list
    final_list = []
    for pk, binfo in aggregator.items():
        final_list.append({
            "pattern_key": pk,
            "aggregated_score": binfo["aggregated_score"],
            "patterns_detail": binfo["patterns"],
            "sections_encountered": list(binfo["sections_encountered"]),
            "modules_encountered": list(binfo["modules_encountered"]),
            "sets_encountered": list(binfo["sets_encountered"])
        })

    final_list.sort(key=lambda x: x["aggregated_score"], reverse=True)

    return {
        "analysis_timestamp": datetime.datetime.now().isoformat(),
        "aggregated_patterns": final_list
    }

Analytical Module Design Internal Audit
This document reviews the design and integration of the analytical modules discussed so far (Pattern Extractor, Long String Reduction, and Hot Zones) and evaluates how their outputs flow into the Analysis/Aggregator stage. The goal is to ensure that every feature these modules produce is preserved and leveraged, and to identify any gaps or improvements to optimize the overall pipeline.
Module Feature Extraction Summary
Pattern Extractor: Identifies recurring patterns in the data and outputs a set of canonical patterns (standardized pattern representations). For each pattern, it provides its V-Trac equivalent (an alternate mapped representation of the pattern), the locations or positions where the pattern occurs, and a score indicating the pattern's significance or frequency. It also attaches special flags such as a consensus flag (if the pattern is corroborated by multiple sources or criteria) and a double flag (if the pattern contains repeating elements or a doubled sequence segment).
Long String Reduction: Detects excessively long or repetitive sequences and compresses them into a simpler form. It outputs the core repeated sequence that was extracted from the long string, along with details like the original length of the sequence and how many times it repeats. This module may also provide a reduction score or ratio indicating how much the sequence was condensed, and possibly a flag if a particularly long repetition (e.g., an unusually long run) was found. The result is a condensed representation of patterns that might otherwise dominate the analysis due to length, ensuring subsequent analysis focuses on the unique pattern rather than its lengthy repetition.
Hot Zones: Analyzes the data for clusters or “hot” areas of activity. It assigns hot zone levels to elements or positions in the dataset, categorizing regions by intensity (for example, Level 1 might indicate normal activity, while higher levels like 2 or 3 mark increasingly active or significant zones). The output includes each identified hot zone region with its corresponding level classification. In addition, the module might provide a heat score or threshold value that justifies why a region was classified at a certain level. This highlights which parts of the data are most active or noteworthy, serving as cues for potential synergy with pattern findings.
Feature Flow into Aggregator
Pattern Extractor → Aggregator: The aggregator ingests the list of canonical patterns and their associated data (scores, flags, etc.). In the current design, the aggregator uses the pattern scores as a primary input – patterns with higher significance scores are given more weight in the combined analysis. It also carries forward the consensus and double flags: a consensus-tagged pattern can be noted as highly reliable (since multiple criteria agreed on it), and a double-flagged pattern is recognized for its repeated element. However, some detailed attributes like the exact V-Trac representation or the precise location indices of each pattern might not be explicitly utilized in the aggregation logic. These fields are passed along in the pattern data structure, but the aggregator does not yet have specific rules to leverage them. Long String Reduction → Aggregator: The aggregator receives the condensed sequence information from the Long String Reduction module. Instead of a long raw sequence, it obtains the core repeated pattern and possibly the count of repeats or a reduction metric. The aggregator incorporates this by treating the core pattern similarly to other patterns identified – i.e. it may assign a weight or score to that core pattern (potentially using the reduction score to gauge its significance). This ensures that a pattern which was previously one long sequence is now accounted for without over-counting its length. If the module set a special flag for an extremely long sequence, the aggregator can interpret that as a sign of a noteworthy pattern (or conversely, a pattern to down-weight if length was due to noise). One potential issue is that the aggregator might currently just take the reduced pattern’s score and not differentiate it from other patterns, which means context about the original sequence length (beyond what the reduction score conveys) could be underutilized. Hot Zones → Aggregator: The aggregator integrates the hot zone outputs by noting which data regions or elements are marked at higher hot zone levels. In practice, this means the aggregator will boost or flag any patterns or numbers that fall within a Level 2 or Level 3 hot zone, for example. If a pattern from the Pattern Extractor corresponds to numbers or positions that are in a hot zone, the aggregator should recognize that overlap and elevate the importance of that pattern. Currently, the aggregator uses the hot zone level as an additional factor – for instance, it might add extra points to the pattern’s score or mark it for priority if it occurs in a hot zone. All hot zone classifications are passed into the aggregator, but there is a question of how explicitly the aggregator cross-references them with specific patterns. It’s possible that the aggregator is, at this stage, simply keeping the hot zone information alongside patterns without a full cross-link (meaning the data is preserved, but the logic to exploit it fully may be minimal). No hot zone data is dropped, but it may be used in a basic way (e.g. a global rule like “prefer items from higher hot zones”) rather than a fine-grained integration.
Gaps or Redundancies in Current Integration
Underutilization of V-Trac Data: The Pattern Extractor provides a V-Trac representation for each pattern, but the aggregator does not have dedicated logic to use this data. This means the insight that two different patterns share a similar V-Trac form (indicating a structural similarity) could be missed. The V-Trac information is carried along but effectively dormant, representing a lost opportunity to identify patterns that are conceptually the same even if composed of different raw values.
Location Context Lost: While pattern locations are identified, the aggregator does not currently leverage the position context of patterns. For example, knowing a pattern always appears at the end of a sequence versus scattered throughout could be insightful, but the aggregator treats all occurrences equally. This is a gap because certain strategies (like combination forming or predictions) might benefit from positional trends, which aren’t being considered now.
Long Sequence Context Overlooked: The Long String Reduction module flags very long repeating sequences (which might indicate either a strong recurring pattern or just a quirk of the data). If the aggregator only uses the reduced pattern and its basic score, it might not distinguish between a pattern that was naturally short and one that was part of a huge repetitive string. Thus, any special treatment intended for long sequences (either to down-weight overemphasized patterns or to highlight an exceptionally persistent pattern) could be lost. This suggests a gap in how the aggregator differentiates patterns coming from a long sequence context.
Overlap of Hot Zones and Patterns: There is potential redundancy in scoring when a pattern falls in a hot zone. If the aggregator isn’t carefully cross-referencing, a pattern in a hot zone might receive two separate boosts: one from its own pattern score and one from the hot zone level, without the system recognizing it's essentially the same signal reinforced. Conversely, if the aggregator is too simplistic, it might not boost at all, effectively ignoring the double significance of that pattern. This overlap needs explicit synergy logic; otherwise, the integration either double-counts the importance or fails to capitalize on the fact that multiple indicators point to the same item.
Lack of Explicit Synergy Detection: More broadly, the current aggregator design doesn’t have a mechanism to flag when multiple modules concur on an insight (apart from the Pattern Extractor’s internal consensus flag). For instance, if the Hot Zones module and Pattern Extractor both highlight the same number or segment independently, there isn’t a dedicated “cross-module consensus” flag being set. The absence of this kind of synchronization is a gap – the system relies on implicit effects (like combined scoring) rather than explicitly noting “this is a strong signal because multiple analyses agree.”
Data Redundancy Without Consolidation: Some information is effectively duplicated across module outputs (e.g. a certain trend might manifest as both a pattern finding and a hot zone indicator). Without a consolidation step, the aggregator might be handling these pieces separately, which can clutter the analysis or cause one signal to overshadow another if they aren’t normalized. This redundancy indicates a need for more cohesive combination logic to merge related signals into one.

Another approach is to introduce a small intermediate module dedicated to feature consolidation and synergy detection. IMPORTANT*** - WHAT CAUSED THIS? WAS IT ALREADY PRESENT PRIOR TO WHAT SEEMED LIKE ADDITIONAL ANALYSIS MODULES?



Below is a more detailed explanation of how straight vs. boxed patterns (and their permutations), as well as multi-length (3+ digit) patterns and V-Trac analysis, can be integrated into the analysis/aggregator stage—especially if they aren’t already fully leveraged by the earlier modules. I’ll clarify why some of your concern arises (“Why does it seem like the aggregator might lose track of whether a pattern is straight/boxed?”) and how to fix it.

1. Straight vs. Boxed Pattern Recognition
What Should Be Happening
In your Pattern Extractor or Hot Zones module, you may already be tagging patterns as “straight” or “box”:

Straight: The exact digit sequence, e.g. 671 reappears as 671, 671, 671 (in the same order).

Box: A 3-digit set whose digits appear in some rotation, e.g. (761, 167, 617, 176, 671, 716)—all permutations of the same 3-digit set.

If your modules are indeed capturing whether a pattern reappeared “exactly in the same order” (straight) vs. “in any permutation” (boxed), they likely store that in the results—perhaps as a flag:

python
Copy
{
  "pattern": "671",
  "pattern_type": "straight"  # or "boxed"
  "score_breakdown": { ... },
  ...
}
or they might store a canonical pattern with a cyclical-rotation approach to unify “671” and “716” as effectively the same box pattern. In any case, the information about how that pattern was found (straight, box, cyclical shift) presumably already exists in the results.

Where the Concern Arises
If your Analysis/Aggregator module is only picking up a single canonical key (e.g., "pattern_key": "167" as the sorted or minimal rotation) but not storing whether the pattern repeated exactly (straight) or by rotation (box), it may seem like the aggregator is “losing track” of that detail.

The aggregator sees pattern_key="167" for everything that’s a rotation of 167/176/617/671/716/761.

If the aggregator lumps them all together, it might not differentiate which times it was a perfect straight repeat vs. a box (rotation-based) repeat.

Result: You can’t apply different synergy or scoring rules for “straight” patterns vs. “boxed” patterns if the aggregator lumps them into the same group without storing how each occurrence was recognized.

How to Fix It
Option 1: In each earlier module (Pattern Extractor, Hot Zones, etc.), attach an extra field:

python
Copy
{
  "pattern": "671",
  "is_straight": True,         # True if we found the exact same digit order repeating
  "is_boxed": False,           # or True if it’s recognized only by rotation
  ...
}
Then, in the aggregator, merge all occurrences but keep track of how often is_straight=True, how often is_boxed=True.

For example, if you see 3 separate occurrences in different columns that each had "is_straight": True, you can reward additional synergy points for a “triple straight repeat.” If you see 5 total occurrences, 2 are straight and 3 are box, you can do a mix.

Option 2: The aggregator itself tries to parse out from the pattern’s location or rotation logic. This is trickier because you’d have to re-check the strings. Usually it’s easiest if the modules that discovered the pattern in a straight or box fashion simply record that fact as a flag (like above) to pass to the aggregator.

2. Multi-Length Patterns (3+ Digits)
Why It’s Potentially Overlooked
If your Digit Reduction or Long String module extracts a 4-digit or 5-digit cluster (e.g., “6811” or “44188”), the aggregator might currently treat them in the same “3-digit pattern” style. So you lose the nuance that it’s a 4+ digit pattern that might break down into multiple 3-digit subsets or might be significant as a longer pattern itself.

Solutions
Preserve the length (and the fact that it’s 4, 5, or 6 digits) in each module’s result. E.g.:

python
Copy
{
  "pattern": "6811",
  "pattern_length": 4,
  "is_boxed": True,
  ...
}
Score weighting: You might add synergy for a pattern discovered with length ≥4 if your earlier analysis indicates it’s stable (persisted across columns). The aggregator sees "pattern_length":4 and can do +1 synergy for each digit above 3.

If the aggregator sees a 4-digit pattern repeatedly appear in the same exact order (like “6811, 6811, 6811”), you can treat that as an even stronger straight repeat than a 3-digit pattern. Conversely, if you prefer to reduce 4-digit patterns to sub-3-digit combos, you can do that before aggregator, but that’s a design choice.

3. Concerns About Incomplete V-Trac Analysis
The Gap
Your modules produce a V-Trac representation or mention “this pattern is a strong V-Trac match.” However, the aggregator might not do much with that. If the aggregator doesn’t have a step like “check how many patterns share the same V-Trac form,” or “did the module say vtrac_straight_match=True?” then the system’s final synergy logic may ignore it.

Example
Your Pattern Extractor might say:

python
Copy
"pattern": "418",
"vtrac": "963", 
"vtrac_straight": True  # indicates it exactly matched the winning V-Trac structure
But if your aggregator only does:

python
Copy
aggregated_score = pattern_extractor_score + synergy
and never reads "vtrac_straight": True, you lose the extra synergy that a V-Trac straight might deserve.

Making Sure V-Trac Is Used
Option 1: In each pattern’s data, store a flag like "vtrac_straight": True or "vtrac_boxed": True. Then the aggregator checks:

python
Copy
if any(x["vtrac_straight"]==True for x in patterns_list):

Below is a fully-coded example of how you can implement a Comprehensive Analysis Module (“Aggregator”) that merges the outputs from your three previously developed modules—Pattern Extractor, Long String Reduction, and Hot Zones—without discarding or overriding any of the strong analytical logic you have built so far.

This design assumes your existing modules each return structured data in dictionaries/lists (e.g., pattern_extractor_results, long_string_results_area1, hotzones_results, etc.).

We incorporate all relevant features (straight vs. box patterns, 3-value pattern flags, V-Trac info, hot zone levels, stable cluster length, etc.) from those modules and unify them in the aggregator.

The aggregator performs final synergy checks (like consensus, cross-module reinforcement, repeated patterns in hot zones, double triggers, V-Trac alignment) while maintaining the raw data each module provided—so you don’t lose detail.

The aggregator then outputs a final, comprehensive structure that you can feed into future modules (Machine Learning, Combination Forming, or Profit/ROI analysis).

"""

Comprehensive Analysis (Aggregator) that merges outputs from:
 - pattern_extractor.py
 - long_string_reduction.py (area1, area2)
 - hotzones_analysis.py

import datetime
from collections import defaultdict

# 1. Example Mirror Map & Helper (If needed)

    '0':'5','1':'6','2':'7','3':'8','4':'9',
    '5':'0','6':'1','7':'2','8':'3','9':'4'

def canonical_boxed_form(digits_str: str) -> str:
    """
    Returns a 'boxed' canonical form (sorted digits).
    If you prefer a rotation-based canonical for 3-digit sequences, adapt as needed.
    E.g. '461' -> '146'
    """
    return ''.join(sorted(digits_str))

#############################################
# 2. Convert Pattern Extractor Output -> universal
#############################################

def convert_pattern_extractor_results(extractor_data):
    """
    Example format from pattern_extractor:
    {
      "patternA": { 
          "canonical": "461",
          "vtrac": "916", 
          "score": 12,
          "score_breakdown": {...},
          "straight_flag": True,
          "box_flag": False,
          "hot_zone": False,
          "some_feature": ...
      },
      "patternB": {...}
    }
    
    We convert each pattern to a universal list item with a "pattern_key" and all features.
    """
    results_list = []

    for pkey, pinfo in extractor_data.items():
        # For safety, you might gather all relevant fields:
        actual_digits = pinfo.get("canonical", pkey)
        # Boxed representation if the module has it or we can do sorted(digits)
        box_form = canonical_boxed_form(actual_digits)
        
        entry = {
            "pattern_key": pkey,  # or box_form, depends on your aggregator logic
            "actual_digits": actual_digits,
            "vtrac": pinfo.get("vtrac",""),
            "source_module": "PatternExtractor",
            "module_score": pinfo.get("score", 0),
            
            # Pass forward any flags
            "straight_flag": pinfo.get("straight_flag", False),
            "box_flag": pinfo.get("box_flag", False),
            "three_value_flag": pinfo.get("three_value_flag", False),
            "double_flag": pinfo.get("double_flag", False),
            "triple_flag": pinfo.get("triple_flag", False),
            "hot_zone_flag": pinfo.get("hot_zone", False),
            
            # Possibly the entire breakdown or partial
            "score_breakdown": pinfo.get("score_breakdown", {}),
            
            # If there's e.g. location data, we can store it:
            "locations": pinfo.get("locations", []),
            
            # etc.  Add more if needed
        }
        results_list.append(entry)
    
    return results_list

#############################################
# 3. Convert Long String Output -> universal

def convert_longstring_results(longstring_data, area_label="LongStringArea1"):
    """
    Suppose long_string_reduction.py returns:
    {
      "module_name": "LongStringReduction_Area1",
      "analysis_results": [
        {
           "cell_id": "...",
           "metadata": {...}, 
           "notable_patterns": [
              {"pattern": "461", "score": 5, "double_flag": True, ...},
              {"pattern": "944", "score": 2, ...}
           ]
        }, ...
      ]
    }
    We'll create universal entries from 'notable_patterns'.
    """
    out = []
    analysis_results = longstring_data.get("analysis_results", [])
    for cell_info in analysis_results:
        cell_id = cell_info.get("cell_id","")
        meta = cell_info.get("metadata", {})
        
        for npat in cell_info.get("notable_patterns", []):
            digits_str = npat["pattern"]
            box_form = canonical_boxed_form(digits_str)
            
            entry = {
                "pattern_key": box_form,  # or keep the raw pattern as the aggregator key
                "actual_digits": digits_str,
                "source_module": area_label,
                "module_score": npat.get("score", 0),
                
                # flags from long string
                "double_flag": npat.get("double_flag", False),
                "triple_flag": npat.get("triple_flag", False),
                "three_value_flag": npat.get("three_value_flag", False),
                "locations": [cell_id],
                
                # Additional metadata
                "cell_meta_section": meta.get("section"),
                "cell_meta_set": meta.get("set"),
                "cell_meta_draw": meta.get("draw"),
                
                # If you had V-Trac, store it:
                "vtrac": npat.get("vtrac_form", ""),  
                
                # Could store any digit shifting flags, method A/B/C used, etc.
                "method_a_used": npat.get("method_a_used", False),
                # ...
            }
            out.append(entry)
    
    return out

def convert_hotzones_results(hotzones_data):
    """
    Suppose hotzones_analysis.py returns:
    {
      "hotzone_patterns": [
        {
          "pattern": "461",
          "local_hotzone_score": 10,
          "section": "Midday",
          "double_flag": False,
          "vtrac_strong": True,
          ...
        },
        ...
      ]

    out = []
    patterns = hotzones_data.get("hotzone_patterns", [])
    for p in patterns:
        digits_str = p["pattern"]
        box_form = canonical_boxed_form(digits_str)
        
        entry = {
            "pattern_key": box_form,
            "actual_digits": digits_str,
            "source_module": "HotZones",
            "module_score": p.get("local_hotzone_score", 0),
            "section": p.get("section"),
            "draw": p.get("draw"),
            "double_flag": p.get("double_flag", False),
            "triple_flag": p.get("triple_flag", False),
            "three_value_flag": p.get("three_value_flag", False),
            "vtrac": p.get("vtrac_form",""),
            "hot_zone_flag": True,  # obviously
            # If you had star-level hot zone (like 1 or 2 stars),
            # store it too:
            "hot_zone_level": p.get("hot_zone_level", 0),
        }
        out.append(entry)
    return out

#############################################
# 5. The Actual Aggregator
#############################################

def comprehensive_analysis_aggregator(
    pattern_extractor_data, 
    longstring_data_area1,
    longstring_data_area2,
    hotzones_data
):
    """
    Aggregates all patterns from the three modules (plus area2 from longstring),
    computes synergy, and returns final sorted list.

    If you have an existing synergy scoring approach, adapt it here.
    """
    # 1) Convert each to universal
    ext_list = convert_pattern_extractor_results(pattern_extractor_data)
    ls1_list = convert_longstring_results(longstring_data_area1, "LongStringArea1")
    ls2_list = convert_longstring_results(longstring_data_area2, "LongStringArea2")
    hz_list = convert_hotzones_results(hotzones_data)
    
    combined = ext_list + ls1_list + ls2_list + hz_list
    
    # 2) Group by pattern_key (box form or canonical)
    aggregator_dict = defaultdict(lambda: {
        "patterns": [],        # list of occurrences from all modules
        "aggregated_score": 0, 
        "modules_used": set(),
        "sections_used": set(),
        "three_value_flag": False, 
        "double_flag": False,
        "triple_flag": False,
        "hot_zone_flag": False,
        "straight_flag_count": 0,
        "box_flag_count": 0,
        "vtrac_forms": set()
    })
    
    for item in combined:
        pk = item["pattern_key"]
        aggregator_dict[pk]["patterns"].append(item)
        
        # track distinct modules
        aggregator_dict[pk]["modules_used"].add(item["source_module"])
        
        # track sections (midday, evening, combined) if present
        section_val = item.get("section")
        if section_val:
            aggregator_dict[pk]["sections_used"].add(section_val)
        
        # gather flags
        if item.get("three_value_flag"):
            aggregator_dict[pk]["three_value_flag"] = True
        if item.get("double_flag"):
            aggregator_dict[pk]["double_flag"] = True
        if item.get("triple_flag"):
            aggregator_dict[pk]["triple_flag"] = True
        if item.get("hot_zone_flag"):
            aggregator_dict[pk]["hot_zone_flag"] = True
        if item.get("straight_flag"):
            aggregator_dict[pk]["straight_flag_count"] += 1
        if item.get("box_flag"):
            aggregator_dict[pk]["box_flag_count"] += 1
        
        vtrac_str = item.get("vtrac","")
        if vtrac_str:
            aggregator_dict[pk]["vtrac_forms"].add(vtrac_str)
    
    # 3) synergy scoring
    # You can define synergy weights or logic. 
    # e.g. multiple modules => +5, hot zone => +3, double => +2, etc.
    
    for pk, block in aggregator_dict.items():
        base_score = 0
        
        # sum up raw module scores
        base_score += sum(p.get("module_score",0) for p in block["patterns"])
        
        # synergy: multiple modules
        if len(block["modules_used"]) > 1:
            base_score += 5
        
        # synergy: multiple sections (midday/evening/combined)
        if len(block["sections_used"]) > 1:
            base_score += 4
        
        # synergy: if hot zone
        if block["hot_zone_flag"]:
            base_score += 3
        
        # synergy: if it's a double or triple
        if block["double_flag"]:
            base_score += 2
        if block["triple_flag"]:
            base_score += 3
        
        # synergy: if three_value_flag
        if block["three_value_flag"]:
            base_score += 1
        
        # synergy: if multiple vtrac forms found or if vtrac strong
        # (some logic to detect if the pattern was recognized in a mirrored or
        #  vtrac-transformed way across modules)
        if len(block["vtrac_forms"]) > 1:
            base_score += 2
        
        # synergy: if a pattern had at least 1 straight_flag
        if block["straight_flag_count"] > 0:
            base_score += block["straight_flag_count"]
        
        # synergy: box_flag_count
        if block["box_flag_count"] > 0:
            base_score += block["box_flag_count"] * 0.5
        
        block["aggregated_score"] = base_score
    
    # 4) produce final sorted list
    final_list = []
    for pk, binfo in aggregator_dict.items():
        row = {
            "pattern_key": pk,  # the canonical/box form
            "aggregated_score": binfo["aggregated_score"],
            "modules_used": list(binfo["modules_used"]),
            "sections_used": list(binfo["sections_used"]),
            "patterns_detail": binfo["patterns"],  # raw item details from each module
            "three_value_flag": binfo["three_value_flag"],
            "double_flag": binfo["double_flag"],
            "triple_flag": binfo["triple_flag"],
            "straight_flag_count": binfo["straight_flag_count"],
            "box_flag_count": binfo["box_flag_count"],
            "vtrac_forms": list(binfo["vtrac_forms"]),
        }
        final_list.append(row)
    
    final_list.sort(key=lambda x: x["aggregated_score"], reverse=True)
    
    return {
        "analysis_timestamp": datetime.datetime.now().isoformat(),
        "aggregated_patterns": final_list
    }

3. Key Features & Rationales
Preserves Straight vs. Box Info

We pass a straight_flag or box_flag from the original modules. The aggregator sums them up or counts them so we can see how often a pattern was recognized as a straight.

Retains 3-Value (Double, Triple) Flags

If a pattern is recognized as “two identical digits + 1 different,” we store double_flag=True. This aggregator merges if multiple modules flagged that. We add synergy if it’s recognized in more than one place.

V-Trac

Each pattern’s vtrac field is included. If the aggregator sees multiple distinct V-Trac versions for the same canonical box form (meaning in one module it was recognized with vtrac=916, in another maybe vtrac=461, etc.), that can add synergy. Hot Zone

We preserve a hot_zone_flag or even hot_zone_level. If any module item has it set, the aggregator merges them and knows that pattern is “hot.” We then add synergy or a bonus to reflect that.

Cross-Module Synergy

We specifically detect if the pattern came from multiple modules by tracking modules_used. If it is in 2 or 3 modules, we add synergy. This is the “double trigger” or “reinforced pattern” scenario.

Cross-Section

If the pattern is found in Midday, Evening, or Combined data, we track them in sections_used. If it appears in more than one (like midday+evening synergy), we add synergy.

Complete Data

We never discard the original fields. The aggregator output keeps a patterns_detail array, so you can see all sub-entries from the prior modules. That means you don’t lose data like location columns, cluster length, or method used.

By structuring it this way, your Analysis Module becomes a single, final aggregator that does not override or conflict with the strong logic in your prior modules. Instead, it unifies them, applies synergy where patterns overlap, and retains everything so your upcoming steps (Machine Learning, Combination Forming, Profit analysis) can use the full wealth of data.

from collections import defaultdict

def canonical_boxed_form(digits):
    """Sort digits to get a 'boxed' canonical key for grouping."""
    return ''.join(sorted(digits))

def convert_pattern_extractor_results(pattern_extractor_data):
    """
    We expect something like:
    {
       "461": {
          "vtrac": "916",
          "score": 14,
          "straight_flag": True,
          "double_flag": False,
          "consensus_flag": False,
          ...
       },
       ...
    }
    We'll produce a list of dicts with universal fields.
    """
    out = []
    for pat_key, pat_info in pattern_extractor_data.items():
        dig = pat_info.get("canonical", pat_key)
        entry = {
            "pattern_key": canonical_boxed_form(dig),
            "actual_digits": dig,
            "source_module": "PatternExtractor",
            "module_score": pat_info.get("score", 0),
            "vtrac": pat_info.get("vtrac",""),
            # advanced flags
            "straight_flag": pat_info.get("straight_flag", False),
            "box_flag": pat_info.get("box_flag", False),
            "double_flag": pat_info.get("double_flag", False),
            "triple_flag": pat_info.get("triple_flag", False),
            "three_value_flag": pat_info.get("three_value_flag", False),
            "consensus_flag": pat_info.get("consensus_flag", False),
            "vtrac_strong_match": pat_info.get("vtrac_strong_match", False),
            "pattern_length": len(dig),
            # store entire sub-score if you want
            "score_breakdown": pat_info.get("score_breakdown", {}),
        }
        out.append(entry)
    return out

def convert_longstring_results(longstring_dict, label="LongStringArea1"):

                "straight_flag": pinfo.get("straight_flag",False),

def comprehensive_analysis_aggregator(
    pattern_extractor_data,
    longstring_area1_data,
    longstring_area2_data,
    hotzones_data
):
    """
    The aggregator merges everything, assigns synergy scores.
    We won't hardcode synergy weights here, but show examples inline.
    """
    # 1) Convert each to a universal list
    ext_list = convert_pattern_extractor_results(pattern_extractor_data)
    ls1_list = convert_longstring_results(longstring_area1_data, "LongStringArea1")
    ls2_list = convert_longstring_results(longstring_area2_data, "LongStringArea2")
    hz_list = convert_hotzones_results(hotzones_data)

    combined = ext_list + ls1_list + ls2_list + hz_list

    # 2) Group by pattern_key
    aggregator = defaultdict(lambda: {
        "patterns": [],
        "aggregated_score": 0,
        "modules_encountered": set(),
        "sections_encountered": set(),
        "sets_encountered": set(),
        # track synergy counters
        "straight_count": 0,
        "box_count": 0,
        "hotzone_count": 0,
        "any_double": False,
        "any_triple": False,
        "any_three_value": False,
        "vtrac_forms": set(),
        "any_vtrac_strong": False,
        "any_consensus": False,
        "max_len": 3
    })

    for item in combined:
        pk = item["pattern_key"]
        aggregator[pk]["patterns"].append(item)
        aggregator[pk]["modules_encountered"].add(item["source_module"])
        st = item.get("set")
        if st: aggregator[pk]["sets_encountered"].add(st)
        sec = item.get("section")
        if sec: aggregator[pk]["sections_encountered"].add(sec)

        # gather flags
        if item.get("straight_flag"):
            aggregator[pk]["straight_count"] += 1
        if item.get("box_flag"):
            aggregator[pk]["box_count"] += 1
        if item.get("hot_zone_flag"):
            aggregator[pk]["hotzone_count"] += 1
        if item.get("double_flag"):
            aggregator[pk]["any_double"] = True
        if item.get("triple_flag"):
            aggregator[pk]["any_triple"] = True
        if item.get("three_value_flag"):
            aggregator[pk]["any_three_value"] = True
        if item.get("consensus_flag"):
            aggregator[pk]["any_consensus"] = True
        if item.get("vtrac_strong_match"):
            aggregator[pk]["any_vtrac_strong"] = True

        # track vtrac
        vform = item.get("vtrac","")
        if vform:
            aggregator[pk]["vtrac_forms"].add(vform)

        # track max pattern length
        plen = item.get("pattern_length", len(item["actual_digits"]))
        if plen > aggregator[pk]["max_len"]:
            aggregator[pk]["max_len"] = plen

    # 3) synergy scoring
    for pk, block in aggregator.items():
        base_score = 0
        # sum raw module scores
        patterns_arr = block["patterns"]
        base_score += sum(x.get("module_score",0) for x in patterns_arr)

        # multiple modules => +5
        if len(block["modules_encountered"])>1:
            base_score += 5
        # multiple sets => +3
        if len(block["sets_encountered"])>1:
            base_score += 3
        # multiple sections => +4
        if len(block["sections_encountered"])>1:
            base_score += 4

        # if hotzone_count>0 => +3, if >1 => extra bonus
        if block["hotzone_count"]>0:
            base_score += 3
            if block["hotzone_count"]>1:
                base_score += 2  # repeating hotzone synergy

        # if straight_count>0 => +2 per occurrence
        if block["straight_count"]>0:
            base_score += 2*block["straight_count"]

        # box_count => +1 per occurrence
        base_score += block["box_count"]

        # double/triple synergy
        if block["any_double"]:
            base_score += 2
        if block["any_triple"]:
            base_score += 3

        # 3-value => +1
        if block["any_three_value"]:
            base_score += 1

        # consensus => +5
        if block["any_consensus"]:
            base_score += 5

        # vtrac_strong => +3
        if block["any_vtrac_strong"]:
            base_score += 3

        # if multiple vtrac forms => +2
        if len(block["vtrac_forms"])>1:
            base_score += 2

        # pattern length synergy
        if block["max_len"]>3:
            base_score += (block["max_len"] - 3)

        block["aggregated_score"] = base_score

    # 4) produce final sorted list
    final_list = []
    for pk, info in aggregator.items():
        final_list.append({
            "pattern_key": pk,
            "aggregated_score": info["aggregated_score"],
            "modules_encountered": list(info["modules_encountered"]),
            "sections_encountered": list(info["sections_encountered"]),
            "sets_encountered": list(info["sets_encountered"]),
            "straight_count": info["straight_count"],
            "box_count": info["box_count"],
            "hotzone_count": info["hotzone_count"],
            "any_double": info["any_double"],
            "any_triple": info["any_triple"],
            "any_three_value": info["any_three_value"],
            "any_consensus": info["any_consensus"],
            "any_vtrac_strong": info["any_vtrac_strong"],
            "vtrac_forms": list(info["vtrac_forms"]),
            "max_pattern_length": info["max_len"],
            "patterns_detail": info["patterns"]  # raw sub-entries
        })

    final_list.sort(key=lambda x: x["aggregated_score"], reverse=True)

    return {
        "analysis_timestamp": datetime.datetime.now().isoformat(),
        "aggregated_patterns": final_list
    }
In the synergy logic, we used example weights. Tweak them to your preference.

def aggregator_main(all_patterns):
    """
    all_patterns: a list of dicts, each dict is something like:
      {
        "pattern_key": "117", 
        "actual_digits": "117",
        "vtrac_straight": True or False,
        "three_value_flag": True,
        "section": "Midday",
        "module_score": 8,
        "source_module": "HotZones",
        ...
      }
    We'll group them by pattern_key, then compute synergy.
    """

    from collections import defaultdict

    aggregator = defaultdict(lambda: {
        "patterns": [],
        "sections_encountered": set(),
        "modules_used": set()
    })

    # 1) Collect
    for item in all_patterns:
        pk = item["pattern_key"]
        aggregator[pk]["patterns"].append(item)
        if "section" in item:
            aggregator[pk]["sections_encountered"].add(item["section"])
        aggregator[pk]["modules_used"].add(item["source_module"])

    final_list = []
    for pk, block in aggregator.items():
        synergy_score = 0

        # sum local
        synergy_score += sum(x.get("module_score", 0) for x in block["patterns"])

        # cross-section synergy
        if len(block["sections_encountered"]) > 1:
            synergy_score += 5

        # maybe a separate synergy if exactly "Midday" & "Evening"
        if "Midday" in block["sections_encountered"] and "Evening" in block["sections_encountered"]:
            synergy_score += 3

        # vtrac synergy
        #   if any item had vtrac_straight => +3
        if any(x.get("vtrac_straight",False) for x in block["patterns"]):
            synergy_score += 3
        #   if any item had vtrac_boxed => +2
        if any(x.get("vtrac_boxed",False) for x in block["patterns"]):
            synergy_score += 2

        # repeating 3-digit synergy
        if any(x.get("three_digit_repeat_flag",False) for x in block["patterns"]):
            synergy_score += 4

        # multi-module synergy
        if len(block["modules_used"]) > 1:
            synergy_score += 5

        final_list.append({
            "pattern_key": pk,
            "aggregated_score": synergy_score,
            "patterns": block["patterns"],  # raw details
            "sections_used": list(block["sections_encountered"]),
            "modules_used": list(block["modules_used"])
        })

    # sort
    final_list.sort(key=lambda x: x["aggregated_score"], reverse=True)
    return final_list
