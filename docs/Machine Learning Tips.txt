def predictive_model(historical_data, current_features):
    try:
        # Prepare data
        features = historical_data.drop(['Date', 'Result', 'Top 15 Combinations'], axis=1, errors='ignore')
        target = historical_data['Result']
        current_features = current_features.drop(['Date', 'Result', 'Top 15 Combinations'], axis=1, errors='ignore')
    
        # Align features
        missing_cols = set(features.columns) - set(current_features.columns)
        for col in missing_cols:
            current_features[col] = 0
        
        # Train model
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(features, target)
        
        # Predict
        predictions = model.predict(current_features)
        return predictions
    except Exception as e:
        print(f"Error in predictive model: {e}")
        return None

def create_feature_matrix(patterns, scores):
    """
    Creates a feature matrix for machine learning models.

    Parameters:
    - patterns: A list of patterns or combinations.
    - scores: A list or dictionary of scores corresponding to the patterns.

    Returns:
    - X: A DataFrame containing feature vectors.
    """
    import pandas as pd
    feature_list = []
    for pattern in patterns:
        features = {
            'pattern': ''.join(map(str, pattern)),
            'digit_sum': sum(pattern),
            'has_double_digit': int(len(set(pattern)) < len(pattern)),
            'score': scores.get(tuple(pattern), 0)
            # Add more features as needed
        }
        feature_list.append(features)
    X = pd.DataFrame(feature_list)
    return X

def train_random_forest(X, y):
    """
    Trains a Random Forest classifier.

    Parameters:
    - X: Feature matrix.
    - y: Target labels.

    Returns:
    - model: Trained Random Forest model.
    """
    from sklearn.ensemble import RandomForestClassifier
    model = Random ForestClassifier(n_estimators=100, random_state=42)
    model.fit(X, y)
    return model

def train_xgboost(X, y):
    """
    Trains an XGBoost classifier.

    Parameters:
    - X: Feature matrix.
    - y: Target labels.

    Returns:
    - model: Trained XGBoost model.
    """
    import xgboost as xgb
    model = xgb.XGBClassifier(use_label_encoder=False, eval_metric='logloss')
    model.fit(X, y)
    return model

def make_predictions(model, X_new):
    """
    Makes predictions using the trained model.

    Parameters:
    - model: Trained machine learning model.
    - X_new: Feature matrix for new data.

    Returns:
    - predictions: Array of predicted probabilities or classes.

    predictions = model.predict_proba(X_new)[:, 1]
    return predictions

def evaluate_model(model, X_test, y_test):
    """
    Evaluates the model's performance.

    Parameters:
    - model: Trained machine learning model.
    - X_test: Test feature matrix.
    - y_test: True labels for the test set.

    Returns:
    - metrics: Dictionary containing evaluation metrics.
    """
    from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
    y_pred = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred)
    }
    return metrics
