# Core helpers
mirror_map = {0:5, 5:0, 1:6, 6:1, 2:7, 7:2, 3:8, 8:3, 4:9, 9:4}

def to_class(d): return tuple(sorted((d, mirror_map[d])))  # e.g., 9 -> (4,9)
def classes_of_string(s): return {to_class(int(ch)) for ch in s}

def eliminate(s, T): 
    # remove every ch in T from string s (non-destructive)
    return ''.join(ch for ch in s if int(ch) not in T)

def allowed_eliminations(prev_draw_digits):
    E = set(prev_draw_digits)
    E |= {mirror_map[d] for d in prev_draw_digits}
    return E

# Scoring
def score_subset(S7,S6,S5, P, T):
    E_classes = {to_class(d) for d in allowed_eliminations(P)}
    cols = []
    for s in (S7,S6,S5):
        sp = eliminate(s, T)
        cset = classes_of_string(sp)
        k = len(cset)
        cols.append((sp, cset, k))
    count3 = sum(1 for (_,c,k) in cols if k == 3)
    dist   = sum(abs(k-3) for (_,_,k) in cols)
    len_sum = sum(len(sp) for (sp,_,_) in cols)
    overlap = sum(len(cset & E_classes) for (_,cset,_) in cols)
    csets_3 = [c for (_,c,k) in cols if k == 3]
    synergy = len(set.intersection(*csets_3)) if csets_3 else 0
    size_T = len(T)
    score = 40*count3 - 5*dist + 0.5*len_sum - 2*overlap + 10*synergy - 1*size_T
    return score, cols, synergy

mirror = {0:5,5:0,1:6,6:1,2:7,7:2,3:8,8:3,4:9,9:4}

def to_class(d): 
    a,b = d, mirror[d]
    return f"{min(a,b)}{max(a,b)}"  # '16','27','38','49','05'

def classes_of_string(s): 
    return {to_class(int(ch)) for ch in s}

def remove_digits(s, digits):
    D = set(digits)
    return ''.join(ch for ch in s if int(ch) not in D)

def repeat_score(s):
    # simple: +2 per pair of consecutive equal digits (e.g., '55' +2), +4 if triple
    score = 0; run = 1
    for i in range(1,len(s)):
        if s[i]==s[i-1]: run += 1
        else:
            if run==2: score += 2
            elif run>=3: score += 4
            run = 1
    if run==2: score += 2
    elif run>=3: score += 4
    return score

def evaluate_column(s, T, F):
    sp = remove_digits(s, set(T)|set(F))
    cset = classes_of_string(sp)
    k = len(cset)
    feats = dict(after=sp, classes=cset, k=k, length=len(sp), repeat=repeat_score(sp))
    return feats
