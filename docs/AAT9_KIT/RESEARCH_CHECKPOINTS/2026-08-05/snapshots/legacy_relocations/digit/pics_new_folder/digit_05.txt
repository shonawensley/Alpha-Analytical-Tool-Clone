from dataclasses import dataclass
from typing import Dict, List, Tuple

@dataclass(frozen=True)
class Cluster:
    s: str
    unique: str                 # sorted uniques, e.g., "168"
    span: Tuple[int, int]
    run_lengths: Dict[str, int] # max run per digit

def _longest_run(s: str, d: str) -> int:
    mx = cur = 0
    for ch in s:
        if ch == d: cur += 1; mx = max(mx, cur)
        else: cur = 0
    return mx

def extract_clusters(row_digits: str, min_len: int = 3, max_len: int = 12) -> List[Cluster]:
    n = len(row_digits); out: List[Cluster] = []
    for L in range(min_len, min(max_len, n) + 1):
        for i in range(0, n - L + 1):
            w = row_digits[i:i+L]
            uniq = ''.join(sorted(set(w)))
            if 1 < len(uniq) <= 3:
                rl = {d: _longest_run(w, d) for d in uniq}
                out.append(Cluster(s=w, unique=uniq, span=(i, i+L), run_lengths=rl))
    return out

from typing import Dict, Iterable, Set
from .clustering import Cluster

class VtracIndex:
    def __init__(self, exact: Set[str], family: Dict[str, Set[str]]):
        self.exact = exact
        self.family = family  # key: "abc" sorted family -> set of exact 3-grams

    def is_exact(self, gram: str) -> bool:
        return gram in self.exact

    def family_key(self, s: str) -> str:
        return ''.join(sorted(set(s)))

    def in_family(self, gram: str, fam_key: str) -> bool:
        return gram in self.family.get(fam_key, set())

    def cluster_has_membership(self, cl: Cluster) -> Dict[str, bool]:
        fk = cl.unique
        grams = {cl.s[i:i+3] for i in range(max(0, len(cl.s) - 2))}
        return {
            "has_exact": any(self.is_exact(g) for g in grams),
            "has_vtrac": any(self.in_family(g, fk) for g in grams),
        }

from math import tanh

def score_row(feats, w, caps, gates):
    z = 0.0
    # class hierarchy contributions are assumed pre-aggregated into feats
    z += w["w_box_family_density"] * min(feats.box_family_density, caps["density_max"])
    z += w["w_dup_bonus"] * min(feats.dup_bonus, caps["dup_bonus_max"])
    # echoes / consensus
    z += w["w_method_consensus"] * feats.method_consensus
    z += w["w_cols_hit_3"] if feats.cols_hit >= 3 else (w["w_cols_hit_2"] if feats.cols_hit >= 2 else 0)
    z += w["w_variants_hit_3"] if feats.variants_hit >= 3 else (w["w_variants_hit_2"] if feats.variants_hit >= 2 else 0)
    z += w["w_cluster_echo"] * feats.cluster_echo_count
    z += w["w_variant_echo"] * feats.variant_echo_count
    z += w["w_set_echo"] * feats.set_echo_count
    z += w["w_box_pair_agree"] * feats.box_pair_agree
    # drop quality
    z += (w["w_drop_run_len_2_3"] if feats.drop_run_len in (2,3) else
          w["w_drop_run_len_ge4"] if feats.drop_run_len >= 4 else 0)
    # recency
    z += w["w_set2_carryover"] * feats.recency_carryover
    # purity penalty
    z += w["w_residual_purity"] * feats.residual_purity
    # optional early multi-col drop_vtrac micro-bonus applied by caller when condition holds
    s = tanh(z / max(1e-6, gates.get("tanh_scale", 4.0)))
    lock = "lock" if s >= gates["early_lock"] else "unlock" if s <= gates["early_unlock"] else "undecided"
    return float(s), lock
