# scripts/tools/compound_top5.py
import sys, pandas as pd
from pathlib import Path

def show_top5(compound_csv: Path):
    df = pd.read_csv(compound_csv)
    for sec in ["Midday","Evening","Combined"]:
        sdf = df[df["section"] == sec].copy()
        if sdf.empty:
            continue
        sdf = sdf.sort_values(["compound_score","base_max_score"], ascending=[False, False]).head(5)
        print(f"\n== {sec} Top-5 ==")
        cols = ["Canonical","compound_score","base_max_score",
                "set_chain_depth","draw_chain_depth","col1_hits",
                "hot1_count","hot2_count","consensus_hits",
                "hidden3v_hits","vtrac_straight_hits","double_mirror_hits",
                "compound_why"]
        print(sdf[cols].to_string(index=False))

if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.exit("Usage: python scripts/tools/compound_top5.py data/outputs/analysis/patterns/<STATE>/<STATE>_stable_patterns_compound.csv")
    show_top5(Path(sys.argv[1]))

# scripts/checks/print_stable_header.py
import sys, pandas as pd
from pathlib import Path

def header(p): 
    df = pd.read_csv(p, nrows=0)
    print(f"\n{p.name} :: {len(df.columns)} columns")
    print(", ".join(df.columns))

if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.exit("Usage: python scripts/checks/print_stable_header.py data/outputs/analysis/patterns/<STATE>")
    root = Path(sys.argv[1])
    for stem in ["_stable_patterns_scores.csv","_stable_patterns_families.csv","_stable_patterns_compound.csv"]:
        for p in root.glob(f"*{stem}"):
            header(p)
