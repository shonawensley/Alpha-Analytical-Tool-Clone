import itertools

# Mirror mapping
MIRROR_MAP = {'0': '5', '1': '6', '2': '7', '3': '8', '4': '9', 
              '5': '0', '6': '1', '7': '2', '8': '3', '9': '4'}

# Function to apply VTrac mirror rule
def mirror_digit(digit):
    return MIRROR_MAP.get(digit, digit)

# Function to generate mirrored variations for a 3-digit pattern
def generate_mirrored_combos(pattern):
    mirrored = [
        pattern[:2] + mirror_digit(pattern[2]),  # Mirror 3rd digit
        mirror_digit(pattern[0]) + pattern[1:],  # Mirror 1st digit
        pattern[0] + mirror_digit(pattern[1]) + pattern[2]  # Mirror 2nd digit
    ]
    return mirrored

# Function to generate all R2-R8 permutations for a 3-digit set
def generate_permutations(patterns):
    all_permutations = set()
    for pattern in patterns:
        all_permutations.update(set(itertools.permutations(pattern, 3)))  # Generate 3-digit perms
    return sorted([''.join(p) for p in all_permutations])

# Example 3-digit pattern (Identified from R2-R8 breakdown)

# Generate mirrored combinations

# Generate final reduced combination set

def generate_vtrac_combos(pattern):
    vtrac_variations = set()
    for p in itertools.permutations(pattern, 3):
        vtrac_variations.add(''.join([mirror_digit(d) for d in p]))  # Mirror each digit
    return sorted(vtrac_variations)

def detect_consensus_digit(r2, r4, r6, r8):
    """
    Returns consensus digit if all four values match and are 1–2 digits long.
    """
    values = [r2, r4, r6, r8]
    if all(v == values[0] for v in values) and 1 <= len(values[0]) <= 2:
        return values[0]
    return None

from collections import Counter

def generate_consensus_doubles(consensus: str, key_digits: list, stable_patterns: list = None):
    """
    Generate 3-way double-based combinations using consensus digit (or digits)
    and 3 adjacent key digits from R2/R4/R6/R8 structure.

    Args:
        consensus (str): One or two digit consensus string (e.g., '8' or '66')
        key_digits (list): Adjacent support digits (up to 3) for pairing
        stable_patterns (list): Optional stable double-digit patterns to reinforce

    Returns:
        list: Unique consensus-based combinations
    """
    combinations = set()
    
    # Use only the last digit of consensus (e.g., '66' → '6', '88' → '8')
    if not consensus or len(consensus) == 0:
        return []
    
    trigger_digit = consensus[-1]

    for kd in key_digits:
        combinations.update({
            f"{trigger_digit}{trigger_digit}{kd}",
            f"{trigger_digit}{kd}{trigger_digit}",
            f"{kd}{trigger_digit}{trigger_digit}"
        })

    if stable_patterns:
        combinations.update(stable_patterns)

    return sorted(combinations)

    """

def analyze_vtrac_performance():
    """
    Analyzes historical V-Trac success rates to determine when transformations are useful.
    
    Returns:
        dict: Transformation success rates.
    """
    if not os.path.exists(VTRAC_HIT_LOG):
        return {"message": "No V-Trac performance data available."}

    vtrac_success = {}
    total_attempts = {}

    with open(VTRAC_HIT_LOG, mode='r') as file:
        reader = csv.DictReader(file)
        for row in reader:
            transformation = row["V-Trac Transformation"]
            was_successful = row["Hit Success"] == "True"

            if transformation not in vtrac_success:
                vtrac_success[transformation] = 0
                total_attempts[transformation] = 0

            total_attempts[transformation] += 1
            if was_successful:
                vtrac_success[transformation] += 1

    # Calculate success rate
    success_rates = {key: vtrac_success[key] / max(1, total_attempts[key]) for key in vtrac_success}

    return success_rates
✅ Why This Matters?
•	Helps identify which V-Trac transformations work best.
•	Prevents unnecessary V-Trac applications that lead to false predictions.

VTRAC_THRESHOLD = 0.6  # Only apply V-Trac if past success rate is above 60%

def should_apply_vtrac(original_number, vtrac_transformation):
    """
    Determines if a V-Trac transformation should be applied based on past success rates.
    
    Parameters:
        original_number (str): The original number.
        vtrac_transformation (str): The proposed V-Trac transformation.
    
    Returns:
        bool: True if V-Trac should be applied, False otherwise.
    """
    vtrac_success_rates = analyze_vtrac_performance()

    # Default to applying V-Trac if there is no historical data
    if vtrac_transformation not in vtrac_success_rates:
        return True  

    return vtrac_success_rates[vtrac_transformation] >= VTRAC_THRESHOLD
