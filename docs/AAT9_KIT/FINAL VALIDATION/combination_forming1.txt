# 2. Data structures

@dataclass
class CampaignResult:
    """
    One 'campaign' = one combo-pack played across draws until:
    - a hit occurs, or
    - we stop after max_rounds / max_draws.
    This is the atomic unit ML will learn from.
    """

    state: str
    date_started: str       # 'YYYY-MM-DD' (tables date or run date)
    pack_id: str            # some stable identifier for combo-pack
    pack_type: str          # e.g. vt_full_8, consensus_double_9, etc.
    play_mode: PlayMode
    combos_count: int       # how many combos in the pack
    max_draws_planned: int  # target timeframe window for this campaign
    draws_played: int       # how many draws actually played this pack

    # Result flags
    hit_type: Literal["none", "boxed", "straight", "vt_boxed", "vt_straight"]
    hit_draw_offset: Optional[int]  # 1-based draw index where hit occurred (if any)

    # Money
    total_expense: float
    total_payout: float

    # Evidence snapshot (for ML features)
    evidence_score: float          # combined pattern score
    vt_straight_hint: int          # 1 if vt-straight lane evidence, else 0
    consensus_double_hint: int     # 1 if consensus double pack, else 0
    lingering_pattern_hint: int    # 1 if flagged as lingering 3-value pattern, else 0

    @property
    def net_profit(self) -> float:
        return self.total_payout - self.total_expense

    @property
    def roi(self) -> float:
        if self.total_expense <= 0:
            return 0.0
        return self.net_profit / self.total_expense


# -----------------------------
# 3. Helper: theoretical profitability metric
# -----------------------------

def profitability_metric(
    combos_count: int,
    payout: float,
    max_draws_allowed: int,
    cost_per_combo: float = 1.0,
) -> float:
    """
    Your 'profitability metric' in closed form:
    payout - total_investment_for_timeframe. :contentReference[oaicite:35]{index=35}

    This does NOT guarantee real-world profit; it's a planning tool.
    """
    investment_per_draw = combos_count * cost_per_combo
    total_investment = investment_per_draw * max_draws_allowed
    return payout - total_investment


# -----------------------------
# 4. Engine: PerformanceTracker
# -----------------------------

class PerformanceTracker:
    """
    Stores all campaign results and provides roll-ups:
    - per play_mode
    - per pack_type
    - per state
    """

    def __init__(self) -> None:
        self._records: List[CampaignResult] = []

    # --- logging ---

    def log_campaign(self, result: CampaignResult) -> None:
        """Append a finished campaign result."""
        self._records.append(result)

    # --- export & summary ---

    def to_dataframe(self) -> pd.DataFrame:
        if not self._records:
            return pd.DataFrame()
        return pd.DataFrame([asdict(r) | {
            "net_profit": r.net_profit,
            "roi": r.roi,
        } for r in self._records])

    def summary_by_mode(self) -> pd.DataFrame:
        """
        Aggregate stats per play_mode.
        """
        df = self.to_dataframe()
        if df.empty:
            return df

        group = df.groupby("play_mode")
        summary = group.agg(
            campaigns=("pack_id", "nunique"),
            total_expense=("total_expense", "sum"),
            total_payout=("total_payout", "sum"),
            net_profit=("net_profit", "sum"),
            mean_roi=("roi", "mean"),
            hit_rate_any=("hit_type", lambda s: (s != "none").mean()),
            hit_rate_straight=("hit_type", lambda s: (s == "straight").mean()),
            avg_draws_to_hit=("hit_draw_offset", lambda s: s.dropna().mean()),
        )
        return summary.sort_values("mean_roi", ascending=False)

    def summary_by_pack_type(self) -> pd.DataFrame:
        """
        Aggregate stats per combination-pack type.
        """
        df = self.to_dataframe()
        if df.empty:
            return df

        group = df.groupby("pack_type")
        summary = group.agg(
            campaigns=("pack_id", "nunique"),
            total_expense=("total_expense", "sum"),
            total_payout=("total_payout", "sum"),
            net_profit=("net_profit", "sum"),
            mean_roi=("roi", "mean"),
            hit_rate_any=("hit_type", lambda s: (s != "none").mean()),
            hit_rate_straight=("hit_type", lambda s: (s == "straight").mean()),
            avg_evidence_score=("evidence_score", "mean"),
        )
        return summary.sort_values("mean_roi", ascending=False)

    def filter_profitable_strategies(
        self,
        min_roi: float = 0.0,
        max_avg_draws: Optional[int] = None,
        min_hit_rate: float = 0.0,
    ) -> pd.DataFrame:
        """
        Helper to find strategies that meet a target:
        - ROI threshold
        - timeframe threshold
        - hit rate threshold
        """
        by_mode = self.summary_by_mode()
        if by_mode.empty:
            return by_mode

        mask = by_mode["mean_roi"] >= min_roi
        if max_avg_draws is not None:
            mask &= (by_mode["avg_draws_to_hit"] <= max_avg_draws)
        mask &= (by_mode["hit_rate_any"] >= min_hit_rate)
        return by_mode[mask]


# -----------------------------
# 5. Helpers to compute expense/payout per campaign
# -----------------------------

def compute_flat_play_cost(
    combos_count: int,
    play_mode: PlayMode,
    draws_played: int,
    cost_per_combo: float = 1.0,
) -> float:
    """
    Simple (non-progressive) cost = combos * cost_per_combo * draws_played.
    """
    if play_mode in ("progressive_8_boxed", "progressive_8_straight"):
        raise ValueError("Use compute_progressive_cost for progressive modes.")
    return combos_count * cost_per_combo * draws_played


def compute_flat_play_payout(
    play_mode: PlayMode,
    hit_type: str,
    stake_multiplier: float = 1.0,
) -> float:
    """
    For flat play modes, only pay out if hit_type is compatible with the play_mode.
    Boxed play pays for boxed OR straight hits; straight play pays only on straight.
    """
    if play_mode in ("progressive_8_boxed", "progressive_8_straight"):
        raise ValueError("Use compute_progressive_payout for progressive modes.")

    base = 0.0
    if "boxed" in play_mode:
        if hit_type in ("boxed", "straight", "vt_boxed", "vt_straight"):
            base = PAYOUTS[play_mode]
    else:  # straight modes
        if hit_type in ("straight", "vt_straight"):
            base = PAYOUTS[play_mode]

    return base * stake_multiplier


def compute_progressive_cost_and_payout(
    play_mode: PlayMode,
    round_hit: Optional[int],
) -> tuple[float, float]:
    """
    Look up your pre-defined progressive plan and compute:
    - total cost up to (and including) the final round played
    - payout if hit occured at round_hit, else 0

    round_hit: 1-based index of round where hit occurred, or None if no hit.
    """
    if play_mode not in PROGRESSIVE_PLANS:
        raise ValueError(f"No progressive plan defined for {play_mode}")

    plan = PROGRESSIVE_PLANS[play_mode]
    max_round = len(plan)

    if round_hit is None:
        total_cost = sum(r["expense"] for r in plan)
        return total_cost, 0.0

    # Clamp round_hit to [1, max_round]
    round_hit = max(1, min(round_hit, max_round))

    total_cost = sum(r["expense"] for r in plan[:round_hit])
    payout = plan[round_hit - 1]["payout"]
    return total_cost, payout

    pack_id: str         # "A"|"B"|"C"|"D"
    score: float         # evidence score from Aggregator
    roi_est: float       # expected ROI from validation (e.g. 0.25 = 25%)
    hit_prob_cycle: float  # probability of at least one hit within progressive_steps
    ir_est: float        # “information ratio” style: roi / stdev


The Aggregator will produce a list of these PackSignal entries for that state/date, one per lane that qualifies.

4. Budget rules (Part 2 logic)

Inputs to the budgeter:

bankroll – your total bankroll (moving target; can be updated),

daily_budget – what you’re willing to risk per cycle/day,

risk_cap_pct – max % of bankroll you are willing to lose on a single day,

pack_signals – list of PackSignal from aggregator (already filtered to “interesting” lanes).

Core rules (you can tweak thresholds later):

Risk cap:

max_spend_today = min(daily_budget, bankroll * risk_cap_pct)

Eligibility:

Only consider packs with roi_est > 0 and hit_prob_cycle above some minimum (e.g. ≥ 0.4 for A, ≥ 0.25 for C). 

FRAMING PROFIT AND HOT ZONES 2

Priority weight:

Each pack gets a weight like w = roi_est * ir_est * score.

Normalize weights → per‑pack spend

spend_i = max_spend_today * (w_i / sum(w))

Translate spend → stake per combo & progression

For pack P: combos = P.combos_per_ticket

If you allow P.progressive_steps draws before stop, you can compute base stake so that the total cost over those steps ≈ spend_i.

Simple fixed stake per draw (no escalating):

# spend_i is total planned spend across the whole progression window
stake_per_draw_per_combo = spend_i / (PACKS[pack_id].progressive_steps * PACKS[pack_id].combos_per_ticket)


If you want progressive stake (like 1x, 1x, 2x, 3x…):

progression_multipliers = [1, 1, 2, 3]  # example
total_multiplier = sum(progression_multipliers)
stake_base_per_combo = spend_i / (PACKS[pack_id].combos_per_ticket * total_multiplier)


Then stake in draw k = stake_base_per_combo * progression_multipliers[k].

class PackSignal:

@dataclass
class PackAllocation:
    state: str
    date_tag: str
    lane_id: str
    pack_id: str
    combos_per_ticket: int
    play_mode: PlayMode
    total_spend: float
    stake_per_combo_base: float
    progression_multipliers: List[float]
    expected_payout_per_hit: float
    roi_est: float
    hit_prob_cycle: float
    evidence_score: float

    def as_dict(self) -> Dict:
        return asdict(self)


def _default_progression(pack: PackDefinition) -> List[float]:
    """
    Simple progression profile.
    You can tune per pack if needed (e.g. look up from config).
    """
    if pack.pack_id == "A":
        # short window, modest progression
        return [1, 1, 1, 2, 2, 3, 3, 4][: pack.progressive_steps]
    if pack.pack_id == "B":
        return [1] * pack.progressive_steps  # flat stake
    if pack.pack_id == "C":
        return [1, 1, 1, 2, 2, 3] + [3] * max(0, pack.progressive_steps - 6)
    if pack.pack_id == "D":
        return [1, 1, 2, 2] + [2] * max(0, pack.progressive_steps - 4)
    return [1] * pack.progressive_steps


def allocate_budget(
    bankroll: float,
    daily_budget: float,
    risk_cap_pct: float,
    signals: List[PackSignal],
) -> List[PackAllocation]:
    """
    Core Part 2 budgeter.

    - Filters pack signals by per-pack min ROI / hit probability.
    - Computes weights from (roi_est * ir_est * score).
    - Allocates spend under max_spend_today.
    - Converts spend into base stake per combo + progression profile.
    """
    max_spend_today = min(daily_budget, bankroll * risk_cap_pct)
    if max_spend_today <= 0 or not signals:
        return []

    # Filter by pack-specific thresholds
    eligible: List[PackSignal] = []
    for sig in signals:
        pack = PACKS.get(sig.pack_id)
        if not pack:
            continue
        if sig.roi_est < pack.min_roi:
            continue
        if sig.hit_prob_cycle < pack.min_hit_prob:
            continue
        eligible.append(sig)

    if not eligible:
        return []

    # Compute weights
    weights: List[float] = []
    for sig in eligible:
        # simple weight; can be made more complex based on your validation
        w = max(0.0, sig.roi_est) * max(0.0, sig.ir_est) * max(0.0, sig.score)
        weights.append(w)

    weight_sum = sum(weights)
    if weight_sum <= 0:
        return []

    allocations: List[PackAllocation] = []
    for sig, w in zip(eligible, weights):
        pack = PACKS[sig.pack_id]
        payout_curve = PAYOUTS[pack.play_mode]

        # total spend this pack gets today
        total_spend = max_spend_today * (w / weight_sum)

        # progression profile
        multipliers = _default_progression(pack)
        if not multipliers:
            continue
        total_mult = sum(multipliers)

        # base stake per combo for the whole progression window
        if pack.combos_per_ticket * total_mult == 0:
            continue
        stake_base = total_spend / (pack.combos_per_ticket * total_mult)

        allocations.append(
            PackAllocation(
                state=sig.state,
                date_tag=sig.date_tag,
                lane_id=sig.lane_id,
                pack_id=sig.pack_id,
                combos_per_ticket=pack.combos_per_ticket,
                play_mode=pack.play_mode,
                total_spend=round(total_spend, 2),
                stake_per_combo_base=round(stake_base, 4),
                progression_multipliers=multipliers,
                expected_payout_per_hit=payout_curve.payout,
                roi_est=sig.roi_est,
                hit_prob_cycle=sig.hit_prob_cycle,
                evidence_score=sig.score,
            )
        )

    return allocations
