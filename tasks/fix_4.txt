    rows = []
    for c in ba.get("candidates", []):
        t = sum_tags(c.get("combo",""))

            "Tags": " ".join(sorted(c.get("tags", []))),

# --- Blackapple Alerts (All States) ---
try:
    from modules.blackapple import analyze_blackapple, ba_status_label
    import pandas as _pd

    # Choose whichever mapping exists in this page:
    mapping = None
    if "results" in locals() or "results" in globals():
        mapping = results
    elif "state_to_draws" in locals() or "state_to_draws" in globals():
        mapping = {k: {"draws": v} for k, v in state_to_draws.items()}
    else:
        mapping = {}

    rows_ba = []
    for state, data in (mapping or {}).items():
        if not state or state.lower() == "combined": 
            continue
        dr = [str(x).zfill(3) for x in (data.get("draws") or []) 
              if isinstance(x, (int,str)) and str(x).isdigit() and len(str(x)) <= 3]
        if not dr:
            continue
        ba = analyze_blackapple(dr)
        status = ba_status_label(ba.get("score", 0))
        tr = ba.get("triggers", {})
        tparts = []
        if tr.get("mirror"): tparts.append("Mirror")
        roots = tr.get("root_due", [])
        if roots: tparts.append("Root " + "/".join(map(str, roots)))
        pat = tr.get("pattern", {})
        if pat.get("extreme_due"): tparts.append("SSS/TTT")
        if pat.get("mixed_due"):   tparts.append("SST/STS/TSS")
        flt = tr.get("floating", [])
        if flt: tparts.append("Float " + "".join(flt))
        rows_ba.append({
            "State": state,
            "BA-Score": ba.get("score", 0),
            "Status": status,
            "Triggers": ", ".join(tparts),
            "#Candidates": len(ba.get("candidates", [])),
            "Examples": " ".join([c.get("combo", "") for c in ba.get("candidates", [])[:3]]),
        })

    if rows_ba:
        df_ba = _pd.DataFrame(rows_ba).sort_values(
            ["BA-Score", "#Candidates"], ascending=[False, False]
        ).reset_index(drop=True)
        st.subheader("Blackapple Alerts (All States)")
        st.dataframe(df_ba, use_container_width=True)
except Exception as _e:
