P0.3 — Explicit last‑remaining‑3v on the rightmost stable column

File: alpha_analytical/stable/post_pass_families.py (continue in the same function)

# Rightmost Stable: last remaining 3v
last_rows = []
for (sec, setv, draw), box in df.groupby(["section","Set","Draw"]):
    # Pick the rightmost Column that actually has stable coverage
    # (we use rows_cov >= 3 as "stable" like row engine does)
    b = box[box["rows_cov"] >= 3].copy()
    if b.empty: 
        continue
    # Columns are strings '7'..'1' → cast to int
    b["col_int"] = b["Column"].astype(int)
    col_max = b["col_int"].max()
    survivors = b[b["col_int"]==col_max]
    fams = survivors["family_id"].dropna().unique()
    if len(fams) == 1:
        fid = fams[0]
        mask = (df["section"]==sec) & (df["Set"]==setv) & (df["Draw"]==draw) & (df["Column"]==str(col_max)) & (df["family_id"]==fid)
        df.loc[mask, "last_remaining_3v"] = True
        df.loc[mask, "family_score"] += cfg.get("last_remaining_3v_bonus", 3)

df["last_remaining_3v"] = df.get("last_remaining_3v", False).fillna(False)
