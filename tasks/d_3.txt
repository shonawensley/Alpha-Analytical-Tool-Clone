def _lockscore_v2(row: dict, cfg: dict) -> float:
    s = cfg.get("scoring", {})
    # Helper fetchers
    get = lambda k, d=0.0: float(row.get(k, d) or 0.0)
    w  = lambda k: float(s.get(k, 0.0))
    # Evidence features present in per_item.csv
    score_raw = get("score")  # analyzer's baseline
    earliest_exact = get("earliest_exact_step", 99.0)
    earliest_vtrac = get("earliest_vtrac_step", 99.0)
    persistence = max(get("persistence_exact"),
                      get("persistence_family_exact"),
                      get("persistence_vtrac"))
    box_density = get("box_family_density")
    set_echo    = get("set_echo_count")
    variant_echo= get("variant_echo_count")
    recency     = 1.0 if str(row.get("recency_carryover","")).lower()=="true" else 0.0
    v_hot       = get("vtrac.v_hot")  # assumed 0..1
    cols_hit    = int(row.get("cols_hit", 0) or 0)
    section     = (row.get("section") or "").strip()  # Midday/Evening/Combined
    final_val   = str(row.get("final_value",""))
    is_double   = (len(set(final_val))==2)

    # Early-step transforms (earlier step => larger addend)
    early_exact = 1.0/(1.0+earliest_exact) if earliest_exact < 99 else 0.0
    early_vtrac = 1.0/(1.0+earliest_vtrac) if earliest_vtrac < 99 else 0.0

    # Single-column rescue (strict + early)
    rescue = 0.0
    if cols_hit <= int(s.get("t_cols_hit_rescue",1)):
        if min(earliest_exact,earliest_vtrac) <= float(s.get("t_early",3)):
            rescue = w("w_single_col_rescue")

    # Optional mild bias against 'Combined' over-dominance when exclusive
    penalty = 0.0
    if section.lower()=="combined":
        penalty = float(s.get("penalty_combined_bias",0.0))

    lock = (
        w("w_score_raw")        * score_raw
      + w("w_earliest_exact")   * early_exact
      + w("w_earliest_vtrac")   * early_vtrac
      + w("w_persistence")      * persistence
      + w("w_box_density")      * box_density
      + w("w_set_echo")         * set_echo
      + w("w_variant_echo")     * variant_echo
      + w("w_recency")          * recency
      + w("w_vtrac_hot")        * v_hot
      + rescue
      + w("w_double_guard")     * (1.0 if is_double else 0.0)
      - penalty
    )

    # Normalize to [0,1] by a soft cap (keeps YAML linear)
    return max(0.0, min(1.0, lock/ max(1.0, 1.0 + penalty)))

def _attach_lockscore(rows: list[dict], cfg: dict) -> None:
    for r in rows:
        r["lock_score_v2"] = round(_lockscore_v2(r, cfg), 6)
        # Optional 'lock' decision for dev viewing (not changing the bundle contract)
        r["lock_decision_v2"] = ("lock" if r["lock_score_v2"] >= float(
            cfg.get("scoring",{}).get("min_lockscore_to_lock",0.85)) else "hold")

# Call this right after per_item rows are finalized, before write_artifacts(...)
_attach_lockscore(rows, cfg)

#!/usr/bin/env python3
"""
Join Digit-Reduction per_item/top files with winners, compute metrics, and emit
summary CSV + slices CSV + a tiny Markdown dashboard.
"""

from __future__ import annotations
import json, csv, sys
from pathlib import Path
import pandas as pd

def _load_per_item(an_dir: Path, state: str) -> pd.DataFrame:
    pat = f"{state}_analyzer_v2_per_item.csv"
    df  = pd.read_csv(an_dir / pat)
    # Minimal sanity
    needed = ["section","set","draw","col","final_value","score","lock_score_v2"]
    for k in needed:
        if k not in df.columns:
            df[k] = None
    return df

def _load_top(an_dir: Path, state: str) -> pd.DataFrame:
    pat = f"{state}_analyzer_v2_top_candidates.csv"
    return pd.read_csv(an_dir / pat) if (an_dir/pat).exists() else pd.DataFrame()

def _load_winners_map(win_dir: Path, stamp: str, state: str) -> pd.DataFrame:
    # Expect centralized Control Center artifacts: *_winner_flags.csv (Midday/Evening)
    frames = []
    for variant in ("Midday","Evening"):
        p = win_dir / f"{stamp}_{variant}_winner_flags.csv"
        if p.exists():
            df = pd.read_csv(p)
            df["variant"] = variant
            df["state"] = state  # for join stability across states in one folder
            frames.append(df)
    return pd.concat(frames, ignore_index=True) if frames else pd.DataFrame()

def _normalize_cand(s: str)->str:
    s = "".join(ch for ch in str(s) if ch.isdigit())
    return "".join(sorted(s)) if s else s

def evaluate_state(state: str, root: Path, stamp: str, winners_root: Path) -> dict:
    an_dir = root / state / "analyzer_v2"
    per_df = _load_per_item(an_dir, state)
    top_df = _load_top(an_dir, state)
    win_df = _load_winners_map(winners_root, stamp, state)

    if win_df.empty:
        return {"state": state, "rows": 0}

    # Canonicalize candidates and winners for robust matching
    per_df["cand_canon"] = per_df["final_value"].apply(_normalize_cand)
    # winners file assumed to have a 'winner' column; add canon
    if "winner" not in win_df.columns:
        # fallback: if columns exist like 'value' or 'winning_number'
        for c in ("value","winning_number","win"):
            if c in win_df.columns:
                win_df["winner"] = win_df[c]; break
    win_df["win_canon"] = win_df["winner"].apply(_normalize_cand)

    # For each winner row, rank candidates by score and lock_score_v2 within same variant/section
    out_rows = []
    for _, w in win_df.iterrows():
        section = str(w.get("variant","")).strip()  # Midday/Evening
        w_canon = w["win_canon"]
        scope = per_df[ per_df["section"].str.lower()==section.lower() ]
        if scope.empty:
            continue

        # rankings
        scope = scope.copy()
        scope["rank_score"]      = (-scope["score"]).rank(method="min")
        scope["rank_lock_v2"]    = (-scope["lock_score_v2"]).rank(method="min")

        # winner rows
        win_hits = scope[ scope["cand_canon"]==w_canon ]
        if win_hits.empty:
            out_rows.append({
                "state":state,"variant":section,"winner":w.get("winner",""),
                "present":False,"rank_score":None,"rank_lock_v2":None,
                "lock_score_v2":None,"score":None
            })
        else:
            best = win_hits.sort_values("rank_lock_v2").iloc[0]
            out_rows.append({
                "state":state,"variant":section,"winner":w.get("winner",""),
                "present":True,
                "rank_score": int(best["rank_score"]),
                "rank_lock_v2": int(best["rank_lock_v2"]),
                "lock_score_v2": float(best["lock_score_v2"]),
                "score": float(best["score"])
            })

    # Summaries
    rows = pd.DataFrame(out_rows)
    return {
        "state":state,
        "rows":len(rows),
        "hit_at_1_score": int((rows["rank_score"]==1).sum()),
        "hit_at_3_score": int((rows["rank_score"]<=3).sum()),
        "hit_at_1_lock":  int((rows["rank_lock_v2"]==1).sum()),
        "hit_at_3_lock":  int((rows["rank_lock_v2"]<=3).sum()),
        "detail": rows
    }

def main():
    import argparse
    ap = argparse.ArgumentParser()
    ap.add_argument("--states", nargs="+", required=True)
    ap.add_argument("--analysis-root", default="data/outputs/analysis/digit_reduction")
    ap.add_argument("--winners-root",  default="data/outputs/analysis/digit_reduction/Connecticut4/training_sets") # overridden per state if needed
    ap.add_argument("--stamp", required=True, help="e.g., 20251103")
    ap.add_argument("--out", default="dr_validation_out")
    args = ap.parse_args()
    root = Path(args.analysis_root)
    winr = Path(args.winners_root)
    Path(args.out).mkdir(parents=True, exist_ok=True)

    summaries = []
    slices = []
    for st in args.states:
        res = evaluate_state(st, root, args.stamp, winr)
        summaries.append({k:v for k,v in res.items() if k!="detail"})
        if "detail" in res and isinstance(res["detail"], pd.DataFrame):
            df = res["detail"].copy()
            df["state"]=st
            slices.append(df)

    sum_df = pd.DataFrame(summaries)
    sum_df.to_csv(Path(args.out)/f"DR_PERF_SUMMARY_{args.stamp}.csv", index=False)

    if slices:
        sl_df = pd.concat(slices, ignore_index=True)
        sl_df.to_csv(Path(args.out)/f"DR_PERF_SLICES_{args.stamp}.csv", index=False)

    # Tiny Markdown dashboard for ChatGPT review
    with open(Path(args.out)/f"DR_VALIDATION_SUMMARY_{args.stamp}.md","w") as f:
        f.write("|State|Rows|Hit@1 score|Hit@3 score|Hit@1 lock|Hit@3 lock|\n|---|---:|---:|---:|---:|---:|\n")
        for _,r in sum_df.iterrows():
            f.write(f"|{r['state']}|{int(r.get('rows',0))}|{int(r.get('hit_at_1_score',0))}|{int(r.get('hit_at_3_score',0))}|{int(r.get('hit_at_1_lock',0))}|{int(r.get('hit_at_3_lock',0))}|\n")
    print("Done.")

if __name__=="__main__":
    main()

Why these are “final, unstoppable” optimizations

Single‑column rescue directly targets the failure mode we saw (true winners emerging only in one column). We gate it by earliest step to avoid noisy late “finds.”

Earliest‑step premium and persistence/echo tap signals your analyzer already computes; this only reweights them cleanly.

V‑TRAC heat gradient lets DR inherit some family pressure without needing the V‑TRAC page to run first (the features exist in per‑item).

Mild Combined bias penalty prevents Combined from swamping Midday/Evening when those variants carry the sharper signal on their own.

All choices are YAML‑tunable and validated via a single pass runner so we can make a data‑driven keep/adjust call in minutes.
