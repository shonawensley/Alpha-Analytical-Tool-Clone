def initialize_vtrac_lookup() -> None:
    VTRAC_LOOKUP.clear()
    BOXED_LABEL_LOOKUP.clear()   # ← restore this clear
    INDEX_BY_VTRAC.clear()       # ← also clear the fallback map

    for entry in BOXED_VTRAC_REFERENCE:
        idx = int(entry["Index"])
        sample_combo = None
        for key in ("Singles", "Doubles"):
            for combo in entry[key]:
                if sample_combo is None and combo:
                    sample_combo = combo
                for perm in _permutations3(combo):
                    VTRAC_LOOKUP[perm] = idx
                    BOXED_LABEL_LOOKUP[perm] = combo
        if sample_combo:
            vcode = ''.join(sorted(str(DIGIT2V[d]) for d in sample_combo if d in DIGIT2V))
            if len(vcode) == 3:
                INDEX_BY_VTRAC.setdefault(vcode, idx)

def get_vtrac_index(draw: str) -> Optional[int]:
    s = normalize_draw(draw)
    if s is None:
        return None
    # 1) direct hit
    idx = VTRAC_LOOKUP.get(s)
    if idx is not None:
        return idx
    # 2) pure triples are suppressed by design
    if len(set(s)) == 1:
        return None
    # 3) permutation hit
    for perm in _permutations3(s):
        idx = VTRAC_LOOKUP.get(perm)
        if idx is not None:
            return idx
    # 4) fallback by V‑class (e.g., "277" → "333" → 26)
    vcode = ''.join(sorted(str(DIGIT2V.get(ch, 0)) for ch in s))
    return INDEX_BY_VTRAC.get(vcode)


This restores O(1) behavior for the common case and keeps our explicit “no pure triples” policy.

# consensus → doubles support
def _doubles_for_digit(d: str) -> list[str]:
    # 9 canonical doubles given a digit d (e.g., "5" -> ["055","155",...,"955"])
    return [f"{a}{d}{d}" for a in "0123456789" if a != d]  # keep simple v1

df["any_doubles_support"] = False

# For each family box with consensus, check if any derived doubles appear in the same box or adjacent columns
for (sec, Set, Draw, Col, fid), chunk in df.groupby(["section","Set","Draw","Column","family_id"]):
    if not chunk["any_consensus"].any():
        continue
    # consensus digit is embedded in top_canonicals "d:count" — extract best guess
    # fallback: use modal_orders if needed
    cons_digit = None
    parts = (chunk["top_canonicals"].iloc[0] or "").split(";")
    for p in parts:
        k = p.split(":")[0].strip()
        if len(k) == 2 and k.isdigit():
            cons_digit = k[-1]  # tail consensus; adjust if you store separately
            break
    if not cons_digit:
        continue
    candidates = set(_doubles_for_digit(cons_digit))
    # same box or adjacent columns
    col_int = pd.to_numeric(Col, errors="coerce")
    cols_considered = [str(int(col_int))] if not pd.isna(col_int) else [Col]
    if not pd.isna(col_int):
        cols_considered += [str(int(col_int) - 1), str(int(col_int) + 1)]

    mask_near = (
        (df["section"] == sec) & (df["Set"] == Set) & (df["Draw"] == Draw)
        & (df["Column"].isin(cols_considered)) & (df["family_id"] == fid)
    )
    # if any canonical appears that is one of the candidates, flag & bonus
    hit = df.loc[mask_near, "top_canonicals"].astype(str).str.contains(
        "|".join(candidates), regex=True, na=False
    ).any()
    if hit:
        df.loc[(df["section"] == sec) & (df["Set"] == Set) & (df["Draw"] == Draw)
               & (df["Column"] == Col) & (df["family_id"] == fid), "any_doubles_support"] = True

df["family_score"] += df["any_doubles_support"].astype(int) * cfg.get("doubles_trigger_bonus", 1)
