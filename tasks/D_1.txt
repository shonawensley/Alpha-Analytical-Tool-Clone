C) New module: final scoring (copy‑paste file)

File (NEW): alpha_analytical/digit_reduction/analyzer_v2/scoring.py

from __future__ import annotations
from math import exp
from typing import Dict, Any

SAFE_ZERO = 0.0

def _b(x: Any, default: float = 0.0) -> float:
    """Best-effort float conversion with safe default."""
    try:
        if x is None: return default
        if isinstance(x, bool): return 1.0 if x else 0.0
        return float(x)
    except Exception:
        return default

def _sigmoid(z: float) -> float:
    # numerically stable logistic
    if z >= 0:
        ez = exp(-z)
        return 1.0 / (1.0 + ez)
    ez = exp(z)
    return ez / (1.0 + ez)

def compute_final_score(row: Dict[str, Any], w: Dict[str, float]) -> Dict[str, float]:
    """
    Build a calibrated final score and probability from per-item evidence.
    Only uses columns already emitted by Analyzer V2 (lean bundle).
    """
    # Base evidence (existing columns)
    recency   = _b(row.get("recency_carryover"))
    set_echo  = _b(row.get("set_echo_count"))
    var_echo  = _b(row.get("variant_echo_count"))
    cols_hit  = _b(row.get("cols_hit"))
    density   = _b(row.get("box_family_density"))
    dup_bonus = _b(row.get("dup_bonus"))
    v_hot     = _b(row.get("vtrac.v_hot"))
    v_set     = _b(row.get("vtrac.set"))  # 0..15 typical
    mode      = (row.get("mode") or "").lower()
    section   = (row.get("section") or "").lower()

    # Linear blend (small, safe nudges)
    linear = 0.0
    linear += w.get("w_recency_carryover", 0.0) * recency
    linear += w.get("w_set_echo", 0.0) * set_echo
    linear += w.get("w_variant_echo", 0.0) * var_echo
    linear += w.get("w_box_family_density", 0.0) * density
    linear += w.get("w_dup_bonus", 0.0) * dup_bonus

    if cols_hit == 1.0:
        linear += w.get("w_single_column_rescue", 0.0)

    if v_hot > 0.0:
        linear += w.get("w_vtrac_hot", 0.0)

    if v_set > 0.0:
        linear += w.get("w_vtrac_local_set", 0.0) * (v_set / 15.0)

    if mode == "own":
        linear += w.get("w_mode_own_bonus", 0.0)

    if section == "combined":
        linear += w.get("w_section_combined_penalty", 0.0)

    # Logistic map -> probability in [0,1]
    z = w.get("final_logit_bias", 0.0) + w.get("final_logit_scale", 1.0) * linear
    prob = _sigmoid(z)

    return {
        "final_linear": linear,
        "final_prob": prob
    }

#!/usr/bin/env python3
"""
Quick local smoke to ensure final_prob & lock_decision exist
and that top list is prob-sorted.
"""
import sys, json
from pathlib import Path
import pandas as pd

def main(state: str):
    root = Path(f"data/outputs/analysis/digit_reduction/{state}/analyzer_v2")
    per_item = pd.read_csv(root / f"{state}_analyzer_v2_per_item.csv")
    top = pd.read_csv(root / f"{state}_analyzer_v2_top_candidates.csv")

    for col in ["final_linear", "final_prob", "lock_decision"]:
        assert col in per_item.columns, f"missing {col} in per_item"

    assert top["final_prob"].is_monotonic_decreasing, "top not sorted by final_prob desc"

    print(json.dumps({
        "state": state,
        "rows": len(per_item),
        "top": len(top),
        "locks": int((per_item["lock_decision"]=="lock").sum())
    }, indent=2))

if __name__ == "__main__":
    if len(sys.argv)!=2:
        print("Usage: dr_quickcheck.py <StateLabel>")
        sys.exit(2)
    main(sys.argv[1])
