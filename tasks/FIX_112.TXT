from pathlib import Path
import re

DRAW_DIRS = [Path("data/cleaned/draws"), Path("data/cleaned")]  # prefer subfolder

def _canon_draws_basename(state: str) -> str:
    """Normalize project state to draws CSV basename."""
    s = state.strip()
    # normalize trailing '4' and underscores vs spaces, as per Aux docs
    s = re.sub(r"\s+", "_", s)
    s = s[:-1] if s.endswith("4") else s
    return s

def resolve_draw_csv(state: str, variant: str) -> Path | None:
    base = _canon_draws_basename(state)
    names = {
        "combined": f"{base}_draws.csv",
        "midday":   f"{base}_Midday_draws.csv",
        "evening":  f"{base}_Evening_draws.csv",
    }
    for root in DRAW_DIRS:
        p = root / names[variant.lower()]
        if p.exists():
            return p
    return None  # tell the caller exactly what we tried

def load_state_draws(state: str, variant: str):
    p = resolve_draw_csv(state, variant)
    if not p:
        tried = [str(Path(d) / f) for d in DRAW_DIRS for f in {
            f"{_canon_draws_basename(state)}_draws.csv",
            f"{_canon_draws_basename(state)}_Midday_draws.csv",
            f"{_canon_draws_basename(state)}_Evening_draws.csv",
        }]
        raise FileNotFoundError(f"No draws CSV for {state}/{variant}. Tried: {tried}")
    # read file → return newest-first list[str]
    rows = Path(p).read_text(encoding="utf-8").splitlines()
    draws = [r.strip().split(",")[0] for r in rows if r.strip()]  # adjust if you have headers
    return draws, str(p)
