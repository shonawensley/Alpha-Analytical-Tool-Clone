4.1 Extended cluster extraction (family‑aware)
# alpha_analytical/digit_reduction/analyzer_v2/clustering.py
from dataclasses import dataclass
from typing import List, Tuple, Iterable, Dict

@dataclass(frozen=True)
class Cluster:
    s: str                     # original digit string, e.g., "6111188"
    unique: str                # sorted unique digit set, e.g., "168"
    span: Tuple[int, int]      # (start, end) in the parent string
    run_lengths: Dict[str,int] # per-digit max run length within the cluster

def longest_run(s: str, ch: str) -> int:
    m = 0; cur = 0
    for c in s:
        if c == ch:
            cur += 1; m = max(m, cur)
        else:
            cur = 0
    return m

def extract_clusters(row_digits: str, max_window: int = 12, min_len: int = 3) -> List[Cluster]:
    """Scan variable windows to capture *family* clusters (2–3 unique digits),
    not just fixed 3-grams. Supports long clusters like '6111188'."""
    n = len(row_digits)
    out: List[Cluster] = []
    for L in range(min_len, min(max_window, n) + 1):
        for i in range(0, n - L + 1):
            w = row_digits[i:i+L]
            uniq = ''.join(sorted(set(w)))
            if 1 < len(uniq) <= 3:   # 2 or 3 unique digits => family candidate
                rl = {d: longest_run(w, d) for d in uniq}
                out.append(Cluster(s=w, unique=uniq, span=(i, i+L), run_lengths=rl))
    return out

4.2 V‑TRAC crosswalk (Exact & family variants)
# alpha_analytical/digit_reduction/analyzer_v2/vtrac_index.py
from typing import Set, Dict, Iterable
from .clustering import Cluster

class VtracIndex:
    def __init__(self, exact: Set[str], family: Dict[str, Set[str]]):
        """
        exact: canonical set of exact patterns (e.g., '618', '546', ...)
        family: maps family key -> set of 3-grams in that family
                e.g., '168' -> {'618','681','168',...}  (order-insensitive family)
        """
        self.exact = exact
        self.family = family

    @staticmethod
    def family_key(d: str) -> str:
        return ''.join(sorted(set(d)))

    def is_exact(self, trigram: str) -> bool:
        return trigram in self.exact

    def is_vtrac(self, trigram: str) -> bool:
        # V-TRAC membership == in the family closure
        return trigram in self.family.get(self.family_key(trigram), set())

    def cluster_membership(self, c: Cluster) -> Dict[str, bool]:
        """Check a long cluster (e.g., '6111188') for 3-gram evidence both Exact and VTRAC."""
        grams = {c.s[i:i+3] for i in range(0, len(c.s)-2)}
        exact_hit = any(self.is_exact(g) for g in grams)
        vtrac_hit = any(self.is_vtrac(g) for g in grams)
        return {
            "Exact": exact_hit,
            "VTRAC": vtrac_hit,
            "ThreeValueExact": (len(c.unique) == 3) and exact_hit,
            "ThreeValueVTRAC": (len(c.unique) == 3) and vtrac_hit
        }

4.3 Features & scoring (config‑driven, early lock/unlock)
# alpha_analytical/digit_reduction/analyzer_v2/features.py
from dataclasses import dataclass
from typing import Dict, Any, List
from .clustering import extract_clusters
from .vtrac_index import VtracIndex

@dataclass
class FeatureRow:
    state: str; box: str; variant: str; method: str
    pattern: str; family_id: str; is_extended_cluster: bool
    feats: Dict[str, float]

def build_features(
    state: str, box: str, variant: str, method: str,
    digit_string: str, training_trace: Dict[str, Any],
    vtrac: VtracIndex
) -> List[FeatureRow]:
    clusters = extract_clusters(digit_string, max_window=12, min_len=3)
    out: List[FeatureRow] = []
    # Example features; Codex can extend from training_trace as needed
    for c in clusters:
        m = vtrac.cluster_membership(c)
        feats = {
            "run_len_max": max(c.run_lengths.values()),
            "is_three_value": 1.0 if len(c.unique) == 3 else 0.0,
            "exact_evidence": 1.0 if m["Exact"] else 0.0,
            "vtrac_evidence": 1.0 if m["VTRAC"] else 0.0,
            "t_first_three_value": training_trace.get("t_first_three_value", 1e3),
            "late_churn": training_trace.get("late_churn", 0.0),
            "method_consensus": training_trace.get("method_consensus", 0.0),
            "perm_consensus": training_trace.get("perm_consensus", 0.0),
            "own_vs_combined_delta": training_trace.get("own_vs_combined_delta", 0.0),
        }
        out.append(FeatureRow(
            state=state, box=box, variant=variant, method=method,
            pattern=c.s, family_id=c.unique, is_extended_cluster=True, feats=feats
        ))
    return out

# alpha_analytical/digit_reduction/analyzer_v2/scoring.py
from typing import Dict
from math import tanh

def score_row(feats: Dict[str, float], w: Dict[str, float], cfg: Dict) -> Dict:
    # Weighted sum with gentle non-linearities + early lock/unlock
    z = (
        w["exact"] * feats["exact_evidence"] +
        w["vtrac"] * feats["vtrac_evidence"] +
        w["run"]   * feats["run_len_max"] +
        w["3val"]  * feats["is_three_value"] +
        w["cons_m"]* feats["method_consensus"] +
        w["cons_p"]* feats["perm_consensus"] +
        w["own_d"] * feats["own_vs_combined_delta"] -
        w["late"]  * feats["late_churn"]
    )
    # Soft clamping to [-1, 1]
    s = tanh(z / max(1e-6, cfg.get("tanh_scale", 4.0)))
    # Early lock/unlock
    if s >= cfg["lock_threshold"]:
        lock = ("lock", "score>=lock_threshold")
    elif s <= cfg["unlock_threshold"]:
        lock = ("unlock", "score<=unlock_threshold")
    else:
        lock = ("undecided", "")
    return {"score": float(s), "lock": lock[0], "lock_reason": lock[1]}

4.4 Winners overlay (compact flags + detailed hits)
# alpha_analytical/digit_reduction/analyzer_v2/winners_overlay.py
import pandas as pd
from typing import Iterable, Dict

EVIDENCE = ["Exact","VTRAC","DropExact","DropVTRAC","ThreeValueExact","ThreeValueVTRAC"]

def overlay_flags(per_item: pd.DataFrame) -> pd.DataFrame:
    """Return compact boolean/int flags per candidate for the UI colorizer."""
    flags = pd.DataFrame(index=per_item.index)
    for e in EVIDENCE:
        col = f"ev_{e.lower()}"
        flags[col] = per_item[f"feat_{e}"].astype(int) if f"feat_{e}" in per_item.columns else 0
    return flags

def overlay_hits_detailed(per_item: pd.DataFrame) -> pd.DataFrame:
    """Human-readable rows explaining evidence per candidate."""
    rows = []
    for _, r in per_item.iterrows():
        for e in EVIDENCE:
            bit = r.get(f"feat_{e}", 0)
            if bit:
                rows.append({
                    "state": r["state"], "box": r["box"], "variant": r["variant"],
                    "pattern": r["pattern"], "family_id": r["family_id"],
                    "evidence": e, "reason": f"{e} via {r.get('method','?')}"
                })
    return pd.DataFrame(rows)
