def combos_for_pair(combo_miss_maps: dict[str, dict[str, int]], pair: str) -> list[str]:
    """
    combo_miss_maps = {
      'C': {'244': 1042, '455': 1184, ...},
      'M': {'211': 1013, ...},
      'E': {'233': 1260, ...}
    }
    Return a list like ['244C', '744M', '544C'] but with merged badges
    e.g., '455 M/C' if present in multiple maps.
    """
    badges = {}
    p = pair[0]              # '4' for '44'
    def is_double_of_pair(s): 
        return (s.count(p) == 2) and (len(set(s)) == 2)  # exactly two of the same digit
    for badge, dmap in combo_miss_maps.items():
        for combo in dmap:
            if is_double_of_pair(combo):
                badges.setdefault(combo, set()).add(badge)
    # keep only combos actually matching this pair
    out = []
    for combo, bset in badges.items():
        if p in combo and ((combo.count(p) == 2) and (len(set(combo)) == 2)):
            out.append(f"{combo} {'/'.join(sorted(bset))}")
    # sort by lexicographic or miss length if you want; keep stable
    out.sort()
    return out

def build_due_doubles_row(state, variant) -> dict:
    pairs = top4_due_double_pairs_combined(state)     # [('11',75), ('66',47), ('22',39), ('55',38)]
    missC = long_miss_double_combos(state, 'Combined', RED_MISS)
    missM = long_miss_double_combos(state, 'Midday', RED_MISS)
    missE = long_miss_double_combos(state, 'Evening', RED_MISS)
    badge_maps = {'C': missC, 'M': missM, 'E': missE}

    def fmt_pair(pp_gap): 
        pp, gap = pp_gap
        return f"{pp} – {gap}"

    combos_cols = []
    for (pp, _gap) in pairs:
        lst = combos_for_pair(badge_maps, pp)
        combos_cols.append(COMBO_SEP.join(lst) if lst else '—')

    return {
