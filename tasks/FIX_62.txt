def score_row(r: dict, W: dict, P: dict, C: dict) -> float:
    s = 0.0
    # families
    s += W["tail.exact_len3"]            * float(r.get("tail.exact_len3", 0))
    s += W["tail.unique2"]               * float(r.get("tail.unique2", 0))
    s += W["final.len_is1"]              * float(r.get("final.len_is1", 0))
    s += W["final.len_is2"]              * float(r.get("final.len_is2", 0))
    s += W["final.len_is3"]              * float(r.get("final.len_is3", 0))
    s += W["traj.early_terminal"]        * float(r.get("traj.early_terminal", 0))
    s += W["traj.reduction_slope"]       * float(r.get("traj.reduction_slope", 0))
    s += W["pre.mirror_pair"]            * float(r.get("pre.mirror_pair", 0))
    s += W["pre.core3_hint"]             * float(r.get("pre.core3_hint", 0))
    s += W["sec.consensus_any"]          * float(r.get("sec.consensus_any", 0))
    s += W["sec.consensus_strong"]       * float(r.get("sec.consensus_strong", 0))
    s += W["sec.pairwise_jaccard"]       * float(r.get("sec.pairwise_jaccard", 0))
    s += W["mode.agree_core"]            * float(r.get("mode.agree_core", 0))
    s += W["mode.time_to3_delta_abs"]    * float(r.get("mode.time_to3_delta_abs", 0))
    s += W["mode.len_delta_abs"]         * float(r.get("mode.len_delta_abs", 0))
    s += W["methods.core_agreement"]     * float(r.get("methods.core_agreement", 0))
    s += W["methods.early_fraction"]     * float(r.get("methods.early_fraction", 0))
    s += W["method.agree_count"]         * float(r.get("method.agree_count", 0))
    s += W["set.memory_strength"]        * float(r.get("set.memory_strength", 0))
    s += W["set.repeat_new_box"]         * float(r.get("set.repeat_new_box", 0))
    s += W["xcol.agree_count"]           * float(r.get("xcol.agree_count", 0))
    s += W["stability.order_cue"]        * float(r.get("stability.order_cue", 0))
    s += W["stability.horiz_persist"]    * float(r.get("stability.horiz_persist", 0))
    s += W["stability.survival_frac3"]   * float(r.get("stability.survival_frac3", 0))
    s += W["perm.density"]               * float(r.get("perm.density", 0))
    # penalties
    s += P["pen.degenerate_empty"]       * (1.0 if r.get("final.value","")=="" else 0.0)
    s += P["pen.tail_wobble"]            * float(r.get("tail.wobble", 0))
    s += P["pen.mode_only_one"]          * float(r.get("mode.only_one", 0))
    # optional V‑TRAC synergy
    s += W.get("vtrac.v_hot", 0.0)       * float(r.get("vtrac.v_hot", 0.0))
    # clamp
    lo = C.get("score_min", 0.0); hi = C.get("score_max", 100.0)
    return float(max(lo, min(hi, s*10.0)))

    """
