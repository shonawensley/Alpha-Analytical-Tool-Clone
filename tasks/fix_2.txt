# -------- Config knobs --------

def _has_mirror_pair(s: str) -> bool:

def _pairs_from_draw(d: str) -> Set[str]:

            out.add(f"{i}{j}")

def _remaining_pairs(draws: List[str], target: int) -> Set[str]:
    """Cross off pairs newest->older until ~target remain (27–29 method)."""

    mirror_event = _had_mirror_in_last_k(draws, MIRROR_LOOKBACK)
    due_roots    = _longest_out_root_sums(draws)
    pat_ds       = _pattern_draws_since(draws)
    extreme_due  = (pat_ds["SSS"] >= EXTREME_PATTERN_OVERDUE) or (pat_ds["TTT"] >= EXTREME_PATTERN_OVERDUE)

    triggers = {
        "mirror": mirror_event,
        "root_due": due_roots,
        "pattern": {"extreme_due": extreme_due, "mixed_due": mixed_due},
        "floating": floating,
        "pairs": {"remaining_count": len(rem_pairs)}

          + (1 if (extreme_due or mixed_due) else 0) \
          + (1 if floating_on else 0) \
          + (1 if len(rem_pairs) <= REMAINING_PAIRS_TARGET + 2 else 0)
