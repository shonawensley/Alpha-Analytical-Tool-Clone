    """Return the V-TRAC index (0..999) for a 3-digit winner like '579'."""
    s = "".join(ch for ch in winner if ch.isdigit())
    if len(s) != 3:
        raise ValueError(f"winner must be 3 digits, got {winner!r}")
    # placeholder deterministic mapping (replace with your real table):

# src/reporting/winner_report_full.py
from __future__ import annotations
from pathlib import Path
import datetime as dt
import pandas as pd

from modules.vtrac_reference import (
    get_vtrac_index, get_index_set, get_index_straights
)

def build_and_save_full_report(
    state4: str,
    draw: str,            # "Midday" | "Evening" | "Combined"
    date_yyyy_mm_dd: str, # "2025-09-16"
    winner: str,          # "579"
    outputs_root: str | Path = "data/outputs/analysis/winners",
    tables_root: str | Path = "data/outputs/tables",
) -> Path:
    """Builds a 3-pane analyzer-style HTML for a specific (state, draw, date, winner)."""
    state4 = str(state4)
    draw = str(draw)
    date_yyyy_mm_dd = str(date_yyyy_mm_dd)

    tables_dir = Path(tables_root) / state4
    input_csv = tables_dir / f"{state4}_{draw}_combined.csv"
    if not input_csv.exists():
        raise FileNotFoundError(f"Missing table: {input_csv}")

    # String-safe CSV read (no NA coercion, preserve leading zeros):
    df = pd.read_csv(input_csv, dtype=str, keep_default_na=False, na_filter=False)

    # V-TRAC overlays
    idx = get_vtrac_index(winner)
    purple_set = set(get_index_set(idx))
    green_set  = set(get_index_straights(winner))

    # --- Very-light HTML render (you can swap in a Jinja2 template later) ---
    # Policy: green overlay should override purple where both apply
    def colorize(cell: str) -> str:
        val = str(cell)
        cls = "none"
        if val in purple_set: cls = "purple"
        if val in green_set:  cls = "green"
        return f'<td class="{cls}">{val}</td>'

    # Assume df is a rectangular “numbers” grid; adapt if your schema differs.
    rows_html = "\n".join(
        "<tr>" + "".join(colorize(x) for x in row) + "</tr>"
        for row in df.fillna("").values.tolist()
    )

    css = """
    <style>
      td { padding: 2px 6px; font-family: ui-monospace, Consolas, monospace; }
      td.purple { background: rgba(128, 0, 128, .18); }
      td.green  { background: rgba(0, 128,   0, .28); font-weight: 700; }
    </style>
    """

    meta = f"<p><strong>{state4} — {draw}</strong> · Date {date_yyyy_mm_dd} · Winner {winner} · Index {idx:03d}</p>"
    html = f"<!doctype html><html><head><meta charset='utf-8'>{css}</head><body>{meta}<table>{rows_html}</table></body></html>"

    out_dir = Path(outputs_root) / state4
    out_dir.mkdir(parents=True, exist_ok=True)
    out = out_dir / f"{state4}_{draw}_{date_yyyy_mm_dd}_winner_{winner}_analyzer.html"
    out.write_text(html, encoding="utf-8")
    return out
