# --- AUX window helpers (canonical; safe to call from src/app.py) ---

def _slice(draws, n):
    try:
        return list(draws)[:n]
    except Exception:
        return []

def build_aux_windows(draws, pairs_window=360, vtrac_index_window=1000):
    """
    Always return two lists: (draws_100, draws_1000).
    - If fewer than N draws exist, we return the shorter list (never None).
    - This matches the intent you've said out loud: "if it's not in the list, that *is* the 1000-draw view."
    """
    d100  = _slice(draws, pairs_window)
    d1000 = _slice(draws, vtrac_index_window)
    return d100, d1000
