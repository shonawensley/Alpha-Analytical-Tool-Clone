# modules/module_d_auxiliary_tools/refactored/bootstrap_imports.py
from __future__ import annotations
import sys
from pathlib import Path
import importlib.util

def _load_module_from(path: Path, as_name: str):
    spec = importlib.util.spec_from_file_location(as_name, str(path))
    if spec and spec.loader:
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)
        sys.modules[as_name] = mod
        return mod
    return None

def init() -> None:
    """
    Make legacy modules importable for Aux tools.

    Registers BOTH names so legacy & refactor agree:
      "vtrac_reference"   and "modules.vtrac_reference"
      "analyze_pairs"     and "modules.analyze_pairs"
    """
    root = Path(__file__).resolve().parents[1]  # .../module_d_auxiliary_tools
    candidates = [
        root / "legacy_2" / "modules",
        root / "legacy_modules_backup" / "modules",
        root / "legacy_modules_backup",
    ]
    for p in candidates:
        if p.exists():
            p_str = str(p)
            if p_str not in sys.path:
                sys.path.insert(0, p_str)

    for name in ("vtrac_reference", "analyze_pairs"):
        try:
            __import__(name)
        except Exception:
            for base in candidates:
                file = base / f"{name}.py"
                if file.exists():
                    _load_module_from(file, name)
                    break
        if name in sys.modules:
            sys.modules[f"modules.{name}"] = sys.modules[name]

__all__ = ["init"]

from __future__ import annotations
from pathlib import Path
import hashlib, json, time
from typing import Dict, Any

RUNS_DIR = Path("runs")  # add to .gitignore

def _short_md5(b: bytes) -> str:
    return hashlib.md5(b).hexdigest()[:8]

def compute_run_id(date_label: str, file_bytes: bytes) -> str:
    return f"{date_label}_{_short_md5(file_bytes)}"

def start_run(date_label: str, excel_path: Path) -> str:
    file_bytes = excel_path.read_bytes()
    run_id = compute_run_id(date_label, file_bytes)
    run_dir = RUNS_DIR / run_id
    run_dir.mkdir(parents=True, exist_ok=True)

    # Persist canonical inputs/metadata
    meta = {
        "run_id": run_id,
        "date_label": date_label,
        "ts": int(time.time()),
        "excel_name": excel_path.name,
        "excel_size": len(file_bytes),
    }
    (run_dir / "run.json").write_text(json.dumps(meta, indent=2))

    # Initialize checklist
    checklist = {
        "core_done": False,
        "aux_done": False,
        "combined_done": False,
        "winners_posted": False
    }
    (run_dir / "checklist.json").write_text(json.dumps(checklist, indent=2))
    return run_id

def load_checklist(run_id: str) -> Dict[str, bool]:
    return json.loads((RUNS_DIR / run_id / "checklist.json").read_text())

def mark_complete(run_id: str, key: str, value: bool = True):
    p = RUNS_DIR / run_id / "checklist.json"
    chk = json.loads(p.read_text())
    chk[key] = value
    p.write_text(json.dumps(chk, indent=2))

def save_state_outputs(run_id: str, state: str, data: Dict[str, Any]):
    d = RUNS_DIR / run_id / state
    d.mkdir(parents=True, exist_ok=True)
    (d / "results.json").write_text(json.dumps(data, indent=2))

def save_combined_outputs(run_id: str, data: Dict[str, Any]):
    d = RUNS_DIR / run_id / "combined"
    d.mkdir(parents=True, exist_ok=True)
    (d / "results.json").write_text(json.dumps(data, indent=2))

from __future__ import annotations
from typing import Dict, List, Tuple, Optional
from collections import defaultdict

def is_double(draw: str) -> bool:
    # exactly one repeated digit (exclude triples)
    if len(draw) != 3 or not draw.isdigit(): return False
    a,b,c = draw
    return (a==b!=c) or (a==c!=b) or (b==c!=a)

def is_triple(draw: str) -> bool:
    return len(draw)==3 and draw.isdigit() and draw[0]==draw[1]==draw[2]

def double_base(draw: str) -> Optional[str]:
    # return '00'..'99' for doubles & triples; None for singles
    if len(draw)!=3 or not draw.isdigit(): return None
    a,b,c = draw
    if a==b==c:     return a*2
    if a==b!=c:     return a*2
    if a==c!=b:     return a*2
    if b==c!=a:     return b*2
    return None

def draws_since_last_double(draws_newest_first: List[str]) -> int:
    for i, d in enumerate(draws_newest_first):
        if is_double(d) or is_triple(d):
            return i
    return len(draws_newest_first)

def draws_since_last_double_base(draws_newest_first: List[str], base: str) -> int:
    for i, d in enumerate(draws_newest_first):
        if double_base(d) == base:
            return i
    return len(draws_newest_first)

def per_state_summary(state_draws: Dict[str, List[str]]) -> List[Dict]:
    rows = []
    for state, draws in state_draws.items():
        ds = draws_since_last_double(draws)
        latest = None
        for d in draws:
            if is_double(d) or is_triple(d):
                latest = d
                break
        rows.append({"State": state, "DrawsSinceLastDouble": ds, "LatestDoubleOrTriple": latest})
    return rows

def aggregated_double_bases(state_draws: Dict[str, List[str]]) -> List[Dict]:
    bases = [f"{d}{d}" for d in "0123456789"]
    acc: List[Dict] = []
    for b in bases:
        per_state = []
        for state, draws in state_draws.items():
            ds = draws_since_last_double_base(draws, b)
            per_state.append((state, ds))
        per_state.sort(key=lambda x: -x[1])  # longest gap first
        max_gap = per_state[0][1]
        leaders = [s for s, ds in per_state if ds == max_gap]
        acc.append({
            "Base": b, "MaxGap": max_gap, "Leaders": ", ".join(leaders),
            "MedianGap": sorted(ds for _, ds in per_state)[len(per_state)//2],
            "MeanGap": round(sum(ds for _, ds in per_state)/max(1,len(per_state)), 1)
        })
    acc.sort(key=lambda x: -x["MaxGap"])
    return acc

# modules/blackapple.py  (v1.2)
from typing import List, Dict, Set

# ------------------- Config knobs -------------------
# Coverage vs. precision
USE_FOUNDATION_AS_HARD_BASE = False   # False = HYBRID (wide search), True = STRICT (foundation-only)

# Trigger thresholds
FLOATING_OUT_THRESHOLD    = 5         # digits absent >= this many draws are "floating"
EXTREME_PATTERN_OVERDUE   = 25        # SSS or TTT gap
MIXED_GROUP_OVERDUE       = 12        # no SST/STS/TSS for this many draws

# Remaining-pairs method
REMAINING_PAIRS_TARGET    = 28        # aim 27–29 left after cross-offs
TOP_N_CANDIDATES          = 12

# Mirror lookback (how many recent draws count as a "mirror event")
MIRROR_LOOKBACK           = 1         # 1 = just the last draw; raise to 2–3 if you want a wider mirror window

# Weights for scoring (Hybrid mode)
WEIGHTS = dict(
    PAIR_FOUNDATION = 2,  # bonus if combo's 3 pairs ⊆ remaining set
    MIR = 2,              # combo has a mirror pair when a mirror event is active
    RS  = 2,              # combo's root sum in due set
    PAT = 1,              # combo matches due pattern bucket
    FLT = 1,              # combo contains any floating digit
    SD  = 1,              # (optional) combo's short-sum in due set (currently off by default)
)

SPIKES  = set("234567")
TROUGHS = set("0189")
MIRROR  = {"0":"5","1":"6","2":"7","3":"8","4":"9","5":"0","6":"1","7":"2","8":"3","9":"4"}

# ---------- sums helpers ----------
def _sum_digits(combo: str) -> int: return sum(int(c) for c in combo)

def _digital_root_sum(combo: str) -> int:
    s = _sum_digits(combo)
    while s > 9: s = sum(int(c) for c in str(s))
    return s

def _short_sum(combo: str) -> int: return _sum_digits(combo) % 10

# ---------- pattern / mirror ----------
def _st_pattern(combo: str) -> str:
    return ''.join('S' if d in SPIKES else 'T' for d in combo)

def _has_mirror_pair(combo: str) -> bool:
    s = set(combo)
    for d in s:
        if MIRROR[d] in s and MIRROR[d] != d:
            return True
    return False

def _had_mirror_in_last_k(draws: List[str], k: int) -> bool:
    upto = min(k, len(draws))
    for i in range(upto):
        d = draws[i]
        if len(d) == 3 and _has_mirror_pair(d):
            return True
    return False

# ---------- pairs universe ----------
def _pairs_from_draw(draw: str) -> Set[str]:
    a,b,c = draw[0], draw[1], draw[2]
    return { ''.join(sorted(ab)) for ab in [a+b, b+c, a+c] }

def _all_pairs_universe() -> Set[str]:
    out = set()
    for i in range(10):
        for j in range(i,10):
            out.add(f"{i}{j}")   # 45 non-repeat + 10 doubles = 55 pairs
    return out

def _remaining_pairs(draws: List[str], target: int) -> Set[str]:
    """Cross off pairs newest->older until ~target remain (27–29 method)."""
    U = _all_pairs_universe().copy()
    for d in draws:
        if len(U) <= target: break
        if len(d) != 3: continue
        for p in _pairs_from_draw(d):
            U.discard(p)
    return U

def _singles_from_pairs(remaining_pairs: Set[str]) -> List[str]:
    """Boxed singles whose 3 internal pairs are all in remaining_pairs."""
    combos = []
    for i in range(10):
        for j in range(i+1,10):
            for k in range(j+1,10):
                a,b,c = str(i), str(j), str(k)
                if {a+b, a+c, b+c} <= remaining_pairs:
                    combos.append(a+b+c)
    return combos

def _all_singles() -> List[str]:
    """All 120 boxed singles (three distinct digits)."""
    return [f"{i}{j}{k}"
            for i in range(10)
            for j in range(i+1,10)
            for k in range(j+1,10)]

# ---------- draws-since helpers ----------
def _longest_out_root_sums(draws: List[str]) -> List[int]:
    last = {r: len(draws) for r in range(10)}
    for i, d in enumerate(draws):
        if len(d) != 3: continue
        r = _digital_root_sum(d)
        if last[r] == len(draws): last[r] = i
    max_gap = max(last.values())
    return [r for r,g in last.items() if g == max_gap]

def _longest_out_short_sums(draws: List[str]) -> List[int]:
    last = {s: len(draws) for s in range(10)}  # sD 0..9
    for i, d in enumerate(draws):
        if len(d) != 3: continue
        sD = (int(d[0])+int(d[1])+int(d[2])) % 10
        if last[sD] == len(draws): last[sD] = i
    max_gap = max(last.values())
    return [s for s,g in last.items() if g == max_gap]

def _pattern_draws_since(draws: List[str]) -> Dict[str, int]:
    patterns = ["SSS","SST","STS","TSS","TTS","TST","STT","TTT"]
    last = {p: len(draws) for p in patterns}
    for i, d in enumerate(draws):
        if len(d) != 3: continue
        p = _st_pattern(d)
        if last[p] == len(draws): last[p] = i
    return last

def _group_draws_since_sst_sts_tss(draws: List[str]) -> int:
    last = len(draws)
    for i, d in enumerate(draws):
        if len(d) != 3: continue
        if _st_pattern(d) in {"SST","STS","TSS"}:
            last = i
            break
    return last

def _floating_digits(draws: List[str], threshold: int) -> List[str]:
    last = {str(d): len(draws) for d in range(10)}
    for i, d in enumerate(draws):
        if len(d) != 3: continue
        for ch in set(d):
            if last[ch] == len(draws):
                last[ch] = i
    return [d for d,g in last.items() if g >= threshold]

# ---------- main ----------
def analyze_blackapple(draws: List[str]) -> Dict:
    if not draws:
        return {"score": 0, "triggers": {}, "candidates": []}

    # Mirror event in last K draws?
    mirror_event = _had_mirror_in_last_k(draws, MIRROR_LOOKBACK)

    # Due sets
    due_roots = _longest_out_root_sums(draws)
    pat_ds    = _pattern_draws_since(draws)
    extreme_due = (pat_ds["SSS"] >= EXTREME_PATTERN_OVERDUE) or (pat_ds["TTT"] >= EXTREME_PATTERN_OVERDUE)
    mixed_due   = (_group_draws_since_sst_sts_tss(draws) >= MIXED_GROUP_OVERDUE)
    floating    = _floating_digits(draws, FLOATING_OUT_THRESHOLD)
    floating_on = (len(floating) >= 2)

    # Remaining-pairs foundation
    rem_pairs  = _remaining_pairs(draws, REMAINING_PAIRS_TARGET)
    foundation = _singles_from_pairs(rem_pairs)

    # Optional short-sum due (kept off unless you enable in scoring)
    due_short  = _longest_out_short_sums(draws)

    # Triggers summary
    triggers = {
        "mirror": mirror_event,
        "root_due": due_roots,
        "pattern": {"extreme_due": extreme_due, "mixed_due": mixed_due},
        "floating": floating,
        "pairs": {"remaining_count": len(rem_pairs)}
    }

    # BA-Score = count of active trigger families
    score = (1 if mirror_event else 0) \
          + (1 if due_roots else 0) \
          + (1 if (extreme_due or mixed_due) else 0) \
          + (1 if floating_on else 0) \
          + (1 if len(rem_pairs) <= REMAINING_PAIRS_TARGET + 2 else 0)

    # Choose base set
    base = foundation if USE_FOUNDATION_AS_HARD_BASE else _all_singles()

    # Build candidates with weighted scoring
    candidates = []
    floating_set = set(floating)
    for combo in base:
        tags  = set()
        score_c = 0

        # Foundation bonus (even in HYBRID mode)
        if {combo[0]+combo[1], combo[0]+combo[2], combo[1]+combo[2]} <= rem_pairs:
            tags.add("PAIR"); score_c += WEIGHTS["PAIR_FOUNDATION"]

        if mirror_event and _has_mirror_pair(combo):
            tags.add("MIR");  score_c += WEIGHTS["MIR"]

        if due_roots and _digital_root_sum(combo) in set(due_roots):
            tags.add("RS");   score_c += WEIGHTS["RS"]

        if (extreme_due or mixed_due):
            pat = _st_pattern(combo)
            if (extreme_due and pat in {"SSS","TTT"}) or (mixed_due and pat in {"SST","STS","TSS"}):
                tags.add("PAT"); score_c += WEIGHTS["PAT"]

        if floating_on and any(d in floating_set for d in combo):
            tags.add("FLT");  score_c += WEIGHTS["FLT"]

        # Optional short-sum due
        # if _short_sum(combo) in set(due_short):
        #     tags.add("SD");   score_c += WEIGHTS["SD"]

        if score_c > 0:
            candidates.append({"combo": combo, "score": score_c, "tags": tags})

    candidates.sort(key=lambda x: (-x["score"], x["combo"]))
    return {"score": score, "triggers": triggers, "candidates": candidates[:TOP_N_CANDIDATES]}

def ba_status_label(score: int) -> str:
    if score >= 3: return "ALERT"
    if score == 2: return "WATCH"
    return "OFF"

def sum_tags(combo: str) -> Dict[str,int]:
    s = _sum_digits(combo)
    return {"Sigma": s, "sD": s % 10, "RS": _digital_root_sum(combo)}
