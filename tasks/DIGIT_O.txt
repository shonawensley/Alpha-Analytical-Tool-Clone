# scripts/experiments/dr_make_playset.py
import argparse, pandas as pd
from pathlib import Path
from yaml import safe_load

VMAP={"0":"1","5":"1","1":"2","6":"2","2":"3","7":"3","3":"4","8":"4","4":"5","9":"5"}
def to_vtrac(s): return "".join(VMAP.get(ch,"") for ch in "".join(sorted(str(s))))

def main():
    ap=argparse.ArgumentParser()
    ap.add_argument("stamp"); ap.add_argument("--analysis-root", default="data/outputs/analysis/digit_reduction")
    ap.add_argument("--config", default="alpha_analytical/digit_reduction/analyzer_v2/config.yml")
    ap.add_argument("--states", nargs="*")
    ap.add_argument("--max-n", type=int, default=10)
    args=ap.parse_args()

    cfg=safe_load(Path(args.config).read_text())
    # resolve score_field as in validator
    def score_field(df):
        if cfg.get("lockscore",{}).get("enabled") and cfg["lockscore"].get("use_for_top") and "lockscore_prob" in df.columns: return "lockscore_prob"
        if cfg.get("scoring_v2",{}).get("enabled") and cfg["scoring_v2"].get("use_for_top") and "score_v2" in df.columns: return "score_v2"
        if cfg.get("scoring_linear",{}).get("enabled") and cfg["scoring_linear"].get("use_for_top") and "final_prob" in df.columns: return "final_prob"
        return "score"

    out_root=Path("reports/DR")/args.stamp/"playsets"; out_root.mkdir(parents=True, exist_ok=True)
    for state_dir in sorted((Path(args.analysis_root)).glob("*")):
        state=state_dir.name
        if args.states and state not in args.states: continue
        p=state_dir/"analyzer_v2"/f"{state}_analyzer_v2_per_item.csv"
        if not p.exists(): continue
        df=pd.read_csv(p, dtype=str).fillna("")
        df["final_value"]=df["final_value"].str.replace(r"\D","",regex=True).str.zfill(3)
        df["cand_vtrac"]=df["final_value"].map(to_vtrac)
        key=score_field(df)
        for variant, block in df.groupby(df["section"].str.upper()):
            fam = block.sort_values([key,"score"], ascending=False).groupby("cand_vtrac").head(50)
            if fam.empty: continue
            top_family=fam.iloc[0]["cand_vtrac"]
            pick = (block[block["cand_vtrac"]==top_family]
                    .sort_values([key,"score"], ascending=False)
                    .drop_duplicates(subset=["final_value"]).head(args.max_n))
            if not pick.empty:
                pick.to_csv(out_root/f"{state}_{variant}_playset.csv", index=False)

if __name__=="__main__": main()
