# at top
import json

def _write_metrics_json(out_dir, state, df_scores, df_families, df_raw=None, df_fam_spot=None, winners=None):
    out = {
        "state": state,
        "rows": int(len(df_scores or [])),
        "families": int(len(df_families or [])),
        "winners": list(winners or []),
        "spotlight_rate_raw": float((len(df_raw) / max(1, len(df_scores))) if df_raw is not None else 0.0),
        "spotlight_rate_families": float((len(df_fam_spot) / max(1, len(df_families))) if df_fam_spot is not None else 0.0),
        "top_winner_family_rank": _best_rank_for_winners(df_families, winners),
        "family_compression_ratio": float((len(df_families) / max(1, len(df_scores))) if (df_scores is not None and df_families is not None) else 0.0),
        "hot_density_top10_mean": float((df_families.sort_values("family_score", ascending=False)
                                         .head(10)["hot_density"].mean()) if df_families is not None and "hot_density" in df_families else 0.0),
    }
    metrics_path = out_dir / f"{state}_stable_patterns_metrics.json"
    metrics_path.write_text(json.dumps(out, indent=2))
    return metrics_path

def _best_rank_for_winners(df_families, winners):
    try:
        if df_families is None or not len(winners):
            return None
        ranks = []
        for fid in set(_winner_to_family_ids(winners)):  # your existing converter if any
            sub = df_families[df_families["family_id"] == fid]
            if not sub.empty:
                rank = int(sub["family_score"].rank(ascending=False, method="min").min())
                ranks.append(rank)
        return min(ranks) if ranks else None
    except Exception:
        return None
