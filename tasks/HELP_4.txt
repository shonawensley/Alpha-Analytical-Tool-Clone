# alpha_analytical/stable/post_pass_families.py (near top)
from collections import defaultdict
from itertools import combinations, permutations

# must match the extractor's map
DIGIT2V = {'0':1,'5':1,'1':2,'6':2,'2':3,'7':3,'3':4,'8':4,'4':5,'9':5}

def _rep3_from_classes(canon: str) -> str | None:
    """Derive a deterministic 3-digit representative for a 3-value canonical."""
    if not canon or len(canon) < 3 or not canon.isdigit():
        return None

    # group digits by V-class
    by_class: dict[int, list[str]] = defaultdict(list)
    for ch in canon:
        v = DIGIT2V.get(ch)
        if v is None:
            return None
        by_class[v].append(ch)

    classes = sorted(by_class.keys())
    k = len(classes)

    # smallest digit helper per class
    smallest = {c: sorted(by_class[c])[0] for c in classes}

    if k == 3:
        rep = sorted([smallest[c] for c in classes])
        return "".join(rep)

    if k == 2:
        # choose smallest from each class
        seeds = [smallest[classes[0]], smallest[classes[1]]]
        # duplicate the class with the larger count; tie -> smaller smallest-digit
        c0, c1 = classes
        cnt0, cnt1 = len(by_class[c0]), len(by_class[c1])
        if cnt0 > cnt1:
            dup = smallest[c0]
        elif cnt1 > cnt0:
            dup = smallest[c1]
        else:
            dup = min(smallest[c0], smallest[c1])
        rep = sorted(seeds + [dup])
        return "".join(rep)

    if k == 1:
        d = smallest[classes[0]]
        return d*3

    # if somehow more than 3 classes slipped through, bail
    return None

def derive_vtrac_index_for_canonical(canon: str, get_vtrac_index) -> int | None:
    """Robustly map any 3-value canonical (len>=3) to a V-Trac index."""
    if not canon or len(canon) < 3:
        return None
    rep3 = _rep3_from_classes(canon)
    if rep3:
        vti = get_vtrac_index(rep3)
        if vti is not None:
            return vti

    # fallback: try all 3-combinations of digits (with permutations)
    digits = list(canon)
    seen = set()
    for combo in combinations(digits, 3):
        for perm in permutations(combo, 3):
            s = "".join(perm)
            if s in seen: 
                continue
            seen.add(s)
            vti = get_vtrac_index(s)
            if vti is not None:
                return vti
    return None
