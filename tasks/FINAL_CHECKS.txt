def summarize_vtrac_indexes(vstat: dict) -> dict:
    """
    Build a simple summary over the VTRAC index stats:
    - ordered lists of most overdue / most recent indexes
    - a 'due' flag based on the SSOT threshold.
    """
    from core import aux_config as AC

    entries = []
    for idx, meta in vstat.items():
        # Let Codex inspect meta to find the real key name for draws-since.
        ds = meta.get("draws_since", meta.get("DS", 0))
        entries.append({"index": idx, "ds": ds})

    # Filter out indexes with missing ds
    entries = [e for e in entries if e["ds"] is not None]

    overdue_sorted = sorted(entries, key=lambda e: e["ds"], reverse=True)
    recent_sorted  = sorted(entries, key=lambda e: e["ds"])

    top_overdue = overdue_sorted[:AC.VTRAC_INDEX_TOPK]
    top_recent  = recent_sorted[:AC.VTRAC_INDEX_TOPK]

    most_overdue = top_overdue[0] if top_overdue else None
    most_recent  = top_recent[0] if top_recent else None

    due_indices = [
        e for e in overdue_sorted
        if e["ds"] >= getattr(AC, "VTRAC_INDEX_DUE_THRESHOLD", e["ds"] + 1)
    ]

    return {
        "most_overdue": most_overdue,
        "most_recent": most_recent,
        "top_overdue": top_overdue,
        "top_recent": top_recent,
        "due_indices": due_indices,
    }
