# modules/vtrac_matchers.py
from itertools import permutations
import re
from typing import Iterable, Set, Tuple

_DIGITS = re.compile(r"\d")

def _digits_only(s: str) -> str:
    return re.sub(r"\D", "", s or "")

def _perms(d3: str) -> Set[str]:
    # Unique set of all 3-digit permutations (e.g., '894' -> {'894','849','984','948','489','498'})
    return {''.join(p) for p in set(permutations(d3, 3))}

def _gap1_patterns(pat: str) -> Tuple[re.Pattern, re.Pattern]:
    # For 'abc' allow exactly one extra digit either between a&b or b&c
    a, b, c = pat
    # Note: we search *within* the digits-only token, but we wrap the whole token for simplicity/robustness
    return (
        re.compile(fr"{a}\d{b}{c}"),  # a ? b c
        re.compile(fr"{a}{b}\d{c}"),  # a b ? c
    )

def tag_cell_html(cell_html: str, *, winner3: str, family3: Iterable[str]) -> str:
    """
    Scan the textual content of a table cell and wrap:
      - winner straights        -> <span class="hit-winner">...</span>
      - winner gap-1            -> <span class="hit-winner-gap">...</span>
      - family straights        -> <span class="hit-family">...</span>
      - family gap-1            -> <span class="hit-family-gap">...</span>
    Strategy: we split cell content into "digit tokens" vs "other" and only wrap entire digit tokens
    that contain the requested pattern (contiguous or gap-1).
    This is conservative but very fast and visually clear.
    """
    if not cell_html:
        return cell_html

    # Build detection sets/patterns
    w_perms = _perms(winner3)
    f_set   = set(family3)  # already canonical 3-digit codes

    # Precompile regexes for gap-1 of all perms
    w_gap = [gp for p in w_perms for gp in _gap1_patterns(p)]
    f_gap = [gp for p in f_set   for gp in _gap1_patterns(p)]

    # Split the cell into alternating non-digits / digits tokens so we can wrap digits safely
    parts = re.split(r"(\d+)", cell_html)  # keeps delimiters
    for i in range(1, len(parts), 2):  # every odd index is a digits-token
        token = parts[i]
        d = token  # already digits by split()

        # Winner contiguous?
        if any(p in d for p in w_perms):
            parts[i] = f'<span class="hit-winner">{token}</span>'
            continue
        # Winner gap-1?
        if any(rx.search(d) for rx in w_gap):
            parts[i] = f'<span class="hit-winner-gap">{token}</span>'
            continue
        # Family contiguous?
        if any(p in d for p in f_set):
            parts[i] = f'<span class="hit-family">{token}</span>'
            continue
        # Family gap-1?
        if any(rx.search(d) for rx in f_gap):
            parts[i] = f'<span class="hit-family-gap">{token}</span>'
            continue

    return "".join(parts)

# modules/vtrac_matchers.py
import re
from itertools import permutations
from typing import Dict, Iterable, List

_DIGITS_RE = re.compile(r"\d+")

def _digits_only(text: str) -> str:
    # pass 1: keep only digits, collapse separators
    return "".join(_DIGITS_RE.findall(text or ""))

def winner_perms(winner: str) -> List[str]:
    w = "".join(ch for ch in (winner or "") if ch.isdigit())[:3]
    return sorted({ "".join(p) for p in permutations(w, 3) })

def _gap1_patterns(seq: str) -> List[str]:
    # allow exactly one digit between neighbors (9.3 4 or 9 3.4)
    a,b,c = seq
    return [f"{a}\\d{b}{c}", f"{a}{b}\\d{c}"]

def build_winner_regexes(winner: str) -> Dict[str, List[re.Pattern]]:
    regs = {"win": [], "win_gap": []}
    for p in winner_perms(winner):
        regs["win"].append(re.compile(p))
        for pat in _gap1_patterns(p):
            regs["win_gap"].append(re.compile(pat))
    return regs

def build_family_regexes(family: Iterable[str]) -> Dict[str, List[re.Pattern]]:
    regs = {"fam": [], "fam_gap": []}
    uniq = sorted({ s for s in (family or []) if s and s.isdigit() and len(s)==3 })
    if not uniq: 
        return regs
    regs["fam"] = [re.compile("|".join(uniq))]
    gap_list = []
    for t in uniq:
        a,b,c = t[0], t[1], t[2]
        gap_list.append(f"{a}\\d{b}{c}")
        gap_list.append(f"{a}{b}\\d{c}")
    regs["fam_gap"] = [re.compile("|".join(gap_list))]
    return regs

def find_hits(cell_text: str, winner: str, family: Iterable[str]) -> Dict[str, List[str]]:
    """Double pass: raw cell, then digits‑only string."""
    out = {"win":[], "win_gap":[], "fam":[], "fam_gap":[]}

    joined = _digits_only(cell_text)
    wregs = build_winner_regexes(winner)
    fregs = build_family_regexes(family)

    for lbl, patterns in (("win", wregs["win"]), ("win_gap", wregs["win_gap"])):
        for rg in patterns:
            out[lbl].extend(rg.findall(joined))

    for lbl, patterns in (("fam", fregs["fam"]), ("fam_gap", fregs["fam_gap"])):
        for rg in patterns:
            out[lbl].extend(rg.findall(joined))

    # de‑dup and prefer longer matches first for safe replacement
    for k in out:
        out[k] = sorted(set(out[k]), key=len, reverse=True)
    return out

from modules.vtrac_reference import get_index_set  # your canonical API

def _annotate_cell(text: str, winner: str, index_set) -> str:
    hits = find_hits(text, winner, index_set)

    # overlay order: winner strict > winner gap > family strict > family gap
    def wrap(t, cls): return f'<span class="{cls}">{t}</span>'
