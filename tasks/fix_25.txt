# Immutable thresholds (keep these hard-coded)
THR_PENDING = 25

# Repeating: 00, 11, ...
THR_REP_BLUE = 71
THR_REP_RED  = 107

# Non-repeating: 01, 23, ...
THR_NONREP_BLUE = 37
THR_NONREP_RED  = 56

def band_repeating(ds: int) -> str | None:
    if ds >= THR_REP_RED:  return "red"     # 107+
    if ds >= THR_REP_BLUE: return "blue"    # 71..106
    if ds >= THR_PENDING:  return "purple"  # 25..70
    return None

def band_nonrepeating(ds: int) -> str | None:
    if ds >= THR_NONREP_RED:  return "red"     # 56+
    if ds >= THR_NONREP_BLUE: return "blue"    # 37..55
    if ds >= THR_PENDING:     return "purple"  # 25..36
    return None

No color logic involved.

3) Is hard‑coding better? (Yes—when it’s a policy)

Hard‑coding policy constants (like “Repeating BLUE = 71+”) is a good thing. It prevents “threshold drift,” keeps different pages consistent, and makes tests trivial.
