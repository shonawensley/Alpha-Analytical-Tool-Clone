6) Winners join helper (tiny utility to avoid drift)

Add a small helper so every winners surface can enrich with Stable evidence the same way:

# alpha_analytical/stable/winners_join.py
from __future__ import annotations
import pandas as pd
from alpha_analytical.vtrac import get_vtrac_index

def add_stable_evidence_to_winners(winners_df: pd.DataFrame, families_df: pd.DataFrame) -> pd.DataFrame:
    """
    winners_df: must have a 'Winner' column (string like '277')
    families_df: Stable families CSV
    Returns winners_df with family_id, family_score, section_count, progression_flag, last_remaining_3v, any_doubles_support, hot_density
    """
    winners = winners_df.copy()
    winners["family_id"] = winners["Winner"].map(get_vtrac_index)
    core = families_df[[
        "family_id","family_score","section_count","progression_flag","last_remaining_3v",
        "any_doubles_support","hot_density"
    ]].copy()
    # Pick top family row per family_id by score
    core = (core.sort_values("family_score", ascending=False)
                 .drop_duplicates(subset=["family_id"], keep="first"))
    return winners.merge(core, on="family_id", how="left")

3) “Winners Evidence Bus” (minimal, high‑leverage integration)

Create one tiny helper used by all winners tables to attach Stable evidence:

# alpha_analytical/stable/winners_enrich.py
from __future__ import annotations
import pandas as pd
from pathlib import Path

REQUIRED_STABLE_COLS = {
    "family_id","family_score","hot_density","section_count","progression_flag",
    "last_remaining_3v","any_doubles_support"
}

def attach_stable_evidence(winners_df: pd.DataFrame,
                           stable_families_csv: Path,
                           stable_scores_csv: Path | None = None) -> pd.DataFrame:
    fams = pd.read_csv(stable_families_csv)
    missing = REQUIRED_STABLE_COLS - set(fams.columns)
    if missing:
        raise ValueError(f"Stable families CSV missing: {sorted(missing)}")
    # Normalize columns we’ll use for joins
    fams["family_id"] = pd.to_numeric(fams["family_id"], errors="coerce").astype("Int64")
    out = winners_df.copy()
    # Expect winners_df to have either family_id or a canonical draw; if canonical, map to family_id from scores
    if "family_id" not in out.columns:
        if stable_scores_csv is None:
            raise ValueError("Need stable_scores_csv to map Canonical → family_id")
        scores = pd.read_csv(stable_scores_csv)[["Canonical","family_id","score"]]
        scores = scores.sort_values("score", ascending=False).drop_duplicates(["Canonical"])
        out = out.merge(scores, how="left", left_on="Canonical", right_on="Canonical")
    # Attach family evidence
    out = out.merge(
        fams[["family_id","family_score","hot_density","section_count","progression_flag",
              "last_remaining_3v","any_doubles_support"]],
        how="left", on="family_id"
    )
    return out
