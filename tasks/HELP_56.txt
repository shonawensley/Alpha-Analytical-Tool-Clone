2. Add a tiny debug / summary script for Stable

Here is a complete, no‑placeholder Python script you can hand to Codex or run yourself to sanity‑check any state’s Stable outputs:

import json
import sys
from pathlib import Path

import pandas as pd


def canonical(num_str: str) -> int:
    """Return the digit-sorted canonical triple as an int (e.g. '517' -> 157)."""
    return int("".join(sorted(num_str)))


def summarize_stable_state(state: str, root: str) -> None:
    base = Path(root)
    metrics_path = base / f"{state}_metrics.json"
    spot_raw_path = base / f"{state}_winner_family_spotlight_raw.csv"
    compound_path = base / f"{state}_stable_patterns_compound.csv"

    if not metrics_path.exists():
        raise FileNotFoundError(f"Missing metrics file: {metrics_path}")
    if not spot_raw_path.exists():
        raise FileNotFoundError(f"Missing spotlight file: {spot_raw_path}")
    if not compound_path.exists():
        raise FileNotFoundError(f"Missing compound file: {compound_path}")

    metrics = json.loads(metrics_path.read_text())
    winners = metrics["winners"]
    winner_fams = metrics.get("winner_family_ids", [])

    print(f"State: {state}")
    print(f"Winners: {winners}")
    print(f"Winner family_ids: {winner_fams}")

    canon_map = {w: canonical(w) for w in winners}
    print(f"Canonical mapping (literal -> canonical): {canon_map}")

    spot = pd.read_csv(spot_raw_path)
    comp = pd.read_csv(compound_path)

    for w in winners:
        c = canon_map[w]
        rows_spot = spot[spot["Canonical"] == c]
        rows_comp = comp[comp["Canonical"] == c]

        print(f"\nWinner {w} (canonical {c})")
        print(f"  Spotlight rows: {len(rows_spot)}")
        print(f"  Compound rows : {len(rows_comp)}")

        if not rows_spot.empty:
            eb = int(rows_spot["is_exact_boxed"].sum())
            es = int(rows_spot["is_exact_straight"].sum())
            vb = int(rows_spot["is_vtrac_boxed"].sum())
            sections = rows_spot["section"].value_counts().to_dict()
            print(f"  Spotlight flags: exact_boxed={eb}, exact_straight={es}, vtrac_boxed={vb}")
            print(f"  Spotlight sections: {sections}")

        if not rows_comp.empty:
            best_score = rows_comp["compound_score"].max()
            print(f"  Best compound_score: {best_score}")

    print()


def main() -> None:
    if len(sys.argv) < 3:
        msg = (
            "Usage: python summarize_stable_state.py <STATE_NAME> <OUTPUTS_ROOT>\n"
            "Example: python summarize_stable_state.py OntarioCanada4 "
            "sharepacks/2025-06-21/OntarioCanada4/stable/OntarioCanada4"
        )
        print(msg)
        sys.exit(1)

    state = sys.argv[1]
    root = sys.argv[2]
    summarize_stable_state(state, root)


if __name__ == "__main__":
    main()
