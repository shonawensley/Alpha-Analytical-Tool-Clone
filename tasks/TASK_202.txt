#!/usr/bin/env python3
# tools/make_pro_payload.py
# Universal "Pro Share" packer – V-TRAC implementation.
from __future__ import annotations
import json, csv, zipfile
from pathlib import Path
from datetime import datetime

VTRAC_VAL_BASE = Path("data/outputs/analysis/vtrac_validation")
ZIP_NAME = "vtrac_pro_payload.zip"

SECTIONS = ("Combined", "Midday", "Evening")

def _load_json(p: Path) -> dict:
    try:
        return json.loads(p.read_text(encoding="utf-8"))
    except Exception:
        return {}

def _overlap_tokens(sec: dict) -> list[str]:
    if not isinstance(sec, dict): return []
    winners_top = set((sec.get("signals", {}) or {}).get("top_vtrac_box_signatures", []) or [])
    analyzer    = set(sec.get("analyzer_signatures", []) or [])
    return sorted(winners_top & analyzer)

def _hot(sec: dict) -> tuple[int, int]:
    s = (sec.get("signals", {}) if isinstance(sec, dict) else {}) or {}
    return int(s.get("hot", 0)), int(s.get("superhot", 0))

def _consensus(sec: dict) -> tuple[bool, bool]:
    s = (sec.get("signals", {}) if isinstance(sec, dict) else {}) or {}
    return bool(s.get("consensus_col1", False)), bool(s.get("consensus_col2", False))

def _stable_cols(sec: dict) -> list[str]:
    s = (sec.get("signals", {}) if isinstance(sec, dict) else {}) or {}
    cols = s.get("stable_columns", []) or []
    return [str(c) for c in cols]

def _summarize_state(state_dir: Path) -> dict:
    rep = state_dir / "validation_report.json"
    data = _load_json(rep)
    state = data.get("state", state_dir.name)
    out = {
        "state": state,
        # combined fields will be filled in loop below
        "picked_html": data.get("picked_html", ""),
        "analyzer_json": data.get("analyzer_json", ""),
    }
    sections = data.get("sections", {}) if isinstance(data, dict) else {}
    for label in SECTIONS:
        sec = sections.get(label, {}) or {}
        key = label.lower()
        toks = _overlap_tokens(sec)
        hot, sh = _hot(sec)
        c1, c2 = _consensus(sec)
        stab = _stable_cols(sec)
        out[f"{key}_overlap_count"]  = len(toks)
        out[f"{key}_overlap_tokens"] = ",".join(toks) if toks else ""
        out[f"{key}_hot"]            = hot
        out[f"{key}_superhot"]       = sh
        out[f"{key}_consensus_c1"]   = c1
        out[f"{key}_consensus_c2"]   = c2
        out[f"{key}_stable_cols"]    = ",".join(stab) if stab else ""
    return out

def build_summaries(base: Path) -> tuple[Path, Path, list[dict]]:
    if not base.exists():
        raise SystemExit(f"Missing folder: {base}")
    rows = []
    for p in sorted(base.iterdir()):
        if p.is_dir() and (p / "validation_report.json").exists():
            rows.append(_summarize_state(p))

    # CSV
    csv_path = base / "summary.csv"
    fieldnames = ["state"]
    for s in SECTIONS:
        k = s.lower()
        fieldnames += [
            f"{k}_overlap_count", f"{k}_overlap_tokens",
            f"{k}_hot", f"{k}_superhot",
            f"{k}_consensus_c1", f"{k}_consensus_c2",
            f"{k}_stable_cols"
        ]
    fieldnames += ["picked_html","analyzer_json"]

    with csv_path.open("w", newline="", encoding="utf-8") as fh:
        w = csv.DictWriter(fh, fieldnames=fieldnames)
        w.writeheader()
        for r in rows:
            w.writerow(r)

    # Markdown
    md_path = base / "summary.md"
    lines = [ "# V-TRAC Validator Summary",
              f"_Generated: {datetime.utcnow().isoformat()}Z_",
              "" ]
    for r in rows:
        lines.append(f"## {r['state']}")
        for s in SECTIONS:
            k = s.lower()
            lines.append(
              f"- **{s}** overlap: {r[f'{k}_overlap_count']} "
              f"({r[f'{k}_overlap_tokens'] or '—'}); "
              f"hot/superhot: {r[f'{k}_hot']}/{r[f'{k}_superhot']}; "
              f"consensus c1/c2: {r[f'{k}_consensus_c1']}/{r[f'{k}_consensus_c2']}; "
              f"stable cols: {r[f'{k}_stable_cols'] or '—'}")
        lines.append(f"- Picked HTML: {r['picked_html'] or '—'}")
        lines.append(f"- Analyzer JSON: {r['analyzer_json'] or '—'}")
        lines.append("")
    md_path.write_text("\n".join(lines), encoding="utf-8")
    return csv_path, md_path, rows

def build_zip(base: Path, rows: list[dict]) -> Path:
    zip_path = base / ZIP_NAME
    with zipfile.ZipFile(zip_path, "w", compression=zipfile.ZIP_DEFLATED) as z:
        # always include summaries
        for p in (base/"summary.csv", base/"summary.md"):
            if p.exists(): z.write(p, p.relative_to(base.parent))
        # include root matrix/findings if present
        for name in ("matrix.csv","findings.md"):
            rp = base / name
            if rp.exists():
                z.write(rp, rp.relative_to(base.parent))
        # include per-state reports + referenced analyzer bundles when present
        for state_dir in sorted(base.iterdir()):
            if not state_dir.is_dir(): continue
            vrj = state_dir / "validation_report.json"
            vrm = state_dir / "validation_report.md"
            for p in (vrj, vrm):
                if p.exists(): z.write(p, p.relative_to(base.parent))
        # attempt to include analyzer_json files referenced in summary
        for r in rows:
            aj = r.get("analyzer_json")
            if aj:
                p = Path(aj)
                if not p.is_absolute():
                    p = base.parent / p  # allow relative paths
                if p.exists():
                    z.write(p, p.relative_to(base.parent))
    return zip_path

def main():
    VTRAC_VAL_BASE.mkdir(parents=True, exist_ok=True)
    csv_path, md_path, rows = build_summaries(VTRAC_VAL_BASE)
    zip_path = build_zip(VTRAC_VAL_BASE, rows)
    print(f"Wrote {csv_path}")
    print(f"Wrote {md_path}")
    print(f"Wrote {zip_path}")

if __name__ == "__main__":
    main()
