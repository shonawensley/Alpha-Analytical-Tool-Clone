# alpha_analytical/stable/compound.py
from __future__ import annotations
import pandas as pd

COMPOUND_COLUMNS = [
    'section','Canonical','family_id','compound_score','base_max_score',
    'set_chain_depth','draw_chain_depth','hot1_count','hot2_count','col1_hits',
    'consensus_hits','hidden3v_hits','vtrac_straight_hits','double_mirror_hits',
    'rows_covered','examples','compound_why'
]

def compute_compound_scores(rows: pd.DataFrame, feature_cfg: dict) -> pd.DataFrame:
    """
    Aggregate per-row evidence into one record per (section, Canonical).
    Expects Stable rows CSV columns:
      section, Set, Draw, Column, Canonical, score, family_id, hot,
      score_cons, hidden3v, score_vtrac_straight, score_double_mirror, why, ...
    Returns a DataFrame sorted by (section, compound_score desc, base_max_score desc).
    """
    if rows is None or len(rows) == 0:
        return pd.DataFrame(columns=COMPOUND_COLUMNS)

    df = rows.copy()
    # Normalize booleans
    df['hot_level'] = df.get('hot', 0)
    df['is_hot1'] = df['hot_level'].eq(1)
    df['is_hot2'] = df['hot_level'].eq(2)
    df['is_consensus'] = df.get('score_cons', 0).astype(float).gt(0) | \
                         df.get('cons_full', '').astype(str).str.upper().eq('Y') | \
                         df.get('cons_3v', '').astype(str).str.upper().eq('Y')
    df['is_hidden3v'] = df.get('hidden3v', '').astype(str).str.upper().eq('Y')
    df['is_vtrac_straight'] = df.get('score_vtrac_straight', 0).astype(float).gt(0)
    df['is_double_mirror'] = df.get('score_double_mirror', 0).astype(float).gt(0) | \
                             df.get('double_mirror', '').astype(str).str.upper().eq('Y')

    out_rows = []
    for (section, canonical), g in df.groupby(['section', 'Canonical'], sort=False):
        base_max = float(g['score'].max())
        sets = g['Set'].dropna().unique().tolist()
        set_chain = len(sets)
        draw_chain = int(g.loc[g['Set'].eq('Set1'), 'Draw'].nunique())
        col1_hits = int((g['Column'] == 1).sum())
        hot1 = int(g['is_hot1'].sum())
        hot2 = int(g['is_hot2'].sum())
        cons = int(g['is_consensus'].sum())
        hid  = int(g['is_hidden3v'].sum())
        vstr = int(g['is_vtrac_straight'].sum())
        dblm = int(g['is_double_mirror'].sum())

        w = feature_cfg or {}
        bonus = (
            w.get('compound.set_chain_bonus', 2.0) * max(0, set_chain - 1) +
            w.get('compound.draw_chain_bonus', 0.5) * draw_chain +
            w.get('compound.col1_bonus', 1.0) * col1_hits +
            w.get('compound.hot1_bonus', 1.0) * hot1 +
            w.get('compound.hot2_bonus', 2.0) * hot2 +
            w.get('compound.consensus_bonus', 1.0) * cons +
            w.get('compound.hidden_core_bonus', 1.0) * hid +
            w.get('compound.vtrac_straight_bonus', 0.5) * vstr +
            w.get('compound.double_mirror_bonus', 0.5) * dblm
        )
        score_compound = round(base_max + bonus, 3)

        why_bits = []
        if set_chain > 1:     why_bits.append(f"set_chain{set_chain}")
        if draw_chain > 0:    why_bits.append(f"draw_chain{draw_chain}")
        if col1_hits:         why_bits.append(f"col1x{col1_hits}")
        if hot1:              why_bits.append(f"hot1x{hot1}")
        if hot2:              why_bits.append(f"hot2x{hot2}")
        if cons:              why_bits.append(f"consensusx{cons}")
        if hid:               why_bits.append(f"hidden3vx{hid}")
        if vstr:              why_bits.append(f"vstrx{vstr}")
        if dblm:              why_bits.append(f"dblmirrorx{dblm}")

        out_rows.append({
            'section': section,
            'Canonical': str(canonical),
            'family_id': int(g['family_id'].dropna().astype(int).mode().iat[0]) if 'family_id' in g and not g['family_id'].dropna().empty else None,
            'compound_score': score_compound,
            'base_max_score': base_max,
            'set_chain_depth': int(set_chain),
            'draw_chain_depth': int(draw_chain),
            'hot1_count': hot1,
            'hot2_count': hot2,
            'col1_hits': col1_hits,
            'consensus_hits': cons,
            'hidden3v_hits': hid,
            'vtrac_straight_hits': vstr,
            'double_mirror_hits': dblm,
            'rows_covered': int(len(g)),
            'examples': ";".join(g.sort_values('score', ascending=False).head(2)['why'].astype(str).tolist()),
            'compound_why': "|".join(why_bits)
        })
    return pd.DataFrame(out_rows).sort_values(['section','compound_score','base_max_score'], ascending=[True, False, False]).reset_index(drop=True)

from alpha_analytical.stable.compound import compute_compound_scores

def test_compound_bonus_and_sorting():
    rows = pd.DataFrame([
        # Same Canonical across sets/draws with hot1/hot2, col1 hits, etc.
        dict(section='Combined', Set='Set1', Draw='Draw5', Column=1, Canonical='244', family_id=355,
             score=30.0, hot=2, score_cons=1.0, hidden3v='Y', score_vtrac_straight=0.5, score_double_mirror=0.0, why='example A'),
        dict(section='Combined', Set='Set1', Draw='Draw3', Column=2, Canonical='244', family_id=355,

        dict(section='Combined', Set='Set2', Draw='Draw1', Column=1, Canonical='244', family_id=553,
             score=25.0, hot=1, score_cons=0.0, hidden3v='Y', score_vtrac_straight=0.5, score_double_mirror=0.0, why='example C'),
    ])
    cfg = {
        'compound.set_chain_bonus': 2.0,
        'compound.draw_chain_bonus': 0.5,
        'compound.col1_bonus': 1.0,
        'compound.hot1_bonus': 1.0,
        'compound.hot2_bonus': 2.0,
        'compound.consensus_bonus': 1.0,
        'compound.hidden_core_bonus': 1.0,
        'compound.vtrac_straight_bonus': 0.5,
        'compound.double_mirror_bonus': 0.5,

    df = compute_compound_scores(rows, cfg)
    r = df.iloc[0]
    assert r['Canonical'] == '244'

    assert abs(r['compound_score'] - 44.5) < 1e-6
    assert r['set_chain_depth'] == 2
    assert r['draw_chain_depth'] == 2
    assert r['hot2_count'] == 1
