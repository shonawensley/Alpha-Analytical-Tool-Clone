Minimal, safe UI hook (DEV-only expander)

Add this small block to the Dig# inside show_digit_reduction_page(state) near the bottom
import streamlit as st
from pathlib import Path

with st.expander("Analyzer V2 (DEV)", expanded=False):
    st.caption("Runs the unified analyzer and writes CSV/JSON beside the reducer outputs.")

    if st.button("Run Analyzer V2 for this state", type="primary"):
        try:
            # SSOT path: data/outputs/analysis/
            from utils.path_handler import get_analysis_output_dir
            from alpha_analytical.digit_reduction.analyzer_v2 import run as run_v2

            analysis_root = Path(get_analysis_output_dir())  # → data/outputs/analysis
            info = run_v2(state, analysis_root=analysis_root)

            st.success(f"Wrote {info.get('rows',0)} rows → {info.get('out_dir','')}")
            out_dir = Path(info["out_dir"])
            # quick links
            for patt in (
                f"{state}_analyzer_v2_per_item.csv",
                f"{state}_analyzer_v2_own_vs_combined_delta.csv",
                f"{state}_analyzer_v2_meta.json",
            ):
                p = out_dir / patt
                if p.exists():
                    st.markdown(f"• [{p.name}]({p.as_posix()})")
        except Exception as e:
            st.error(f"Analyzer V2 failed: {e}")
