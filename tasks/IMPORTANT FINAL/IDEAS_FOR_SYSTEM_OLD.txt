def analyze_gaps(draws_df):
    last_seen = {i: None for i in range(1, 36)}
    gaps_record = {i: [] for i in range(1, 36)}

    for row in draws_df.itertuples(index=False):
        draw_num, pick3_str = row.draw_number, row.pick3_result
        idx = pick3_to_index(pick3_str)
        if idx:
            if last_seen[idx] is not None:
                gaps_record[idx].append(draw_num - last_seen[idx])
            last_seen[idx] = draw_num
    return gaps_record

class ProfitabilityMetrics:
    def __init__(self, payout, combo_count, ticket_cost):
        self.payout = payout
        self.combo_count = combo_count
        self.ticket_cost = ticket_cost
        self.history = []

    def log_result(self, draws_played, hit_occurred):
        expense = draws_played * self.combo_count * self.ticket_cost
        profit = self.payout - expense if hit_occurred else -expense
        self.history.append({
            'draws_played': draws_played,
            'expense': expense,
            'profit': profit,
            'hit': hit_occurred
        })

    def get_performance_summary(self):
        total_profit = sum(item['profit'] for item in self.history)
        hits = sum(1 for item in self.history if item['hit'])
        total_rounds = len(self.history)
        return {
            'total_profit': total_profit,
            'hit_rate': hits / total_rounds if total_rounds else 0,
            'avg_profit_per_round': total_profit / total_rounds if total_rounds else 0
        }

# Usage Example:
boxed_metric = ProfitabilityMetrics(payout=160, combo_count=8, ticket_cost=1)
boxed_metric.log_result(draws_played=10, hit_occurred=True)
summary = boxed_metric.get_performance_summary()
