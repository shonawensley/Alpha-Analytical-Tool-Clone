# src/core/vtrac_families.py
from collections import defaultdict
from typing import Dict, Iterable, List, Tuple, Set

# 0↔5, 1↔6, 2↔7, 3↔8, 4↔9
MIRROR = {0:5, 1:6, 2:7, 3:8, 4:9, 5:0, 6:1, 7:2, 8:3, 9:4}

def mirror(d: int) -> int:
    return MIRROR[d]

def pair_id(d: int) -> int:
    """Use the smaller digit of a mirror pair as the pair id (0,1,2,3,4)."""
    m = MIRROR[d]
    return min(d, m)

def canon_combo(digs: Iterable[int]) -> str:
    return ''.join(str(x) for x in sorted(digs))

def is_double_combo(combo: str) -> bool:
    ds = list(map(int, combo))
    return len(set(ds)) == 2 and (ds.count(ds[0]) == 2 or ds.count(ds[1]) == 2 or ds.count(ds[2]) == 2)

def family_key(combo: str) -> Tuple[int, int]:
    """Key identifying the family as a pair of mirror-pair ids (sorted)."""
    a, b, c = sorted(map(int, combo))
    # Find repeated digit r and singleton o
    r = a if a == b else c
    o = c if a == b else a
    return tuple(sorted((pair_id(r), pair_id(o))))

def family_members(combo: str) -> List[str]:
    """
    All 8 boxed doubles in the same V-TRAC double family as `combo`,
    returned as canonical sorted strings.
    """
    a, b, c = sorted(map(int, combo))
    r = a if a == b else c
    o = c if a == b else a
    A = {r, mirror(r)}
    B = {o, mirror(o)}
    fam: Set[str] = set()
    for rr in A:
        for oo in B:
            fam.add(canon_combo((rr, rr, oo)))
    for rr in B:
        for oo in A:
            fam.add(canon_combo((rr, rr, oo)))
    return sorted(fam)

# ---------- Ranking for Control Center ----------

# Expected input record per combo (you already have these in Aux analysis):
# { 'combo': '044', 'variant': 'Combined'|'Midday'|'Evening',
#   'status': 'red'|'blue', 'draws_since': int, 'index': int }

def rank_double_families(
    items: Iterable[Dict],
    hot_indices: Set[int] = frozenset(),
    top_n: int = 5
) -> List[Dict]:
    """
    Group red/blue doubles into families and compute a simple score per family.
    Returns top_n families with their badges ready for rendering.
    """
    buckets: Dict[Tuple[int,int], Dict] = defaultdict(lambda: {
        'score': 0, 'badges': set(), 'combos': set(), 'ds': []
    })

    for rec in items:
        combo = rec['combo']
        if not is_double_combo(combo):
            continue
        key = family_key(combo)
        color = rec.get('status', '').lower()
        variant = rec.get('variant', '')
        ds = int(rec.get('draws_since', 0))
        idx = rec.get('index', None)

        w_color = 3 if color == 'red' else 1 if color == 'blue' else 0
        v_var   = 1 if variant == 'Combined' else 0
        r_rep   = 1 if (idx is not None and idx in hot_indices) else 0

        buckets[key]['score'] += (w_color + v_var + r_rep)
        buckets[key]['badges'].add(f"{combo}{variant[:1] or ''}")  # e.g., 044E
        buckets[key]['combos'].add(combo)
        buckets[key]['ds'].append(ds)

    # small recency bonus if member ds >= family median
    for key, obj in buckets.items():
        if not obj['ds']:
            continue
        sorted_ds = sorted(obj['ds'])
        med = sorted_ds[len(sorted_ds)//2]
        obj['score'] += sum(1 for v in obj['ds'] if v >= med)

    rows = []
    for key, obj in buckets.items():
        rows.append({
            'family': key,                        # e.g., (0,4) means {0,5}×{4,9}
            'members': sorted(obj['badges']),     # badges like 044E, 599C ...
            'score': obj['score'],
            'sample': sorted(obj['combos'])[:3],  # optional preview
        })

    rows.sort(key=lambda r: (-r['score'], r['family']))
    return rows[:top_n]

1) What to call this

Twin‑Index Double Family (TIDF) — a pair of V‑TRAC indices of the form iij and jji (with i≠j) that share the same two digit‑classes but flip which class carries the double.

Example (your case): digit‑classes 1={0,5} and 5={4,9}

Index 115 (double from class 1): 004 009 455 559

Index 155 (double from class 5): 044 099 445 599
Together they form one TIDF: {1,5} ↔ (115, 155).

Why it matters: a TIDF collapses “nearly‑identical” doubles that differ only by which side is doubled. That reduces coverage and often co‑moves in time (you’ll see both sides go late/very‑late together more often than chance, because they live in the same two class bands).

2) How to detect it (and where to use it)

Digit‑class map (V‑TRAC 5‑way):

1 → {0,5}

2 → {1,6}

3 → {2,7}

4 → {3,8}

5 → {4,9}

Definition (computable):

For an unordered class pair {i,j} (with i<j), build two indices:

iij = doubles from class i with a single from class j

jji = doubles from class j with a single from class i

Each side has exactly 4 boxed doubles (2 choices for the doubled digit × 2 choices for the single).

CLASSES = {
    1: (0,5), 2: (1,6), 3: (2,7), 4: (3,8), 5: (4,9)
}

def family_for_classes(i, j):
    # returns dict with indices and 8 boxed doubles split by side
    i, j = sorted((i, j))
    def canon(a,b,c): return ''.join(map(str, sorted((a,b,c))))
    doubles_i = sorted({canon(da,da,db) for da in CLASSES[i] for db in CLASSES[j]})
    doubles_j = sorted({canon(db,db,da) for da in CLASSES[i] for db in CLASSES[j]})
    triple_iij = f"{i}{i}{j}"  # index label, you already know how to map to row #
    triple_jji = f"{j}{j}{i}"
    return {
        "classes": (i, j),
        "iij": triple_iij, "jji": triple_jji,
        "doubles_i": doubles_i, "doubles_j": doubles_j
    }
