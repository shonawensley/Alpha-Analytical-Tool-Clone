Pipeline hook (add to alpha_analytical/digit_reduction/analyzer_v2/pipeline.py):

def _apply_post_score(rows, cfg):
    s2 = cfg.get("scoring_v2", {})
    if not s2.get("enabled", False):
        return
    w = s2.get("weights", {})
    guards = s2.get("guards", {})
    for r in rows:
        base = r.get("score_raw", r.get("score", 0.0))
        score2 = float(base)

        # Recency & persistence
        if r.get("recency_carryover"): score2 += w.get("recency_bonus", 0.0)
        fam_persist = float(r.get("persistence_family_exact", 0) or 0) + float(r.get("persistence_family_vtrac", 0) or 0)
        score2 += w.get("family_persistence", 0.0) * fam_persist

        # Single-column rescue (when evidence is concentrated but dense)
        if int(r.get("cols_hit", 0) or 0) == 1 and float(r.get("box_family_density", 0) or 0) >= w.get("density_min", 0.0):
            score2 += w.get("singlecol_rescue", 0.0)

        # Late arrival penalty
        late = min(7, max(0, int(r.get("earliest_exact_step", 7) or 7)))
        score2 -= w.get("late_penalty", 0.0) * (late / 7.0)

        # V-HOT contribution
        try:
            score2 += w.get("vhot", 0.0) * float(r.get("vtrac.v_hot", 0) or 0)
        except Exception:
            pass

        # Guards when a row literally matches the day’s winner flags
        if r.get("final_exact_match"):  score2 += guards.get("boost_exact_match", 0.0)
        if r.get("final_vtrac_match"):  score2 += guards.get("boost_vtrac_match", 0.0)

        r["score_v2"] = round(score2, 6)

# call it just after you finish computing rows/top rows, before ranking:

def _top_candidates(rows, cfg=None):

    key = "score_v2" if use_v2 and any("score_v2" in r for r in rows) else "score"

scripts/experiments/digit_reduction_validate.py

#!/usr/bin/env python3
import csv, json, sys
from pathlib import Path
import pandas as pd
import yaml

# ---- Repo contracts & paths ----
# Analyzer outputs live under data/outputs/analysis/digit_reduction/<STATE>/analyzer_v2/
# Winners are centralized by Control Center (see KIT docs); we accept either a CSV (date,state,variant,win) OR existing map/flags CSVs.

AN_ROOT = Path("data/outputs/analysis/digit_reduction")
REPORT_ROOT = Path("reports/DR")

def _per_item_path(state: str) -> Path:
    return AN_ROOT / state / "analyzer_v2" / f"{state}_analyzer_v2_per_item.csv"

def _top_path(state: str) -> Path:
    return AN_ROOT / state / "analyzer_v2" / f"{state}_analyzer_v2_top_candidates.csv"

def _load_csv(p: Path) -> pd.DataFrame:
    if not p.exists():
        raise FileNotFoundError(p)
    return pd.read_csv(p)

def _canon3(s: str) -> str:
    s = "".join(ch for ch in str(s) if ch.isdigit())
    return "".join(sorted(s)) if len(s)==3 else ""

def _family_match(row_canon: str, win_canon: str) -> bool:
    # boxed family match (same sorted triple)
    return row_canon == win_canon

def evaluate_state_variant(df_per_item: pd.DataFrame, win_str: str, variant: str):
    if df_per_item.empty:
        return None
    win_canon = _canon3(win_str)
    # choose score column for ranking
    key = "score_v2" if "score_v2" in df_per_item.columns else "score"
    # filter relevant variant
    sub = df_per_item[df_per_item["section"].str.lower() == variant.lower()]
    if sub.empty:
        return None

    sub = sub.copy()
    # canonicalize final triple value if present; else fallback to features_json/regex as needed
    if "final_value" in sub.columns:
        sub["canon"] = sub["final_value"].astype(str).str.replace(r"\D","", regex=True).apply(_canon3)
    else:
        sub["canon"] = ""  # guard; future parser could be added

    sub.sort_values(key, ascending=False, inplace=True, kind="mergesort")
    sub["rank"] = range(1, len(sub)+1)

    # metrics
    exact_hits = sub[(sub.get("final_exact_match", False)==True)]
    fam_hits   = sub[sub["canon"] == win_canon]

    def first_rank(frame: pd.DataFrame):
        return int(frame.iloc[0]["rank"]) if not frame.empty else None

    r_exact = first_rank(exact_hits)
    r_fam   = first_rank(fam_hits)

    k_list = [1,3,5,10]
    out = {f"recall_at_{k}": int((r_exact and r_exact<=k) or (r_fam and r_fam<=k)) for k in k_list}
    out.update({
        "mrr": (1.0 / min([r for r in [r_exact, r_fam] if r])) if any([r_exact, r_fam]) else 0.0,
        "rank_exact": r_exact or "",
        "rank_family": r_fam or "",
        "top_score": sub.iloc[0][key] if len(sub) else "",
        "win_row_score": float(sub.iloc[r_exact-1][key]) if r_exact else (float(sub.iloc[r_fam-1][key]) if r_fam else ""),
        "gap_to_top": (float(sub.iloc[0][key]) - float(sub.iloc[r_exact-1][key])) if r_exact else (float(sub.iloc[0][key]) - float(sub.iloc[r_fam-1][key]) if r_fam else "")
    })

    # top near-misses around the winner family (helps tuning)
    near = sub.head(10)[["rank","section","set","draw","col_rank","score","score_v2"] if "score_v2" in sub.columns else ["rank","section","set","draw","col_rank","score"]]
    return out, near

def main():
    if len(sys.argv) < 3:
        print("Usage: digit_reduction_validate.py <STAMP:YYYYMMDD> <winners.csv>", file=sys.stderr)
        sys.exit(2)

    stamp = sys.argv[1]
    winners_csv = Path(sys.argv[2])
    winners = pd.read_csv(winners_csv)

    report_dir = (REPORT_ROOT / stamp)
    report_dir.mkdir(parents=True, exist_ok=True)
    metrics_rows = []
    misses_rows  = []

    for _, row in winners.iterrows():
        state   = str(row["state"])
        variant = str(row["variant"])
        win     = str(row["win"])

        per_item = _per_item_path(state)
        if not per_item.exists():
            print(f"[SKIP] per_item not found for {state}", file=sys.stderr)
            continue

        df = _load_csv(per_item)
        eval_out = evaluate_state_variant(df, win, variant)
        if not eval_out:
            continue
        metrics, near = eval_out
        metrics_rows.append({
            "state": state, "variant": variant, "win": win, **metrics
        })
        near = near.copy()
        near.insert(0, "state", state)
        near.insert(1, "variant", variant)
        misses_rows.append(near)

    if metrics_rows:
        pd.DataFrame(metrics_rows).to_csv(report_dir / "digit_reduction_metrics.csv", index=False)
    if misses_rows:
        pd.concat(misses_rows, ignore_index=True).to_csv(report_dir / "digit_reduction_top_misses.csv", index=False)

    # summary markdown for humans
    md = []
    dfm = pd.DataFrame(metrics_rows)
    if not dfm.empty:
        rec1 = dfm["recall_at_1"].mean()
        rec3 = dfm["recall_at_3"].mean()
        rec5 = dfm["recall_at_5"].mean()
        mrr  = dfm["mrr"].mean()
        md += [
          f"# Digit-Reduction Performance (stamp {stamp})",
          "",
          f"- Recall@1: **{rec1:.2%}**  |  Recall@3: **{rec3:.2%}**  |  Recall@5: **{rec5:.2%}**",
          f"- Mean Reciprocal Rank: **{mrr:.3f}**",
          "",
          "## Notes",
          "- Score_v2 used when present; otherwise score.",
          "- Gaps in `digit_reduction_top_misses.csv` suggest where rescues/penalties will matter."
        ]
    else:
        md += ["# Digit-Reduction Performance", "", "_No rows evaluated (check paths and winners input)._"]
    (report_dir / "DR_Perf_Summary.md").write_text("\n".join(md), encoding="utf-8")

if __name__ == "__main__":
    main()

2) Optional: weight sweep (tiny grid) to auto‑pick better defaults

scripts/experiments/dr_grid_search.py

#!/usr/bin/env python3
import itertools, json, sys
from pathlib import Path
import yaml
import subprocess
import pandas as pd

CFG_PATH = Path("alpha_analytical/digit_reduction/analyzer_v2/config.yml")
VALIDATE = Path("scripts/experiments/digit_reduction_validate.py")

GRID = {
  "recency_bonus":    [0.3, 0.5, 0.7],
  "family_persistence":[0.15,0.20,0.30],
  "singlecol_rescue": [0.25,0.35,0.45],
  "density_min":      [0.35,0.40,0.45],
  "vhot":             [0.10,0.15,0.20],
  "late_penalty":     [0.20,0.30,0.40]
}

def write_cfg_variant(base, weights):
    base = dict(base)
    base.setdefault("scoring_v2", {}).setdefault("weights", {}).update(weights)
    CFG_PATH.write_text(yaml.safe_dump(base, sort_keys=False))

def main():
    if len(sys.argv) < 3:
        print("Usage: dr_grid_search.py <STAMP> <winners.csv>")
        sys.exit(2)
    stamp = sys.argv[1]
    winners_csv = sys.argv[2]
    base = yaml.safe_load(CFG_PATH.read_text())
    combos = list(itertools.product(*GRID.values()))
    keys = list(GRID.keys())
    results = []
    for vals in combos:
        w = dict(zip(keys, vals))
        write_cfg_variant(base, w)
        # rerun validator (expects analyzer artifacts already present)
        subprocess.check_call([sys.executable, str(VALIDATE), stamp, winners_csv])
        df = pd.read_csv(Path("reports/DR")/stamp/"digit_reduction_metrics.csv")
        score = df["recall_at_3"].mean() + 0.5*df["mrr"].mean()
        results.append((score, w))
    best = sorted(results, key=lambda x: x[0], reverse=True)[0]
    print("BEST:", best[1], "score=", best[0])

if __name__ == "__main__":
    main()
