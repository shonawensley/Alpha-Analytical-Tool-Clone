# --- Winners overlay search helpers (strict + 1-gap tolerant) ---
import re
from typing import Iterable, Tuple, List, Dict

_NON_DIGITS = re.compile(r"[^0-9]+")

def _digits(s: str) -> str:
    return _NON_DIGITS.sub("", str(s or ""))

def build_targets_for_winner(winner: str, vtrac_api) -> Dict[str, List[str]]:
    w = _digits(winner)
    straights = sorted({"".join(p) for p in set(__import__("itertools").permutations(w, 3))})
    idx = vtrac_api.get_vtrac_index(w)
    family = sorted(set(map(_digits, vtrac_api.get_index_set(idx))))
    return {"straights": straights, "family": family}

def compile_gap1_regexes(target: str) -> re.Pattern:
    # allow a single digit gap between any adjacent pair; keep order
    a, b, c = target[0], target[1], target[2]
    return re.compile(fr"{a}\d?{b}\d?{c}")

def find_hits(cell: str, targets: Iterable[str], allow_gap: bool) -> Tuple[set, set]:
    s = _digits(cell)
    strict, gap1 = set(), set()
    if not s:
        return strict, gap1
    for t in targets:
        if t in s:
            strict.add(t)
        elif allow_gap and compile_gap1_regexes(t).search(s):
            gap1.add(t)
    return strict, gap1

import re

_PUNCT_RE = re.compile(r"[^\d]+")  # anything that's not 0–9

def _digits_only(s: str) -> str:
    """Return just the digits from a cell (preserves order). Handles *, ' , thin spaces, etc."""
    if not s:
        return ""
    return _PUNCT_RE.sub("", str(s))


2) Build both contiguous and “gap‑1” regexes for the winner straights

from itertools import permutations

def _winner_regexes(winner: str):
    # All 3-digit straights for the winner
    perms = {''.join(p) for p in permutations(winner, 3)}
    # e.g., '934|943|349|394|439|493'
    strict = re.compile('|'.join(perms))
    # Allow a single *digit* between each pair: 9\d?3\d?4 etc.
    gap1  = re.compile('|'.join(f"{a}\\d?{b}\\d?{c}" for (a, b, c) in {tuple(p) for p in perms}))
    return strict, gap1


If the “purple” V‑Trac index set is also triads (it is in your doc), build the same pair of regexes for those combos once, not per‑cell.

3) Match strategy per cell

def _hits_cell(cell_text: str, strict_rx, gap1_rx) -> bool:
    s = _digits_only(cell_text)         # normalize once
    if not s:
        return False
    # Fast path: contiguous first
    if strict_rx.search(s):
        return True
    # Second pass: allow one extra digit interrupting the pattern
    return gap1_rx.search(s) is not None
