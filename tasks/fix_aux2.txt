B. Make the loader use it

modules/aux_loaders.py – minimal robust loader:

from pathlib import Path
import csv
from utils.path_handler import get_cleaned_draws_dir, get_cleaned_data_dir

def _normalize(label: str) -> str:
    s = label.strip().replace(" ", "").replace("_", "")
    return s[:-1] if s.endswith("4") else s  # Connecticut4 -> Connecticut

def _names(name: str, variant: str):
    v = variant.lower()
    if v == "combined":
        yield f"{name}_draws.csv"
    elif v == "midday":
        yield f"{name}_Midday_draws.csv"
    elif v == "evening":
        yield f"{name}_Evening_draws.csv"
    else:
        raise ValueError(f"unknown variant: {variant}")

def _read_csv(path: Path):
    out = []
    with path.open("r", encoding="utf-8", newline="") as f:
        for row in csv.reader(f):
            if row:
                s = row[0].strip()
                if s:
                    out.append(s)
    return out

def load_state_draws(state_label: str, variant: str = "combined"):
    name = _normalize(state_label)
    for root in (Path(get_cleaned_draws_dir()), Path(get_cleaned_data_dir())):  # new -> legacy
        for fn in _names(name, variant):
            p = root / fn
            if p.exists():
                return _read_csv(p), str(p)
    # last-resort glob
    for root in (Path(get_cleaned_draws_dir()), Path(get_cleaned_data_dir())):
        hits = list(root.glob(f"{name}*draws*.csv"))
        if hits:
            p = hits[0]
            return _read_csv(p), str(p)
    return [], f"NOT FOUND: {state_label} ({variant})"
