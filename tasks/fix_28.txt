# modules/winner_report_full.py
from __future__ import annotations
import os, re
from datetime import datetime
from typing import Dict, Set

from src.utils import path_handler as ph
from src.core import module_c_vtrac as vtrac   # analyzer module

def _winner_straights(win: str) -> Set[str]:
    win = (win or "").strip()
    a,b,c = win[0], win[1], win[2]
    return {a+b+c, a+c+b, b+a+c, b+c+a, c+a+b, c+b+a}  # 6 perms (if your map expects 8, include those too)

def _inject_green(html: str, straights: Set[str]) -> str:
    if not html or not straights:
        return html
    css = ('<style>.hit-straight{background:#d6f5d6;color:#0a7a0a;'
           'font-weight:600;border-radius:3px;padding:0 2px;}</style>')
    html = html.replace("</head>", css + "</head>") if "</head>" in html else css + html
    pat = re.compile(r"\b(\d{3})\b")
    return pat.sub(lambda m: f'<span class="hit-straight">{m.group(1)}</span>'
                             if m.group(1) in straights else m.group(0), html)

def _load_tables(state: str) -> Dict[str, "pd.DataFrame"]:
    # prefer analyzer helper if present
    loader = getattr(vtrac, "load_state_data", None)
    if callable(loader):
        return loader(state)
    # fallback: direct CSV read (strings only)
    import pandas as pd
    base = ph.get_tables_output_dir()                 # -> data/outputs/tables
    sdir = os.path.join(base, state)
    out = {}
    for key in ("Midday_combined","Evening_combined","Combined_combined"):
        p = os.path.join(sdir, f"{state}_{key}.csv")
        if os.path.exists(p): out[key] = pd.read_csv(p, dtype=str).fillna("")
    return out

def write_winner_full_report(state: str, winner: str) -> str:
    state = (state or "").strip()
    win   = (winner or "").strip()
    assert len(win) == 3 and win.isdigit(), "Winning number must be 3 digits"

    # 1) compute index + full pattern set like analyzer does
    idx_finder = getattr(vtrac, "find_vtrac_index_and_combos", None)
    if callable(idx_finder):
        info = idx_finder(win)
        idx  = info.get("index")
        pats = set(info.get("combos", []))
    else:
        from modules.vtrac_reference import get_vtrac_index, get_combos_for_index
        idx  = get_vtrac_index(win)
        pats = set(get_combos_for_index(idx))

    # 2) load tables as strings
    tables = _load_tables(state)
    if not tables:
        raise RuntimeError(f"No combined tables available for {state}")

    # 3) locate the analyzer renderer robustly
    _RENDER = None
    for name in ("generate_index_html_report","render_index_html_report",
                 "build_index_html_report","export_index_html_report"):
        f = getattr(vtrac, name, None)
        if callable(f):
            _RENDER = f; break
    if _RENDER is None:
        raise RuntimeError("Analyzer HTML renderer not found in src/core/module_c_vtrac.*")

    # 4) render + overlay green for the winner straights
    ts   = datetime.now().strftime("%Y%m%d_%H%M%S")
    html = _RENDER(state, idx, pats, tables, score=0, rank=0, timestamp=ts)
    html = _inject_green(html, _winner_straights(win))

    # 5) write to date-bucketed winners path
    date_str  = datetime.now().strftime("%Y-%m-%d")
    out_root  = ph.get_winners_output_dir()           # -> data/outputs/winners/<date>
    target    = os.path.join(out_root, "vtrac_reports", state)
    os.makedirs(target, exist_ok=True)
    out_path  = os.path.join(target, f"{state}_vtrac{idx}_winner_{win}_{ts}.html")
    with open(out_path, "w", encoding="utf-8") as fh: fh.write(html)
    return out_path
