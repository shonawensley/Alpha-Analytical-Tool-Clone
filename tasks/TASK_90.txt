Boxing and gap

def canon3(s: str) -> str:            # '689' -> '689', '986' -> '689'
    return ''.join(sorted(s.strip().zfill(3)))

def draws_since(draws: list[str], combo: str) -> tuple[int, bool]:
    """returns (gap, unseen) where unseen=True if not found in window."""
    tgt = canon3(combo)
    for i, d in enumerate(draws):
        if canon3(d) == tgt:
            return i, False
    return len(draws), True  # your rule: unseen in 1,000 → treat as RED; we’ll flag it.

DOUBLE_BLUE = 667
DOUBLE_RED  = 1000

def classify_double_gap(gap: int) -> str|None:
    if gap >= DOUBLE_RED:  return "R"
    if gap >= DOUBLE_BLUE: return "B"
    return None
