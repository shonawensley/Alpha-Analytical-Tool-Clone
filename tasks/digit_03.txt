def extract_clusters(s, min_len=3, max_len=12):
    # yield (start, end, token, run_profile) for all substrings length L
    for L in range(min_len, min(max_len+1, len(s)+1)):
        for i in range(0, len(s)-L+1):
            token = s[i:i+L]
            yield (i, i+L, token, summarize_runs(token))

def vtrac_family(token):  # your existing map/family lookup
    return map_to_vtrac_family(token)

def classes_for(token):
    fam = vtrac_family(token)
    return {
        "exact": token,
        "vtrac": fam,
        "drop_exact": {drop_digit(token, d) for d in possible_drop_digits()},
        "drop_vtrac": {vtrac_family(t) for t in drop_exact_set},
        "family_exact": collapse_runs(token),           # e.g., 661111888 -> 618
        "family_vtrac": vtrac_family(collapse_runs(token)),
    }

def scan_box(box):  # has R2, R4, R6, R8 strings, metadata, etc.
    feats = init_feature_accumulator()
    for step_name in ["R2","R4","R6","R8"]:
        s = box[step_name]
        seen_step = set()
        for _,_,tok,run in extract_clusters(s):
            for cls, reps in classes_for(tok).items():
                reps = {reps} if isinstance(reps, str) else reps
                for rep in reps:
                    key = (cls, rep)
                    feats.earliest_step[key] = min(feats.earliest_step.get(key, step_name), step_name)
                    feats.persistence[key] = feats.persistence.get(key, 0) + (0 if key in seen_step else 1)
                    seen_step.add(key)
                    feats.dup_density_accum[key].append(run.dup_metric)
                    feats.residual_purity_accum[key].append(residual_purity(s))
                    feats.drop_digit_tallies[key].update(run.drop_digit_candidates)
    return finalize(feats)
