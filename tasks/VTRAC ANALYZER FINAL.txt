from __future__ import annotations
from typing import Dict, Iterable, List, Tuple, Set
from collections import defaultdict

from modules.vtrac_enhanced.types import Cell

def slide_triad_windows(s: str) -> List[str]:
    """All contiguous 3-digit windows inside s."""
    s = "".join(ch for ch in s if ch.isdigit())
    return [s[i:i+3] for i in range(0, max(0, len(s)-2))]

def mirror_string(s: str, pairs: Dict[str,str]) -> str:
    return "".join(pairs.get(ch, ch) for ch in s)

def count_survival_across_columns(cols: List[str]) -> Dict[str, int]:
    """
    For every triad seen in the grid, return the longest consecutive run length across columns (right-aligned).
    We align to the rightmost columns (cols[-1] is col1).
    """
    runs: Dict[str, int] = defaultdict(int)
    # build presence flags per column
    col_tris: List[Set[str]] = [set(slide_triad_windows(c)) for c in cols]
    # compute longest consecutive run ending at each position; store max per triad
    for i in range(len(col_tris)-1, -1, -1):
        present = col_tris[i]
        for tri in present:
            # count backward consecutive presence
            length = 1
            j = i-1
            while j >= 0 and tri in col_tris[j]:
                length += 1
                j -= 1
            if length > runs[tri]:
                runs[tri] = length
    return runs

def straight_chain_strength(s: str, winner_hint_index: int|None = None) -> Dict[str, int]:
    """
    Heuristic: if the same triad recurs multiple times inside a single cell, count 'links' as repeats-1.
    Returns triad -> links (?0).
    """
    triads = slide_triad_windows(s)
    counts: Dict[str, int] = defaultdict(int)
    for t in triads:
        counts[t] += 1
    return {t: max(0, c-1) for t, c in counts.items()}

def index_has_double(combo: str) -> bool:
    return (combo[0] == combo[1]) or (combo[0] == combo[2]) or (combo[1] == combo[2])

from typing import Dict, List, Tuple, Set
from collections import defaultdict

from modules.vtrac_enhanced.types import EngineInput, IndexScore, Evidence
from modules.vtrac_enhanced.config import EnhancedVtracConfig
from modules.vtrac_enhanced.features import (
    slide_triad_windows, mirror_string, count_survival_across_columns,
    straight_chain_strength, index_has_double

# Import V-TRAC reference helpers from your repo
from modules.vtrac_reference import get_vtrac_index, get_index_straights, get_index_set, normalize_draw  # type: ignore

def _add_ev(sc: IndexScore, feat: str, ev: Evidence, emit: bool) -> None:
    sc.add_ev(feat, ev, emit)

def _col_weight_for(r_key: str, cfg: EnhancedVtracConfig, cols: List[str]) -> List[float]:
    base = cfg.col_weights.get(r_key, [1.0] * len(cols))
    if len(base) != len(cols):
        if len(cols) < len(base):
            base = base[-len(cols):]  # align right
        else:
            base = base + [base[-1]] * (len(cols) - len(base))
    return base

def _hot_multiplier(cell) -> float:
    mul = 1.0
    if getattr(cell, "hot", False):
        mul += cfg.h_hot  # will be bound later in context
    if getattr(cell, "superhot", False):
        mul += cfg.h_superhot
    return mul

def score_vtrac_indices(inp: EngineInput, cfg: EnhancedVtracConfig) -> Dict[int, IndexScore]:
    """
    Core scoring over R2/R4/R6/R8 pattern strings across sections.
    Implements the explicit weighted-sum + bonuses design.
    """
    scores: Dict[int, IndexScore] = {}
    section_seen: Dict[int, Set[str]] = defaultdict(set)  # cross-section consensus
    first_hit_col: Dict[int, int] = {}                    # earliest (closest to col1) presence
    persistence_cols: Dict[int, int] = defaultdict(int)   # max consecutive survival

    # Precompute assist contexts
    recent0_digits: Set[int] = set(int(d) for d in normalize_draw(inp.recent_draws[0])) if inp.recent_draws else set()
    mirror_pairs = cfg.mirror_pairs

    # pass 1: presence + ring/col/set/section weighting + hot boosts + survival + straight chains
    for sec in inp.sections:
        sec_w = cfg.section_weights.get(sec.section, 1.0)
        set_w = cfg.set_weights.get(sec.set_name, 1.0)
        for r_key, cells in sec.patterns.by_r.items():
            ring_w = cfg.ring_weights.get(r_key, 1.0)
            raw_cols = [c.digits for c in cells]
            col_weights = _col_weight_for(r_key, cfg, raw_cols)
            survival_map = count_survival_across_columns(raw_cols)

            for c_idx, cell in enumerate(cells):
                cw = col_weights[c_idx] * sec_w * set_w * ring_w
                # triads at this cell
                triads = slide_triad_windows(cell.digits)
                chain = straight_chain_strength(cell.digits, inp.winner_hint)

                for triad in triads:
                    idx = get_vtrac_index(triad)
                    if idx is None:
                        continue
                    sc = scores.setdefault(idx, IndexScore(index=idx))

                    # base presence
                    mul = 1.0
                    if cell.hot:
                        mul += cfg.h_hot
                    if cell.superhot:
                        mul += cfg.h_superhot
                    sc.add(cfg.w_index_presence * cw * mul)
                    _add_ev(sc, "presence", Evidence("presence", {
                        "section": sec.section, "set": sec.set_name, "ring": r_key, "col": c_idx+1,
                        "triad": triad, "cw": cw, "hot": int(cell.hot), "superhot": int(cell.superhot)
                    }), cfg.emit_evidence)
                    sc.ring_hits[r_key] += 1
                    section_seen[idx].add(sec.section)

                    # first-hit tracking (near col1 => smaller c_idx from right)
                    if idx not in first_hit_col or c_idx > first_hit_col[idx]:
                        first_hit_col[idx] = c_idx

                    # survival (consecutive across columns)
                    surv = survival_map.get(triad, 1)
                    if surv >= cfg.persistence_min_cols:
                        sc.add(cfg.w_column_survival * (surv - 1) * cw)
                        _add_ev(sc, "survival", Evidence("survival", {
                            "triad": triad, "cols": surv, "ring": r_key, "col": c_idx+1, "cw": cw
                        }), cfg.emit_evidence)
                        if surv > persistence_cols[idx]:
                            persistence_cols[idx] = surv

                    # straight chain within cell
                    links = chain.get(triad, 0)
                    if links > 0:
                        sc.add(cfg.w_straight_chain * links * cw)
                        _add_ev(sc, "straight_chain", Evidence("straight_chain", {
                            "triad": triad, "links": links, "ring": r_key, "col": c_idx+1, "cw": cw
                        }), cfg.emit_evidence)

                # optional straight clue vs. last winner's perm set (tiny nudge, conservative)
                if inp.winner_hint:
                    winner_perms = get_index_straights(inp.winner_hint) or set()
                    for triad in triads:
                        if triad in winner_perms:
                            sc = scores.setdefault(get_vtrac_index(triad), IndexScore(index=get_vtrac_index(triad)))
                            sc.add(cfg.w_straight_clue * cw)
                            _add_ev(sc, "straight_clue", Evidence("straight_clue", {
                                "triad": triad, "ring": r_key, "col": c_idx+1, "cw": cw
                            }), cfg.emit_evidence)

    # pass 2: cross-section consensus + first-hit + persistence bonuses
    for idx, sc in scores.items():
        secs = section_seen.get(idx, set())
        if len(secs) >= 2:
            sc.add(cfg.bonus_cross_section)
            _add_ev(sc, "consensus", Evidence("consensus", {"sections": sorted(list(secs))}), cfg.emit_evidence)
        # first-hit bonus (col1..first_hit_cols are the rightmost; c_idx 0-based from left, but we aligned weights)
        # We used c_idx growing left->right; reinterpret as distance from right:
        c_from_right = 7 - first_hit_col.get(idx, 6)  # rough; columns count may vary; we used 7 default
        if c_from_right <= cfg.first_hit_cols:
            sc.add(cfg.bonus_first_hit)
            _add_ev(sc, "first_hit", Evidence("first_hit", {"col_from_right": c_from_right}), cfg.emit_evidence)

        if (persistence_cols.get(idx, 0) >= cfg.persistence_min_cols):
            sc.add(cfg.bonus_persistence)
            _add_ev(sc, "persistence", Evidence("persistence", {"cols": persistence_cols[idx]}), cfg.emit_evidence)

    # pass 3: assists (reduction & mirror) and doubles affinity
    # Reduction: remove digits from most recent draw; if index still present in any rightmost cells, add bonus.
    reduction_agreement: Dict[int, int] = defaultdict(int)
    if cfg.enable_reduction_assist and recent0_digits:
        for sec in inp.sections:
            for r_key, cells in sec.patterns.by_r.items():
                # consider rightmost 1–2 columns to keep assist conservative
                for c_idx in range(max(0, len(cells)-2), len(cells)):
                    cell = cells[c_idx]
                    red = "".join(ch for ch in cell.digits if (not ch.isdigit()) or (int(ch) not in recent0_digits))
                    for tri in slide_triad_windows(red):
                        idx = get_vtrac_index(tri)
                        if idx is not None:
                            reduction_agreement[idx] = 1

    for idx, sc in scores.items():
        if reduction_agreement.get(idx, 0) == 1:
            sc.add(cfg.bonus_reduction)
            _add_ev(sc, "reduction_survives", Evidence("reduction_survives", {"agree": True}), cfg.emit_evidence)

        # mirror assist: if mirroring reveals same-index triads in rightmost cells, nudge
        if cfg.enable_mirror_assist:
            mirror_hit = 0
            for sec in inp.sections:
                for r_key, cells in sec.patterns.by_r.items():
                    for c_idx in range(max(0, len(cells)-2), len(cells)):
                        mir = mirror_string(cells[c_idx].digits, mirror_pairs)
                        for tri in slide_triad_windows(mir):
                            m_idx = get_vtrac_index(tri)
                            if m_idx == idx:
                                mirror_hit = 1
                                break
                    if mirror_hit:
                        break
                if mirror_hit:
                    break
            if mirror_hit:
                # integrate mirror assist as a small part of reduction bonus to stay conservative
                sc.add(0.5 * cfg.bonus_reduction)
                _add_ev(sc, "mirror_confirm", Evidence("mirror_confirm", {"agree": True}), cfg.emit_evidence)

        # doubles affinity: measure share of doubles combos in index; if ? 4/8, nudge
        boxed = list(get_index_set(idx) or [])
        if boxed:
            dbl = sum(1 for c in boxed if index_has_double(c))
            if dbl >= 4:
                sc.add(cfg.bonus_doubles)
                _add_ev(sc, "doubles_affinity", Evidence("doubles_affinity", {"double_combos": dbl, "of": len(boxed)}), cfg.emit_evidence)

    return scores

def _load_engine_input(path: Path) -> EngineInput:
    obj = json.loads(path.read_text())
    sections = []
    for sec in obj["sections"]:
        by_r = {}
        for r, cells in sec["by_r"].items():
            by_r[r] = [Cell(**c) for c in cells]
        sections.append(SectionData(section=sec["section"], set_name=sec["set_name"], patterns=PatternsGrid(by_r=by_r)))

def _hot_flags_for_cell(digits: str, hot_k: int = 2, superhot_k: int = 1) -> Tuple[bool, bool]:

def _load_from_tables_json(tables_dir: Path) -> Dict[str, Dict[str, List[Cell]]]:
    """
    Expect a compact JSON at: tables_dir / 'vtrac_tables.json'
    Shape:
    {
      "Midday":   {"R2": ["col7str",...,"col1str"], "R4":[...],"R6":[...],"R8":[...]},
      "Evening":  {"R2": [...], "R4":[...], "R6":[...], "R8":[...]},

    """

        return {}

    for section, ring_map in obj.items():
        out[section] = {}
        for r, cols in ring_map.items():
            cells: List[Cell] = []
            for s in cols:
                hot, superhot = _hot_flags_for_cell(str(s), hot_k=2, superhot_k=1)
                cells.append(Cell(digits=str(s), hot=hot, superhot=superhot))
            out[section][r] = cells
    return out

# ----------- CSV fallback loader (heuristic but safe) -----------
def _try_csv_parse(tables_dir: Path, base_name: str) -> Dict[str, List[Cell]]:
    """
    Attempt to parse a single section CSV like 'Midday_Combined.csv'.
    Heuristics:
      - Look for columns named exactly '7','6','5','4','3','2','1' OR prefixed by R-key rows.
      - If the file has a 'RowType' column (values R2/R4/R6/R8) and numeric columns 7..1, 
        we build cells by taking the row for each R-key and reading the digit-strings per column.
    """
    if pd is None:
        return {}
    # seek files that match '<base>_Combined.csv'
    candidates = [p for p in tables_dir.glob(f"{base_name}_Combined.csv")]
    if not candidates:
        return {}
    df = pd.read_csv(candidates[0], dtype=str).fillna("")
    cols_num = [c for c in df.columns if c.strip() in list("7654321")]
    if "RowType" in df.columns and cols_num:
        out: Dict[str, List[Cell]] = {}
        for r in ("R2", "R4", "R6", "R8"):
            sub = df[df["RowType"].astype(str).str.strip() == r]
            if sub.empty:
                continue
            # Prefer the last Draw row (right-most / latest context) if a 'Draw' col exists
            if "Draw" in sub.columns:
                sub = sub.tail(1)
            # Build left->right as 7..1, but we keep them in that order and the engine
            # treats the final element as col1 (most recent).
            cells: List[Cell] = []
            # enforce correct order 7..1
            for c in ["7","6","5","4","3","2","1"]:
                if c in sub.columns:
                    s = str(sub.iloc[0][c])
                    hot, superhot = _hot_flags_for_cell(s)
                    cells.append(Cell(digits=s, hot=hot, superhot=superhot))
            if cells:
                out[r] = cells
        return out
    return {}

def load_engine_input_from_tables(state: str) -> EngineInput:
    """
    Build EngineInput for the Enhanced V-TRAC engine using:
     1) vtrac_tables.json if present (preferred),
     2) otherwise CSV heuristics from Midday/Evening/Combined _Combined.csv.
    We assign Set1 to the loaded sections (Set2/Set3 can be added later if you export them).

    tdir = _tables_dir_for_state(state)

        raise FileNotFoundError(f"Tables directory not found for state '{state}': {tdir}")

    # 1) JSON preferred
    json_map = _load_from_tables_json(tdir)
    sections: List[SectionData] = []

    if json_map:

            ring_map = json_map.get(sec, {})
            if not ring_map:
                continue
            pg = PatternsGrid(by_r=ring_map)
            sections.append(SectionData(section=sec, set_name="Set1", patterns=pg))

        ring_map = _try_csv_parse(tdir, base_name=sec)
        if ring_map:
            pg = PatternsGrid(by_r=ring_map)
            sections.append(SectionData(section=sec, set_name="Set1", patterns=pg))

    if not sections:
        raise RuntimeError(
            f"No section data could be loaded for '{state}'. "
            f"Provide vtrac_tables.json in {tdir} or ensure *_Combined.csv have RowType and columns 7..1."
        )

    recent = []
    rd = tdir / "recent_draws.json"

        try:
            recent = list(json.loads(rd.read_text(encoding="utf-8")))
        except Exception:
            recent = []

    return EngineInput(sections=sections, recent_draws=recent)

# modules/vtrac_enhanced/permutation_ranker.py
from __future__ import annotations
from typing import Dict, List, Tuple, Set
from math import exp
from collections import defaultdict

from modules.vtrac_enhanced.types import EngineInput, Cell
from modules.vtrac_enhanced.config import EnhancedVtracConfig
from modules.vtrac_enhanced.features import slide_triad_windows
from modules.vtrac_reference import get_index_set  # boxed combos for an index

def _col_from_right(total: int, c_idx: int) -> int:
    # c_idx is 0..N-1 left->right; col1 (rightmost) should be 1
    return total - c_idx

def _ensure_unique_perms(s: str) -> List[str]:
    a,b,c = s[0], s[1], s[2]
    perms = {a+b+c, a+c+b, b+a+c, b+c+a, c+a+b, c+b+a}
    return sorted(perms)

def _has_subseq_gap1(s: str, perm: str) -> bool:
    """digits appear in order with ?1 gap between successive matches."""
    s = "".join(ch for ch in s if ch.isdigit())
    if not s:
        return False
    # contiguous
    if perm in s:
        return True
    # subseq with ?1 gap: i<j<k and j-i<=2, k-j<=2
    d1, d2, d3 = perm[0], perm[1], perm[2]
    idxs = [i for i,ch in enumerate(s) if ch==d1]
    for i in idxs:
        for j in range(i+1, min(len(s), i+3)):   # <=1 gap ? j-i<=2
            if s[j] != d2: 
                continue
            for k in range(j+1, min(len(s), j+3)):  # <=1 gap ? k-j<=2
                if s[k] == d3:
                    return True
    return False

def _hot_multiplier(cell: Cell, cfg: EnhancedVtracConfig) -> float:
    mul = 1.0
    if getattr(cell, "hot", False):
        mul += cfg.h_hot
    if getattr(cell, "superhot", False):
        mul += cfg.h_superhot
    return mul

def rank_permutations_for_indices(
    inp: EngineInput,
    cfg: EnhancedVtracConfig,
    index_scores: Dict[int, float],
    top_indices: List[int]
) -> Dict[int, List[Dict[str, object]]]:
    """
    For each selected index, compute per-permutation probabilities using order evidence.
    Returns: index -> list of {"perm": "xyz", "perm_score": float, "why": [str,...]}
    """
    out: Dict[int, List[Dict[str, object]]] = {}

    for idx in top_indices:
        boxed = list(get_index_set(idx) or [])
        if not boxed:
            out[idx] = []
            continue

        # generate unique straights (permutations) across all boxed triples
        perms: Set[str] = set()
        for triple in boxed:
            t = "".join(ch for ch in triple if ch.isdigit())
            if len(t) != 3:
                continue
            for p in _ensure_unique_perms(t):
                perms.add(p)

        # raw scores and why-tags
        raw: Dict[str, float] = defaultdict(float)
        why: Dict[str, List[str]] = defaultdict(list)

        # evidence scan
        for sec in inp.sections:
            sec_w = cfg.section_weights.get(sec.section, 1.0)
            set_w = cfg.set_weights.get(sec.set_name, 1.0)
            for r_key, cells in sec.patterns.by_r.items():
                ring_w = cfg.ring_weights.get(r_key, 1.0)
                # column weights aligned to cell length
                colw = cfg.col_weights.get(r_key, [1.0]*len(cells))
                if len(colw) != len(cells):
                    if len(cells) < len(colw):
                        colw = colw[-len(cells):]
                    else:
                        colw = colw + [colw[-1]]*(len(cells)-len(colw))

                for c_idx, cell in enumerate(cells):
                    base_w = colw[c_idx] * sec_w * set_w * ring_w * _hot_multiplier(cell, cfg)
                    col_from_right = _col_from_right(len(cells), c_idx)
                    s = cell.digits if isinstance(cell.digits, str) else str(cell.digits)

                    for p in perms:
                        if not p.isdigit() or len(p)!=3:
                            continue
                        if p in s:
                            raw[p] += base_w
                            why[p].append(f"contig {sec.section}/{r_key}@c{col_from_right}")
                        else:
                            if _has_subseq_gap1(s, p):
                                raw[p] += base_w
                                why[p].append(f"subseq1 {sec.section}/{r_key}@c{col_from_right}")

        # ? same-order echo (?2 rings or ?2 sections, near the right edge)
        for p in perms:
            ring_seen: Set[str] = set()
            sect_seen: Set[str] = set()
            for sec in inp.sections:
                for r_key, cells in sec.patterns.by_r.items():
                    for c_idx, cell in enumerate(cells):
                        cfr = _col_from_right(len(cells), c_idx)
                        if cfr > 2:  # last 2 columns only
                            continue
                        s = cell.digits
                        if (p in s) or _has_subseq_gap1(s, p):
                            ring_seen.add(r_key)
                            sect_seen.add(sec.section)
            if len(ring_seen) >= 2 or len(sect_seen) >= 2:
                raw[p] += cfg.kappa_same_order_echo
                why[p].append("echo(?)")

        # ? tail anchor (last digit repeatedly at col1 across rings/sections)
        for p in perms:
            tail = p[-1]
            tail_hits = 0
            for sec in inp.sections:
                for r_key, cells in sec.patterns.by_r.items():
                    if not cells:
                        continue
                    col1 = cells[-1].digits
                    if isinstance(col1, str) and len(col1) > 0 and col1[-1] == tail:
                        tail_hits += 1
            if tail_hits >= 2:  # anchor shows in ?2 places
                raw[p] += cfg.lambda_tail_anchor
                why[p].append("tail@col1(?)")

        # soft-normalize to per-index probabilities
        values = list(raw.values())
        if not values or all(v == 0 for v in values):
            # uniform fallback
            prob = 1.0 / max(1, len(perms))
            ranked = [{"perm": p, "perm_score": prob, "why": why.get(p, [])} for p in sorted(perms)]
        else:
            m = max(values)
            exps = {p: exp(raw[p] - m) for p in perms}
            z = sum(exps.values())
            ranked = [{"perm": p, "perm_score": exps[p]/z, "why": why.get(p, [])} for p in sorted(perms)]

        # prioritize by IndexScore × PermutationProb (per redesign)
        idx_score = index_scores.get(idx, 1.0)
        ranked.sort(key=lambda d: (idx_score * float(d["perm_score"])), reverse=True)

        # cap output per config

from __future__ import annotations
from typing import Dict, List, Set, Tuple
from collections import defaultdict

from modules.vtrac_enhanced.types import IndexScore
from modules.vtrac_enhanced.config import EnhancedVtracConfig

def _presence_iter(sc: IndexScore):
    for ev in sc.ev.get("presence", []):
        yield ev.payload

def _sections_seen(sc: IndexScore) -> Set[str]:
    secs = set()
    for p in _presence_iter(sc):
        sec = p.get("section")
        if sec:
            secs.add(sec)
    return secs

def _has_set1_presence(sc: IndexScore) -> bool:
    for p in _presence_iter(sc):
        if p.get("set") == "Set1":
            return True
    return False

def _has_recent_hot(sc: IndexScore) -> bool:
    for p in _presence_iter(sc):
        if p.get("set") == "Set1" and (p.get("hot", 0) == 1 or p.get("superhot", 0) == 1):
            return True
    return False

def _presence_count(sc: IndexScore) -> int:
    return len(sc.ev.get("presence", []))

def _passes(sc: IndexScore, cfg: EnhancedVtracConfig) -> bool:
    if sc.score < cfg.floor_min_score:
        return False
    if cfg.require_set1_presence and not _has_set1_presence(sc):
        return False
    if cfg.require_recent_hot_hit and not _has_recent_hot(sc):
        return False
    if _presence_count(sc) < cfg.min_presence_events:
        return False
    if len(_sections_seen(sc)) < cfg.require_min_sections:
        return False
    return True

def select_indices(ranked: List[IndexScore], cfg: EnhancedVtracConfig, target_n: int | None = None) -> List[int]:
    """
    Apply policy gates with progressive relaxation to guarantee at least N indices.
    Relaxation order: recent_hot -> sections -> set1_presence -> presence_min -> floor
    """
    N = int(target_n) if target_n is not None else int(cfg.select_top_n_default)
    if N <= 0:
        return []

    # Pass 1: strict
    chosen = [sc.index for sc in ranked if _passes(sc, cfg)]
    if len(chosen) >= N:
        return chosen[:N]

    # Progressive relaxations
    relax = [
        ("require_recent_hot_hit", False),
        ("require_min_sections", 1),
        ("require_set1_presence", False),
        ("min_presence_events", 1),
        ("floor_min_score", float("-inf")),
    ]
    saved = (cfg.require_recent_hot_hit, cfg.require_min_sections, cfg.require_set1_presence,
             cfg.min_presence_events, cfg.floor_min_score)

    for key, val in relax:
        setattr(cfg, key, val)
        chosen = [sc.index for sc in ranked if _passes(sc, cfg)]
        if len(chosen) >= N:
            # restore cfg then return
            (cfg.require_recent_hot_hit, cfg.require_min_sections, cfg.require_set1_presence,
             cfg.min_presence_events, cfg.floor_min_score) = saved
            return chosen[:N]

    # Final fallback: just take top N
    (cfg.require_recent_hot_hit, cfg.require_min_sections, cfg.require_set1_presence,
     cfg.min_presence_events, cfg.floor_min_score) = saved
    return [sc.index for sc in ranked[:N]]

5) Backtest CLI — Index@K and Straight@M
File: tools/vtrac_backtest.py
# tools/vtrac_backtest.py
"""
Backtest Enhanced V-TRAC predictions against a winners ledger.

Usage:
  python tools/vtrac_backtest.py --state CT \
    --winners_csv data/winners/ct_winners.csv \
    --pred_dir data/outputs/predictions/CT \
    --top_k 3 --top_m 8

The winners CSV must have headers: date,state,winner
  - date: YYYY-MM-DD
  - state: e.g., CT (must match pred_dir state)
  - winner: 3-digit string (e.g., 691)

Metrics:
  - Index@K: whether the winner's V-TRAC index is among top K indices by score
  - Straight@M: whether the exact straight winner is among the aggregated straights of the selected indices
Outputs:
  - per_day.csv with daily outcomes and ranks
  - summary.json with aggregates
  - written under data/outputs/analysis/vtrac/<STATE>/backtest_<timestamp>/
"""
from __future__ import annotations
import argparse, json, csv
from pathlib import Path
from datetime import datetime

from modules.vtrac_reference import get_vtrac_index  # index lookup for a straight

def _load_winners(fp: Path) -> dict:
    out = {}
    with fp.open("r", encoding="utf-8") as f:
        rd = csv.DictReader(f)
        for row in rd:
            d = row["date"].strip()
            st = row["state"].strip()
            w = "".join(ch for ch in row["winner"] if ch.isdigit())
            if len(w) == 3:
                out[(st, d)] = w
    return out

def _pred_files(pred_dir: Path):
    for p in sorted(pred_dir.glob("*_vtrac_predictions.json")):
        yield p

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--state", required=True, help="State code matching predictions dir")
    ap.add_argument("--winners_csv", required=True, help="CSV with date,state,winner")
    ap.add_argument("--pred_dir", required=True, help="Folder with *_vtrac_predictions.json files")
    ap.add_argument("--top_k", type=int, default=3, help="Top K indices considered")
    ap.add_argument("--top_m", type=int, default=8, help="Straights per selected index considered")
    args = ap.parse_args()

    state = args.state
    winners = _load_winners(Path(args.winners_csv))
    pdir = Path(args.pred_dir)
    assert pdir.exists(), f"pred_dir not found: {pdir}"

    # output folder
    ts = datetime.now().strftime("%Y%m%d%H%M%S")
    out_root = Path("data/outputs/analysis") / "vtrac" / state / f"backtest_{ts}"
    out_root.mkdir(parents=True, exist_ok=True)

    # metrics
    total = 0
    idx_hits = 0
    straight_hits = 0

    per_day_rows = []

    for pf in _pred_files(pdir):
        # filename assumed: YYYY-MM-DD_vtrac_predictions.json
        date_key = pf.name.split("_vtrac_predictions.json")[0]
        key = (state, date_key)
        if key not in winners:
            continue
        total += 1
        winner = winners[key]
        w_idx = get_vtrac_index(winner)

        obj = json.loads(pf.read_text(encoding="utf-8"))
        ranked = list(obj.get("indices_ranked", []))
        # sort defensively by score if not already sorted
        ranked.sort(key=lambda r: float(r.get("score", 0.0)), reverse=True)

        # Index@K
        top_indices = [int(r["index"]) for r in ranked[:max(1,args.top_k)]]
        idx_hit = int((w_idx in top_indices) if w_idx is not None else 0)
        idx_hits += idx_hit

        # Straight@M across selected indices' straights
        cand = set()
        for r in ranked[:max(1,args.top_k)]:
            for s in r.get("straights", [])[:max(1,args.top_m)]:
                p = str(s.get("perm",""))
                if len(p) == 3:
                    cand.add(p)
        straight_hit = int(winner in cand)
        straight_hits += straight_hit

        per_day_rows.append({
            "date": date_key,
            "winner": winner,
            "winner_index": w_idx if w_idx is not None else -1,
            "index@K": idx_hit,
            "straight@M": straight_hit,
            "top_indices": ",".join(f"{i:02d}" for i in top_indices),
        })

    # write per_day.csv
    with (out_root / "per_day.csv").open("w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=["date","winner","winner_index","index@K","straight@M","top_indices"])
        w.writeheader()
        for row in per_day_rows:
            w.writerow(row)

    # write summary.json
    summary = {
        "state": state,
        "days": total,
        "Index@K": {"K": args.top_k, "hits": idx_hits, "rate": (idx_hits/total if total else 0.0)},
        "Straight@M": {"M": args.top_m, "hits": straight_hits, "rate": (straight_hits/total if total else 0.0)},
        "pred_dir": str(pdir),
        "winners_csv": str(Path(args.winners_csv).resolve())
    }
    (out_root / "summary.json").write_text(json.dumps(summary, indent=2), encoding="utf-8")
    print(json.dumps(summary, indent=2))

if __name__ == "__main__":
    main()

from __future__ import annotations
from dataclasses import asdict, fields

from pathlib import Path
import json

try:
    import yaml  # type: ignore
    _HAS_YAML = True
except Exception:
    _HAS_YAML = False

from modules.vtrac_enhanced.config import EnhancedVtracConfig

DEFAULT_CFG_PATH = Path("configs/vtrac_enhanced.yml")

def _coerce(value: Any, to_type: type) -> Any:
    try:
        if to_type is bool:
            if isinstance(value, bool):
                return value
            s = str(value).strip().lower()
            return s in ("1","true","yes","on")
        if to_type is int:
            return int(value)
        if to_type is float:
            return float(value)
        if to_type is dict:
            return dict(value)
        if to_type is list:
            return list(value)
        return value
    except Exception:
        return value

def dict_to_config(d: Dict[str, Any]) -> EnhancedVtracConfig:
    base = EnhancedVtracConfig()
    base_dict = asdict(base)
    for f in fields(EnhancedVtracConfig):
        name = f.name
        if name in d:
            val = d[name]
            if isinstance(base_dict.get(name), dict) and isinstance(val, dict):
                # shallow merge for dicts
                merged = dict(base_dict[name])
                merged.update(val)
                setattr(base, name, merged)
            else:
                setattr(base, name, _coerce(val, f.type))  # type: ignore
    return base

def config_to_dict(cfg: EnhancedVtracConfig) -> Dict[str, Any]:
    return asdict(cfg)

def load_config(path: Path | None = None) -> EnhancedVtracConfig:
    fp = path or DEFAULT_CFG_PATH
    if not fp.exists():
        return EnhancedVtracConfig()
    try:
        if _HAS_YAML and fp.suffix in (".yml",".yaml"):
            with fp.open("r", encoding="utf-8") as f:
                obj = yaml.safe_load(f) or {}
        else:
            with fp.open("r", encoding="utf-8") as f:
                obj = json.loads(f.read() or "{}")
    except Exception:
        return EnhancedVtracConfig()
    if not isinstance(obj, dict):
        return EnhancedVtracConfig()
    return dict_to_config(obj)

def save_config(cfg: EnhancedVtracConfig, path: Path | None = None) -> Path:
    fp = path or DEFAULT_CFG_PATH
    fp.parent.mkdir(parents=True, exist_ok=True)
    data = config_to_dict(cfg)
    if _HAS_YAML and fp.suffix in (".yml",".yaml"):
        with fp.open("w", encoding="utf-8") as f:
            yaml.safe_dump(data, f, sort_keys=True)  # type: ignore
    else:
        # write JSON if YAML unavailable or suffix not YAML
        with fp.open("w", encoding="utf-8") as f:
            f.write(json.dumps(data, indent=2))
    return fp

def _flatten_index_row(date_str: str, state: str, r: Dict[str, Any], winner: str | None) -> Dict[str, Any]:
    idx = int(r["index"])
    score = float(r.get("score", 0.0))
    ev = r.get("evidence", {}) or {}
    # basic features
    presence = ev.get("presence", [])
    survival = ev.get("survival", [])
    consensus = ev.get("consensus", [])
    flags = {
        "reduction": 1 if ev.get("reduction_survives") else 0,
        "mirror":    1 if ev.get("mirror_confirm") else 0,
        "doubles":   1 if ev.get("doubles_affinity") else 0,
    }
    # derive consensus sections count
    cons_sections = 0
    if consensus:
        seen = set()
        for e in consensus:
            for s in e.get("sections", []):
                seen.add(s)
        cons_sections = len(seen) if seen else len(consensus)

    w_idx = get_vtrac_index(winner) if winner else None
    label = 1 if (w_idx is not None and int(w_idx) == idx) else 0

    return {
        "date": date_str, "state": state, "index": idx, "score": score,
        "presence_events": len(presence), "survival_events": len(survival), "consensus_sections": cons_sections,
        "flag_reduction": flags["reduction"], "flag_mirror": flags["mirror"], "flag_doubles": flags["doubles"],
        "label_is_winning_index": label
    }

def _flatten_straights_rows(date_str: str, state: str, r: Dict[str, Any], winner: str | None) -> List[Dict[str, Any]]:

from __future__ import annotations
from dataclasses import asdict, fields
from typing import Any, Dict
from pathlib import Path
import json

try:
    import yaml  # type: ignore
    _HAS_YAML = True
except Exception:
    _HAS_YAML = False

from modules.vtrac_enhanced.config import EnhancedVtracConfig

DEFAULT_CFG_PATH = Path("configs/vtrac_enhanced.yml")

def _coerce(value: Any, to_type: type) -> Any:
    try:
        if to_type is bool:
            if isinstance(value, bool):
                return value
            s = str(value).strip().lower()
            return s in ("1","true","yes","on")
        if to_type is int:
            return int(value)
        if to_type is float:
            return float(value)
        if to_type is dict:
            return dict(value)
        if to_type is list:
            return list(value)
        return value
    except Exception:
        return value

def dict_to_config(d: Dict[str, Any]) -> EnhancedVtracConfig:
    base = EnhancedVtracConfig()
    base_dict = asdict(base)
    for f in fields(EnhancedVtracConfig):
        name = f.name
        if name in d:
            val = d[name]
            if isinstance(base_dict.get(name), dict) and isinstance(val, dict):
                # shallow merge for dicts
                merged = dict(base_dict[name])
                merged.update(val)
                setattr(base, name, merged)
            else:
                setattr(base, name, _coerce(val, f.type))  # type: ignore
    return base

def config_to_dict(cfg: EnhancedVtracConfig) -> Dict[str, Any]:
    return asdict(cfg)

def load_config(path: Path | None = None) -> EnhancedVtracConfig:
    fp = path or DEFAULT_CFG_PATH
    if not fp.exists():
        return EnhancedVtracConfig()
    try:
        if _HAS_YAML and fp.suffix in (".yml",".yaml"):
            with fp.open("r", encoding="utf-8") as f:
                obj = yaml.safe_load(f) or {}
        else:
            with fp.open("r", encoding="utf-8") as f:
                obj = json.loads(f.read() or "{}")
    except Exception:
        return EnhancedVtracConfig()
    if not isinstance(obj, dict):
        return EnhancedVtracConfig()
    return dict_to_config(obj)

def save_config(cfg: EnhancedVtracConfig, path: Path | None = None) -> Path:
    fp = path or DEFAULT_CFG_PATH
    fp.parent.mkdir(parents=True, exist_ok=True)
    data = config_to_dict(cfg)
    if _HAS_YAML and fp.suffix in (".yml",".yaml"):
        with fp.open("w", encoding="utf-8") as f:
            yaml.safe_dump(data, f, sort_keys=True)  # type: ignore
    else:
        # write JSON if YAML unavailable or suffix not YAML
        with fp.open("w", encoding="utf-8") as f:
            f.write(json.dumps(data, indent=2))
    return fp

def _flatten_index_row(date_str: str, state: str, r: Dict[str, Any], winner: str | None) -> Dict[str, Any]:
    idx = int(r["index"])
    score = float(r.get("score", 0.0))
    ev = r.get("evidence", {}) or {}
    # basic features
    presence = ev.get("presence", [])
    survival = ev.get("survival", [])
    consensus = ev.get("consensus", [])
    flags = {
        "reduction": 1 if ev.get("reduction_survives") else 0,
        "mirror":    1 if ev.get("mirror_confirm") else 0,
        "doubles":   1 if ev.get("doubles_affinity") else 0,
    }
    # derive consensus sections count
    cons_sections = 0
    if consensus:
        seen = set()
        for e in consensus:
            for s in e.get("sections", []):
                seen.add(s)
        cons_sections = len(seen) if seen else len(consensus)

    w_idx = get_vtrac_index(winner) if winner else None
    label = 1 if (w_idx is not None and int(w_idx) == idx) else 0

    return {
        "date": date_str, "state": state, "index": idx, "score": score,
        "presence_events": len(presence), "survival_events": len(survival), "consensus_sections": cons_sections,
        "flag_reduction": flags["reduction"], "flag_mirror": flags["mirror"], "flag_doubles": flags["doubles"],
        "label_is_winning_index": label
    }

def _flatten_straights_rows(date_str: str, state: str, r: Dict[str, Any], winner: str | None) -> List[Dict[str, Any]]:
    idx = int(r["index"])
    straights = r.get("straights", []) or []
    rows: List[Dict[str, Any]] = []
    for s in straights:
        perm = str(s.get("perm",""))
        ps = float(s.get("perm_score", 0.0))
        label = 1 if (winner is not None and perm == winner) else 0
        rows.append({
            "date": date_str, "state": state, "index": idx,
            "perm": perm, "perm_score": ps,
            "label_is_winning_straight": label
        })
    return rows

def write_capsules_for_file(state: str, pred_file: Path, winner: str | None = None) -> Tuple[Path, Path]:

    date_str = pred_file.name.split("_vtrac_predictions.json")[0]

    for r in ranked:

        st_rows.extend(_flatten_straights_rows(date_str, state, r, winner))

    idx_fp = out_dir / "capsule_indices.csv"

        w.writeheader()
        for row in idx_rows:

    with st_fp.open("w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=list(st_rows[0].keys()) if st_rows else ["date","state","index","perm","perm_score"])
        w.writeheader()
        for row in st_rows:
            w.writerow(row)
    return idx_fp, st_fp

def write_capsules_for_dir(state: str, winners_csv: Path | None = None) -> Path:
    pdir = _pred_dir(state)
    out_root = Path("data/outputs/analysis") / "vtrac" / state / f"capsules_{datetime.now().strftime('%Y%m%d%H%M%S')}"
    out_root.mkdir(parents=True, exist_ok=True)

        import csv as _csv
        with winners_csv.open("r", encoding="utf-8") as f:

    for pred in sorted(pdir.glob("*_vtrac_predictions.json")):

        winner = win_map.get((state, date_str))
        write_capsules_for_file(state, pred, winner=winner)

    return out_root
