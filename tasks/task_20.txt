import re

def _parse_winners_input(s: str) -> list[str]:
    if not s:
        return []
    # accept commas, spaces, semicolons, pipes; normalize to 3 digits with leading zeros
    toks = re.split(r"[\s,;|]+", s.strip())
    return [t.zfill(3) for t in toks if re.fullmatch(r"\d{1,3}", t)]
