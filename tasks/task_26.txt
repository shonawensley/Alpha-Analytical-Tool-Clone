# modules/winner_report.py
from pathlib import Path
import os

def write_winner_index_html(state_name: str,
                            date_str: str,
                            winning_number: str,
                            tables: dict,
                            out_root: str = "data/outputs/winners") -> str:
    """
    Create a V‑TRAC report for the winning number's index and save it as:
        data/outputs/winners/<STATE>/<STATE>_<DATE>_WIN_index_<IDX>.html
    Returns the full path to the saved HTML.
    """
    # Use the same mapping as the analyzer
    from modules.vtrac_reference import get_vtrac_index, VTRAC_DISPLAY

    win = str(winning_number).strip()
    if len(win) != 3 or not win.isdigit():
        raise ValueError(f"Winning number '{winning_number}' must be 3 digits (got '{winning_number}').")

    idx = get_vtrac_index(win)

    # Gather combos for that index
    singles, doubles = [], []
    for row in VTRAC_DISPLAY:
        if int(row.get("Index", 0)) == int(idx):
            singles = row.get("Singles", []) or []
            doubles = row.get("Doubles", []) or []
            break

    # Prefer the analyzer's full HTML if available
    html = None
    try:
        # Adjust import path if your project keeps it elsewhere
        from src.core.module_c_vtrac import generate_index_html_report
        html = generate_index_html_report(
            state_name=state_name,
            index=idx,
            patterns={"Singles": singles, "Doubles": doubles},
            tables=tables,          # {"Midday": df, "Evening": df, "Combined": df} if you have them
            score=0,
            rank=0,
        )
    except Exception:
        # Lightweight fallback HTML (still correct index & combos)
        def _ul(title, seq):
            return "<h3>{}</h3><ul>{}</ul>".format(title, "".join(f"<li>{c}</li>" for c in seq))
        html = f"""<!doctype html>
<html><head><meta charset="utf-8"><title>{state_name} WIN Index {idx}</title></head>
<body>
<h1>{state_name} — Winning V‑TRAC Index {idx}</h1>
<p><strong>Winning Number:</strong> {win}</p>
{_ul("Singles", singles)}
{_ul("Doubles", doubles)}
</body></html>"""

    out_dir = Path(out_root) / state_name
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / f"{state_name}_{date_str}_WIN_index_{idx}.html"
    with open(out_path, "w", encoding="utf-8") as f:
        f.write(html)
    return str(out_path)

import os, re, datetime

# ---- Optional: centralize the winners vtrac report path
def winners_vtrac_dir(date_str: str, state_name: str) -> Path:
    # date_str like "2025-09-10"
    return Path("data") / "outputs" / "winners" / date_str / "vtrac_reports" / state_name

def _timestamp():
    return datetime.datetime.now().strftime("%Y%m%d_%H%M")

def _highlight_straights_in_html(html: str, straights: list[str]) -> str:
    """
    Post-process: color the V‑TRAC straight combos in green, preserving the analyzer's HTML.
    We inject a .straight CSS and wrap exact 3-digit hits with <span class="straight">...</span>.
    """
    if not straights:
        return html
    css = (
        "<style>"
        ".straight{background:#d8f5d6;color:#0b7a0b;font-weight:600;"
        "border-radius:3px;padding:0 2px}"
        "</style>"
    )
    # add CSS into <head>
    if "</head>" in html:
        html = html.replace("</head>", css + "</head>")
    else:
        html = css + html

    # replace 3-digit tokens with span; ensure we don't match inside longer numbers
    for s in sorted(set(straights), reverse=True):
        patt = rf'(?<!\d){re.escape(s)}(?!\d)'
        html = re.sub(patt, f'<span class="straight">{s}</span>', html)
    return html

def compute_vtrac_index_and_display(winning_number: str):
    """
    Returns (index:int, patterns:dict, vstraights:list[str]).
    patterns is what analyzer's generator expects: {"Singles":[...], "Doubles":[...]}.
    vstraights is the 8-combo "V‑TRAC straight" set for the specific v-code that includes winning_number.
    """
    win = str(winning_number).strip().zfill(3)
    # Try the same refs the analyzer uses
    idx = None
    patterns = {"Singles": [], "Doubles": []}
    try:
        # common analyzer reference
        from modules.vtrac_reference import get_vtrac_index, VTRAC_DISPLAY
        idx = get_vtrac_index(win)
        for row in VTRAC_DISPLAY:
            if int(row.get("Index", -1)) == int(idx):
                patterns["Singles"] = row.get("Singles", []) or []
                patterns["Doubles"] = row.get("Doubles", []) or []
                break
    except Exception:
        # Fallback if names differ in your branch
        from src.core.module_c_vtrac import find_vtrac_index_and_combos  # adjust if needed
        idx, combos = find_vtrac_index_and_combos(win)
        patterns = combos

    # V‑TRAC straight mapping (winner's 8-combo set). We load from a small module you’ll add next.
    from modules.vtrac_straight_map import VSTRAIGHTS  # see step 2
    vstraights = []
    for vcode, combos in VSTRAIGHTS.items():
        if win in combos:
            vstraights = combos[:]  # the full 8
            break
    return int(idx), patterns, vstraights

def write_winner_index_html(state_name: str,
                            date_str: str,
                            winning_number: str,
                            tables: dict) -> str:
    """
    Build analyzer-style HTML (Midday/Evening/Combined) for the winner’s V‑TRAC index
    and save it under winners/<DATE>/vtrac_reports/<STATE>/...WIN_index_<IDX>_<STAMP>.html
    Returns the full path.
    """
    idx, patterns, vstraights = compute_vtrac_index_and_display(winning_number)

    # Prefer the analyzer's rich generator
    try:
        from src.core.module_c_vtrac import generate_index_html_report
        html = generate_index_html_report(
            state_name=state_name,
            index=idx,
            patterns=patterns,   # {"Singles": [...], "Doubles":[...]}
            tables=tables,       # {"Midday":df, "Evening":df, "Combined":df} or {}
            score=0,
            rank=0,

    except Exception:

        html = f"""<!doctype html><html><head><meta charset="utf-8">
<title>{state_name} WIN index {idx}</title></head>
<body>
<h1>{state_name} — WIN index {idx}</h1>
<p><strong>Winning:</strong> {str(winning_number).zfill(3)}</p>
<h3>Singles</h3><p>{", ".join(patterns.get("Singles", []))}</p>
<h3>Doubles</h3><p>{", ".join(patterns.get("Doubles", []))}</p>
</body></html>"""

    # Green-mark the V‑TRAC straights
    html = _highlight_straights_in_html(html, vstraights)

    out_path.write_text(html, encoding="utf-8")
