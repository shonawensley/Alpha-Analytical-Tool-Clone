def get_columns_for(state: str, category: str = 'combined'):
    """
    category: 'combined' | 'midday' | 'evening' | 'nite' | 'morning'
    returns: list[str] of Excel column letters
    """
    s = state  # state names use spaces, e.g., 'New Jersey', 'West Virginia'
    category = category.lower()

    if category == 'combined':
        col = COMBINED_LOCKED.get(s)
        return [col] if col else []

    if category == 'midday':
        # Noon is normalized into MIDDAY_COL for TX; WV/VA may be absent by design
        col = MIDDAY_COL.get(s)
        return [col] if col else []

    if category == 'evening':
        return EVENING_COLS.get(s, [])

    # specials (only where present)
    extra = EXTRA_COLS.get(s, {})
    if category == 'nite':
        c = extra.get('Nite')
        return [c] if c else []
    if category == 'morning':
        c = extra.get('Morning')
        return [c] if c else []
    if category == 'noon':
        c = extra.get('Noon')
        return [c] if c else []

    return []

    df = load_draw_data(excel_path)
    os.makedirs(outdir, exist_ok=True)

    for s in states:
        # 1) Combined (locked) — keep your current Aux behavior untouched
        c_cols = get_columns_for(s, 'combined')
        if c_cols:
            draws = extract_draws_from_columns(df, c_cols, max_draws=1000)
            pd.DataFrame(draws, columns=['Date','Draw']).to_csv(
                os.path.join(outdir, f"{s}_draws.csv"), index=False
            )

        # 2) Midday
        m_cols = get_columns_for(s, 'midday')
        if m_cols:
            draws = extract_draws_from_columns(df, m_cols, max_draws=1000)
            pd.DataFrame(draws, columns=['Date','Draw']).to_csv(
                os.path.join(outdir, f"{s}_Midday_draws.csv"), index=False
            )

        # 3) Evening (includes Nite where applicable)
        e_cols = get_columns_for(s, 'evening')
        if e_cols:
            draws = extract_draws_from_columns(df, e_cols, max_draws=1000)

                os.path.join(outdir, f"{s}_Evening_draws.csv"), index=False

        for special in ('Morning','Noon','Nite'):
