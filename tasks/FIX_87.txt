What’s wrong in the new family code (from your log + UI symptoms)

A. Add a small, explicit classifier (use SSOT constants)
# src/app.py  (or src/core/aux_config.py if you prefer constants there)
COMBO_SINGLE_LATE = 334
COMBO_SINGLE_VERY_LATE = 501
COMBO_DOUBLE_LATE = 667
COMBO_DOUBLE_VERY_LATE = 1000

def _canon_boxed(s: str) -> str:
    return "".join(sorted((s or "").strip().zfill(3)))

def _combo_type(boxed: str) -> str:
    a, b, c = boxed[0], boxed[1], boxed[2]
    if a == b == c: return "triple"
    if a == b or b == c: return "double"
    return "single"

def _classify_gap(ds: int, boxed: str) -> str | None:
    t = _combo_type(boxed)
    if t == "double":
        if ds >= COMBO_DOUBLE_VERY_LATE: return "R"
        if ds >= COMBO_DOUBLE_LATE:      return "B"
        return None
    if t == "single":
        if ds >= COMBO_SINGLE_VERY_LATE: return "R"
        if ds >= COMBO_SINGLE_LATE:      return "B"
        return None
    # triples (rare): treat like doubles for overdue signaling
    if ds >= COMBO_DOUBLE_VERY_LATE: return "R"
    if ds >= COMBO_DOUBLE_LATE:      return "B"
    return None

B. Build per‑variant gaps → per‑variant combo classes; filter strictly
# assume variant_draws = {"combined": [...], "midday": [...], "evening": [...]}
_VARIANT_BADGE = {"combined": "C", "midday": "M", "evening": "E"}

def _draws_since(draws: list[str], boxed_combo: str) -> int:
    tgt = _canon_boxed(boxed_combo)
    for i, d in enumerate(draws):
        if _canon_boxed(d) == tgt:
            return i
    return len(draws)

def _all_boxed_doubles() -> tuple[str, ...]:
    s = set()
    for r in "0123456789":
        for o in "0123456789":
            if r == o: 
                continue
            s.add(_canon_boxed(r+r+o))
            s.add(_canon_boxed(r+o+r))
            s.add(_canon_boxed(o+r+r))
    return tuple(sorted(s))

def _variant_combo_classes(variant_draws: dict[str, list[str]]) -> tuple[dict[str, dict[str, int]],
                                                                         dict[str, dict[str, str|None]]]:
    all_doubles = _all_boxed_doubles()
    gaps: dict[str, dict[str, int]] = {}
    cls:  dict[str, dict[str, str|None]] = {}
    for vkey, draws in variant_draws.items():
        gmap = {c: _draws_since(draws, c) for c in all_doubles}
        cmap = {c: _classify_gap(gmap[c], c) for c in all_doubles}
        gaps[vkey], cls[vkey] = gmap, cmap
    return gaps, cls


Filter rule: a combo qualifies for the family strip only if it is R or B in at least one of C/M/E.

C. Group by V‑TRAC mirror family and render variant‑specific tokens
_MIRROR = {0:0,5:0, 1:1,6:1, 2:2,7:2, 3:3,8:3, 4:4,9:4}

def _family_key_for_double(boxed: str) -> tuple[int,int]:
    # boxed is sorted; doubles: middle is a repeated digit
    a,b,c = int(boxed[0]), int(boxed[1]), int(boxed[2])
    if not (a==b or b==c):
        raise ValueError("expected a double")
    repeated = b
    singleton = a if a!=b else c
    return (_MIRROR[repeated], _MIRROR[singleton])

def _rank_double_families(variant_draws: dict[str, list[str]], limit: int=5):
    gaps, cls = _variant_combo_classes(variant_draws)
    families: dict[tuple[int,int], list[str]] = {}
    tokens_by_family: dict[tuple[int,int], list[str]] = {}
    score_by_family: dict[tuple[int,int], int] = {}
    bestgap_by_family: dict[tuple[int,int], int] = {}

    for c in _all_boxed_doubles():
        # keep only if any variant is hot
        v_hits = [(v, cls[v][c]) for v in ("combined","midday","evening") if cls[v][c] in ("R","B")]
        if not v_hits:
            continue
        fkey = _family_key_for_double(c)
        families.setdefault(fkey, []).append(c)
        for v, k in v_hits:
            badge = _VARIANT_BADGE[v]
            token = f"{k}{c}{badge}"      # e.g., R225C or B225M
            tokens_by_family.setdefault(fkey, []).append(token)
            score_by_family[fkey] = score_by_family.get(fkey, 0) + (2 if k=="R" else 1)
            g = gaps[v][c]
            bestgap_by_family[fkey] = min(bestgap_by_family.get(fkey, g), g)

    # order tokens within a family: R before B, then larger gap first
    def _tok_sort_key(tok: str):
        k = 0 if tok[0]=="R" else 1
        combo = tok[1:4]
        v = {"C":"combined","M":"midday","E":"evening"}[tok[-1]]
        return (k, -gaps[v][combo], tok)

    rows = []
    for fkey, toks in tokens_by_family.items():
        toks.sort(key=_tok_sort_key)
        rows.append({
            "family": fkey, 
            "display": " ".join(toks),
            "score": score_by_family.get(fkey, 0),
            "best_gap": bestgap_by_family.get(fkey, 10**9),
        })
    rows.sort(key=lambda r: (-r["score"], r["best_gap"]))
    return rows[:limit]

def _sum_flags(ds, z, late, very_late):
    flags = []
    if ds >= very_late: flags.append("red")
    elif ds >= late:    flags.append("blue")
    if z <= -1.5:       flags.append("purple")
    return " ".join(flags)

row["deficit"] = row["hit_pct"] - row["exp_pct"]

def _index_heat(draws, resolver, window=1000, k=200):
    # returns dict[index] = metrics
    hits = {i: [] for i in range(1, 36)}
    last_seen = {i: None for i in range(1, 36)}
    for pos, draw in enumerate(draws):
        idx = resolver(draw)  # 1..35
        if last_seen[idx] is not None:
            hits[idx].append(pos - last_seen[idx])
        last_seen[idx] = pos
    out = {}
    for i in range(1, 36):
        gaps = hits[i][-k:] if k else hits[i]
        freq_k = len([g for g in gaps if g <= k])
        avg_gap = (sum(gaps)/len(gaps)) if gaps else None
        q80 = sorted(gaps)[int(0.8*len(gaps))] if gaps else None
        out[i] = {
            "ds": (len(draws) - (last_seen[i] or len(draws))),
            "freq_k": freq_k,
            "avg_gap_k": avg_gap,
            "q80_gap": q80,
            "hazard": (0 if not avg_gap else 1/avg_gap),
        }
    return out
