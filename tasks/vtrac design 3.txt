from __future__ import annotations

from dataclasses import dataclass, field
from typing import Dict, List, Optional, Any


# A single cell in the grid: a digits string at a given ring/column with hot flags.
@dataclass
class Cell:
    digits: str
    hot: bool = False      # hot-zone boost
    superhot: bool = False # stronger boost


# Patterns for one section (Midday/Evening/Combined):
# R2/R4/R6/R8 -> List[Cell] aligned right (cols: index 0==col7 … index 6==col1/newest)
@dataclass
class PatternsGrid:
    by_r: Dict[str, List[Cell]]  # keys: "R2","R4","R6","R8"; values: 7 columns (left..right). Rightmost is newest.


# One section wrapper.
@dataclass
class SectionData:
    section: str               # "Midday" | "Evening" | "Combined"
    set_name: str              # "Set1"/"Set2"/"Set3" (recency weight)
    patterns: PatternsGrid
    last_draw: Optional[str] = None  # last draw in this section, for transit/mirror pressure


# Engine input for a single run across 1–3 sections (Midday/Evening/Combined).
@dataclass
class EngineInput:
    sections: List[SectionData]
    recent_draws: List[str] = field(default_factory=list)  # most recent first (e.g., ["690","123","..."])
    winner_hint: Optional[int] = None                      # V‑TRAC index of the last winner (if known)
    state: Optional[str] = None                            # for optional JSON output


# Evidence record per index (each observation contributes to score).
@dataclass
class EvidenceRec:
    section: str
    ring: str
    col: int           # 7..1 (displayed as col number; 1 is newest)
    feature: str       # e.g., "hit", "hot", "echo_straight", "hidden3"
    weight: float      # signed weight that will be summed into index score
    snippet: str       # short token that matched (e.g., "241" or "2.?2.?5")
    meta: Dict[str, Any] = field(default_factory=dict)


# Evidence bundle per V‑TRAC index
@dataclass
class EvidenceIndex:
    index: int
    score: float
    records: List[EvidenceRec]


# Config knobs (safe to tweak without changing logic)
@dataclass
class AnalyzerConfig:
    # Column weights (rightmost survival): col 1 is newest
    column_weights: Dict[int, float] = field(default_factory=lambda: {7: 0.65, 6: 0.8, 5: 0.9, 4: 1.0, 3: 1.2, 2: 1.6, 1: 2.0})
    # Ring weights (R8 strongest by default)
    ring_weights: Dict[str, float] = field(default_factory=lambda: {"R2": 0.6, "R4": 0.9, "R6": 1.2, "R8": 1.6})
    # Set recency: Set1 (today) > Set2 > Set3 (older)
    set_weights: Dict[str, float] = field(default_factory=lambda: {"Set1": 1.0, "Set2": 0.75, "Set3": 0.6})
    # Hot zone bonuses
    hot_bonus: float = 0.3
    superhot_bonus: float = 0.45
    # Cross‑section reinforcement (index appears in 2+ sections near the tail)
    cross_section_bonus: float = 0.6
    # Hidden‑behind‑one‑digit hit (regex like 2.?2.?5)
    hidden3_bonus: float = 0.35
    # Straight‑order echo (exact 3‑digit straight inside strings)
    straight_echo_bonus: float = 0.25
    # Transit/mirror pressure (downweight indices dominated by last draw digits)
    mirror_pressure: float = 0.35
    # Reduction agreement (external filters concur; used in Part 2 wiring)
    reduction_agreement: float = 0.4
    # Minimum distinct evidence hits before an index is considered
    min_evidence_hits: int = 2
    # Max indices to keep from the scorer (ranked)
    max_indices: int = 12


DEFAULT_CONFIG = AnalyzerConfig()

def normalize3(s: str) -> Optional[str]:
    """Normalize a token to a 3‑digit string preserving leading zeroes."""
    val = normalize_draw(s)
    return val if val and len(val) == 3 else None


def vtrac_index_of(token3: str) -> Optional[int]:
    """Return V‑TRAC index (1..35) for a 3‑digit token (or None for triples)."""
    return get_vtrac_index(token3)


def family_for_index(index: int) -> Set[str]:
    """All boxed combos (singles + doubles) for a V‑TRAC index."""
    return get_index_set(index)


def straights_for_index(index: int) -> Set[str]:
    """All straight permutations for every combo in the index (union)."""
    all_tokens = get_index_set(index)
    acc: Set[str] = set()
    for tok in all_tokens:
        acc |= get_index_straights(tok)
    return acc


def sliding_trigrams(s: str) -> List[str]:
    """All length‑3 substrings (overlapping) inside a digit string."""
    s = "".join(ch for ch in s if ch.isdigit())
    n = len(s)
    return [s[i:i+3] for i in range(n - 2)] if n >= 3 else []


def contains_hidden_triplet(s: str, triplet: str) -> bool:
    """
    Hidden‑behind‑one‑digit pattern (e.g., '225' found as '2.?2.?5').
    Matches order‑preserving with a single wildcard between digits.
    """
    if len(triplet) != 3:
        return False
    s = "".join(ch for ch in s if ch.isdigit())
    a, b, c = triplet[0], triplet[1], triplet[2]
    # a .*? b .*? c with max 1 gap char between successive hits
    i = s.find(a)
    while i != -1 and i + 1 < len(s):
        j_range = range(i + 1, min(i + 3, len(s)))  # allow at most 1 wildcard between a and b
        for j in j_range:
            if s[j] == b:
                k_range = range(j + 1, min(j + 3, len(s)))  # at most 1 wildcard between b and c
                for k in k_range:
                    if s[k] == c:
                        return True
        i = s.find(a, i + 1)
    return False

# modules/vtrac_enhanced/features.py
from __future__ import annotations

from typing import Dict, List, Set, Tuple
from collections import defaultdict

from .types import Cell, PatternsGrid, SectionData, EngineInput, EvidenceRec, EvidenceIndex, AnalyzerConfig
from .mapping import sliding_trigrams, vtrac_index_of, family_for_index, straights_for_index, normalize3, contains_hidden_triplet


def _col_weight(col: int, cfg: AnalyzerConfig) -> float:
    return cfg.column_weights.get(int(col), 1.0)


def _ring_weight(r: str, cfg: AnalyzerConfig) -> float:
    return cfg.ring_weights.get(r, 1.0)


def _set_weight(set_name: str, cfg: AnalyzerConfig) -> float:
    return cfg.set_weights.get(set_name, 1.0)


def _mirror_set_from_last(last_draw: str) -> Set[str]:
    """
    Pressure digits from the last draw and its mirrors (0↔5, 1↔6, 2↔7, 3↔8, 4↔9),
    encoded as single chars to test trigrams quickly.
    """
    if not last_draw or len(last_draw) != 3 or not last_draw.isdigit():
        return set()
    mirrors = {"0": "5", "1": "6", "2": "7", "3": "8", "4": "9",
               "5": "0", "6": "1", "7": "2", "8": "3", "9": "4"}
    acc: Set[str] = set()
    for ch in last_draw:
        acc.add(ch)
        acc.add(mirrors[ch])
    return acc


def _penalize_by_transit(tri: str, bad: Set[str]) -> float:
    """Return negative weight if the trigram is composed only of pressured digits."""
    return -1.0 if len(tri) == 3 and all(c in bad for c in tri) else 0.0


def _scan_cell(section: str, set_name: str, ring: str, col: int, cell: Cell,
               cfg: AnalyzerConfig, last_draw: str | None,
               features_by_index: Dict[int, List[EvidenceRec]]) -> None:
    """
    Scan one cell and append evidence records (positive or negative) to the features map.
    """
    digits = "".join(ch for ch in cell.digits if ch.isdigit())
    if not digits:
        return

    cw = _col_weight(col, cfg) * _ring_weight(ring, cfg) * _set_weight(set_name, cfg)
    hot_bonus = (cfg.hot_bonus if cell.hot else 0.0) + (cfg.superhot_bonus if cell.superhot else 0.0)

    # 1) Direct trigram hits -> index
    for tri in sliding_trigrams(digits):
        tok = normalize3(tri)
        if not tok:
            continue
        idx = vtrac_index_of(tok)
        if not idx:
            continue
        base = 1.0 * cw
        w = base + hot_bonus
        features_by_index[idx].append(EvidenceRec(
            section=section, ring=ring, col=col, feature="hit", weight=w, snippet=tok,
            meta={"base": base, "hot": hot_bonus}
        ))

        # Straight‑order echo (same exact order)
        features_by_index[idx].append(EvidenceRec(
            section=section, ring=ring, col=col, feature="echo_straight",
            weight=cfg.straight_echo_bonus * cw, snippet=tok
        ))

    # 2) Hidden‑behind‑a‑digit recovery for every family token implied by trigrams inside the cell
    #    (Keep cheap: only test the concrete trigrams we actually saw.)
    seen_tris: Set[str] = set(sliding_trigrams(digits))
    candidate_indices: Set[int] = set()
    for tri in seen_tris:
        tok = normalize3(tri)
        if tok:
            idx = vtrac_index_of(tok)
            if idx:
                candidate_indices.add(idx)

    for idx in candidate_indices:
        for fam_tok in family_for_index(idx):
            if contains_hidden_triplet(digits, fam_tok):
                features_by_index[idx].append(EvidenceRec(
                    section=section, ring=ring, col=col,
                    feature="hidden3",
                    weight=cfg.hidden3_bonus * cw,
                    snippet=fam_tok
                ))

    # 3) Transit/mirror pressure (downweight saturated cells)
    if last_draw:
        bad = _mirror_set_from_last(last_draw)
        for tri in seen_tris:
            tok = normalize3(tri)
            if not tok:
                continue
            idx = vtrac_index_of(tok)
            if not idx:
                continue
            pen = _penalize_by_transit(tok, bad)
            if pen:
                features_by_index[idx].append(EvidenceRec(
                    section=section, ring=ring, col=col,
                    feature="mirror_pressure",
                    weight=pen * _set_weight(set_name, cfg),  # only set-weighted
                    snippet=tok
                ))


def build_index_features(inp: EngineInput, cfg: AnalyzerConfig) -> Dict[int, List[EvidenceRec]]:
    """
    Extract per‑index evidence from all sections.
    """
    out: Dict[int, List[EvidenceRec]] = defaultdict(list)

    # Scan every cell in every section
    for sec in inp.sections:
        last = sec.last_draw or (inp.recent_draws[0] if inp.recent_draws else None)
        for ring, columns in sec.patterns.by_r.items():
            # columns: 7 entries (left..right). Rightmost is newest (col==1).
            n = len(columns)
            for i, cell in enumerate(columns):
                colnum = (n - i)  # so index 0 => col n; index n-1 => col 1
                _scan_cell(sec.section, sec.set_name, ring, colnum, cell, cfg, last, out)

    # Cross‑section reinforcement: if an index collected tail evidence in 2+ sections, add a bump.
    # We treat "tail" as columns in {1,2,3}.
    by_index_tail_sections: Dict[int, Set[str]] = defaultdict(set)
    for idx, recs in out.items():
        for rec in recs:
            if rec.col in (1, 2, 3) and rec.feature in ("hit", "hidden3", "echo_straight"):
                by_index_tail_sections[idx].add(rec.section)

    for idx, sects in by_index_tail_sections.items():
        if len(sects) >= 2:
            # attach a single global record
            out[idx].append(EvidenceRec(
                section="ALL", ring="*", col=1,
                feature="cross_section_tail",
                weight=cfg.cross_section_bonus,
                snippet="+".join(sorted(sects))
            ))

    return out

# modules/vtrac_enhanced/score.py
from __future__ import annotations

from typing import Dict, List
from collections import defaultdict

from .types import EvidenceRec, EvidenceIndex, AnalyzerConfig


def score_indices(evidence: Dict[int, List[EvidenceRec]], cfg: AnalyzerConfig) -> List[EvidenceIndex]:
    """
    Sum weights per index and return a ranked list (desc) with evidence kept.
    """
    totals: Dict[int, float] = defaultdict(float)
    cleaned: Dict[int, List[EvidenceRec]] = {}

    for idx, recs in evidence.items():
        # optionally: drop tiny negative noise so small penalties don’t bury good signals
        keep = [r for r in recs if abs(r.weight) > 1e-6]
        cleaned[idx] = keep
        for r in keep:
            totals[idx] += r.weight

    ranked = sorted(
        (EvidenceIndex(index=i, score=totals[i], records=cleaned[i]) for i in cleaned),
        key=lambda x: x.score,
        reverse=True,
    )

    # Filter by minimum evidence hits and clip to configured max
    filtered = [ri for ri in ranked if len(ri.records) >= cfg.min_evidence_hits]
    return filtered[: cfg.max_indices]

def rank_indices(inp: EngineInput, cfg: AnalyzerConfig | None = None) -> List[EvidenceIndex]:
    """
    Main entry: build features and score all V‑TRAC indices.
    """
    cfg = cfg or DEFAULT_CONFIG
    ev = build_index_features(inp, cfg)
    return score_indices(ev, cfg)


def rank_straights_for_index(index: int, end_sequence_bias: float = 0.0,
                             same_order_echo: float = 0.0,
                             rs_due_set: Optional[str] = None) -> List[Dict[str, float]]:
    """
    Rank the eight V‑TRAC straights for a selected index.  Keeps it simple:
    returns [{"straight": "241", "weight": 1.0}, ...] sorted by weight.
    Signals like end_sequence_bias/same_order_echo can be layered from Part 2
    when we have section‑tail context and R‑set due counters ready.
    """
    base = 1.0
    rows = [{"straight": s, "weight": base} for s in sorted(straights_for_index(index))]
    # placeholder for small tie‑breakers; exact heuristics wired in Part 2.
    if end_sequence_bias:
        for r in rows:
            if r["straight"].endswith(("1", "2", "3")):  # trivial default
                r["weight"] += end_sequence_bias
    if same_order_echo:
        for r in rows:
            # nudge numeric order like 123/234/345 etc
            seq = r["straight"]
            if int(seq[1]) - int(seq[0]) == int(seq[2]) - int(seq[1]):
                r["weight"] += same_order_echo
    return sorted(rows, key=lambda z: z["weight"], reverse=True)


def predict(inp: EngineInput, top_n: int = 6, cfg: AnalyzerConfig | None = None) -> Dict[str, object]:
    """
    Full prediction: rank indices + per‑index straight recommendations.
    Returns a JSON‑serializable dict with evidence.
    """
    cfg = cfg or DEFAULT_CONFIG
    ranked = rank_indices(inp, cfg)

    # top‑N indices + straights
    top = ranked[: max(1, top_n)]
    out_rows: List[Dict[str, object]] = []
    for i, ri in enumerate(top, start=1):
        straights = rank_straights_for_index(ri.index)
        out_rows.append({
            "rank": i,
            "index": ri.index,
            "score": round(float(ri.score), 6),
            "straights": [s["straight"] for s in straights],
            "evidence": [asdict(r) for r in ri.records],
        })

    return {
        "generated_at": datetime.utcnow().isoformat(timespec="seconds") + "Z",
        "state": inp.state or "",
        "predictions": out_rows,
        "config": asdict(cfg),
    }


def _predictions_dir() -> Path:
    """
    Resolve data/outputs/predictions using your canonical outputs root.
    If path_handler is not importable (unit test), fall back to local ./data/outputs/predictions.

    else:
        root = Path("data") / "outputs"  # safe default
    p = root / "predictions"
    p.mkdir(parents=True, exist_ok=True)
    return p


def write_predictions_json(payload: Dict[str, object], state: str | None = None) -> str:
    """
    Save prediction payload to data/outputs/predictions/<STATE>_<YYYY-MM-DD>_predictions.json
    (Sits alongside winners/analysis/bundles in your outputs layout.)
    """
    state_name = (state or payload.get("state") or "UNKNOWN")  # type: ignore[union-attr]
    today = datetime.now().strftime("%Y-%m-%d")
    out_path = _predictions_dir() / f"{state_name}_{today}_predictions.json"
    with open(out_path, "w", encoding="utf-8") as f:
        json.dump(payload, f, ensure_ascii=False, indent=2)
    return str(out_path)

from typing import Dict, Iterable, List, Optional, Tuple
from collections import defaultdict

    get_vtrac_index, get_index_straights, get_index_set, normalize_draw

MIRROR_MAP = str.maketrans({"0":"5","1":"6","2":"7","3":"8","4":"9","5":"0","6":"1","7":"2","8":"3","9":"4"})

def slide_triad_windows(s: str) -> List[str]:
    """Unique triads when sliding a window of 3 across s (skip triples like 000/111)."""
    out: List[str] = []
    n = len(s)
    for i in range(n - 2):
        tri = s[i:i+3]
        if len(set(tri)) == 1:  # ignore triples
            continue
        out.append(tri)
    return out

def triads_to_indices(triads: Iterable[str]) -> List[Tuple[str, Optional[int]]]:
    return [(t, get_vtrac_index(t)) for t in triads]

def mirror_string(s: str) -> str:
    return s.translate(MIRROR_MAP)

def digits_to_remove_from_draw(last_draw: Optional[str]) -> set[str]:
    """Transit digit reduction: remove each digit in last_draw and its mirror partner."""
    if not last_draw:
        return set()
    s = normalize_draw(last_draw) or last_draw
    rm = set(s)
    rm |= set(mirror_string(s))
    return rm

def reduce_by_digits(s: str, rm: set[str]) -> str:
    if not rm:
        return s
    return "".join(ch for ch in s if ch not in rm)

def count_survival_across_columns(columns: List[str]) -> Dict[str, int]:
    """For left→right column strings, count how many distinct columns a triad appears in."""
    seen = defaultdict(set)
    for c_idx, col in enumerate(columns):
        for tri in set(slide_triad_windows(col)):
            seen[tri].add(c_idx)
    return {tri: len(cols) for tri, cols in seen.items()}

def index_frequency(pairs: Iterable[Tuple[str, Optional[int]]]) -> Dict[int, int]:
    c = defaultdict(int)
    for _, idx in pairs:
        if idx is not None:
            c[idx] += 1
    return dict(c)

def combos_for_index(index: int) -> List[str]:
    """Canonical 3‑digit boxed combos belonging to an index."""
    return sorted(get_index_set(index))

def normalize_permutation_list(perms: Iterable[str]) -> List[str]:
    """Return unique, order‑aware permutations list (keep appearance order)."""
    seen = set()
    out: List[str] = []
    for p in perms:
        if p not in seen:
            out.append(p)
            seen.add(p)
    return out

def _add_ev(acc: Dict[str, List[Evidence]], key: str, ev: Evidence, emit: bool):
    if emit:
        acc.setdefault(key, []).append(ev)

def score_vtrac_indices(inp: EngineInput, cfg: EnhancedVtracConfig) -> Dict[int, IndexScore]:
    """
    Core evidence aggregation:
      - scan R2/R4/R6/R8 columns across sections & sets
      - presence, survival, hot/superhot weights
      - reduction & mirror assist flags (supportive only)
      - cross‑section consensus
    """
    scores: Dict[int, IndexScore] = {}
    # For cross-section consensus:
    index_seen_sections: Dict[int, set] = defaultdict(set)
    # For straight-order hints (per index we count ordered triads evidence):
    order_hits: Dict[int, Dict[str, float]] = defaultdict(lambda: defaultdict(float))

    # Precompute reduction/mirror helpers:
    rm_digits = digits_to_remove_from_draw(inp.recent_draws[0] if inp.recent_draws else None)

    # Iterate sections -> sets -> rings -> columns
    for sec in inp.sections:
        sec_w = cfg.section_weights.get(sec.section, 1.0) * cfg.set_weights.get(sec.set_name, 1.0)

        for r_key, cells in sec.patterns.by_r.items():
            # Prepare column weights aligned right
            col_weights = cfg.col_weights.get(r_key, [1.0] * 7)
            if len(col_weights) != len(cells):
                if len(cells) < len(col_weights):
                    col_weights = col_weights[-len(cells):]
                else:
                    col_weights += [col_weights[-1]] * (len(cells) - len(col_weights))

            # Survival map over these columns (raw digits)
            cols_raw = [c.digits for c in cells]
            survival = count_survival_across_columns(cols_raw)

            # Process each column
            for c_idx, cell in enumerate(cells):
                base_w = col_weights[c_idx] * sec_w
                hot_boost = (1.0 + (cfg.w_hot if cell.hot else 0.0) + (cfg.w_superhot if cell.superhot else 0.0))

                triads = slide_triad_windows(cell.digits)
                pairs = triads_to_indices(triads)

                # mirror & reduction variants (assist only)
                reduced = reduce_by_digits(cell.digits, rm_digits)
                mirrored = mirror_string(cell.digits)

                triads_reduced = set(slide_triad_windows(reduced))
                triads_mir = set(slide_triad_windows(mirrored))

                for triad, idx in pairs:
                    if idx is None:
                        continue

                    sc = scores.setdefault(idx, IndexScore(index=idx))
                    # presence
                    sc.score += cfg.w_presence * base_w * hot_boost
                    _add_ev(sc.ev, "presence", Evidence("presence", {
                        "section": sec.section, "set": sec.set_name, "ring": r_key, "col": c_idx,
                        "triad": triad, "weight": round(base_w * hot_boost, 4)
                    }), cfg.emit_evidence)

                    # survival bonus for this triad (how many distinct cols in this ring sequence)
                    surv_cols = max(1, survival.get(triad, 1))
                    if surv_cols > 1:
                        sc.score += cfg.w_survival * (surv_cols - 1) * base_w
                        _add_ev(sc.ev, "survival", Evidence("survival", {
                            "triad": triad, "ring": r_key, "cols": surv_cols, "add": round(cfg.w_survival * (surv_cols - 1) * base_w, 4)
                        }), cfg.emit_evidence)

                    # first-hit bonus (rightmost cols have larger col_weights already; here we tag)
                    if c_idx >= len(cells) - 2:  # col2 or col1
                        sc.score += cfg.w_first_hit_bonus * (0.5 if c_idx == len(cells) - 2 else 1.0) * base_w
                        _add_ev(sc.ev, "first_hit", Evidence("first_hit", {
                            "col": c_idx, "ring": r_key, "add": round(cfg.w_first_hit_bonus * (0.5 if c_idx == len(cells)-2 else 1.0) * base_w, 4)
                        }), cfg.emit_evidence)

                    # persistence bonus if this index shows again in next col (simple check)
                    if c_idx < len(cells) - 1:
                        next_tris = set(slide_triad_windows(cells[c_idx + 1].digits))
                        if any(get_vtrac_index(t) == idx for t in next_tris):
                            sc.score += cfg.w_persistence_bonus * base_w
                            _add_ev(sc.ev, "persistence", Evidence("persistence", {
                                "ring": r_key, "col": c_idx, "add": round(cfg.w_persistence_bonus * base_w, 4)
                            }), cfg.emit_evidence)

                    # assists—only if the *same* triad also exists after transforms
                    if triad in triads_reduced:
                        sc.score += cfg.w_reduction_assist * base_w
                        _add_ev(sc.ev, "reduction_assist", Evidence("reduction_assist", {
                            "ring": r_key, "col": c_idx, "triad": triad
                        }), cfg.emit_evidence)
                    if triad in triads_mir:
                        sc.score += cfg.w_mirror_assist * base_w
                        _add_ev(sc.ev, "mirror_assist", Evidence("mirror_assist", {
                            "ring": r_key, "col": c_idx, "triad": triad
                        }), cfg.emit_evidence)

                    # Straight-order hint accumulation for permutation ranking
                    # (the observed triad order is taken as-is)
                    order_hits[idx][triad] += base_w * hot_boost

                    # mark section presence for cross-section consensus later
                    index_seen_sections[idx].add(sec.section)

    # Cross-section consensus and doubles bias
    for idx, sc in scores.items():
        sec_count = len(index_seen_sections.get(idx, set()))
        if sec_count >= 2:
            add = cfg.w_cross_section * (sec_count - 1)
            sc.score += add
            _add_ev(sc.ev, "cross_section", Evidence("cross_section", {"sections": sec_count, "add": add}), cfg.emit_evidence)

        if inp.doubles_context or idx in cfg.doubles_lean_indices:
            sc.score += cfg.w_doubles_bias
            _add_ev(sc.ev, "doubles_bias", Evidence("doubles_bias", {"add": cfg.w_doubles_bias}), cfg.emit_evidence)

    # Normalize to dict[int, IndexScore]
    return scores

def rank_index_combos(scores: Dict[int, IndexScore],
                      cfg: EnhancedVtracConfig,
                      order_hits: Optional[Dict[int, Dict[str, float]]] = None
                     ) -> Dict[int, List[ComboScore]]:
    """
    Rank ordered triads (straight candidates) within each index. If 'order_hits' is not provided,
    derive it from presence evidence.
    """
    by_index: Dict[int, List[ComboScore]] = {}

    # Build order evidence if not supplied
    if order_hits is None:
        order_hits = defaultdict(lambda: defaultdict(float))
        for idx, sc in scores.items():
            for ev in sc.ev.get("presence", []):
                tri = str(ev.details.get("triad", ""))
                if tri and get_vtrac_index(tri) == idx:
                    order_hits[idx][tri] += float(ev.details.get("weight", 1.0))

    for idx, tri_map in order_hits.items():
        # Sort permutations by weight
        perm_items = sorted(tri_map.items(), key=lambda kv: kv[1], reverse=True)
        # Convert to ComboScore
        ranked = [ComboScore(index=idx, combo=p, score=float(w), tags=["order"]) for p, w in perm_items]
        # keep top N
        by_index[idx] = ranked[:cfg.per_index_top_straights]

    return by_index

def select_top_indices(scores: Dict[int, IndexScore], top_n: int = 3) -> List[IndexScore]:
    """Return top‑N indices by score, sorted descending."""
    return sorted(scores.values(), key=lambda s: s.score, reverse=True)[:top_n]

def _cells_from_list(digits_list: List[str], draw_key: str) -> List[Cell]:
    # left→right provided; mark hot flags from the right
    n = len(digits_list)
    hot_n, super_n = _HOT_RULES.get(draw_key, (0, 0))
    hot_from = max(0, n - hot_n)
    super_from = max(0, n - super_n)
    out: List[Cell] = []
    for i, s in enumerate(digits_list):
        out.append(Cell(
            digits=str(s),
            hot=(i >= hot_from),
            superhot=(i >= super_from)
        ))
    return out

def load_from_training_json(json_path: Path) -> EngineInput:
    """
    Load the example structure you shared (state_name -> sections -> sets -> draws -> pattern_variations).
    We ingest the newest context: Set1 / Draw1 by default, and also Set2/Set3 if present (Draw1).
    """
    data = json.loads(Path(json_path).read_text(encoding="utf-8"))
    sections: List[SectionData] = []

    def build_section(section_name: str, set_key: str, draw_key: str, pv: Dict[str, List[str]]):
        by_r = {}
        for r_key in ("R2", "R4", "R6", "R8"):
            if r_key in pv:
                by_r[r_key] = _cells_from_list(pv[r_key], draw_key)
            else:
                by_r[r_key] = []
        sections.append(SectionData(
            section=section_name,
            set_name=set_key,
            patterns=PatternsGrid(by_r=by_r)
        ))

    for section_name, section_body in data.get("sections", {}).items():
        sets = section_body.get("sets", {})
        # Prefer Draw1 as “frontline” for each Set
        for set_key in ("Set1", "Set2", "Set3"):
            draw1 = sets.get(set_key, {}).get("draws", {}).get("Draw1")
            if not draw1:
                continue
            pv = draw1.get("pattern_variations", {})
            build_section(section_name, set_key, "Draw1", pv)

    # optional doubles context flag from the JSON?
    doubles_ctx = data.get("analysis_guide", {}).get("doubles_day", False)
    return EngineInput(sections=sections, doubles_context=bool(doubles_ctx))

# SPDX-License-Identifier: MIT
from __future__ import annotations
from dataclasses import dataclass
from typing import Dict, List, Optional, Tuple
import csv
import json
from pathlib import Path

from .scoring_types import PatternCell, PatternSet, SectionPatterns, EngineInput
from modules.vtrac_reference import normalize_draw

# Some combined tables are CSV, others (training exports) are JSON. We support both.
# Expected columns (CSV): r_key in {"R2","R4","R6","R8"} with 7 columns left→right (C1..C7) *or*
# one wide cell per r_key that encodes 7 cols as a JSON array of strings.
#
# Fallback strategy:
# - Try JSON first (like your Set/Draw exports).
# - Else, look for per-ring columns: R2_C1..R2_C7, R4_C1.., etc.
# - Else, look for R2..R8 columns that already contain 7 strings as comma-joined payload.

R_KEYS = ("R2", "R4", "R6", "R8")

@dataclass
class SectionFiles:
    section: str              # "Midday" | "Evening" | "Combined"
    path: Path                # path to the section table file

def _read_text(p: Path) -> str:
    return p.read_text(encoding="utf-8")

def _parse_json_table(text: str) -> Dict[str, List[str]]:
    """
    JSON table shape we accept:
    {
      "R2": ["559244133877", "5924133877", ...],  # left→right (7->1 or 1->7, we’ll normalize)
      "R4": [...], "R6": [...], "R8": [...]
    }
    """
    j = json.loads(text)
    out: Dict[str, List[str]] = {}
    for r in R_KEYS:
        v = j.get(r) or j.get(r.lower())
        if not v: continue
        out[r] = [str(x) for x in v]
    return out

def _parse_csv_table(text: str) -> Dict[str, List[str]]:
    # Heuristics: look for R2_C1..R2_C7 headers; else R2 single with JSON payload; else joined strings.
    lines = list(csv.DictReader(text.splitlines()))
    if not lines:
        return {}

    # Case A: headers like R2_C1..R2_C7
    headers = lines[0].keys()
    have_block = all(any(h.startswith(f"{r}_C") for h in headers) for r in R_KEYS)
    if have_block:
        out: Dict[str, List[str]] = {r: [] for r in R_KEYS}
        # single row with all columns is expected; if multiple, take the last (most recent)
        row = lines[-1]
        for r in R_KEYS:
            cols = [row.get(f"{r}_C{i}", "") for i in range(1, 8)]
            out[r] = [c.strip() for c in cols if c is not None]
        return out

    # Case B: headers like R2, R4... each cell stores JSON array
    out_json: Dict[str, List[str]] = {}
    for r in R_KEYS:
        if r in lines[-1] and lines[-1][r]:
            try:
                out_json[r] = [str(x) for x in json.loads(lines[-1][r])]
            except Exception:
                pass
    if out_json:
        return out_json

    # Case C: last resort—assume each ring cell is a comma-separated list of 7 strings
    out_fallback: Dict[str, List[str]] = {}
    row = lines[-1]
    for r in R_KEYS:
        cell = (row.get(r) or "").strip()
        if not cell:
            continue
        parts = [p.strip() for p in cell.split(",")]
        out_fallback[r] = parts
    return out_fallback

def _normalize_cols_left_to_right(cols: List[str]) -> List[str]:
    # We expect left→right oldest→newest. If we detect 1..7 newest→oldest, reverse.
    # Heuristic: favor the version where the rightmost string tends to be shorter (end-strings).
    if not cols:
        return cols
    a = cols
    b = list(reversed(cols))
    def right_tail_len(v: List[str]) -> int:
        return len(v[-1]) if v and isinstance(v[-1], str) else 999
    return a if right_tail_len(a) <= right_tail_len(b) else b

def _cells_from_table(section: str, ring_to_cols: Dict[str, List[str]]) -> List[PatternCell]:
    cells: List[PatternCell] = []
    for r_key in R_KEYS:
        cols = ring_to_cols.get(r_key, [])
        cols = _normalize_cols_left_to_right(cols)
        for c_idx, s in enumerate(cols):
            # Hot zone heuristic: rightmost columns are hotter; we set a boolean here.
            hot = c_idx >= max(0, len(cols) - 3)
            cells.append(PatternCell(r_key=r_key, column_idx=c_idx, digits=str(s), hot=hot, section=section))
    return cells

def _pattern_set(ring_to_cols: Dict[str, List[str]], section: str) -> SectionPatterns:
    by_r: Dict[str, List[PatternCell]] = {r: [] for r in R_KEYS}
    for cell in _cells_from_table(section, ring_to_cols):
        by_r[cell.r_key].append(cell)
    return SectionPatterns(section=section, patterns=PatternSet(by_r=by_r))

def load_from_combined_tables(midday: Optional[Path], evening: Optional[Path], combined: Optional[Path],
                              last_draw: Optional[str] = None,
                              doubles_context: bool = False) -> EngineInput:
    sections: List[SectionPatterns] = []
    for section, p in (("Midday", midday), ("Evening", evening), ("Combined", combined)):
        if not p: 
            continue
        text = _read_text(p)
        ring_to_cols: Dict[str, List[str]] = {}
        # Try JSON then CSV
        try:
            ring_to_cols = _parse_json_table(text)
        except Exception:
            pass
        if not ring_to_cols:
            ring_to_cols = _parse_csv_table(text)
        if not ring_to_cols:
            continue
        sections.append(_pattern_set(ring_to_cols, section))

    return EngineInput(sections=sections, last_draw=normalize_draw(last_draw or ""), doubles_context=doubles_context)

from __future__ import annotations
import json
from pathlib import Path
from typing import Dict, Any, Optional, List, Tuple
from datetime import date as dt_date

def ensure_dir(p: Path) -> None:
    p.mkdir(parents=True, exist_ok=True)

def predictions_path(state: str, d: Optional[str]) -> Path:
    dd = d or dt_date.today().isoformat()
    return Path("data/outputs/predictions") / state / f"{dd}_vtrac_predictions.json"

def write_predictions_json(state: str, d: Optional[str], payload: Dict[str, Any]) -> Path:
    out = predictions_path(state, d)
    ensure_dir(out.parent)
    out.write_text(json.dumps(payload, indent=2), encoding="utf-8")
    return out

# modules/vtrac_enhanced/adapters.py
# SPDX-License-Identifier: MIT
from __future__ import annotations
from dataclasses import dataclass
from pathlib import Path
from typing import Dict, List, Optional, Tuple
import json
import re

import pandas as pd

from .scoring_types import PatternCell, PatternSet, SectionPatterns, EngineInput

# Heuristics to find ring/column headers in combined tables.
# We match "R2", "R4", "R6", "R8" headers followed by a column number (either _C1.._C7 or just 1..7).
_RING_PATTERNS = {"R2": re.compile(r"^R?2[_\-]?(C)?([1-7])$"),
                  "R4": re.compile(r"^R?4[_\-]?(C)?([1-7])$"),
                  "R6": re.compile(r"^R?6[_\-]?(C)?([1-7])$"),
                  "R8": re.compile(r"^R?8[_\-]?(C)?([1-7])$")}

@dataclass
class TablesPaths:
    root: Path
    midday_csv: Path
    evening_csv: Path
    combined_csv: Path

def _detect_ring_columns(df: pd.DataFrame) -> Dict[str, List[str]]:
    """
    Return ordered column names per ring {R2:[...], R4:[...], R6:[...], R8:[...]}
    Columns are returned left->right (1→7).
    We accept headers like: R2_C7, R2-7, R2 7, 2_7, etc.
    """
    per_ring_cols: Dict[str, List[Tuple[int, str]]] = {"R2": [], "R4": [], "R6": [], "R8": []}
    for col in df.columns:
        c = str(col).strip().replace(" ", "").upper()
        for ring, rx in _RING_PATTERNS.items():
            m = rx.match(c)
            if m:
                idx = int(m.group(2))
                per_ring_cols[ring].append((idx, col))
                break
    # sort asc, then keep left->right order
    ordered = {}
    for ring, items in per_ring_cols.items():
        ordered[ring] = [c for _, c in sorted(items, key=lambda t: t[0])]
    return ordered

def _to_pattern_set(df: pd.DataFrame, section: str) -> PatternSet:
    """
    Produces PatternSet by reading the recognized R2/R4/R6/R8 columns from the DataFrame.
    Each 'column' is a PatternCell with digits string; hot-zone flags are left False here.
    Hot zones can be injected by your metadata JSON if available.
    """
    cols = _detect_ring_columns(df)
    by_r: Dict[str, List[PatternCell]] = {}
    for ring, names in cols.items():
        cells: List[PatternCell] = []
        # Ensure we traverse left->right; leftmost is older (col7) → rightmost is newest (col1)
        # If your CSV is 7→1 or 1→7, the detection sorts numerically, so we get a canonical left->right list.
        for i, colname in enumerate(names):
            series = df[colname].dropna().astype(str)
            # Join the column into one long string (as your excel images show).
            # If your CSV stores each row as a separate "stream", we concatenate—mirrors your visual table.
            digits = "".join(series.tolist())
            cells.append(PatternCell(r_key=ring, column_idx=i, digits=digits, hot=False, section=section))
        by_r[ring] = cells
    return PatternSet(by_r=by_r)

def load_from_tables(table_paths: TablesPaths, inject_hot_metadata: Optional[Dict] = None,
                     last_draw: Optional[str] = None, doubles_context: bool = False) -> EngineInput:
    """
    Reads Midday/Evening/Combined CSV tables and returns EngineInput for the engine.
    `inject_hot_metadata` (optional) can mark hot/superhot trailing segments by ring/column.
    """
    sections: List[SectionPatterns] = []
    for section, p in [("Midday", table_paths.midday_csv),
                       ("Evening", table_paths.evening_csv),
                       ("Combined", table_paths.combined_csv)]:
        df = pd.read_csv(p)
        pset = _to_pattern_set(df, section=section)
        # Optional: inject hot metadata (if you store it in JSON). Otherwise, engine will apply defaults.
        if inject_hot_metadata and section in inject_hot_metadata:
            md = inject_hot_metadata[section]
            for ring, cells in pset.by_r.items():
                hot_cols = md.get(ring, {})
                for cell in cells:
                    # mark column hot if present
                    cell.hot = bool(hot_cols.get(str(cell.column_idx), False))
        sections.append(SectionPatterns(section=section, patterns=pset))
    return EngineInput(sections=sections, last_draw=last_draw, doubles_context=doubles_context)

def load_from_training_json(json_path: Path, state_hint: Optional[str] = None) -> EngineInput:
    """
    Ingests the “training JSON” structure you shared (Midday/Evening/Combined with Set1/2/3→Draws→R2/4/6/8 arrays).
    Collapses them into PatternCells consistent with our engine input.
    """
    blob = json.loads(Path(json_path).read_text(encoding="utf-8"))
    sections: List[SectionPatterns] = []
    for section_name in blob["sections"].keys():
        section = blob["sections"][section_name]
        # By design we prioritise Set1/Draw1..7 progression (current day), then Set2, Set3
        # We concatenate each ring’s strings from older→newer columns.
        by_r: Dict[str, List[PatternCell]] = {"R2": [], "R4": [], "R6": [], "R8": []}
        # Build columns left->right (older to newer) by stacking Draw7..Draw1 (if present)
        def push_cells(set_obj, sec_label):
            if "draws" not in set_obj:
                return
            # Collect per ring the column strings in left->right layout.
            # We will normalize to exactly 7 columns if available.
            ring_cols: Dict[str, List[str]] = {"R2": [], "R4": [], "R6": [], "R8": []}
            # Draw1 expected to be 7 columns, Draw2 → 6, ... etc. We append in decreasing length to form 7 columns.
            for draw_name, draw_obj in set_obj["draws"].items():
                pvars = draw_obj.get("pattern_variations", {})
                for ring in ["R2", "R4", "R6", "R8"]:
                    arr = pvars.get(ring, [])
                    # Append strings left->right
                    for s in arr:
                        ring_cols[ring].append(s)
            # Now create PatternCells with column_idx increasing left->right
            for ring, arr in ring_cols.items():
                for i, s in enumerate(arr):
                    by_r[ring].append(PatternCell(r_key=ring, column_idx=i, digits=str(s), section=sec_label))
        # Set3→Set2→Set1
        for set_name in ["Set3", "Set2", "Set1"]:
            if set_name in section.get("sets", {}):
                push_cells(section["sets"][set_name], section_name)
        sections.append(SectionPatterns(section=section_name, patterns=PatternSet(by_r=by_r)))
    last_draw = None
    # Try to fetch a most-recent draw from blob (optional)
    try:
        last_draw = blob["sections"]["Midday"]["sets"]["Set1"]["draws"]["Draw1"]["draw_data"][-1]
    except Exception:
        pass
    return EngineInput(sections=sections, last_draw=last_draw, doubles_context=False)

# src/core/vtrac_input_adapter.py
# SPDX-License-Identifier: MIT
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Dict, List, Optional, Tuple

import json
import pandas as pd
import yaml

# Engine types from the enhanced package (Part 1–2)
from modules.vtrac_enhanced.scoring_types import (
    PatternCell, PatternSet, SectionPatterns, EngineInput
)

# ---------------------------------------------------------------------
# YAML schema (configs/tables_schema.yml) defines, per state & section:
# - rings: R2/R4/R6/R8
# - for each ring: ordered list of column names (left -> right) in the CSV
#   e.g., sections: Midday: { R2: [r2_c7, ..., r2_c1], R4: [...], ... }
# This decouples code from any sheet-specific column naming.
# ---------------------------------------------------------------------

@dataclass
class TableSchema:
    rings_by_section: Dict[str, Dict[str, List[str]]]

    @staticmethod
    def load(schema_path: Path, state: str) -> "TableSchema":
        with open(schema_path, "r", encoding="utf-8") as fh:
            raw = yaml.safe_load(fh)
        # allow a "default" mapping as fallback
        if state in raw:
            return TableSchema(rings_by_section=raw[state]["sections"])
        elif "default" in raw:
            return TableSchema(rings_by_section=raw["default"]["sections"])
        raise KeyError(f"No schema found for state '{state}' in {schema_path}")

# ------------------------------- Helpers ------------------------------

def _mk_cells_from_csv(df: pd.DataFrame,
                       section: str,
                       ring: str,
                       col_names_left_to_right: List[str],
                       hot_spec: Optional[Dict[int, str]] = None) -> List[PatternCell]:
    """
    Build PatternCells for this section/ring using the CSV column order.
    col_names_left_to_right is left -> right (oldest -> newest); we will keep that.
    hot_spec (optional): map of rightmost indexes (1-based) to hot tags ("HOT"/"SUPER").
    """
    cells: List[PatternCell] = []
    # Defensive: some tables store one row per ring; others store multiple rows. We concatenate all rows for a column.
    for cidx, col_name in enumerate(col_names_left_to_right):
        if col_name not in df.columns:
            # If the column is absent, treat as empty cell.
            raw = ""
        else:
            series = df[col_name].dropna().astype(str)
            # Join multiple rows vertically into a single column string (commonly how the charts are exported)
            raw = "".join(series.tolist())
        # Determine hotness from rightmost perspective:
        # hot_spec keyed by "distance from right end" (1 = rightmost), but our list is left -> right.
        # Compute position from right:
        right_pos = len(col_names_left_to_right) - cidx
        hot_flag = False
        if hot_spec:
            tag = hot_spec.get(right_pos)
            hot_flag = tag in ("HOT", "SUPER")
        cells.append(PatternCell(r_key=ring, column_idx=cidx, digits=raw, hot=hot_flag, section=section))
    return cells

def _mk_section_patterns_from_csv(df_map: Dict[str, pd.DataFrame],
                                  schema: TableSchema,
                                  section: str,
                                  hot_spec: Optional[Dict[int, str]]) -> SectionPatterns:
    rings = schema.rings_by_section[section]  # dict of ring -> [col names]
    by_r: Dict[str, List[PatternCell]] = {}
    for ring, cols in rings.items():
        # Pick the right CSV table for the ring if you export rings to separate sheets; else use the same df.
        # Convention: df_map may have a separate key per ring, else a single 'main' DataFrame.
        df = df_map.get(ring, df_map.get("main"))
        if df is None:
            # fallback to an empty frame
            df = pd.DataFrame({c: [] for c in cols})
        by_r[ring] = _mk_cells_from_csv(df, section=section, ring=ring,
                                        col_names_left_to_right=cols,
                                        hot_spec=hot_spec)
    return SectionPatterns(section=section, patterns=PatternSet(by_r=by_r))

# ------------------------------- Public -------------------------------

def load_engine_input_from_csv(
    state: str,
    csv_paths_by_section: Dict[str, Dict[str, Path]],
    schema_path: Path,
    last_draw: Optional[str] = None,
    doubles_context: bool = False,
    hot_spec_by_section: Optional[Dict[str, Dict[int, str]]] = None,
) -> EngineInput:
    """
    csv_paths_by_section:
      {
        "Midday":   {"main": Path(".../Midday_Combined.csv")},
        "Evening":  {"main": Path(".../Evening_Combined.csv")},
        "Combined": {"main": Path(".../Combined_Combined.csv")}
      }
    If you export rings to separate CSVs, use keys "R2"/"R4"/"R6"/"R8" instead of "main".
    hot_spec_by_section (optional):
      Per section, a map of rightmost column positions to hotness:
        e.g., { "Midday": {1: "SUPER", 2: "HOT", 3: "HOT"}, ... }
      We keep this flexible so you can tune hot windows per set if needed later.
    """
    schema = TableSchema.load(schema_path, state)
    sections: List[SectionPatterns] = []

    for section, file_map in csv_paths_by_section.items():
        df_map: Dict[str, pd.DataFrame] = {}
        for key, p in file_map.items():
            if not p.exists():
                # Build empty DataFrame to keep shape predictable
                df_map[key] = pd.DataFrame()
            else:
                df_map[key] = pd.read_csv(p)
        hot_spec = (hot_spec_by_section or {}).get(section)
        sections.append(_mk_section_patterns_from_csv(df_map, schema, section, hot_spec))

    return EngineInput(sections=sections, last_draw=last_draw, doubles_context=doubles_context)

def load_engine_input_from_training_json(
    json_path: Path,
    last_draw: Optional[str] = None,
    doubles_context: bool = False
) -> EngineInput:
    """
    Alternate loader for the JSON ‘training’ shape you shared (Set1..Set3, Draw1..).
    Useful for offline validation and unit tests.
    """
    with open(json_path, "r", encoding="utf-8") as fh:
        data = json.load(fh)

    # Expect structure similar to your example; adapt minimally:
    sections: List[SectionPatterns] = []
    for section_name in ("Midday", "Evening", "Combined"):
        if section_name not in data.get("sections", {}):
            continue
        sec = data["sections"][section_name]
        # Flatten R2/R4/R6/R8 arrays of strings (left->right)
        by_r = {}
        for ring in ("R2", "R4", "R6", "R8"):
            cols = sec.get(ring, [])
            cells = []
            for cidx, s in enumerate(cols):
                hot = False
                # A very light heuristic: flag the last 1–3 columns as hot in Draw1/Set1 context
                if len(cols) - cidx in (1, 2, 3):
                    hot = True
                cells.append(PatternCell(r_key=ring, column_idx=cidx, digits=s, hot=hot, section=section_name))
            by_r[ring] = cells
        sections.append(SectionPatterns(section=section_name, patterns=PatternSet(by_r=by_r)))

    return EngineInput(sections=sections, last_draw=last_draw, doubles_context=doubles_context)

def run_vtrac_enhanced(
    state: str,
    d: Optional[_date] = None,
    schema_path: Optional[Path] = None,
    cfg: Optional[EnhancedVtracConfig] = None,
    last_draw: Optional[str] = None,
    doubles_context: bool = False,
    write_debug: bool = True
) -> Tuple[Path, Dict]:
    """
    End-to-end: load tables -> score indices -> rank straights -> write predictions JSON.
    Returns (predictions_path, predictions_dict).
    """
    d = d or _date.today()
    cfg = cfg or DEFAULT_CONFIG
    schema_path = schema_path or Path("configs") / "tables_schema.yml"

    # Load tables into EngineInput
    csv_paths = _resolve_table_paths(state)
    # Optional: per-section hot specs (rightmost columns)
    hot_spec = {
        "Midday":   {1: "SUPER", 2: "HOT", 3: "HOT"},
        "Evening":  {1: "SUPER", 2: "HOT", 3: "HOT"},
        "Combined": {1: "SUPER", 2: "HOT", 3: "HOT"},
    }
    eng_inp: EngineInput = load_engine_input_from_csv(
        state=state,
        csv_paths_by_section=csv_paths,
        schema_path=schema_path,
        last_draw=last_draw,
        doubles_context=doubles_context,
        hot_spec_by_section=hot_spec,
    )

    # Score indices (evidence features include: ring/column presence, hot-zones, cross-section consensus,
    # persistence / first-hit, doubles affinity, optional overlays). Then select top indices and rank straights.
    idx_scores = score_vtrac_indices(eng_inp, cfg)
    top_df = select_top_indices(idx_scores, top_k=3, cfg=cfg)  # keeps full ranked list internally
    straight_map = rank_index_combos(eng_inp, top_df, cfg)     # top straights per index with tags/why

    # Assemble predictions payload (bundle-friendly shape)
    payload = {
        "date": d.isoformat(),
        "state": state,
        "engine": "vtrac_enhanced@2.0.0",
        "indices_ranked": [],
    }

    # Top list (but keep all scores in debug)
    for _, row in top_df.iterrows():
        index = int(row["index"])
        entry = {
            "index": index,
            "score": float(row["score"]),
            "evidence": row["evidence"],   # explain tags/markers from the engine
            "straights": [],
            "vtrac_boxed": sorted(list(row["boxed"])),
        }
        for combo_score in straight_map.get(index, []):
            entry["straights"].append({
                "perm": combo_score.combo,
                "perm_score": float(combo_score.score),
                "why": combo_score.tags,
            })
        payload["indices_ranked"].append(entry)

    # Write artifacts
    pred_path = _resolve_predictions_path(state, d)
    with open(pred_path, "w", encoding="utf-8") as fh:
        json.dump(payload, fh, indent=2)

    if write_debug:
        analysis_dir = _resolve_analysis_dir(state, d)
        # Full ranked table (CSV) for audit
        try:
            top_df.to_csv(analysis_dir / "indices_ranked.csv", index=False)
        except Exception:
            pass

    return pred_path, payload

from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Dict, Any, Optional, List, Tuple
import json

# Prefer your canonical V-TRAC reference (already in your repo)
try:
    from modules.vtrac_reference import get_vtrac_index, normalize_draw
except Exception:
    # Safe fallbacks if the import path differs in your tree
    def normalize_draw(s: str) -> Optional[str]:
        if s is None: return None
        s = str(s).strip().replace(" ", "").replace("-", "").replace(",", "")
        if not s.isdigit(): return None
        return s.zfill(3) if len(s) <= 3 else None
    def get_vtrac_index(draw: str) -> Optional[int]:
        # Minimal fallback if the reference cannot be imported (returns None for triples)
        s = normalize_draw(draw)
        if s is None: return None
        if len(set(s)) == 1: return None
        return None  # fallback has no mapping; caller should tolerate None

# Path handler (preferred) with safe fallbacks
try:
    from utils import path_handler
except Exception:
    path_handler = None

def _predictions_dir_for_state(state: str) -> Path:
    if path_handler and hasattr(path_handler, "get_predictions_output_dir"):
        return Path(path_handler.get_predictions_output_dir(state))
    # fallback: project-relative path
    return Path("data/outputs/predictions") / state

def _analysis_dir_for_day(state: str, date_str: str) -> Path:
    if path_handler and hasattr(path_handler, "get_analysis_output_dir"):
        return Path(path_handler.get_analysis_output_dir("vtrac", state)) / date_str
    return Path("data/outputs/analysis/vtrac") / state / date_str

def _load_predictions(state: str, date_str: str) -> Tuple[Path, Dict[str, Any]]:
    pdir = _predictions_dir_for_state(state)
    p = pdir / f"{date_str}_vtrac_predictions.json"
    payload = json.loads(p.read_text(encoding="utf-8"))
    return p, payload

def _winner_index(winner: str) -> Optional[int]:
    try:
        return get_vtrac_index(winner)
    except Exception:
        return None

def _hit_at_k(indices_ranked: List[Dict[str, Any]], winner_index: Optional[int], k: int) -> bool:
    if winner_index is None: return False
    tops = [int(e["index"]) for e in indices_ranked[:k]]
    return winner_index in tops

def _straight_hit(indices_ranked: List[Dict[str, Any]], winner: str) -> bool:
    w = normalize_draw(winner)
    if not w: return False
    for e in indices_ranked:
        for sc in e.get("straights", []):
            if normalize_draw(sc.get("perm")) == w:
                return True
    return False

def attach_winner_and_evaluate(
    state: str,
    date_str: str,
    winner: str,
    predictions_path: Optional[Path] = None
) -> Path:
    """
    Joins a day's predictions with the provided winner, computes quick metrics,
    and writes an evaluation JSON to the analysis tree:
      data/outputs/analysis/vtrac/<STATE>/<DATE>/evaluation.json
    """
    # Load predictions
    pred_path, payload = _load_predictions(state, date_str) if not predictions_path \
                         else (predictions_path, json.loads(predictions_path.read_text(encoding="utf-8")))
    indices = payload.get("indices_ranked", [])
    w_norm = normalize_draw(winner)
    vtrac_idx = _winner_index(w_norm) if w_norm else None

    eval_obj: Dict[str, Any] = {
        "date": date_str,
        "state": state,
        "winner": w_norm,
        "winner_vtrac_index": vtrac_idx,
        "engine": payload.get("engine", "unknown"),
        "metrics": {
            "hit_at_1": _hit_at_k(indices, vtrac_idx, 1),
            "hit_at_3": _hit_at_k(indices, vtrac_idx, 3),
            "straight_hit": _straight_hit(indices, w_norm or ""),
        },
        "top": [
            {"index": int(e["index"]), "score": float(e.get("score", 0.0))}
            for e in indices[:3]
        ],
    }

    # Save under analysis folder for the day
    out_dir = _analysis_dir_for_day(state, date_str)
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / "evaluation.json"
    out_path.write_text(json.dumps(eval_obj, indent=2), encoding="utf-8")
    return out_path

# tools/vtrac_backtest_cli.py
# SPDX-License-Identifier: MIT
from __future__ import annotations

import argparse
import csv
from pathlib import Path
from typing import Dict, Any
from src.core.vtrac_winner_eval import attach_winner_and_evaluate

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--state", required=True)
    ap.add_argument("--winners-csv", required=True, help="CSV with columns: date, winner")
    ap.add_argument("--out", default="vtrac_backtest_results.csv")
    args = ap.parse_args()

    state = args.state
    winners = []
    with open(args.winners_csv, "r", encoding="utf-8") as fh:
        rdr = csv.DictReader(fh)
        for row in rdr:
            winners.append({"date": row["date"].strip(), "winner": row["winner"].strip()})

    rows = []
    for w in winners:
        try:
            ev_path = attach_winner_and_evaluate(state, w["date"], w["winner"])
            ev = json.loads(Path(ev_path).read_text(encoding="utf-8"))
            m = ev["metrics"]
            rows.append({
                "date": ev["date"],
                "winner": ev["winner"],
                "winner_vtrac_index": ev["winner_vtrac_index"],
                "hit_at_1": int(m["hit_at_1"]),
                "hit_at_3": int(m["hit_at_3"]),
                "straight_hit": int(m["straight_hit"]),
                "top1_index": ev["top"][0]["index"] if ev["top"] else "",
                "top2_index": ev["top"][1]["index"] if len(ev["top"]) > 1 else "",
                "top3_index": ev["top"][2]["index"] if len(ev["top"]) > 2 else "",
            })
        except Exception as ex:
            rows.append({
                "date": w["date"], "winner": w["winner"], "winner_vtrac_index": "",
                "hit_at_1": "", "hit_at_3": "", "straight_hit": "",
                "top1_index": "", "top2_index": "", "top3_index": "", "error": str(ex)
