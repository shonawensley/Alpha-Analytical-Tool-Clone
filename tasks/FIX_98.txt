# modules/vtrac_matchers.py

from typing import Iterable, List, Tuple

# V‑TRAC map (5 buckets): 0/5→1, 1/6→2, 2/7→3, 3/8→4, 4/9→5
VT = {'0':'1','5':'1','1':'2','6':'2','2':'3','7':'3','3':'4','8':'4','4':'5','9':'5'}

def vt_stream(digits: str) -> str:
    return ''.join(VT.get(ch, '') for ch in digits if ch.isdigit())

def _windows(s: str, kmin: int, kmax: int) -> Iterable[Tuple[int,int,str]]:
    n = len(s)
    for k in range(kmin, min(kmax, n) + 1):
        for i in range(0, n - k + 1):
            yield i, i+k, s[i:i+k]

def match_vt_straight_strict(cell: str, vt_winner: str) -> List[Tuple[int,int]]:
    """3‑digit substring whose vTRAC equals vt_winner exactly."""
    hits = []
    for i, j, sub in _windows(cell, 3, 3):
        if vt_stream(sub) == vt_winner:
            hits.append((i, j))
    return hits

def _reduce_runs_to_len3(vs: str) -> List[str]:
    """
    Given a vTRAC stream (e.g., '444555' from '336644'), return all length‑3
    sequences obtainable by deleting duplicate run elements (preserving order).
    Example: '4455' -> ['455']; '444555' -> ['455'].
    """
    if not vs:
        return []
    # Compress runs but keep counts
    runs, counts = [], []
    last = vs[0]; cnt = 1
    for ch in vs[1:]:
        if ch == last:
            cnt += 1
        else:
            runs.append(last); counts.append(cnt)
            last, cnt = ch, 1
    runs.append(last); counts.append(cnt)

    # If runs already 3, join directly
    if len(runs) == 3:
        return [''.join(runs)]

    # If runs > 3, not obtainable by dropping duplicates only
    if len(runs) > 3:
        return []

    # If runs == 2, we can only get to len=3 by duplicating one of the runs –
    # we treat “value‑blocked” as keeping BOTH of the trailing run’s elements if present.
    # So produce [runs[0] + runs[1] + runs[1]]
    if len(runs) == 2:
        return [runs[0] + runs[1] + runs[1]]

    # len(runs) == 1 cannot reach 3
    return []

def match_vt_straight_tolerant(cell: str, vt_winner: str) -> List[Tuple[int,int]]:
    """
    4..8 digit window whose vTRAC stream can be reduced to vt_winner by deleting
    duplicate run elements while preserving order (value‑blocked case).
    """
    hits = []
    for i, j, sub in _windows(cell, 4, 8):
        vt = vt_stream(sub)
        for cand in _reduce_runs_to_len3(vt):
            if cand == vt_winner:
                hits.append((i, j))
                break
    return hits


Why this catches your examples

3344 → vt_stream('3344') == '4455' → _reduce_runs_to_len3('4455') == ['455'] → matches vt_winner='455'.

336644 → '444555' → reduces to '455' → matches.

(If you also want Winner (value), copy the tolerant approach but compare to the digit permutations of the winner after collapsing duplicate runs of the gap digit. Same idea; different alphabet.)
