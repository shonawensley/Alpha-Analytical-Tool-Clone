In package_training_bundle(... ) (or the helper that gathers winners files), add a parameter and filter:

def package_training_bundle(..., include_combined: bool = False, ...):
    packaged_variants = (
        ("Midday", "Evening", "Combined") if include_combined
        else PACKAGED_VARIANTS_DEFAULT
    )

    winners_src = analysis_root / state / "analyzer_v2" / "winners"
    winners_dst = bundle_root / "winners"
    winners_dst.mkdir(parents=True, exist_ok=True)

    for f in winners_src.glob(f"{stamp}_*_winner_*.*"):
        m = ARTIFACT_RE.match(f.name)
        if not m:
            continue
        if m["variant"] not in packaged_variants:
            continue
        shutil.copy2(f, winners_dst / f.name)
        manifest["files"].append(str(Path("winners") / f.name))

    # also record what we chose
    manifest["packaged_variants"] = list(packaged_variants)


Important: Do not delete or suppress the Combined files under analyzer_v2/winners/. We’re only filtering what the bundle copies.
