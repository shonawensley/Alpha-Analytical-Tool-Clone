
Part 1 — Hot Zones core (models + V‑TRAC + JSON loader + scanner + CLI)

What this implements

Data model & loader for your JSON table environment (Sections→Sets→Draws with R2/R4/R6/R8 strings + metadata).

V‑TRAC engine (digit↔V mapping; VT family canonicalization; VT‑straight detection).

Hot‑zone scanner: mines Set1 “late columns” (super‑hot) and starred boxes for family clusters, vertical (R2/R4/R6/R8) support, horizontal survival, cross‑variant reinforcement, plus VT‑straight/boxed cues.

CLI: validate (quick table/recency checks) and scan-json (produce hotzones_candidates.{json,csv}).

Design notes (why this fits AAT9)

The Control Center remains the only writer of winner maps/flags/overlays; analyzers emit lean “brain” artifacts. This module follows that contract so it slots cleanly into your current repo layout.

We reuse Digit‑Reduction/V‑TRAC feature ideas already present in your per‑item CSVs (e.g., vt_only_lane, funnel_precol1, ls_col_42) when we add richer scoring in Part 2; the code below captures the same concepts (pre‑col1 funnels, VT‑lane evidence), so the Aggregator can join on shared fields later.

# hotzones/vtrac.py

from __future__ import annotations
from typing import Iterable, List, Tuple, Dict, Set

# V‑TRAC map: (0,5)->1, (1,6)->2, (2,7)->3, (3,8)->4, (4,9)->5
DIGIT_TO_V: Dict[str, int] = {
    "0": 1, "5": 1,
    "1": 2, "6": 2,
    "2": 3, "7": 3,
    "3": 4, "8": 4,
    "4": 5, "9": 5,
}

def v_of_digit(d: str) -> int:
    """Map a single decimal digit to its V‑TRAC class (1..5)."""
    if d not in DIGIT_TO_V:
        raise ValueError(f"Invalid digit '{d}' for V‑TRAC mapping")
    return DIGIT_TO_V[d]

def map_digits_to_v(s: str) -> List[int]:
    """Map a string of digits (e.g., '611888') to V‑TRAC classes, preserving order."""
    return [v_of_digit(ch) for ch in s if ch.isdigit()]

def _top_three_by_frequency(digits: Iterable[str]) -> List[str]:
    """Pick up to 3 digits by descending frequency; tie‑break by numeric order."""
    from collections import Counter
    c = Counter([d for d in digits if d.isdigit()])
    if not c:
        return []
    # Sort by (-count, digit) then take up to 3
    ordered = sorted(c.items(), key=lambda kv: (-kv[1], int(kv[0])))
    return [k for k, _ in ordered[:3]]

def canonical_vtriad_from_string(s: str) -> Tuple[Tuple[int, ...], Tuple[str, ...]]:
    """
    Infer the 3‑value V‑TRAC family from an arbitrary pattern string:

    Strategy:
      1) If the string has <=3 unique digits, use those.
      2) Otherwise, take top‑3 digits by frequency.
      3) Return (vt_triad_sorted, raw_digits_sorted) for transparency.

    Example: '611888' -> digits {'6','1','8'} -> vt [2,2,4,4,4]? -> triad (2,4,2?) → canonical (2,4,?) => (2,4,?) sorted => (2,4,?)
    We store the set (up to 3 uniques) as an ordered triple with multiplicity flattened by sort.
    """
    uniq = sorted(set(ch for ch in s if ch.isdigit()), key=int)
    if len(uniq) == 0:
        return tuple(), tuple()
    if len(uniq) <= 3:
        base = uniq
    else:
        base = _top_three_by_frequency(s)

    vtriad = tuple(sorted(v_of_digit(d) for d in base))
    raw_sorted = tuple(sorted(base, key=int))
    return vtriad, raw_sorted

def vt_family_id(vtriad: Tuple[int, ...]) -> str:
    """Human‑friendly family id, e.g., (2,4,1) -> 'v124' (sorted already)."""
    if not vtriad:
        return "v—"
    return "v" + "".join(str(v) for v in vtriad)

def vt_permutations_for_digits(d1: str, d2: str, d3: str) -> Set[Tuple[str, str, str]]:
    """
    All straight permutations for a 3‑digit combo (handles singles/doubles).
    Example: ('6','1','8') -> 6! or fewer if doubles -> return unique 3‑tuples.
    """
    from itertools import permutations
    perm = set(permutations([d1, d2, d3], 3))
    return perm

def has_vt_straight_lane(s: str, vt_sequence: Tuple[int, int, int]) -> bool:
    """
    Does 's' contain a contiguous length‑3 subsequence whose V‑TRAC map equals vt_sequence?
    This captures VT‑straight cues even when literal digits differ by mirrors.
    """
    if len(vt_sequence) != 3:
        return False
    seq = map_digits_to_v(s)
    if len(seq) < 3:
        return False
    target = list(vt_sequence)
    for i in range(0, len(seq) - 2):
        if seq[i:i+3] == target:
            return True
    return False

# hotzones/models.py
from __future__ import annotations
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional, Tuple, Iterable
import json
from pathlib import Path

# --------------------------
# JSON data model
# --------------------------

@dataclass
class PatternBox:
    """R2/R4/R6/R8 arrays of pattern strings for a single column/box."""
    R2: List[str] = field(default_factory=list)
    R4: List[str] = field(default_factory=list)
    R6: List[str] = field(default_factory=list)
    R8: List[str] = field(default_factory=list)

@dataclass
class DrawEntry:
    """One Draw node with draw_data, pattern_variations and metadata."""
    draw_id: str                      # "Draw1".."Draw7"
    draw_data: List[str]              # e.g., ["010","564","775",...]
    patterns: PatternBox
    is_hot_zone: bool = False
    hot_zone_count: int = 0

@dataclass
class SetEntry:
    set_id: str                       # "Set1".."Set3"
    draws: List[DrawEntry]

@dataclass
class SectionEntry:
    name: str                         # "Midday"/"Evening"/"Combined"
    sets: List[SetEntry]

@dataclass
class TableEnv:
    state_name: str
    sections: List[SectionEntry]
    # Optional analysis guide (ignored by code, retained for completeness)
    analysis_guide: Dict = field(default_factory=dict)

# --------------------------
# Hot candidate model
# --------------------------

@dataclass
class HotCandidate:
    state: str
    vt_family: str                    # e.g., 'v124'
    raw_digits: Tuple[str, ...]       # canonical raw digits contributing (e.g., ('1','6','8'))
    variants: Dict[str, int]          # counts per section (Midday/Evening/Combined)
    set1_superhot_hits: int           # hits in Set1 Draw1/2
    set1_hot_hits: int                # hits in Set1 Draw3..5
    vertical_support: int             # how many of R2/R4/R6/R8 contributed (max per box)
    horizontal_span: int              # longest Streak across Set1 Draws (late cols)
    vt_straight_evidence: int         # number of lanes detected
    vt_boxed_presence: int            # number of boxes with VT family present
    boxes_visited: int                # total starred boxes carrying the family
    examples: List[Dict]              # sample evidence rows
    score_basic: float                # simple baseline score (Part 2 will extend)

@dataclass
class HotScanConfig:
    """Tuning knobs for Part 1 (kept small; Part 2 will extend)."""
    use_metadata_hot_flags: bool = True
    superhot_draw_ids: Tuple[str, ...] = ("Draw1", "Draw2")      # C1 vertical equivalents
    hot_draw_ids: Tuple[str, ...] = ("Draw3", "Draw4", "Draw5")  # late columns that still matter
    consider_sections: Tuple[str, ...] = ("Midday", "Evening", "Combined")
    min_boxes_for_candidate: int = 2                             # avoid single‑box noise

# --------------------------
# JSON loader
# --------------------------

def _patterns_from_obj(obj: Dict) -> PatternBox:
    pv = obj.get("pattern_variations", {})
    return PatternBox(
        R2=list(pv.get("R2", [])),
        R4=list(pv.get("R4", [])),
        R6=list(pv.get("R6", [])),
        R8=list(pv.get("R8", [])),
    )

def _draw_from_obj(draw_id: str, obj: Dict) -> DrawEntry:
    meta = obj.get("metadata", {})
    return DrawEntry(
        draw_id=draw_id,
        draw_data=list(obj.get("draw_data", [])),
        patterns=_patterns_from_obj(obj),
        is_hot_zone=bool(meta.get("is_hot_zone", False)),
        hot_zone_count=int(meta.get("hot_zone_count", 0) or 0),
    )

def _set_from_obj(set_id: str, obj: Dict) -> SetEntry:
    draws_node = obj.get("draws", {})
    # Keep Draw1..Draw7 order if present
    ordered = []
    for k in sorted(draws_node.keys(), key=lambda x: int(x.replace("Draw", ""))):
        ordered.append(_draw_from_obj(k, draws_node[k]))
    return SetEntry(set_id=set_id, draws=ordered)

def _section_from_obj(name: str, obj: Dict) -> SectionEntry:
    sets_node = obj.get("sets", {})
    ordered = []
    for set_id in ["Set1", "Set2", "Set3"]:
        if set_id in sets_node:
            ordered.append(_set_from_obj(set_id, sets_node[set_id]))
    return SectionEntry(name=name, sets=ordered)

def load_table_env_from_json(path: str | Path) -> TableEnv:
    p = Path(path)
    data = json.loads(p.read_text(encoding="utf-8"))
    sections_node = data.get("sections", {})
    sections: List[SectionEntry] = []
    for sec_name in ["Midday", "Evening", "Combined"]:
        if sec_name in sections_node:
            sections.append(_section_from_obj(sec_name, sections_node[sec_name]))
    return TableEnv(
        state_name=data.get("state_name", "UNKNOWN"),
        sections=sections,
        analysis_guide=data.get("analysis_guide", {}),
    )

# hotzones/scanner.py
from __future__ import annotations
from typing import Dict, List, Tuple, Iterable
from dataclasses import asdict
from collections import defaultdict
from .models import (
    TableEnv, SectionEntry, SetEntry, DrawEntry,
    HotCandidate, HotScanConfig,
)
from .vtrac import canonical_vtriad_from_string, vt_family_id, has_vt_straight_lane

def _iter_pattern_strings(draw: DrawEntry) -> Iterable[Tuple[str, str]]:
    """
    Yield (row_name, pattern_string) across R2/R4/R6/R8 for a box.
    We iterate from shortest strings first to emphasize end‑of‑progression cues.
    """
    rows = {
        "R2": draw.patterns.R2,
        "R4": draw.patterns.R4,
        "R6": draw.patterns.R6,
        "R8": draw.patterns.R8,
    }
    for row_name, arr in rows.items():
        # Sort by length ASC to surface '13', '132' style end strings early
        for s in sorted(arr, key=len):
            yield (row_name, s)

def _is_draw_in_hot(config: HotScanConfig, draw: DrawEntry) -> Tuple[bool, bool]:
    """
    Decide if a Draw is in 'superhot' (Draw1/2) or 'hot' (Draw3..5).
    We favor metadata if present; else fall back to draw_id grouping.
    """
    if config.use_metadata_hot_flags:
        if draw.is_hot_zone and draw.draw_id in config.superhot_draw_ids:
            return True, False
        if draw.is_hot_zone and draw.draw_id in config.hot_draw_ids:
            return False, True
        # If metadata says no hot zone, fall back to heuristic:
    is_super = draw.draw_id in config.superhot_draw_ids
    is_hot = draw.draw_id in config.hot_draw_ids
    return is_super, is_hot

class HotZoneScanner:
    """
    Scans the JSON table environment to extract late‑column (hot‑zone) 3‑value / V‑TRAC families
    with vertical (R2/R4/R6/R8) and horizontal (Draw‑to‑Draw) support.

    Part 1 keeps a simple, transparent scoring (counts + few boosts); Part 2 will add
    richer weights/cluster density, col‑1 funnels, and cross‑tool features.
    """

    def __init__(self, env: TableEnv, config: HotScanConfig | None = None):
        self.env = env
        self.cfg = config or HotScanConfig()

    def scan(self) -> List[HotCandidate]:
        family = defaultdict(lambda: {
            "variants": defaultdict(int),
            "set1_superhot_hits": 0,
            "set1_hot_hits": 0,
            "vertical_support": 0,      # max observed per box (R2/R4/R6/R8 hits)
            "horizontal_keys": set(),   # (section, set, draw_id)
            "vt_straight_evidence": 0,
            "vt_boxed_presence": 0,
            "boxes_visited": 0,
            "examples": [],
        })

        # For horizontal span measurement
        per_family_draws = defaultdict(lambda: defaultdict(set))  # fam -> section -> {draw_ids}

        for section in self.env.sections:
            if section.name not in self.cfg.consider_sections:
                continue
            for set_entry in section.sets:
                # We only mine Set1 for hot/superhot late columns;
                # Set2/Set3 can contribute evidence but do not define hot‑zone status.
                for draw in set_entry.draws:
                    is_super, is_hot = _is_draw_in_hot(self.cfg, draw)
                    if set_entry.set_id != "Set1" and not (is_super or is_hot):
                        # Outside Set1, still allow counting VT boxed presence/lanes but do not mark as hot.
                        pass

                    # Collect per‑box contributions
                    seen_vfamilies_in_box = set()
                    vt_lane_hits_in_box = 0
                    vt_boxed_in_box = 0
                    vertical_hits = 0

                    # Prefer shortest strings (end‑of‑progression) first
                    for row_name, s in _iter_pattern_strings(draw):
                        vt, raw = canonical_vtriad_from_string(s)
                        if not vt:
                            continue
                        fam_id = vt_family_id(vt)
                        # Update per‑box vertical participation (count each row at most once per family)
                        if (row_name, fam_id) not in seen_vfamilies_in_box:
                            vertical_hits += 1
                            seen_vfamilies_in_box.add((row_name, fam_id))

                        # Boxed presence (family shows up at all in this row)
                        vt_boxed_in_box += 1

                        # VT straight lane evidence (contiguous v‑sequence)
                        if len(vt) == 3 and has_vt_straight_lane(s, vt):
                            vt_lane_hits_in_box += 1

                        # Record family‑level tallies
                        rec = family[fam_id]
                        rec["variants"][section.name] += 1
                        if set_entry.set_id == "Set1":
                            if is_super:
                                rec["set1_superhot_hits"] += 1
                            if is_hot:
                                rec["set1_hot_hits"] += 1
                            per_family_draws[fam_id][section.name].add(draw.draw_id)
                            rec["horizontal_keys"].add((section.name, set_entry.set_id, draw.draw_id))

                            # Save compact example (first few per family)
                            if len(rec["examples"]) < 12:
                                rec["examples"].append({
                                    "section": section.name,
                                    "set": set_entry.set_id,
                                    "draw": draw.draw_id,
                                    "row": row_name,
                                    "sample": s[:30],
                                    "vt": list(vt),
                                    "raw_digits": list(raw),
                                    "is_superhot": bool(is_super),
                                    "is_hot": bool(is_hot),
                                })

                    # After traversing rows in this box:
                    if seen_vfamilies_in_box:
                        for (row_name, fam_id) in {k for k in seen_vfamilies_in_box}:
                            rec = family[fam_id]
                            rec["vertical_support"] = max(rec["vertical_support"], vertical_hits)
                            rec["boxes_visited"] += 1
                            rec["vt_boxed_presence"] += vt_boxed_in_box
                            rec["vt_straight_evidence"] += vt_lane_hits_in_box

        # Build candidates with a minimal, transparent score
        candidates: List[HotCandidate] = []
        for fam_id, rec in family.items():
            # horizontal_span = max unique Draws in Set1 late columns per section
            horiz_span = 0
            for sec_name, draw_ids in per_family_draws[fam_id].items():
                horiz_span = max(horiz_span, len(draw_ids))

            # simple baseline score (Part 2 adds proper weights / cluster density):
            score = (
                1.5 * rec["set1_superhot_hits"] +
                1.0 * rec["set1_hot_hits"] +
                0.8 * rec["vertical_support"] +
                0.6 * horiz_span +
                0.7 * rec["vt_straight_evidence"] +
                0.3 * rec["vt_boxed_presence"] / 4.0
            )

            if rec["boxes_visited"] < self.cfg.min_boxes_for_candidate:
                continue  # suppress single‑box noise

            # derive a representative raw digit triple from first example (purely cosmetic)
            if rec["examples"]:
                raw = tuple(rec["examples"][0].get("raw_digits", []))
            else:
                raw = tuple()

            candidates.append(HotCandidate(
                state=self.env.state_name,
                vt_family=fam_id,
                raw_digits=raw,
                variants={k: int(v) for k, v in rec["variants"].items()},
                set1_superhot_hits=int(rec["set1_superhot_hits"]),
                set1_hot_hits=int(rec["set1_hot_hits"]),
                vertical_support=int(rec["vertical_support"]),
                horizontal_span=int(horiz_span),
                vt_straight_evidence=int(rec["vt_straight_evidence"]),
                vt_boxed_presence=int(rec["vt_boxed_presence"]),
                boxes_visited=int(rec["boxes_visited"]),
                examples=list(rec["examples"]),
                score_basic=float(round(score, 3)),
            ))

        # Sort: stronger first
        candidates.sort(key=lambda c: (-c.score_basic, -c.set1_superhot_hits, -c.set1_hot_hits, -c.vertical_support))
        return candidates

# hotzones/validate.py
from __future__ import annotations
from dataclasses import dataclass
from pathlib import Path
from typing import Dict, List, Tuple
import json
import time

@dataclass
class ValidationReport:
    ok: bool
    messages: List[str]

class TableValidator:
    """
    Lightweight guardrails to prevent stale/missing table runs from polluting analysis.
    Part 1 focuses on:
      - presence checks for JSON mirrors
      - optional freshness check vs. a given XLSX/XLSM source
    Part 2 can add deeper schema/consistency checks against analyzer 'brain' bundles.
    """

    def __init__(self, base_dir: str | Path):
        self.base = Path(base_dir)

    def check_json_exists(self, json_path: str | Path) -> ValidationReport:
        p = Path(json_path)
        msgs: List[str] = []
        if not p.exists():
            msgs.append(f"Missing JSON table env: {p}")
            return ValidationReport(False, msgs)
        try:
            data = json.loads(p.read_text(encoding="utf-8"))
            if "sections" not in data or not data["sections"]:
                msgs.append("JSON lacks 'sections' node.")
                return ValidationReport(False, msgs)
            msgs.append("JSON table env loaded OK.")
            return ValidationReport(True, msgs)
        except Exception as e:
            return ValidationReport(False, [f"JSON parse error: {e}"])

    def check_fresh_vs_source(self, json_path: str | Path, source_excel: str | Path) -> ValidationReport:
        """
        Warn if the JSON mirror is older than the Excel workbook (indicating stale tables).
        """
        jp = Path(json_path)
        sp = Path(source_excel)
        msgs: List[str] = []
        if not jp.exists() or not sp.exists():
            return ValidationReport(False, [f"Missing: {jp} or {sp}"])
        json_mtime = jp.stat().st_mtime
        src_mtime = sp.stat().st_mtime
        if json_mtime + 1 < src_mtime:
            msgs.append(f"Stale JSON: {jp.name} ({time.ctime(json_mtime)}) older than source {sp.name} ({time.ctime(src_mtime)})")
            return ValidationReport(False, msgs)
        msgs.append("Freshness OK: JSON newer than source.")
        return ValidationReport(True, msgs)

# hotzones/cli.py
from __future__ import annotations
import argparse
import json
from pathlib import Path
from typing import Any, Dict
import csv

from .models import load_table_env_from_json, HotScanConfig
from .scanner import HotZoneScanner
from .validate import TableValidator

def _write_jsonl(out_path: Path, objs):
    out_path.parent.mkdir(parents=True, exist_ok=True)
    with out_path.open("w", encoding="utf-8") as f:
        for obj in objs:
            f.write(json.dumps(obj, ensure_ascii=False) + "\n")

def _write_csv(out_path: Path, candidates):
    out_path.parent.mkdir(parents=True, exist_ok=True)
    fields = [
        "state", "vt_family", "raw_digits",
        "variants", "set1_superhot_hits", "set1_hot_hits",
        "vertical_support", "horizontal_span",
        "vt_straight_evidence", "vt_boxed_presence",
        "boxes_visited", "score_basic",
    ]
    with out_path.open("w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=fields)
        w.writeheader()
        for c in candidates:
            row = {
                "state": c.state,
                "vt_family": c.vt_family,
                "raw_digits": "".join(c.raw_digits) if c.raw_digits else "",
                "variants": json.dumps(c.variants, ensure_ascii=False),
                "set1_superhot_hits": c.set1_superhot_hits,
                "set1_hot_hits": c.set1_hot_hits,
                "vertical_support": c.vertical_support,
                "horizontal_span": c.horizontal_span,
                "vt_straight_evidence": c.vt_straight_evidence,
                "vt_boxed_presence": c.vt_boxed_presence,
                "boxes_visited": c.boxes_visited,
                "score_basic": c.score_basic,
            }
            w.writerow(row)

def cmd_validate(args):
    tv = TableValidator(base_dir=".")
    rep = tv.check_json_exists(args.json)
    for m in rep.messages:
        print(m)
    if args.source:
        rep2 = tv.check_fresh_vs_source(args.json, args.source)
        for m in rep2.messages:
            print(m)
        ok = rep.ok and rep2.ok
    else:
        ok = rep.ok
    if not ok:
        raise SystemExit(2)

def cmd_scan_json(args):
    env = load_table_env_from_json(args.json)
    cfg = HotScanConfig(
        use_metadata_hot_flags=not args.ignore_meta_hot,
        superhot_draw_ids=tuple(args.super.split(",")) if args.super else ("Draw1", "Draw2"),
        hot_draw_ids=tuple(args.hot.split(",")) if args.hot else ("Draw3", "Draw4", "Draw5"),
        consider_sections=tuple(args.sections.split(",")) if args.sections else ("Midday","Evening","Combined"),
        min_boxes_for_candidate=max(1, int(args.min_boxes)),
    )
    scanner = HotZoneScanner(env, cfg)
    cans = scanner.scan()

    out_dir = Path(args.out or ".")
    jsonl_path = out_dir / f"{env.state_name}_hotzones_candidates.jsonl"
    csv_path = out_dir / f"{env.state_name}_hotzones_candidates.csv"

    _write_jsonl(jsonl_path, [c.__dict__ for c in cans])
    _write_csv(csv_path, cans)
    print(f"Wrote {jsonl_path}")
    print(f"Wrote {csv_path}")
    if cans:
        print(f"Top candidate: {cans[0].vt_family} score={cans[0].score_basic}")
    else:
        print("No candidates found (try lowering --min-boxes).")

def main(argv=None):
    ap = argparse.ArgumentParser(prog="hotzones", description="AAT9 Hot Zones core tools (Part 1)")

# Mirror mapping used across AAT9 (0↔5, 1↔6, 2↔7, 3↔8, 4↔9)
MIRROR_MAP = {
    '0': '5', '5': '0',
    '1': '6', '6': '1',
    '2': '7', '7': '2',
    '3': '8', '8': '3',
    '4': '9', '9': '4',
}

# V-TRAC group mapping: (0,5)->'1', (1,6)->'2', (2,7)->'3', (3,8)->'4', (4,9)->'5'
VT_GROUP = {
    '0': '1', '5': '1',
    '1': '2', '6': '2',
    '2': '3', '7': '3',
    '3': '4', '8': '4',
    '4': '5', '9': '5',
}

def mirror_digit(d: str) -> str:
    return MIRROR_MAP.get(d, d)

def vt_of_digit(d: str) -> str:
    return VT_GROUP.get(d, d)

def vt_encode(s: str) -> str:
    return ''.join(vt_of_digit(ch) for ch in s if ch.isdigit())

def mirror_string(s: str) -> str:
    return ''.join(mirror_digit(ch) for ch in s if ch.isdigit())

def canonical_triad(triad: str) -> str:
    """Sort digits to canonical form while preserving duplicates, e.g., '769'->'679', '667'->'667'."""
    d = [c for c in triad if c.isdigit()]
    if len(d) != 3:
        raise ValueError(f"Expected 3-digit triad, got: {triad}")
    return ''.join(sorted(d))

def triad_permutations(triad: str) -> List[str]:
    """All unique permutations of a 3-char triad (handles duplicates)."""
    return sorted(set(''.join(p) for p in permutations(triad, 3)))

def vt_triad(triad: str) -> str:
    """Map a 3-digit triad to its V-TRAC triple, e.g., '269' -> vt '3','5','5' -> '355'."""
    return ''.join(vt_of_digit(ch) for ch in triad)

def mirror_triad(triad: str) -> str:
    return ''.join(mirror_digit(ch) for ch in triad)

def _list_columns(row_list: List[str]) -> List[Tuple[int, str]]:
    """
    Map variation array to (column_index, string).
    Column 1 == newest (last column), but JSON arrays list left→right.
    We map so that last element becomes column_index=1.
    """
    cols = []
    # Array like ["....(col7)", "(col6)", ..., "(col1)"]
    # We'll reverse enumerate to assign 1..len
    for i, s in enumerate(reversed(row_list), start=1):
        cols.append((i, ''.join(ch for ch in s if ch.isdigit())))
    return cols

def _transit_set(draw_data: List[str]) -> Set[str]:
    """
    Transit digits set: digits from most recent published draw (last in draw_data)
    plus their mirror digits. This follows your training examples where removing
    last draw digits + mirrors revealed cores in Combined/Midday funnels.
    """
    if not draw_data:
        return set()
    most_recent = draw_data[-1]
    digits = set(ch for ch in most_recent if ch.isdigit())
    mirrors = {mirror_digit(ch) for ch in digits}
    return digits | mirrors

def load_boxes_from_json(payload: Dict, settings: EngineSettings) -> List[BoxData]:

        if section not in settings.variant_names:

            draws = set_obj.get("draws", {})
            for draw_name, draw_obj in draws.items():
                draw_data: List[str] = draw_obj.get("draw_data", [])

def _contains_in_order_with_max_gap(s: str, pattern: str, max_gap: int) -> bool:
    """
    Check if 'pattern' digits appear in 's' in order with each consecutive pair
    separated by at most max_gap extraneous characters.
    """
    if len(pattern) != 3:
        return False
    # Find first char
    i = s.find(pattern[0])
    while i != -1:
        # find second within i+1..i+1+max_gap+len
        start_j = i + 1
        end_j = min(len(s), start_j + 1 + max_gap)
        j_found = -1
        for j in range(start_j, end_j):
            if j < len(s) and s[j] == pattern[1]:
                j_found = j
                break
        if j_found != -1:
            # find third within j+1..j+1+max_gap
            start_k = j_found + 1
            end_k = min(len(s), start_k + 1 + max_gap)
            for k in range(start_k, end_k):
                if k < len(s) and s[k] == pattern[2]:
                    return True
        # move i to next occurrence of pattern[0]
        i = s.find(pattern[0], i + 1)
    return False

def detect_triad_in_box(triad: str, box: BoxData, settings: EngineSettings) -> Tuple[bool, bool]:
    """
    Returns (has_straight_lane, has_vt_straight_lane) for this triad in this box.
    A "straight lane" is present if any permutation appears as contiguous or with gaps≤max_gap.
    VT lane checks the VT-encoded string vs the VT-encoded triad permutations.
    """
    perms = triad_permutations(triad)
    vt_tri = vt_triad(triad)
    vt_perms = triad_permutations(vt_tri)

    # Consider both raw and reduced (clutter removal) strings for straight
    s_candidates = [box.s_raw, box.s_reduced]
    has_straight = False
    for s in s_candidates:
        if not s:
            continue
        # 1) contiguous substr
        if any(p in s for p in perms):
            has_straight = True
            break
        # 2) near-straight (gap≤1 default)
        if any(_contains_in_order_with_max_gap(s, p, settings.max_gap_for_straight) for p in perms):
            has_straight = True
            break

    # VT straight in VT-encoded string (raw only; reduced removal already reflected visually by VT lanes)
    s_vt = box.s_vt
    has_vt_straight = False
    if s_vt:
        if any(p in s_vt for p in vt_perms):
            has_vt_straight = True
        elif any(_contains_in_order_with_max_gap(s_vt, p, settings.max_gap_for_straight) for p in vt_perms):
            has_vt_straight = True

    return has_straight, has_vt_straight

def extract_candidate_triads_from_string(s: str, max_gap: int = 1) -> Set[str]:
    """
    Mine triad candidates from a digit string by scanning contiguous 3-digit windows
    plus near-straight gap≤1 subsequences.
    Returns canonicalized triads (sorted 3-digit strings).
    """
    cands: Set[str] = set()
    # contiguous windows
    for i in range(0, max(0, len(s) - 2)):
        tri = s[i:i+3]
        if len(tri) == 3 and tri.isdigit():
            cands.add(canonical_triad(tri))

    # near-straight with gap≤1: (i, i+1..i+2, i+2..i+4)
    # simple pass: i, j=i+1 or i+2, k=j+1 or j+2
    n = len(s)
    for i in range(n):
        a = s[i]
        if not a.isdigit():
            continue
        for j in (i+1, i+2):
            if j >= n: continue
            b = s[j]
            if not b.isdigit():
                continue
            for k in (j+1, j+2):
                if k >= n: continue
                c = s[k]
                if not c.isdigit():
                    continue
                tri = canonical_triad(a+b+c)
                cands.add(tri)
    return cands

def mine_candidates(boxes: List[BoxData], settings: EngineSettings) -> Dict[str, List[Evidence]]:
    """
    Iterate starred boxes (optional) and collect Evidence per triad per (section, set, draw, column).
    Returns dict triad -> list of Evidence.
    """
    triad_to_evs: Dict[str, List[Evidence]] = defaultdict(list)

    # Organize by grid position to compute vertical/horizontal cues
    # Build an index: (section,set,draw,column) -> list[BoxData] for rows R2/R4/R6/R8
    grid: Dict[Tuple[str, str, str, int], List[BoxData]] = defaultdict(list)
    for b in boxes:
        grid[(b.ref.section, b.ref.set_name, b.ref.draw_name, b.ref.column_index)].append(b)

    for key, row_boxes in grid.items():
        section, set_name, draw_name, col_idx = key
        # Skip non-starred if configured
        if settings.scan_only_starred and not any(b.is_starred for b in row_boxes):
            continue

        # collect a superset of candidate triads present in any row string (raw & reduced)
        triad_cands: Set[str] = set()
        for b in row_boxes:
            triad_cands |= extract_candidate_triads_from_string(b.s_raw, settings.max_gap_for_straight)
            if b.s_reduced:
                triad_cands |= extract_candidate_triads_from_string(b.s_reduced, settings.max_gap_for_straight)

        # evaluate each candidate triad across the vertical stack (R2/R4/R6/R8) at this column
        for triad in sorted(triad_cands):
            vt_tri = vt_triad(triad)
            row_hits: Dict[str, bool] = {}
            has_any_straight = False
            has_any_vt = False
            vt_only_lane = False

            # row-wise
            for rb in row_boxes:
                s_hit, vt_hit = detect_triad_in_box(triad, rb, settings)
                row_hits[rb.ref.row_name] = (s_hit or vt_hit)
                has_any_straight = has_any_straight or s_hit
                has_any_vt = has_any_vt or vt_hit

            vt_only_lane = (has_any_vt and not has_any_straight)

            # context booleans (funnel, late columns, Set1 bias, etc.)
            is_set1 = any(rb.is_set1 for rb in row_boxes)
            is_starred = any(rb.is_starred for rb in row_boxes)
            star_count = max((rb.star_count for rb in row_boxes), default=0)
            is_superhot_slot = any(rb.is_superhot_slot for rb in row_boxes)

            col1_arrival = (col_idx == 1) and (has_any_straight or has_any_vt)
            precol1_funnel = (col_idx in (2, 3)) and (has_any_straight or has_any_vt)
            ls_col_42 = (col_idx in (4, 2)) and (has_any_straight or has_any_vt)
            ls2_lane = (is_set1 and col_idx in (3, 1) and (has_any_straight or has_any_vt))

            triad_to_evs[triad].append(Evidence(
                triad=triad,
                vt_triad=vt_tri,
                section=section,
                set_name=set_name,
                draw_name=draw_name,
                column_index=col_idx,
                row_hits=row_hits,
                has_straight=has_any_straight,
                has_vt_straight=has_any_vt,
                vt_only_lane=vt_only_lane,
                col1_arrival=col1_arrival,
                precol1_funnel=precol1_funnel,
                ls_col_42=ls_col_42,
                ls2_lane=ls2_lane,
                is_starred=is_starred,
                star_count=star_count,
                is_superhot_slot=is_superhot_slot,
                is_set1=is_set1
            ))

    return triad_to_evs

def _score_evidence(e: Evidence, weights: HotZoneWeights) -> Tuple[float, List[str]]:
    reasons = []
    score = 0.0

    if e.is_starred:
        score += weights.w_hot_star * max(1, e.star_count)
        reasons.append(f"hot{e.star_count}")

    if e.is_superhot_slot:
        score += weights.w_superhot_set1
        reasons.append("superhot_set1")

    # Vertical support = number of rows R2/R4/R6/R8 where triad/VT appeared
    vert = sum(1 for k, v in e.row_hits.items() if v)
    if vert:
        score += weights.w_vertical_support * vert
        reasons.append(f"vertical{vert}")

    # Column position signals
    if e.precol1_funnel:
        score += weights.w_precol1_funnel
        reasons.append("funnel_precol1")
    if e.col1_arrival:
        score += weights.w_col1_arrival
        reasons.append("col1")
    if e.ls_col_42:
        score += weights.w_ls_col_42
        reasons.append("ls_col_42")
    if e.ls2_lane:
        score += weights.w_ls2_lane
        reasons.append("ls2_lane")

    # VT lanes / straights
    if e.has_vt_straight:
        score += weights.w_vt_straight
        reasons.append("vt_straight")
    if e.has_straight:
        score += weights.w_exact_straight_lane
        reasons.append("straight_lane")
    if e.vt_only_lane:
        score += weights.w_vt_only_lane_bonus
        reasons.append("vt_only_lane")

    # Recency (Set1 bias)
    if e.is_set1:
        score += weights.w_set1_bias
        reasons.append("set1_bonus")

    return score, reasons

def aggregate_and_score(triad_to_evs: Dict[str, List[Evidence]],
                        weights: HotZoneWeights) -> Tuple[List[PerItemRow], List[TopCandidateRow]]:
    per_items: List[PerItemRow] = []
    tops: List[TopCandidateRow] = []

    # Precompute horizontal/set/variant spans per triad using Evidence fields
    for triad, evs in triad_to_evs.items():
        # Build indices for spans
        section_set = {(e.section, e.set_name) for e in evs}
        section_only = {e.section for e in evs}
        set_only = {e.set_name for e in evs}
        columns_in_set1 = {e.column_index for e in evs if e.set_name == "Set1"}
        columns_all = {e.column_index for e in evs}

        # Compute per-item rows with scores
        for e in evs:
            vertical_support = sum(1 for k, v in e.row_hits.items() if v)
            # horizontal span: distinct columns in the same set/section for this triad
            horiz_span = len({ev.column_index for ev in evs
                              if ev.section == e.section and ev.set_name == e.set_name})
            set_span = len(set_only)
            variant_echo = len(section_only)

            s, reasons = _score_evidence(e, weights)
            per_items.append(PerItemRow(
                section=e.section,
                set_name=e.set_name,
                draw_name=e.draw_name,
                column_index=e.column_index,
                triad=e.triad,
                vt_triad=e.vt_triad,
                vertical_support=vertical_support,
                horizontal_span=horiz_span,
                set_span=set_span,
                variant_echo=variant_echo,
                has_straight=int(e.has_straight),
                has_vt_straight=int(e.has_vt_straight),
                vt_only_lane=int(e.vt_only_lane),
                col1_arrival=int(e.col1_arrival),
                precol1_funnel=int(e.precol1_funnel),
                ls_col_42=int(e.ls_col_42),
                ls2_lane=int(e.ls2_lane),
                is_starred=int(e.is_starred),
                star_count=int(e.star_count),
                is_superhot_slot=int(e.is_superhot_slot),
                is_set1=int(e.is_set1),
                score=s,
                reasons="|".join(reasons)
            ))

        # Aggregate top rows
        hot_hits = sum(1 for e in evs if e.is_starred)
        superhot_hits = sum(1 for e in evs if e.is_superhot_slot)
        vertical_hits = sum(1 for e in evs if sum(1 for k, v in e.row_hits.items() if v) >= 2)
        set1_hits = sum(1 for e in evs if e.is_set1)
        col1_hits = sum(1 for e in evs if e.col1_arrival)
        precol1_hits = sum(1 for e in evs if e.precol1_funnel)
        vt_straight_hits = sum(1 for e in evs if e.has_vt_straight)
        vt_only_lane_hits = sum(1 for e in evs if e.vt_only_lane)

        scores = [per.score for per in per_items if per.triad == triad]
        score_mean = statistics.fmean(scores) if scores else 0.0
        score_max = max(scores) if scores else 0.0

        tops.append(TopCandidateRow(
            triad=triad,
            vt_triad=vt_triad(triad),
            support_count=len(evs),
            hot_hits=hot_hits,
            superhot_hits=superhot_hits,
            vertical_hits=vertical_hits,
            set1_hits=set1_hits,
            col1_hits=col1_hits,
            precol1_hits=precol1_hits,
            vt_straight_hits=vt_straight_hits,
            vt_only_lane_hits=vt_only_lane_hits,
            variant_span=len(section_only),
            set_span=len(set_only),
            column_span=len(columns_all),
            score_mean=round(score_mean, 5),
            score_max=round(score_max, 5),
            evidence_tags=_mk_evidence_tags(hot_hits, superhot_hits, col1_hits,
                                            precol1_hits, vt_straight_hits, vt_only_lane_hits)
        ))

    # Sort tops by score_max then by score_mean then support_count
    tops.sort(key=lambda r: (r.score_max, r.score_mean, r.support_count), reverse=True)
    return per_items, tops

def _mk_evidence_tags(hot: int, superhot: int, col1: int,
                      precol1: int, vt: int, vt_only: int) -> str:
    tags = []
    if superhot: tags.append("superhot")
    if hot: tags.append("hot")
    if col1: tags.append("col1")
    if precol1: tags.append("precol1")
    if vt: tags.append("vt")
    if vt_only: tags.append("vt_only")
    return "|".join(tags)

# V-TRAC bins (per your training guide):

VT_BIN_OF_DIGIT = {
    0: 1, 5: 1,
    1: 2, 6: 2,
    2: 3, 7: 3,
    3: 4, 8: 4,
    4: 5, 9: 5
}

def digit_to_vt(d: int) -> int:
    return VT_BIN_OF_DIGIT[d]

def str3_to_vt_triplet(s: str) -> Tuple[int, int, int]:
    """Map a 3-char digit string like '691' -> (2,5,2) preserving order."""
    return tuple(digit_to_vt(int(ch)) for ch in s)

def canonical_3v_sorted(s: str) -> Tuple[int, int, int]:
    """Sorted VT family for triad grouping (orderless family identity)."""
    return tuple(sorted(str3_to_vt_triplet(s)))

def canonical_3d_sorted(s: str) -> str:
    """Sorted literal-digit family identity (e.g., '691' -> '169')."""
    return ''.join(sorted(s))

def iter_triads_from_digit_string(s: str) -> Iterable[str]:
    """
    Extract ordered triads from a digit string by:
      1) contiguous windows length=3,
      2) simple gap‑1 subsequences (i, i+2, i+3) when available.
    """
    out = set()
    # contiguous
    for i in range(0, max(0, len(s) - 2)):
        triad = s[i:i+3]
        if len(triad) == 3 and triad.isdigit():
            out.add(triad)
    # gap‑1 (e.g., '24138' -> '248' via indices [0,2,4] if digits exist)
    for i in range(0, max(0, len(s) - 4)):
        triad = s[i] + s[i+2] + s[i+4]
        if triad.isdigit() and len(triad) == 3:
            out.add(triad)
    return out

def late_column_weight(row_values: List[str], col_idx: int) -> int:
    """
    Approximate recency proxy from column position.
    Larger index = later column (col1). Return 3 for col1, 2 for col2, 1 for col3, else 0.
    """
    last = len(row_values) - 1
    if col_idx == last:
        return 3
    if col_idx == last - 1:
        return 2
    if col_idx == last - 2:
        return 1
    return 0

def set_bucket(set_name: str) -> int:
    """Set1 (recency), Set2 (yesterday), Set3 (older)."""
    return 1 if set_name == "Set1" else (2 if set_name == "Set2" else 3)

class HotZonesEngine:
    def __init__(self, weights: ScoringWeights = DEFAULT_W):
        self.w = weights

    def run(self, state_json: Dict, out_dir_root: str) -> None:
        state_name = state_json.get("state_name", "UNKNOWN")
        sections = state_json.get("sections", {})
        evidence: Dict[str, TriadEvidence] = {}
        # pass 1: collect literal triads from starred boxes, accumulate features
        for section_name, section in sections.items():
            sets = section.get("sets", {})
            for set_name, set_obj in sets.items():
                draws = set_obj.get("draws", {})
                for draw_name, draw_obj in draws.items():
                    meta = draw_obj.get("metadata", {})
                    is_hot = bool(meta.get("is_hot_zone", False))
                    hz_count = int(meta.get("hot_zone_count", 0))
                    pat = draw_obj.get("pattern_variations", {})
                    for row in ("R2","R4","R6","R8"):
                        row_values: List[str] = pat.get(row, [])
                        for col_idx, s in enumerate(row_values):
                            # we parse every column but boost late columns via recency weights
                            triads = iter_triads_from_digit_string(s)
                            for t in triads:
                                key = (section_name, set_name, draw_name, row, col_idx, t)
                                self._accumulate(evidence, key, is_hot, hz_count, row_values)

        # pass 2: derive secondary flags (vt_only_lane, funnel_precol1, ls2_lane) and score
        for ev in evidence.values():
            self._derive_lane_flags(ev)
            ev.score_raw = self._score(ev)
        # normalize scores per state
        scores = [ev.score_raw for ev in evidence.values()]
        mn, mx = (min(scores), max(scores)) if scores else (0.0, 0.0)
        for ev in evidence.values():
            ev.score_norm = 0.0 if mx == mn else (ev.score_raw - mn) / (mx - mn)

        # write outputs (lean bundle)
        out_dir = os.path.join(out_dir_root, state_name)
        os.makedirs(out_dir, exist_ok=True)
        per_item_path = os.path.join(out_dir, f"{state_name}_hotzones_per_item.csv")
        top_path = os.path.join(out_dir, f"{state_name}_hotzones_top_candidates.csv")
        meta_path = os.path.join(out_dir, f"{state_name}_hotzones_meta.json")

        self._write_per_item(per_item_path, evidence)
        self._write_top(top_path, evidence)
        self._write_meta(meta_path, state_json, len(evidence))

        print(f"[HotZones] Wrote:\n  {per_item_path}\n  {top_path}\n  {meta_path}")

    # ---- feature accumulation ----

    def _accumulate(
        self,
        evidence: Dict[str, TriadEvidence],
        key: Tuple[str,str,str,str,int,str],
        is_hot: bool,
        hot_zone_count: int,
        row_values: List[str]
    ):
        section_name, set_name, draw_name, row, col_idx, triad = key
        fam3d = canonical_3d_sorted(triad)
        fam3v = canonical_3v_sorted(triad)
        vt_trip = str3_to_vt_triplet(triad)
        ev_key = f"{triad}|{section_name}|{set_name}"
        ev = evidence.get(ev_key)
        if ev is None:
            ev = TriadEvidence(
                triad=triad, vt_triplet=vt_trip, fam3d=fam3d, fam3v=fam3v
            )
            evidence[ev_key] = ev

        # update sets/sections/rows presence
        ev.sections_present.add(section_name)
        ev.rows_present.add(row)
        # set lane presence
        bucket = set_bucket(set_name)
        if bucket == 1:
            ev.set1_presence_cols.add(col_idx)
        elif bucket == 2:
            ev.set2_presence_cols.add(col_idx)
        else:
            ev.set3_presence_cols.add(col_idx)

        # vertical density (row stacking within same column)
        # simple: +1 per row occurrence in this column index
        ev.counts_vertical += 1

        # horizontal survival proxy: we assume if a triad appears at more than one column
        # within the same set lane, we'll count survival when we later see multiple col indices
        # (finalization will count unique cols across Set1)
        # Here, add a little incremental reward for late columns directly
        ev.counts_horizontal += late_column_weight(row_values, col_idx)

        # recency / hotness
        if bucket == 1:
            # smaller is better; col1 ~ 0 using weight mapping
            # map weight 3->0, 2->1, 1->2, else 5 (far)
            lw = late_column_weight(row_values, col_idx)
            rec = {3:0, 2:1, 1:2}.get(lw, 5)
            ev.last_seen_recency = min(ev.last_seen_recency, rec)
        if is_hot:
            if hot_zone_count >= 2:
                ev.hot2_hits += 1
            elif hot_zone_count >= 1:
                ev.hot1_hits += 1

    # ---- lane flags ----

    def _derive_lane_flags(self, ev: TriadEvidence) -> None:
        # ls2 lane: Set1 has late columns col3/1 hits
        has_col1 = (0 in self._to_late_ranks(ev.set1_presence_cols))
        has_col2 = (1 in self._to_late_ranks(ev.set1_presence_cols))
        has_col3 = (2 in self._to_late_ranks(ev.set1_presence_cols))
        ev.ls2_lane = bool(has_col1 or (has_col2 and has_col3))
        # funnel pre-col1: concentrated on col3/2, with col1 nearby
        ev.funnel_precol1 = bool(has_col2 and has_col3)

        # vt_only_lane heuristic:
        # if literal triad lacks col1 in Set1 but appears across multiple sections/rows,
        # treat as VT-only leaning lane (literal path weak in the hottest column).
        literal_col1 = has_col1
        ev.vt_only_lane = bool((not literal_col1) and (len(ev.sections_present) >= 2) and (len(ev.rows_present) >= 3))
        # reasons
        if ev.vt_only_lane:
            ev.reasons.append("vt_only_lane")
        if ev.funnel_precol1:
            ev.reasons.append("funnel_precol1")
        if ev.ls2_lane:
            ev.reasons.append("ls2_lane")

    @staticmethod
    def _to_late_ranks(cols: Set[int]) -> Set[int]:
        """
        Translate raw col indices (unknown width) to generic late ranks:
        assume the JSON arrays shrink like 7..1 from left to right; we only need to
        detect presence near tail. We approximate by mapping the largest seen index to 'col1'.
        """
        if not cols:
            return set()
        mx = max(cols)
        out = set()
        for c in cols:
            if c == mx:
                out.add(0)      # col1
            elif c == mx - 1:
                out.add(1)      # col2
            elif c == mx - 2:
                out.add(2)      # col3
        return out

    # ---- scoring ----

    def _score(self, ev: TriadEvidence) -> float:
        w = self.w
        score = 0.0
        # vertical/horizontal
        score += w.w_vertical * ev.counts_vertical
        score += w.w_horizontal * ev.counts_horizontal

        # recency (Set1)
        late_ranks = self._to_late_ranks(ev.set1_presence_cols)
        if 0 in late_ranks: score += w.w_recency_set1_col1
        if 1 in late_ranks: score += w.w_recency_set1_col2
        if 2 in late_ranks: score += w.w_recency_set1_col3

        # hotness
        score += w.w_hot2 * ev.hot2_hits
        score += w.w_hot1 * ev.hot1_hits

        # row stacking
        score += w.w_rows_stack * (len(ev.rows_present) - 1)  # extra rows beyond first

        # sections echo
        score += w.w_sections_echo * (len(ev.sections_present) - 1)

        # Set2/Set3 echoes
        if ev.set2_presence_cols:
            score += w.w_set2_echo
        if ev.set3_presence_cols:
            score += w.w_set3_echo

        # lane flags
        if ev.vt_only_lane: score += w.w_vt_only_lane
        if ev.funnel_precol1: score += w.w_funnel_precol1
        if ev.ls2_lane: score += w.w_ls2_lane

        return score

    # ---- writers ----

    def _write_per_item(self, path: str, evidence: Dict[str, TriadEvidence]) -> None:
        # Schema mirrors your Analyzer per_item CSV shape (field names adapted to HotZones)
        # ref: Analyzer Lean Output Spec (per_item/per_top/meta contraction)
        # We include reasons_json for human QA.
        fields = [

        ]
        with open(path, "w", newline="") as f:
            w = csv.DictWriter(f, fieldnames=fields)

            for ev in sorted(evidence.values(), key=lambda e: e.score_norm, reverse=True):
                lock = "consider" if ev.score_norm >= 0.7 else ("maybe" if ev.score_norm >= 0.5 else "no")
                w.writerow({
                    "triad": ev.triad,

                    "counts_vertical": ev.counts_vertical,
                    "counts_horizontal": ev.counts_horizontal,
                    "hot1_hits": ev.hot1_hits,
                    "hot2_hits": ev.hot2_hits,
                    "rows_present": ",".join(sorted(ev.rows_present)),

                })

    def _write_top(self, path: str, evidence: Dict[str, TriadEvidence], top_k: int = 120) -> None:
        # Aggregate by VT family + literal family to reduce duplication,

        agg = collections.defaultdict(list)

        fields = [
            "family_vt","family_3d","members","score_mean","score_max",

            score_mean = statistics.fmean([x.score_norm for x in items]) if items else 0.0
            score_max = max(x.score_norm for x in items) if items else 0.0
            sections = sorted(set.union(*[x.sections_present for x in items])) if items else []
            rows_present = sorted(set.union(*[x.rows_present for x in items])) if items else []
            set1_union = sorted(set.union(*[x.set1_presence_cols for x in items])) if items else []
            hot2_total = sum(x.hot2_hits for x in items)
            hot1_total = sum(x.hot1_hits for x in items)
            vt_only_any = int(any(x.vt_only_lane for x in items))
            funnel_any = int(any(x.funnel_precol1 for x in items))
            ls2_any = int(any(x.ls2_lane for x in items))
            members = sorted(set(x.triad for x in items))
            evidence_tags = []
            if vt_only_any: evidence_tags.append("vt_only_lane")
            if funnel_any: evidence_tags.append("funnel_precol1")
            if ls2_any: evidence_tags.append("ls2_lane")
            steps_summary = f"Set1 cols={set1_union}; sections={sections}; rows={rows_present}"

            rows.append({

                "family_3d": fam3d,

                "set1_cols_union": ",".join(map(str, set1_union)),
                "hot2_total": hot2_total,
                "hot1_total": hot1_total,
                "vt_only_lane_any": vt_only_any,
                "funnel_precol1_any": funnel_any,
                "ls2_lane_any": ls2_any,
                "evidence_tags": ",".join(evidence_tags),
                "steps_summary": steps_summary
            })

        rows.sort(key=lambda r: (r["score_max"], r["score_mean"], r["hot2_total"], r["hot1_total"]), reverse=True)
        with open(path, "w", newline="") as f:
            w = csv.DictWriter(f, fieldnames=fields)
            w.writeheader()
            for r in rows[:top_k]:
                w.writerow(r)

Assumptions encoded

Transit‑digit reduction: remove last published draw digits and their mirrors from starred strings (single pass, clutter‑removal).

# -*- coding: utf-8 -*-

"""
Hot Zones — Part 3: Scoring & Candidate Ranking Engine
------------------------------------------------------
Reads a JSON "string table" environment (sections -> sets -> draws -> R2/R4/R6/R8),
extracts triad candidates from starred boxes, computes multi-signal evidence,
scores and ranks candidates, and emits CSV/JSON outputs.

Usage:
  python hotzones_part3_scoring.py --input path/to/env.json --outdir out --topk 50

Outputs:
  out/hotzones_candidates.csv
  out/hotzones_candidates.json
"""

from __future__ import annotations
import argparse
import json
import math
import os
import sys
import csv
import itertools
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Tuple, Set, Optional, Iterable, Any, DefaultDict
from collections import defaultdict, Counter

# ----------------------------
# VT / mirror / small helpers
# ----------------------------

# Mirror pairs: 0<->5, 1<->6, 2<->7, 3<->8, 4<->9
MIRROR = {'0':'5','1':'6','2':'7','3':'8','4':'9','5':'0','6':'1','7':'2','8':'3','9':'4'}

# V-TRAC mapping (digit -> vt digit in '1'..'5')
VT_MAP = {'0':'1','5':'1','1':'2','6':'2','2':'3','7':'3','3':'4','8':'4','4':'5','9':'5'}

DIGITS = set('0123456789')

def vt_of_digit(d: str) -> str:
    return VT_MAP[d]

def mirror_of(d: str) -> str:
    return MIRROR[d]

def canon_triad(triad: str) -> str:
    """Canonicalize a 3-digit triad with sorting *but preserving doubles/unique*."""
    if len(triad) != 3 or any(c not in DIGITS for c in triad):
        raise ValueError(f"Invalid triad: {triad}")
    return ''.join(sorted(triad))

def vt_triple(triad: str) -> str:
    return ''.join(VT_MAP[c] for c in triad)

def all_permutations_with_dupes(triad: str) -> Set[str]:
    """All 3! / (dup!) permutations as strings."""
    return set(''.join(p) for p in itertools.permutations(triad, 3))

def subseq_in(s: str, seq: str) -> bool:
    """Check if `seq` is a subsequence of `s`."""
    it = iter(s)
    return all(ch in it and (lambda x: True)(next((c for c in it if c == ch), ch)) for ch in seq)

def unordered_cover(s: str, triad: str) -> bool:
    """All triad digits appear at least once in s (any order)."""
    return all(ch in s for ch in set(triad))

def vt_subseq_in(s: str, vt_seq: str) -> bool:
    """Check subsequence after mapping s digits to VT digits."""
    vt_s = ''.join(VT_MAP[ch] for ch in s)
    # strict subsequence in VT space
    idx = 0
    for ch in vt_s:
        if ch == vt_seq[idx]:
            idx += 1
            if idx == len(vt_seq):
                return True
    return False

def last_draw_of_section(section_blob: Dict[str, Any]) -> Optional[str]:
    """
    Best-effort: take Set1 -> Draw7 (last column) if present; else the last available draw's draw_data[0].
    Fallback to None if unavailable.
    """
    sets = section_blob.get('sets', {})
    set1 = sets.get('Set1', {})
    draws = set1.get('draws', {})
    # prefer Draw7, then Draw6, ..., Draw1
    for i in range(7, 0, -1):
        d = draws.get(f'Draw{i}')
        if d and d.get('draw_data'):
            # draw_data is a list like ["632","267","898"], last draw is the rightmost recent?
            # In your tables, draw_data is listed oldest -> newest across columns;
            # for comparison we take the *most recent* (rightmost) which is the last element.
            return str(d['draw_data'][-1])
    # fallback: try any set/draw
    for setname, sblob in sets.items():
        for dname, dblob in sblob.get('draws', {}).items():
            if dblob.get('draw_data'):
                return str(dblob['draw_data'][-1])
    return None

def star_level(meta: Dict[str, Any]) -> str:
    """
    Map metadata.hot_zone_count to level:
      >=5 => 'superhot'
      >=2 => 'hot'
      else 'none'
    """
    hz = int(meta.get('hot_zone_count', 0) or 0)
    if hz >= 5:
        return 'superhot'
    elif hz >= 2:
        return 'hot'
    return 'none'

def draw_to_column_index(set_name: str, draw_name: str) -> int:
    """
    Columns: Set1 Draw1->7, Draw2->6, ... Draw7->1.
    For Set2/Set3 we consider Draw1 as col7 (they act as C2 snapshots).
    """
    if set_name == 'Set1':
        try:
            n = int(draw_name.replace('Draw', ''))
        except Exception:
            return 7
        # Draw1 -> 7, Draw2 -> 6, ..., Draw7 -> 1
        return max(1, min(7, 8 - n))
    else:
        return 7  # previous days’ C2 snapshots modeled as col7

# ----------------------------
# Transit-digit reduction (clutter removal)
# ----------------------------

def reduce_by_transit(s: str, transit_digits: Set[str]) -> str:
    """
    Remove transit digits and their mirrors from string s (single pass).
    """
    remove = set(transit_digits) | {mirror_of(d) for d in transit_digits}
    return ''.join(ch for ch in s if ch not in remove)

# Data containers
# ----------------------------

@dataclass
class BoxRef:
    section: str             # 'Midday' | 'Evening' | 'Combined'
    set_name: str            # 'Set1' | 'Set2' | 'Set3'
    draw_name: str           # 'Draw1'..'Draw7'
    column: int              # 1..7
    row: str                 # 'R2' | 'R4' | 'R6' | 'R8'
    star: str                # 'none' | 'hot' | 'superhot'
    raw: str                 # the string value in the box (e.g., '559244133877')

@dataclass
class Evidence:
    # counts & flags accumulated for a triad
    star_counts: Counter = field(default_factory=Counter)  # {'hot':x,'superhot':y}
    late_cols: Counter = field(default_factory=Counter)    # {3:cnt,2:cnt,1:cnt}
    vertical_hits: int = 0
    horizontal_span: Set[Tuple[str, int]] = field(default_factory=set)  # (section, column)
    cross_variants: Set[str] = field(default_factory=set)  # sections where starred evidence found
    vt_lane_hits: int = 0
    straight_order_hits: int = 0
    unordered_hits: int = 0
    pending_sections: Set[str] = field(default_factory=set)
    reasons: List[str] = field(default_factory=list)

    def to_reason_string(self) -> str:
        return '; '.join(self.reasons)

@dataclass
class Candidate:
    triad: str                 # canonical triad, e.g. '169', '355'
    vt: str                    # vt triple (e.g. '252')
    score: float               # total score
    evidence: Evidence
    by_section: Dict[str, Dict[str, Any]] = field(default_factory=dict)  # finer breakdown

@dataclass
class Weights:
    # Star / recency
    w_superhot: float = 3.0
    w_hot: float = 1.5
    w_col3: float = 1.0
    w_col2: float = 1.5
    w_col1: float = 2.5

    # Structure & spread
    w_vertical_per_row: float = 0.4
    w_horizontal_per_col: float = 0.3

    # Cross-variant reinforcement
    w_cross_variant_2: float = 2.0
    w_cross_variant_3: float = 3.5

    # Pattern “types”
    w_unordered: float = 0.3
    w_straight: float = 0.6
    w_vt_lane: float = 1.2

    # Pending
    w_pending_per_section: float = 0.5

def iter_boxes(env: Dict[str, Any], do_transit_reduction: bool = True) -> Iterable[BoxRef]:
    """
    Yield every (starred and non-starred) R2/R4/R6/R8 string as a BoxRef.
    Optionally apply transit-digit reduction using the section's most recent draw.
    """
    sections = env.get('sections', {})
    for section_name, section_blob in sections.items():
        transit = set(last_draw_of_section(section_blob)) if do_transit_reduction and last_draw_of_section(section_blob) else set()
        sets_blob = section_blob.get('sets', {})
        for set_name, set_blob in sets_blob.items():
            draws_blob = set_blob.get('draws', {})
            for draw_name, draw_blob in draws_blob.items():
                meta = draw_blob.get('metadata', {}) or {}
                star = star_level(meta)
                col = draw_to_column_index(set_name, draw_name)
                pv = draw_blob.get('pattern_variations', {}) or {}
                for row in ('R2','R4','R6','R8'):
                    if row in pv:
                        seqs = pv[row]
                        if not isinstance(seqs, list):
                            continue
                        for s in seqs:
                            s_str = ''.join(ch for ch in str(s) if ch in DIGITS)
                            if do_transit_reduction and transit:
                                s_str = reduce_by_transit(s_str, transit)
                            yield BoxRef(
                                section=section_name,
                                set_name=set_name,
                                draw_name=draw_name,
                                column=col,
                                row=row,
                                star=star,
                                raw=s_str
                            )

def triads_from_string(s: str) -> Set[str]:
    """
    Extract candidate triads from a string by taking combinations of distinct digits present.
    We avoid a combinatorial explosion by *not* enumerating subsequences; we use the set of digits.
    """
    ds = sorted(set(ch for ch in s if ch in DIGITS))
    triads = set()
    if len(ds) >= 3:
        for a, b, c in itertools.combinations(ds, 3):
            triads.add(canon_triad(a+b+c))
    # include doubles if any digits repeat substantially (heuristic: if length>8 and a digit occurs >=3)
    cnt = Counter(s)
    for d, k in cnt.items():
        if d in DIGITS and k >= 3:
            # generate double patterns like d d x (x != d)
            for x in ds:
                if x != d:
                    triads.add(canon_triad(d+d+x))
    return triads

def discover_candidates(env: Dict[str, Any]) -> Set[str]:
    """
    Discover candidate triads by scanning **starred** boxes with emphasis on Set1 late columns.
    """
    candidates: Set[str] = set()
    for box in iter_boxes(env, do_transit_reduction=True):
        if box.star == 'none':
            continue
        # Late column emphasis: col<=3 strongly preferred; we still collect from hot/superhot anywhere
        triads = triads_from_string(box.raw)
        if box.column <= 3 or box.star in ('hot','superhot'):
            candidates.update(triads)
    return candidates

def accumulate_evidence_for_triad(env: Dict[str, Any], triad: str) -> Candidate:
    """
    Walk all boxes and accumulate evidence for one triad.
    """
    ev = Evidence()
    by_section: Dict[str, Dict[str, Any]] = defaultdict(lambda: {
        'hot': 0, 'superhot': 0, 'cols': set(), 'rows_hit': set(),
        'unordered': 0, 'straight': 0, 'vt_lane': 0
    })

    # precompute VT target straight in VT-space for the triad as canonical order variants
    vt_target = vt_triple(triad)

    # pending proxy per section
    sections_blob = env.get('sections', {})
    section_last: Dict[str, Optional[str]] = {name: last_draw_of_section(blob) for name, blob in sections_blob.items()}

    # to count vertical density per (section, set, draw, column): rows that match
    vertical_seen: DefaultDict[Tuple[str,int], Set[str]] = defaultdict(set)

    # to compute horizontal span, track (section, column) where we matched at least once
    horizontal_cols: Set[Tuple[str,int]] = set()

    # We count matches using two modes:
    #  - unordered cover (all digits appear)
    #  - straight (any permutation appears as subsequence)
    perms = all_permutations_with_dupes(triad)

    for box in iter_boxes(env, do_transit_reduction=True):
        matched_unordered = unordered_cover(box.raw, triad)
        matched_straight = any(subseq_in(box.raw, p) for p in perms)
        matched_vt_lane = vt_subseq_in(box.raw, vt_target)

        if matched_unordered or matched_straight or matched_vt_lane:
            # star / late columns
            if box.star in ('hot','superhot'):
                ev.star_counts[box.star] += 1
                ev.cross_variants.add(box.section)
                if box.star == 'superhot':
                    ev.reasons.append(f"{box.section}/{box.set_name}/{box.draw_name}/{box.row}@col{box.column}: superhot")
                else:
                    ev.reasons.append(f"{box.section}/{box.set_name}/{box.draw_name}/{box.row}@col{box.column}: hot")

            if box.column in (3,2,1):
                ev.late_cols[box.column] += 1

            # vertical/horizontal
            vertical_seen[(box.section, box.column)].add(box.row)
            horizontal_cols.add((box.section, box.column))

            # type tallies
            if matched_unordered:
                ev.unordered_hits += 1
                by_section[box.section]['unordered'] += 1
            if matched_straight:
                ev.straight_order_hits += 1
                by_section[box.section]['straight'] += 1
            if matched_vt_lane:
                ev.vt_lane_hits += 1
                by_section[box.section]['vt_lane'] += 1

            # section breakdown
            by_section[box.section]['cols'].add(box.column)
            by_section[box.section]['rows_hit'].add(box.row)
            if box.star == 'hot':
                by_section[box.section]['hot'] += 1
            elif box.star == 'superhot':
                by_section[box.section]['superhot'] += 1

    # finalize vertical density
    ev.vertical_hits = sum(len(rows) for rows in vertical_seen.values())
    ev.horizontal_span = horizontal_cols

    # pending proxy
    for sec, lastn in section_last.items():
        if not lastn:
            continue
        # if triad equals literal last draw (any permutation), consider satisfied in that section
        if any(''.join(p) == lastn for p in itertools.permutations(triad, 3)):
            # satisfied -> no pending tag
            pass
        else:
            ev.pending_sections.add(sec)

    # score assembly
    w = Weights()
    score = 0.0
    score += ev.star_counts['superhot'] * w.w_superhot
    score += ev.star_counts['hot'] * w.w_hot
    score += ev.late_cols.get(3,0) * w.w_col3
    score += ev.late_cols.get(2,0) * w.w_col2
    score += ev.late_cols.get(1,0) * w.w_col1
    score += ev.vertical_hits * w.w_vertical_per_row
    score += len(ev.horizontal_span) * w.w_horizontal_per_col

    if len(ev.cross_variants) == 2:
        score += w.w_cross_variant_2
    elif len(ev.cross_variants) >= 3:
        score += w.w_cross_variant_3

    score += ev.unordered_hits * w.w_unordered
    score += ev.straight_order_hits * w.w_straight
    score += ev.vt_lane_hits * w.w_vt_lane

    score += len(ev.pending_sections) * w.w_pending_per_section

    return Candidate(
        triad=triad,
        vt=vt_triple(triad),
        score=round(score, 4),
        evidence=ev,
        by_section=by_section
    )

def default_convert(obj):
    if isinstance(obj, set):
        return sorted(list(obj))
    if hasattr(obj, '__dict__'):
        return asdict(obj)
    return str(obj)

def write_json(outdir: str, ranked: List[Candidate]) -> str:
    path = os.path.join(outdir, 'hotzones_candidates.json')
    ensure_dir(outdir)
    blob = []
    for i, c in enumerate(ranked, 1):
        blob.append({
            'rank': i,
            'triad': c.triad,
            'vt_triple': c.vt,
            'score': c.score,
            'evidence': {
                'star_counts': c.evidence.star_counts,
                'late_cols': c.evidence.late_cols,
                'vertical_hits': c.evidence.vertical_hits,
                'horizontal_span': sorted(list(c.evidence.horizontal_span)),
                'cross_variants': sorted(list(c.evidence.cross_variants)),
                'unordered_hits': c.evidence.unordered_hits,
                'straight_order_hits': c.evidence.straight_order_hits,
                'vt_lane_hits': c.evidence.vt_lane_hits,
                'pending_sections': sorted(list(c.evidence.pending_sections)),
                'reasons': c.evidence.reasons
            },
            'by_section': {
                k: {
                    'hot': v['hot'],
                    'superhot': v['superhot'],
                    'cols': sorted(list(v['cols'])),
                    'rows_hit': sorted(list(v['rows_hit'])),
                    'unordered': v['unordered'],
                    'straight': v['straight'],
                    'vt_lane': v['vt_lane'],
                } for k, v in c.by_section.items()
            }
        })
    with open(path, 'w', encoding='utf-8') as f:
        json.dump(blob, f, indent=2, ensure_ascii=False)
    return path

def run(input_json: str, outdir: str, topk: int) -> Tuple[str, str]:
    with open(input_json, 'r', encoding='utf-8') as f:
        env = json.load(f)

    cands = discover_candidates(env)
    if not cands:
        print("No candidates discovered from starred boxes. Consider checking JSON or star thresholds.", file=sys.stderr)

    ranked: List[Candidate] = []
    for triad in cands:
        ranked.append(accumulate_evidence_for_triad(env, triad))

    ranked.sort(key=lambda c: (-c.score, c.triad))

    if topk and topk > 0:
        ranked = ranked[:topk]

    csv_path = write_csv(outdir, ranked)
    json_path = write_json(outdir, ranked)
    return csv_path, json_path

VTRAC_MAP = {
    "0": "1", "5": "1",
    "1": "2", "6": "2",
    "2": "3", "7": "3",
    "3": "4", "8": "4",
    "4": "5", "9": "5",

def ordered_unique_digits(s: str) -> str:
    """
    Return ordered unique digits seen left-to-right in s.
    Example: "611888" -> "618"; "559244133877" -> "59241387" -> then trimmed by triad()
    """
    out = []
    seen = set()
    for ch in s:
        if ch.isdigit() and ch not in seen:
            out.append(ch)
            seen.add(ch)
    return "".join(out)

def triad_from_string(s: str) -> Tuple[str, bool]:
    """
    Extract the primary 3-value ordered triad (or double) from a string.
    - We take ordered-unique digits left->right, then:
      * if >=3 uniques: take the first 3 as the 'ordered triad'
      * if exactly 2 uniques: return both (double pattern), flag is_double=True
      * if <2 uniques: return "", False (not useful)
    This follows the training notion that extended strings (e.g., 611888) live in the same
    3-value envelope as their shorter core (618). (Design choice described in logs.) 
    """
    ou = ordered_unique_digits(s)
    if len(ou) >= 3:
        return ou[:3], False
    if len(ou) == 2:
        return ou, True
    return "", False

def vtrac_of_digits(s: str) -> str:
    """Map each digit in s to its VTRAC symbol (1..5) preserving order."""
    return "".join(VTRAC_MAP.get(ch, "?") for ch in s)

def canonical_boxed(s: str) -> str:
    """Return sorted digits (boxed canonical) for 2- or 3-digit strings."""
    return "".join(sorted(s))

def column_number_from_idx(length: int, idx: int) -> int:
    """
    In each DrawX row, arrays shrink 7->1 (Draw1 has 7 entries = columns 7..1).
    For each array entry index idx, the 'column' is (length - idx).
    """
    return length - idx

def is_superhot(set_name: str, column: int) -> bool:
    """
    Treat Set1 columns 1/2 as 'superhot' as per C1 vertical emphasis in training.
    """
    return set_name == "Set1" and column in (1, 2)

class HotZonesScorer:
    def __init__(self) -> None:
        # Accumulators
        self.cells: List[CellEvidence] = []
        # Indexes for features
        self.by_triad_loc: Dict[Tuple[str,str,str,int,str], List[CellEvidence]] = defaultdict(list)
        self.by_triad_all: Dict[Tuple[str,str], List[CellEvidence]] = defaultdict(list)  # (section, triad_o)
        self.by_vt_family: Dict[Tuple[str,str], List[CellEvidence]] = defaultdict(list) # (section, vt_b)
        self.triad_columns_by_set: Dict[Tuple[str,str,str], Set[int]] = defaultdict(set) # (section, set, triad_o)
        self.triad_columns_set1: Dict[Tuple[str,str], Set[int]] = defaultdict(set)       # (section, triad_o)

        # For cross-variant echoes, we’ll also track across sections
        self.triad_sections: Dict[str, Set[str]] = defaultdict(set)   # triad_o -> {Midday, Evening, Combined}
        self.vt_sections: Dict[str, Set[str]] = defaultdict(set)      # vt_b -> sections

        # Track col1/col2 & LS(4/2) hits (per triad_o within Set1)
        self.triad_col_hits_set1: Dict[Tuple[str,str], Counter] = defaultdict(Counter)  # (section, triad_o) -> counts

    # ---------- Parsing ----------
    def ingest_json(self, data: Dict) -> None:
        """
        Ingests the JSON environment (Ontario example structure).
        """
        sections = data.get("sections", {})
        for section_name, section_obj in sections.items():
            if section_name not in SECTION_KEYS:
                # keep all, but section filtering available if needed
                pass
            sets_obj = section_obj.get("sets", {})
            for set_name, set_obj in sets_obj.items():
                draws = set_obj.get("draws", {})
                # Determine draw indices: assume names like "Draw1".."Draw7"
                for draw_name, draw_obj in draws.items():
                    draw_idx = self._parse_draw_index(draw_name)
                    meta = draw_obj.get("metadata", {})
                    is_hot = bool(meta.get("is_hot_zone", False))
                    hz_count = int(meta.get("hot_zone_count", 0))

                    pv = draw_obj.get("pattern_variations", {})
                    for rkey in R_KEYS:
                        arr = pv.get(rkey, [])
                        length = len(arr)
                        for i, literal_string in enumerate(arr):
                            col = column_number_from_idx(length, i)
                            triad_o, is_double = triad_from_string(literal_string)
                            if not triad_o:
                                continue
                            triad_b = canonical_boxed(triad_o)
                            vt_o = vtrac_of_digits(triad_o)
                            vt_b = canonical_boxed(vt_o)
                            sup = is_superhot(set_name, col)

                            cell = CellEvidence(
                                section=section_name, set_name=set_name, draw_name=draw_name,
                                draw_index=draw_idx, rkey=rkey, column=col,
                                is_hot_zone=is_hot, hot_zone_count=hz_count,
                                literal_string=literal_string,
                                triad_o=triad_o, triad_is_double=is_double,
                                triad_b=triad_b, vt_o=vt_o, vt_b=vt_b,
                                superhot=sup
                            )
                            self.cells.append(cell)

                            # Feature indexes
                            k_loc = (section_name, set_name, draw_name, col, triad_o)
                            self.by_triad_loc[k_loc].append(cell)
                            self.by_triad_all[(section_name, triad_o)].append(cell)
                            self.by_vt_family[(section_name, vt_b)].append(cell)

                            self.triad_columns_by_set[(section_name, set_name, triad_o)].add(col)
                            if set_name == "Set1":
                                self.triad_columns_set1[(section_name, triad_o)].add(col)
                                self.triad_col_hits_set1[(section_name, triad_o)][col] += 1

                            self.triad_sections[triad_o].add(section_name)
                            self.vt_sections[vt_b].add(section_name)

    @staticmethod
    def _parse_draw_index(draw_name: str) -> int:
        # "Draw1".."Draw7" -> 1..7 (fallback 0)
        try:
            return int(draw_name.replace("Draw", "").strip())
        except Exception:
            return 0

    # ---------- Feature computation ----------
    def _vertical_density(self, section: str, set_name: str, draw_name: str, col: int, triad_o: str) -> int:
        # count occurrences across R2/R4/R6/R8 for same cell location and triad
        cnt = 0
        for rkey in R_KEYS:
            k = (section, set_name, draw_name, col, triad_o)
            cnt += sum(1 for c in self.by_triad_loc.get(k, []) if c.rkey == rkey)
        return max(1, cnt)  # at least 1

    def _horizontal_span(self, section: str, set_name: str, triad_o: str) -> int:
        cols = self.triad_columns_by_set.get((section, set_name, triad_o), set())
        return (max(cols) - min(cols)) if cols else 0

    def _cross_variant_echo(self, triad_o: str) -> int:
        return len(self.triad_sections.get(triad_o, set()))

    def _ls_col_42(self, section: str, triad_o: str) -> bool:
        cols = self.triad_columns_set1.get((section, triad_o), set())
        return bool(4 in cols or 2 in cols)

    def _funnel_precol1(self, section: str, triad_o: str) -> bool:
        cols = self.triad_columns_set1.get((section, triad_o), set())
        # "funnel" means presence in col 2 or 3, and also col 1
        return bool((2 in cols or 3 in cols) and (1 in cols))

    def _ls2_lane(self, section: str, triad_o: str) -> bool:
        # emphasize Set1 columns 3->1 path
        cols = self.triad_columns_set1.get((section, triad_o), set())
        return bool(3 in cols and 1 in cols)

    def _vt_only_lane(self, section: str, vt_b: str, triad_b: str) -> bool:
        """
        Heuristic: if a VT family appears via multiple distinct literal boxed triads
        in the same section (i.e., same vt_b but many triad_b forms), we mark vt_only_lane.
        This captures 'family marching' even when one exact literal triad isn’t dominant.
        """
        triad_b_set = set(c.triad_b for c in self.by_vt_family.get((section, vt_b), []))
        return len(triad_b_set) > 1 and (len(triad_b_set) >= 3 or triad_b not in triad_b_set)

    def _recency_weight(self, set_name: str, column: int, superhot: bool) -> float:
        # Bias Set1 and col1/2; extra if superhot.
        base = 0.0
        if set_name == "Set1":
            base += 1.0
            if column == 1:
                base += 1.0
            elif column == 2:
                base += 0.6
        if superhot:
            base += 0.4
        return base

    def _score_item(self,
                    section: str, set_name: str, draw_name: str, draw_index: int,
                    rkey: str, column: int, triad_o: str, triad_b: str, vt_b: str,
                    is_hot_zone: bool, superhot: bool) -> Tuple[float, Dict[str, float|int|bool]]:
        vert = self._vertical_density(section, set_name, draw_name, column, triad_o)
        span = self._horizontal_span(section, set_name, triad_o)
        echo = self._cross_variant_echo(triad_o)
        ls42 = self._ls_col_42(section, triad_o)
        funnel = self._funnel_precol1(section, triad_o)
        ls2 = self._ls2_lane(section, triad_o)
        vt_only = self._vt_only_lane(section, vt_b, triad_b)
        rec = self._recency_weight(set_name, column, superhot)

        # Weighted additive score (tuned from your logs’ pain points)
        # - heavier recency/superhot & column-1, moderate vertical density, give VT lanes a lane,
        #   and make LS col(4/2) & funnel add lift.
        score = 0.0
        score += 2.2 * rec
        score += 1.6 * math.log1p(vert)
        score += 0.9 * math.log1p(span)
        score += 0.8 * (echo - 1) if echo > 1 else 0.0
        score += 0.7 if funnel else 0.0
        score += 0.45 if ls2 else 0.0
        score += 0.35 if ls42 else 0.0
        score += 0.5 if vt_only else 0.0
        score += 0.4 if is_hot_zone else 0.0
        score += 0.4 if superhot else 0.0

        feats = dict(
            vertical_density=vert,
            horizontal_span=span,
            cross_variant_echo=echo,
            recency_score=rec,
            ls_col_42=ls42,
            funnel_precol1=funnel,
            ls2_lane=ls2,
            vt_only_lane=vt_only,
        )
        return score, feats

    # ---------- Public API ----------
    def build_per_item(self) -> List[PerItemRow]:
        rows: List[PerItemRow] = []
        raw_scores: List[float] = []

        # Precompute a per-item entry for each cell/triad occurrence (collapsed per location)
        # We collapse over literal_string within the same (section,set,draw,col,rkey,triad_o)
        seen_keys: Set[Tuple[str,str,str,int,str,str]] = set()
        for cell in self.cells:
            k = (cell.section, cell.set_name, cell.draw_name, cell.column, cell.rkey, cell.triad_o)
            if k in seen_keys:
                continue
            seen_keys.add(k)

            sc, feats = self._score_item(
                section=cell.section, set_name=cell.set_name, draw_name=cell.draw_name,
                draw_index=cell.draw_index, rkey=cell.rkey, column=cell.column,
                triad_o=cell.triad_o, triad_b=cell.triad_b, vt_b=cell.vt_b,
                is_hot_zone=cell.is_hot_zone, superhot=cell.superhot
            )
            raw_scores.append(sc)

            # evidence tag string
            tags = []
            if cell.is_hot_zone: tags.append("hot")
            if cell.superhot: tags.append("superhot")
            if feats["funnel_precol1"]: tags.append("funnel_precol1")
            if feats["ls2_lane"]: tags.append("ls2_lane")
            if feats["ls_col_42"]: tags.append("ls_col_42")
            if feats["vt_only_lane"]: tags.append("vt_only_lane")
            if feats["cross_variant_echo"] and feats["cross_variant_echo"] > 1:
                tags.append(f"xvar{int(feats['cross_variant_echo'])}")

            row = PerItemRow(
                section=cell.section,
                set_name=cell.set_name,
                draw_name=cell.draw_name,
                draw_index=cell.draw_index,
                rkey=cell.rkey,
                column=cell.column,
                triad_o=cell.triad_o,
                triad_b=cell.triad_b,
                vt_o=cell.vt_o,
                vt_b=cell.vt_b,
                literal_string=cell.literal_string,
                is_hot_zone=cell.is_hot_zone,
                superhot=cell.superhot,
                ls_col_42=bool(feats["ls_col_42"]),
                funnel_precol1=bool(feats["funnel_precol1"]),
                ls2_lane=bool(feats["ls2_lane"]),
                vt_only_lane=bool(feats["vt_only_lane"]),
                vertical_density=int(feats["vertical_density"]),
                horizontal_span=int(feats["horizontal_span"]),
                cross_variant_echo=int(feats["cross_variant_echo"]),
                recency_score=float(feats["recency_score"]),
                score=sc,
                normalized_score=0.0,  # fill later
                evidence_tags=",".join(tags)
            )
            rows.append(row)

        # Normalize scores to 0..100
        if raw_scores:
            mx = max(raw_scores)
            if mx <= 0:
                for r in rows:
                    r.normalized_score = 0.0
            else:
                for r in rows:
                    r.normalized_score = round((r.score / mx) * 100.0, 3)

        return rows

    def build_top_candidates(self, per_item: List[PerItemRow]) -> List[TopCandidateRow]:
        # Aggregate by (section, vt_b) to capture family consensus across literal forms
        agg: Dict[Tuple[str,str], List[PerItemRow]] = defaultdict(list)
        for r in per_item:
            agg[(r.section, r.vt_b)].append(r)

        out: List[TopCandidateRow] = []
        for (section, vt_b), items in agg.items():
            support = len(items)
            triad_b_counts = Counter(r.triad_b for r in items)
            triad_b_modes = ",".join(tb for tb, _ in triad_b_counts.most_common(3))

            superhot_hits = sum(1 for r in items if r.superhot)
            col1_hits = sum(1 for r in items if r.column == 1)
            col2_hits = sum(1 for r in items if r.column == 2)
            ls_col_42_hits = sum(1 for r in items if r.ls_col_42)
            funnel_hits = sum(1 for r in items if r.funnel_precol1)
            vt_only_lane_hits = sum(1 for r in items if r.vt_only_lane)
            cross_sections = len({r.section for r in items})  # will be 1 here (aggregating per section), but retained

            score_sum = sum(r.score for r in items)
            score_max = max(r.score for r in items) if items else 0.0

            # Aggregate evidence tags (limited)
            tag_counter = Counter()
            for r in items:
                for t in filter(None, r.evidence_tags.split(",")):
                    tag_counter[t] += 1
            tags = ",".join(t for t, _ in tag_counter.most_common(6))

            out.append(TopCandidateRow(
                section=section,
                key_family=vt_b,
                triad_b_modes=triad_b_modes,
                support_count=support,
                superhot_hits=superhot_hits,
                col1_hits=col1_hits,
                col2_hits=col2_hits,
                ls_col_42_hits=ls_col_42_hits,
                funnel_hits=funnel_hits,
                vt_only_lane_hits=vt_only_lane_hits,
                cross_variant_sections=cross_sections,
                score_sum=round(score_sum, 3),
                score_max=round(score_max, 3),
                evidence_tags=tags
            ))

        # Rank implicitly by score_sum then score_max, then superhot/col1
        out.sort(key=lambda r: (r.score_sum, r.score_max, r.superhot_hits, r.col1_hits), reverse=True)
        return out

        os.makedirs(path, exist_ok=True)

                    r.vertical_density, r.horizontal_span, r.cross_variant_echo, f"{r.recency_score:.3f}",

    def write_top_candidates_csv(self, rows: List[TopCandidateRow], out_dir: str, state_name: str) -> str:

"""

    └─ <STATE>_hotzones_meta.json

Design goals:
  - Late-column (c3/c2/c1) emphasis with Superhot bias
  - Vertical clustering (R2/R4/R6/R8) and horizontal persistence (column-span)
  - Cross-variant reinforcement (Midday/Evening/Combined)
  - VTRAC-aware family rollups (consistent with centralized winners taxonomy)
"""

from __future__ import annotations

import argparse

# -------- VTRAC mapping (0/5→1, 1/6→2, 2/7→3, 3/8→4, 4/9→5) --------

VT_MAP: Dict[str, str] = {
    '0': '1', '5': '1',
    '1': '2', '6': '2',
    '2': '3', '7': '3',
    '3': '4', '8': '4',
    '4': '5', '9': '5'
}

SECTIONS_ORDER = ("Midday", "Evening", "Combined")
ROW_ORDER = ("R2", "R4", "R6", "R8")


# --------------------------- Data classes ---------------------------

@dataclass(frozen=True)
class Location:
    state: str
    section: str            # Midday / Evening / Combined
    set_name: str           # Set1 / Set2 / Set3
    draw_name: str          # Draw1..Draw7 (Set1 has Draw1..Draw7; Set2/3 often Draw1)
    row: str                # R2/R4/R6/R8
    col_index_from_left: int   # 1..N (left→right as stored)
    col_index_from_right: int  # 1..N (rightmost is 1 = most recent)
    is_hot_zone_marked: bool   # metadata.is_hot_zone
    hot_zone_count: int        # metadata.hot_zone_count if present

@dataclass(frozen=True)
class TriadKey:
    """Canonical identity for aggregation (value triad + vtrac family)."""
    triad_straight: str      # e.g. "691"
    triad_boxed: str         # sorted digits, e.g. "169"
    vtrac_straight: str      # e.g. "241"
    vtrac_boxed: str         # sorted vtrac code, e.g. "124"

@dataclass
class PerItem:
    """One hot-zone observation (box-level)."""
    loc: Location
    src_string: str               # the raw R2/R4/R6/R8 column string
    triad: TriadKey
    is_hot: bool                  # late columns (<=3 from right) OR metadata
    is_superhot: bool             # late columns (<=2 from right)
    features: Dict[str, float]    # expanded scoring parts
    score: float                  # final per-item score
    evidence_tags: List[str]      # textual tags for auditability

@dataclass
class CandidateAgg:
    """Aggregate across all observations for one triad family."""
    triad: TriadKey
    appearances: int = 0
    hot_hits: int = 0
    superhot_hits: int = 0
    sections: Set[str] = field(default_factory=set)
    sets: Set[str] = field(default_factory=set)
    rows: Set[str] = field(default_factory=set)
    columns_right: Counter = field(default_factory=Counter)  # distribution of right-index (1=most recent)
    column_span: int = 0
    late_columns_covered: Set[int] = field(default_factory=set)  # which of {1,2,3}
    cross_variant_count: int = 0
    vertical_count: int = 0      # how many distinct rows (R2/R4/R6/R8) observed
    normalized_score: float = 0.0
    evidence_tags: List[str] = field(default_factory=list)


# --------------------------- Utilities ------------------------------

def last_three_unique_digits(s: str) -> Optional[str]:
    """Return the right-to-left last three unique digits found in s."""
    seen = []
    for ch in reversed(s):
        if ch.isdigit() and ch not in seen:
            seen.append(ch)
            if len(seen) == 3:
                break
    if not seen:
        return None
    # We want the order as encountered right-to-left reversed → left-to-right
    triad = ''.join(reversed(seen))
    # If shorter than 3 unique digits (double/triple), left-pad with the last seen digit
    if len(triad) < 3:
        triad = (triad[0] * (3 - len(triad))) + triad  # e.g., "66" → "666"; "68" → "668"
    return triad


def triad_key_for(s: str) -> Optional[TriadKey]:
    triad = last_three_unique_digits(s)
    if not triad:
        return None
    boxed = ''.join(sorted(triad))
    vt_straight = ''.join(VT_MAP[d] for d in triad)
    vt_boxed = ''.join(sorted(vt_straight))
    return TriadKey(triad_straight=triad, triad_boxed=boxed,
                    vtrac_straight=vt_straight, vtrac_boxed=vt_boxed)


def iter_boxes(state_json: dict) -> Iterable[Tuple[Location, Dict[str, List[str]]]]:
    """Yield (Location, pattern_variations) across all sections/sets/draws with per-draw metadata."""
    state = state_json.get("state_name", "UNKNOWN")
    sections = state_json.get("sections", {})
    for section in SECTIONS_ORDER:
        sec = sections.get(section)
        if not sec:
            continue
        sets = sec.get("sets", {})
        for set_name, set_body in sets.items():
            draws = set_body.get("draws", {})
            for draw_name, draw_body in draws.items():
                patt = draw_body.get("pattern_variations", {})
                meta = draw_body.get("metadata", {}) or {}
                # Column count uses R2 length if present, else max length among rows
                max_cols = max((len(v) for v in patt.values() if isinstance(v, list)), default=0)
                for row in ROW_ORDER:
                    if row not in patt:
                        continue
                    row_list = patt[row]
                    # For each column string in this row, define location
                    N = len(row_list)
                    for idx_left, col_string in enumerate(row_list, start=1):
                        idx_right = N - idx_left + 1
                        loc = Location(
                            state=state,
                            section=section,
                            set_name=set_name,
                            draw_name=draw_name,
                            row=row,
                            col_index_from_left=idx_left,
                            col_index_from_right=idx_right,
                            is_hot_zone_marked=bool(meta.get("is_hot_zone", False)),
                            hot_zone_count=int(meta.get("hot_zone_count", 0))
                        )
                        yield (loc, {"string": col_string})


# --------------------------- Scoring -------------------------------

def score_per_item(loc: Location,
                   triad: TriadKey,
                   src_string: str) -> Tuple[float, Dict[str, float], List[str], bool, bool]:
    """
    Compute per-item score with explicit parts:
      - recency / late-column emphasis
      - superhot boost
      - hot-zone metadata boost
      - vertical presence (later used in aggregate)
      - column-span prep (later in aggregate)
    """
    # Late-column hot windows
    is_superhot = loc.col_index_from_right <= 2
    is_hot = loc.col_index_from_right <= 3 or loc.is_hot_zone_marked

    parts: Dict[str, float] = {}

    # Base: presence in a hot window
    parts["base_presence"] = 1.0

    # Recency gradient (closer to col1 → higher)
    # col_right = 1 (most recent), we want higher weight; simple reciprocal-like curve
    parts["recency"] = 1.5 / (loc.col_index_from_right ** 0.5)

    # Superhot & Hot boosts
    parts["superhot_bonus"] = 1.25 if is_superhot else 0.0
    parts["hot_bonus"] = 0.75 if is_hot and not is_superhot else 0.0

    # Star-mark boost if metadata flagged the draw as hot_zone
    parts["metadata_hot_mark"] = 0.5 if loc.is_hot_zone_marked else 0.0

    # Column funnel hint: emphasize columns {3,2,1} (sharper near end)
    late = max(0, 4 - loc.col_index_from_right)  # 3→1 map to 1..3; others 0
    parts["late_window_intensity"] = 0.35 * late

    # Compact straight cue: if the last three chars of src_string match triad order (not strict but heuristic)
    tail3 = ''.join([ch for ch in src_string if ch.isdigit()])[-3:]
    parts["straight_tail_hint"] = 0.5 if tail3 == triad.triad_straight else 0.0

    # Sum to final score
    score = sum(parts.values())

    tags = []
    if is_superhot: tags.append("superhot")
    if is_hot and not is_superhot: tags.append("hot")
    if loc.is_hot_zone_marked: tags.append("hot_mark")
    if parts["straight_tail_hint"] > 0: tags.append("straight_tail")

    return score, parts, tags, is_hot, is_superhot


# --------------------------- Aggregation ---------------------------

def aggregate_candidates(per_items: List[PerItem]) -> List[CandidateAgg]:
    agg: Dict[TriadKey, CandidateAgg] = {}

    # Index to compute cross-variant + vertical presence + column-span
    seen_by_triad_columns_right: Dict[TriadKey, Set[int]] = defaultdict(set)

    for item in per_items:
        triad = item.triad
        if triad not in agg:
            agg[triad] = CandidateAgg(triad=triad)
        a = agg[triad]
        a.appearances += 1
        if item.is_hot: a.hot_hits += 1
        if item.is_superhot: a.superhot_hits += 1
        a.sections.add(item.loc.section)
        a.sets.add(item.loc.set_name)
        a.rows.add(item.loc.row)
        a.columns_right[item.loc.col_index_from_right] += 1
        if item.loc.col_index_from_right in (1, 2, 3):
            a.late_columns_covered.add(item.loc.col_index_from_right)
        # Prepare tags
        a.evidence_tags.extend(item.evidence_tags)

        # Track columns for span calc
        seen_by_triad_columns_right[triad].add(item.loc.col_index_from_right)

    # Finalize derived attributes
    for k, a in agg.items():
        if seen_by_triad_columns_right.get(k):
            cols = sorted(seen_by_triad_columns_right[k])
            a.column_span = (cols[-1] - cols[0] + 1) if len(cols) >= 2 else 1
        a.cross_variant_count = len(a.sections)
        a.vertical_count = len(a.rows)

        # Normalize score: blend intensity and coverage
        # Weight late coverage, vertical presence, cross-variant consensus, superhot density
        cover = len(a.late_columns_covered) / 3.0  # 0..1
        vert = a.vertical_count / 4.0              # 0..1
        cross = a.cross_variant_count / 3.0        # 0..1
        super_density = (a.superhot_hits / max(1, a.appearances))
        hot_density = (a.hot_hits / max(1, a.appearances))

        a.normalized_score = (
            2.2 * cover +
            1.8 * vert +
            1.8 * cross +
            1.6 * super_density +
            1.2 * hot_density +
            0.6 * math.log1p(a.column_span)
        )

    # Sort by normalized_score desc, then superhot density, then hot density
    ranked = sorted(agg.values(), key=lambda c: (c.normalized_score,
                                                 c.superhot_hits,
                                                 c.hot_hits,
                                                 c.appearances), reverse=True)
    return ranked


# --------------------------- Winners join (optional) ---------------

def load_winners(path: str) -> Set[str]:
    """
    Load winners as 3-digit strings from a simple CSV/TSV:
      - either a single column with values like 681
      - or a header containing 'winner' / 'winning' / 'number'
    """
    winners: Set[str] = set()
    with open(path, 'r', newline='') as f:
        sample = f.read(1024)
        f.seek(0)
        dialect = csv.Sniffer().sniff(sample) if sample else csv.excel
        reader = csv.reader(f, dialect)
        header = next(reader, [])
        cols = [h.strip().lower() for h in header]
        # If header doesn't look like a header, treat first row as data
        def normalize(n: str) -> Optional[str]:
            n2 = ''.join(ch for ch in n if ch.isdigit())
            if len(n2) == 3:
                return n2
            return None

        if any(k in cols for k in ("winner", "winning", "number", "midday", "evening")):
            # Use all cells as possible winners
            if header:
                for row in [header] + list(reader):
                    for cell in row:
                        val = normalize(cell)
                        if val: winners.add(val)
        else:
            # No header, just values
            if header:
                val = normalize(header[0])
                if val: winners.add(val)
            for row in reader:
                if not row: continue
                val = normalize(row[0])
                if val: winners.add(val)
    return winners


def annotate_four_lanes(winners: Set[str], triad: TriadKey) -> Tuple[bool, bool, bool, bool]:
    """Return (exact_straight, exact_boxed, vt_boxed, vt_straight) against literal winners set."""
    if not winners:
        return (False, False, False, False)
    # Exact straight / boxed
    exact_straight = triad.triad_straight in winners
    exact_boxed = any(''.join(p) in winners for p in set(map(''.join, __permutations(triad.triad_boxed))))
    # VT family: map each winner to vtrac and compare boxed/straight
    vt_winners_straight = {''.join(VT_MAP[d] for d in w) for w in winners if len(w) == 3 and all(c.isdigit() for c in w)}
    vt_winners_box = {''.join(sorted(v)) for v in vt_winners_straight}
    vt_straight = triad.vtrac_straight in vt_winners_straight
    vt_boxed = triad.vtrac_boxed in vt_winners_box
    return (exact_straight, exact_boxed, vt_boxed, vt_straight)


def __permutations(boxed: str) -> Iterable[Tuple[str, str, str]]:
    """All unique permutations of a 3-char string (handles doubles)."""
    a, b, c = boxed[0], boxed[1], boxed[2]
    seen = set()
    for x in (a, b, c):
        for y in (a, b, c):
            for z in (a, b, c):
                if sorted((x, y, z)) == sorted((a, b, c)):
                    key = (x, y, z)
                    if key not in seen:
                        seen.add(key)
                        yield key

def _sha1_of_file(path: str) -> str:
    sha1 = hashlib.sha1()
    with open(path, 'rb') as f:
        while True:
            chunk = f.read(65536)
            if not chunk:
                break
            sha1.update(chunk)
    return sha1.hexdigest()


def _dedupe_in_order(seq: List[str]) -> List[str]:
    seen = set()
    out = []
    for x in seq:
        if x not in seen:
            seen.add(x)
            out.append(x)
    return out

def run_hotzones_pipeline(input_json_path: str,
                          out_root: str = "data/outputs/analysis/hotzones",
                          winners_csv: Optional[str] = None) -> str:
    with open(input_json_path, 'r') as f:
        data = json.load(f)

    state = data.get("state_name", "UNKNOWN")
    out_dir = os.path.join(out_root, state)
    ensure_dir(out_dir)

    # Optional winners
    winners: Optional[Set[str]] = load_winners(winners_csv) if winners_csv else None

    per_items: List[PerItem] = []

    for loc, payload in iter_boxes(data):
        s = payload["string"]
        triad = triad_key_for(s)
        if not triad:
            continue
        score, parts, tags, is_hot, is_superhot = score_per_item(loc, triad, s)
        per_items.append(PerItem(
            loc=loc,
            src_string=s,
            triad=triad,
            is_hot=is_hot,
            is_superhot=is_superhot,
            features=parts,
            score=score,
            evidence_tags=tags
        ))

    # Aggregate
    ranked = aggregate_candidates(per_items)

    # Write bundle
    per_item_path = os.path.join(out_dir, f"{state}_hotzones_per_item.csv")
    top_path = os.path.join(out_dir, f"{state}_hotzones_top_candidates.csv")
    meta_path = os.path.join(out_dir, f"{state}_hotzones_meta.json")

    write_per_item_csv(per_item_path, per_items, winners)
    write_top_csv(top_path, ranked, winners)
    write_meta_json(meta_path, state, input_json_path, per_items)

    return out_dir

# hotzones/guards.py
from __future__ import annotations
import hashlib, json, os, time
from dataclasses import dataclass
from pathlib import Path
from typing import Dict, List, Optional, Tuple

SECTIONS = ("Midday", "Evening", "Combined")
SETS = ("Set1", "Set2", "Set3")
ROWS = ("R2", "R4", "R6", "R8")
COLS = ("C7","C6","C5","C4","C3","C2","C1")

@dataclass
class TablePresence:
    section: str
    set_name: str
    row: str
    path: Path
    mtime: float

@dataclass
class FreshnessReport:
    ok: bool
    missing: List[str]
    stale: List[str]
    present: List[TablePresence]

def _glob_table_candidates(
    tables_root: Path, state: str, section: str, set_name: str
) -> Dict[str, Path]:
    """
    Locate CSVs in either of two shapes:
      A) <root>/<STATE>/<section>/<set_name>_<row>.csv   (e.g., Set1_R2.csv)
      B) <root>/<STATE>/<section>/<set_name>.csv         (stacked: rows R2/R4/R6/R8)
    Returns dict: row -> path  (if stacked, maps 'STACKED' -> path)
    """
    baseA = tables_root / state / section
    candidates = {}
    # A) per-row wide CSVs
    for row in ROWS:
        p = baseA / f"{set_name}_{row}.csv"
        if p.exists():
            candidates[row] = p
    # B) single stacked file
    p_stacked = baseA / f"{set_name}.csv"
    if p_stacked.exists():
        candidates["STACKED"] = p_stacked
    return candidates

def check_tables_freshness(
    tables_root: str | Path,
    state: str,
    min_mtime_sec: Optional[float] = None
) -> FreshnessReport:
    """
    Ensure required CSVs exist for each (section, set, row).
    If a stacked CSV exists for a set, that set doesn't require per-row files.
    Optionally assert recent mtime (e.g., produced during this run).
    """
    tables_root = Path(tables_root)
    missing, stale, present = [], [], []
    for section in SECTIONS:
        for set_name in SETS:
            found = _glob_table_candidates(tables_root, state, section, set_name)
            if not found:
                # Neither per-row nor stacked found
                for row in ROWS:
                    missing.append(f"{state}/{section}/{set_name}_{row}.csv (or {set_name}.csv)")
                continue
            if "STACKED" in found:
                p = found["STACKED"]
                mt = p.stat().st_mtime
                if min_mtime_sec and mt < min_mtime_sec:
                    stale.append(str(p))
                present.append(TablePresence(section, set_name, "STACKED", p, mt))
            else:
                for row, p in found.items():
                    mt = p.stat().st_mtime
                    if min_mtime_sec and mt < min_mtime_sec:
                        stale.append(str(p))
                    present.append(TablePresence(section, set_name, row, p, mt))
    return FreshnessReport(ok=(len(missing)==0 and len(stale)==0), missing=missing, stale=stale, present=present)

def read_meta_hash(meta_path: Path) -> Optional[str]:
    """
    Read a meta file containing config/git SHA (aligning with analyzer lean bundles).
    Returns a short hash string if present; else None.
    """
    if not meta_path.exists():
        return None
    try:
        meta = json.loads(meta_path.read_text())
        # Prefer a config hash or git SHA if available (Analyzer Lean Output Spec)
        # e.g., {"config_hash":"...", "git_sha":"..."}  (hot zones adopts same spirit)
        val = meta.get("config_hash") or meta.get("git_sha")
        return val
    except Exception:
        return None

def compute_dir_signature(paths: List[Path]) -> str:
    """Produce a simple signature from filenames + modified times."""
    h = hashlib.sha256()
    for p in sorted(paths):
        st = p.stat()
        h.update(str(p).encode())
        h.update(str(int(st.st_mtime)).encode())
        h.update(str(st.st_size).encode())
    return h.hexdigest()[:16]

import pandas as pd

COLS = ("C7","C6","C5","C4","C3","C2","C1")

def load_set_tables(
    path_map: Dict[str, Path]
) -> Dict[str, pd.DataFrame]:
    """
    Return dict of row-name -> DataFrame with columns C7..C1 (strings).
    Supports:
      - STACKED: one CSV with column 'row' (values R2/R4/R6/R8) + C7..C1
      - per-row: one CSV per row containing only C7..C1
    All outputs are 1-row DataFrames indexed by 'row'.
    """
    out: Dict[str, pd.DataFrame] = {}
    if "STACKED" in path_map:
        df = pd.read_csv(path_map["STACKED"], dtype=str).fillna("")
        if "row" not in df.columns:
            raise ValueError(f"Stacked CSV {path_map['STACKED']} missing 'row' column.")
        # Keep only expected rows/cols; coerce to strings
        cols = ["row"] + list(COLS)
        df = df[[c for c in cols if c in df.columns]].copy()
        df["row"] = df["row"].str.upper().str.strip()
        df = df[df["row"].isin(["R2","R4","R6","R8"])]
        for r in ["R2","R4","R6","R8"]:
            sub = df[df["row"]==r]
            if sub.empty:
                continue
            row_df = sub[COLS].head(1).copy()
            row_df.index = [r]
            # ensure all string
            for c in COLS:
                row_df[c] = row_df[c].astype(str).fillna("")
            out[r] = row_df
        return out

    # Per-row wide CSVs
    for r, p in path_map.items():
        df = pd.read_csv(p, dtype=str).fillna("")
        # Accept either a single row with C7..C1, or a file that already has head row
        if any(c not in df.columns for c in COLS):
            # maybe the file has just columns and values in first record
            raise ValueError(f"{p} missing expected columns {COLS}")
        row_df = df[COLS].head(1).copy()
        row_df.index = [r]
        for c in COLS:
            row_df[c] = row_df[c].astype(str).fillna("")
        out[r] = row_df
    return out

class DrawMeta:
    is_hot_zone: bool
    hot_zone_count: int

def _progressive_slices(values: List[str]) -> List[List[str]]:
    """
    Given [C7..C1], return the progressive arrays used in JSON:
      Draw1: all 7
      Draw2: last 6
      ...
      Draw7: last 1
    """
    out = []
    n = len(values)
    for k in range(n):
        out.append(values[k:])  # k=0 => 7, k=1 => 6, ..., k=6 => 1
    return out

def _estimate_hot_zone_flags(progressions: List[str]) -> Tuple[bool,int]:
    """
    Heuristic: consider C3/C2/C1 as 'hot' columns; count non-empty strings among them.
    This is conservative but consistent with the late-column funnel (C3→C2→C1).
    """
    hot_cols = progressions[-3:] if len(progressions) >= 3 else progressions
    non_empty = sum(1 for s in hot_cols if s and s.strip())
    return (non_empty > 0, non_empty)

def build_set_structure(set_tables: Dict[str, pd.DataFrame]) -> Dict:
    """
    Build the Set{1..3} JSON block with Draw{1..7} (Set1) and Draw1 (Set2/Set3).
    """
    # Collect R2/R4/R6/R8 strings as [C7..C1]
    rows_as_lists: Dict[str, List[str]] = {}
    for r in ROWS:
        if r in set_tables:
            rows_as_lists[r] = [set_tables[r].iloc[0][c] for c in COLS]
        else:
            rows_as_lists[r] = [""]*7

    # Build progressive slices for Set1: Draw1..Draw7
    set_json = {"draws": {}}
    # Draw data (original numbers) is optional: attach if provided by caller later
    for draw_idx in range(7):
        draw_key = f"Draw{draw_idx+1}"
        pattern_variations = {}
        # For each row (R2..R8), attach progressive slices starting at index draw_idx
        for r in ROWS:
            full = rows_as_lists[r]
            slice_vals = full[draw_idx:]  # down to C1
            pattern_variations[r] = slice_vals
        is_hot, hot_cnt = _estimate_hot_zone_flags([rows_as_lists["R2"][i] for i in range(draw_idx, 7)])
        set_json["draws"][draw_key] = {
            "pattern_variations": pattern_variations,
            "metadata": asdict(DrawMeta(is_hot_zone=is_hot, hot_zone_count=hot_cnt))
        }
    return set_json

def build_set_structure_prior(set_tables: Dict[str, pd.DataFrame]) -> Dict:
    """
    Build Set2/Set3 JSON (Draw1 only): we snapshot the same progressive arrays as 'Draw1'
    to reflect the prior-day 'C2' tables context.
    """
    rows_as_lists: Dict[str, List[str]] = {}
    for r in ROWS:
        if r in set_tables:
            rows_as_lists[r] = [set_tables[r].iloc[0][c] for c in COLS]
        else:
            rows_as_lists[r] = [""]*7
    pattern_variations = {r: rows_as_lists[r] for r in ROWS}
    is_hot, hot_cnt = _estimate_hot_zone_flags(rows_as_lists["R2"])
    return {
        "draws": {
            "Draw1": {
                "pattern_variations": pattern_variations,
                "metadata": asdict(DrawMeta(is_hot_zone=is_hot, hot_zone_count=hot_cnt))
            }
        }
    }

def _read_draws_list(draws_file: Optional[str]) -> Optional[List[str]]:
    if not draws_file:
        return None
    p = Path(draws_file)
    if not p.exists():
        raise FileNotFoundError(f"draws_file not found: {p}")
    vals: List[str] = []
    for line in p.read_text().splitlines():
        s = line.strip().replace(",","").replace("-","")
        if not s:
            continue
        # keep only 3-char tokens
        if len(s) == 3 and s.isdigit():
            vals.append(s)
    # Only keep the last 7
    vals = vals[-7:]
    if len(vals) < 7:
        # pad with empty to keep shape stable
        vals = ([""]*(7-len(vals))) + vals
    return vals

def _sorted_triad(a: str, b: str, c: str) -> str:
    s = ''.join(sorted((a, b, c)))
    return s

def _triad_to_vtrac(triad: str) -> str:
    # triad is 'xyz'
    try:
        vx = _VT_MAP[triad[0]]
        vy = _VT_MAP[triad[1]]
        vz = _VT_MAP[triad[2]]
    except KeyError:
        # Should never happen if triad is digits 0-9
        vx, vy, vz = 9, 9, 9
    return 'v' + ''.join(str(x) for x in sorted((vx, vy, vz)))

def _digit_counts(s: str) -> Counter:
    return Counter(ch for ch in s if ch.isdigit())

def _triads_from_counts(cnt: Counter) -> Set[str]:
    """Generate all triads (including doubles/triples) feasible from digit multiplicities."""
    digits = sorted([d for d, k in cnt.items() if k > 0])
    triads: Set[str] = set()

    # abc (all unique)
    L = len(digits)
    for i in range(L):
        for j in range(i+1, L):
            for k in range(j+1, L):
                triads.add(_sorted_triad(digits[i], digits[j], digits[k]))

    # aab
    for d, c in cnt.items():
        if c >= 2:
            for e in digits:
                if e == d:
                    continue
                t = _sorted_triad(d, d, e)
                triads.add(t)

    # aaa
    for d, c in cnt.items():
        if c >= 3:
            triads.add(_sorted_triad(d, d, d))

    return triads

def _draw_digit_sets(draw_list: List[str]) -> List[Set[str]]:
    """Literal 3-digit numbers -> set of digits for boxed equality checks."""
    out = []
    for x in draw_list:
        # normalize to 3 chars, keep digits
        s = ''.join(ch for ch in x if ch.isdigit())
        if len(s) == 3:
            out.append(set(s))
    return out

def _col_index_from_position(pos: int, total: int) -> int:
    """
    Convert a string position within DrawN's list to a C1/C2-like column number.
    For Draw1 (total = 7), pos 0..6 maps to columns 7..1. For shorter Draws adapt.
    """
    return max(1, total - pos)

class HotZonesBundle:
    """
    Analyze Hot Zones from the JSON table environment and emit the lean bundle.

    The analyzer focuses on strings inside Draw/row structures and records triad/V-TRAC
    occurrences for starred boxes, then aggregates per triad (per-variant) to compute
    funnel/recency/cross-echo features. Scoring emphasizes Set1 col-1 funnels, hot-star sums,
    row coverage, and Set2/Set3 echoes, consistent with your analyzer instincts.
    """

    def __init__(self, state: str, json_path: str, out_root: str,
                 write_steps: bool = True, write_stacked: bool = True) -> None:
        self.state = state
        self.json_path = json_path
        self.out_root = out_root
        self.write_steps = write_steps
        self.write_stacked = write_stacked

        with open(json_path, "r", encoding="utf-8") as f:
            self.data = json.load(f)

        self.occurs: List[Occur] = []
        self.literal_draw_sets: Dict[str, List[Set[str]]] = {}  # per variant
        self.per_item: List[PerItemRow] = []
        self.top: List[TopCandidateRow] = []
        self.meta: Optional[Meta] = None

    # ---- parsing & collection ----

    def collect_occurrences(self) -> None:
        sections = self.data.get("sections", {})
        for variant in _VARIANTS:
            sec = sections.get(variant, {})
            sets = sec.get("sets", {})
            if not sets:
                continue

            # collect literal draw digit sets (boxed identity check)
            literal_sets: List[Set[str]] = []
            # We concatenate all Draws' 'draw_data' lists we encounter in this variant.
            for set_id in _SETS:
                if set_id not in sets:
                    continue
                draws = sets[set_id].get("draws", {})
                for draw_id, dnode in draws.items():
                    draw_data = dnode.get("draw_data", [])
                    literal_sets.extend(_draw_digit_sets(draw_data))
            self.literal_draw_sets[variant] = literal_sets

            # now collect triads from pattern strings
            for set_id in _SETS:
                if set_id not in sets:
                    continue
                draws = sets[set_id].get("draws", {})
                for draw_id, dnode in draws.items():
                    meta = dnode.get("metadata", {})
                    is_hot = bool(meta.get("is_hot_zone", False))
                    hot_ct = int(meta.get("hot_zone_count", 0))

                    pv = dnode.get("pattern_variations", {})
                    for row in _ROWS:
                        seq = pv.get(row, [])
                        total = len(seq)
                        for pos, s in enumerate(seq):
                            # Only consider starred contexts as HotZones input
                            if not is_hot:
                                continue
                            cnt = _digit_counts(s)
                            triads = _triads_from_counts(cnt)
                            col = _col_index_from_position(pos, total)
                            for tri in triads:
                                vt = _triad_to_vtrac(tri)
                                self.occurs.append(Occur(
                                    triad=tri, vtrac=vt, variant=variant,
                                    set_id=set_id, draw_id=draw_id,
                                    col=col, row=row, is_hot_zone=is_hot,
                                    hot_zone_count=hot_ct, source_len=len(s),
                                    source_string=s
                                ))

    # ---- aggregation & features ----

    def build_per_item(self) -> None:
        # Index occurrences by (triad, vtrac, variant)
        by_key: Dict[TriadKey, List[Occur]] = defaultdict(list)
        for oc in self.occurs:
            key = TriadKey(oc.triad, oc.vtrac, oc.variant)
            by_key[key].append(oc)

        # Cross-variant echo detection: track keys across variants
        # We’ll mark echo later while emitting rows.
        triad_variants: Dict[Tuple[str, str], Set[str]] = defaultdict(set)
        for key in by_key:
            triad_variants[(key.triad, key.vtrac)].add(key.variant)

        # Precompute literal boxed presence per variant
        def triad_matches_literal(triad: str, variant: str) -> bool:
            literal_sets = self.literal_draw_sets.get(variant, [])
            tset = set(triad)
            return any(tset == ls for ls in literal_sets)

        per_item_rows: List[PerItemRow] = []

        for key, occs in by_key.items():
            # appearances across rows/sets/cols (Set1 only for col features)
            rows = sorted(set(o.row for o in occs))
            sets = sorted(set(o.set_id for o in occs))
            cols_set1 = sorted({o.col for o in occs if o.set_id == "Set1"}, reverse=True)
            col1 = 1 if 1 in cols_set1 else 0
            col2 = 1 if 2 in cols_set1 else 0
            col3 = 1 if 3 in cols_set1 else 0
            ls2_lane = 1 if (col3 or col2) and not col1 else 0
            funnel_precol1 = 1 if (col2 and col1) else 0
            ls_col_42 = 1 if (4 in cols_set1 and 2 in cols_set1) else 0

            # row coverage
            row_cov = len(rows)

            # hotness
            hot_hits = sum(1 for _ in occs if _.is_hot_zone)
            hot_star_sum = sum(_.hot_zone_count for _ in occs if _.is_hot_zone)

            # cross echoes
            cross_set_echo = 1 if any(s in ("Set2", "Set3") for s in sets) else 0
            cross_variant_echo = 1 if len(triad_variants[(key.triad, key.vtrac)]) >= 2 else 0

            # vt_only_lane = observed in hot zones but never equals any literal draw's digit set
            vt_only_lane = 0 if triad_matches_literal(key.triad, key.variant) else 1

            # persistence in Set1 (# unique cols)
            set1_persistence_cols = len(set(o.col for o in occs if o.set_id == "Set1"))

            # normalized score: crafted from funnel-to-col1, hotness, echoes, coverage
            score = 0.0
            score += 3.0 * col1
            score += 1.5 * funnel_precol1
            score += 0.5 * col2 + 0.25 * col3
            score += 0.75 * cross_set_echo + 0.75 * cross_variant_echo
            score += 0.4 * row_cov
            score += 0.15 * set1_persistence_cols
            score += 0.08 * hot_hits + 0.03 * hot_star_sum
            score += 0.4 * vt_only_lane
            # Clamp to [0, 10] then normalize
            score = max(0.0, min(10.0, score))
            normalized = round(score / 10.0, 6)

            reasons: List[str] = []
            if col1: reasons.append("col1")
            if funnel_precol1: reasons.append("funnel_precol1")
            if ls2_lane: reasons.append("ls2_lane")
            if ls_col_42: reasons.append("ls_col_42")
            if cross_set_echo: reasons.append("cross_set_echo")
            if cross_variant_echo: reasons.append("cross_variant_echo")
            if vt_only_lane: reasons.append("vt_only_lane")
            if row_cov >= 3: reasons.append("row_coverage>=3")
            if set1_persistence_cols >= 3: reasons.append("set1_persistence_cols>=3")

            per_item_rows.append(PerItemRow(
                triad=key.triad, vtrac=key.vtrac, variant=key.variant,
                appears_in_rows=",".join(rows),
                appears_in_sets=",".join(sets),
                appears_in_cols_set1=",".join(str(c) for c in cols_set1),
                col1=col1, col2=col2, col3=col3,
                ls2_lane=ls2_lane,
                funnel_precol1=funnel_precol1,
                ls_col_42=ls_col_42,
                vt_only_lane=vt_only_lane,
                hot_hits=hot_hits,
                hot_star_sum=hot_star_sum,
                cross_set_echo=cross_set_echo,
                cross_variant_echo=cross_variant_echo,
                row_coverage_count=row_cov,
                set1_persistence_cols=set1_persistence_cols,
                normalized_score=normalized,
                reasons_json=json.dumps(reasons, ensure_ascii=False),
            ))

        self.per_item = per_item_rows

    def build_top_candidates(self, topk: int = 200) -> None:
        # sort per-item rows by normalized_score desc then by hot_star_sum desc
        pis = sorted(self.per_item, key=lambda r: (r.normalized_score, r.hot_star_sum), reverse=True)
        top_rows: List[TopCandidateRow] = []
        for r in pis[:topk]:
            tags = []
            if r.col1: tags.append("col1")
            if r.funnel_precol1: tags.append("funnel")
            if r.vt_only_lane: tags.append("vt-only")
            if r.cross_variant_echo: tags.append("x-variant")
            if r.cross_set_echo: tags.append("x-set")
            if r.ls_col_42: tags.append("col4&2")
            if r.ls2_lane and not r.col1: tags.append("ls2")
            top_rows.append(TopCandidateRow(
                triad=r.triad, vtrac=r.vtrac, variant=r.variant,
                score=r.normalized_score,
                evidence_tags=",".join(tags),
                set1_cols=r.appears_in_cols_set1,
                rows=r.appears_in_rows,
                hot_hits=r.hot_hits,
                hot_star_sum=r.hot_star_sum,
            ))
        self.top = top_rows

    # ---- writers ----

    def _ensure_dirs(self) -> Tuple[str, str]:
        out_dir = os.path.join(self.out_root, self.state)
        train_dir = os.path.join(out_dir, "training")
        os.makedirs(out_dir, exist_ok=True)
        os.makedirs(train_dir, exist_ok=True)
        return out_dir, train_dir

    def _write_per_item_csv(self, out_dir: str) -> str:
        path = os.path.join(out_dir, f"{self.state}_hotzones_per_item.csv")
        with open(path, "w", encoding="utf-8", newline="") as f:

                    row.triad, row.vtrac, row.variant, row.appears_in_rows, row.appears_in_sets,

    def _write_top_csv(self, out_dir: str) -> str:

    def run(self) -> None:

# V-TRAC core (deterministic)

# digit -> VTRAC index (1..5)
DIGIT_TO_V = {
    "0": 1, "5": 1,
    "1": 2, "6": 2,
    "2": 3, "7": 3,
    "3": 4, "8": 4,
    "4": 5, "9": 5,
}

# VTRAC index -> the two digits in canonical (low, high) order
V_TO_DIGITS: Dict[int, Tuple[str, str]] = {
    1: ("0", "5"),
    2: ("1", "6"),
    3: ("2", "7"),
    4: ("3", "8"),
    5: ("4", "9"),
}

def mirror_digit(d: str) -> str:
    """Mirror per V-TRAC pair (0<->5, 1<->6, ..., 4<->9)."""
    v = DIGIT_TO_V[d]
    a, b = V_TO_DIGITS[v]
    return b if d == a else a

def triad_to_vcode(triad: Tuple[str, str, str]) -> Tuple[int, int, int]:
    """Map a 3-digit tuple to a V-TRAC code (v1, v2, v3)."""
    return tuple(DIGIT_TO_V[x] for x in triad)  # type: ignore[return-value]

def vcode_to_vtrac_straight(vcode: Tuple[int, int, int]) -> List[str]:
    """
    Generate the 8 V-TRAC straight combinations for a vcode (v1,v2,v3),
    by choosing one digit from each V-TRAC pair in order.
    Deterministic bit-order: 000..111 over (v1,v2,v3).
    """
    v1, v2, v3 = vcode
    d1 = V_TO_DIGITS[v1]
    d2 = V_TO_DIGITS[v2]
    d3 = V_TO_DIGITS[v3]
    out = []
    for b in range(8):  # 0..7
        c1 = d1[(b >> 2) & 1]
        c2 = d2[(b >> 1) & 1]
        c3 = d3[b & 1]
        out.append(c1 + c2 + c3)
    # keep stable unique (though it should already be unique)
    seen, uniq = set(), []
    for s in out:
        if s not in seen:
            seen.add(s)
            uniq.append(s)
    return uniq

def boxed_unique(numbers: Iterable[str]) -> List[str]:
    """
    Reduce to boxed (unordered) unique Pick-3 combinations:
    normalized as sorted digit tuples.
    """
    seen = set()
    out: List[str] = []
    for n in numbers:
        key = "".join(sorted(n))
        if key not in seen:
            seen.add(key)
            out.append(n)  # keep representative as first-seen
    return out

def is_double(n: str) -> bool:
    return len(set(n)) == 2

def is_triple(n: str) -> bool:
    return len(set(n)) == 1

def literal_permutations(triad: Tuple[str, str, str]) -> List[str]:
    """All literal permutations, deduped."""
    perms = {"".join(p) for p in itertools.permutations(triad, 3)}
    return sorted(perms)

def pair_mirror_12(triad: Tuple[str, str, str]) -> List[str]:
    """
    Your training rule:
      - Build three combos from each pair + mirror(missing digit)
      - For each such triple, emit 4 permutations that keep the pair 'visibly adjacent' in some order.
    Deterministic, deduped, returns up to 12.
    """
    a, b, c = triad
    combos = [
        (a, b, mirror_digit(c)),
        (b, c, mirror_digit(a)),
        (a, c, mirror_digit(b)),
    ]
    out: List[str] = []
    for (x, y, z) in combos:
        candidates = [
            x + y + z,  # XY_
            y + x + z,  # YX_
            x + z + y,  # X_Z (keeps X leading)
            z + x + y,  # ZXY
        ]
        for cand in candidates:
            if cand not in out:
                out.append(cand)
    return out

def infer_core_triad_from_string(s: str) -> Optional[Tuple[str, str, str]]:
    """
    Infer a 3-value core (ordered by first appearance) from a long digit string.
    Strategy:
      1) Scan left→right; collect first three distinct digits → triad.
      2) If >3 unique digits, try to compress via V-TRAC class merges to 3 classes;
         choose the earliest majority representatives.
      3) Return None if fewer than 3 unique digits exist (pure doubles/triples).
    Deterministic and safe on noisy strings.
    """
    digits = [ch for ch in s if ch.isdigit()]
    order = []
    seen = set()
    for d in digits:
        if d not in seen:
            seen.add(d)
            order.append(d)
        if len(order) == 3:
            return (order[0], order[1], order[2])
    # if fewer than 3 unique digits, try to expand via mirrors (unlikely useful)
    # but keep it strict: return None and let doubles/triples be handled elsewhere.
    return None

@dataclass(frozen=True)
class Triad:
    d1: str
    d2: str
    d3: str

    def as_tuple(self) -> Tuple[str, str, str]:
        return (self.d1, self.d2, self.d3)

    def vcode(self) -> Tuple[int, int, int]:
        return triad_to_vcode(self.as_tuple())

    def literals(self) -> List[str]:
        return literal_permutations(self.as_tuple())

    def vtrac_straight_8(self) -> List[str]:
        return vcode_to_vtrac_straight(self.vcode())

    def vtrac_boxed(self) -> List[str]:
        return boxed_unique(self.vtrac_straight_8())

    def pair_mirror_12(self) -> List[str]:
        return pair_mirror_12(self.as_tuple())

@dataclass
class BoxEvidence:
    section: str         # Midday/Evening/Combined
    set_id: str          # Set1/Set2/Set3
    draw_id: str         # Draw1..Draw7
    row: str             # R2/R4/R6/R8
    column_index: int    # 1..7 (rightmost=1)
    is_hot_zone: bool
    hot_zone_count: int

@dataclass
class TriadCandidate:
    triad: Triad
    evidences: List[BoxEvidence] = field(default_factory=list)
    superhot_hits: int = 0        # late columns (1/2) appearances
    cross_rows: Set[str] = field(default_factory=set)  # rows (R2..R8) supporting
    cross_sections: Set[str] = field(default_factory=set)
    score: float = 0.0

    def compute_score(self) -> float:
        """
        Transparent heuristic score (kept simple by design):
          - base: count of evidences with is_hot_zone
          - + hot_zone_count sum (scaled)
          - + bonus for late columns (col 1/2)
          - + cross-row diversity
          - + cross-section reinforcement
        Deterministic; no randomness.
        """
        hot_hits = sum(1 for e in self.evidences if e.is_hot_zone)
        hot_star_sum = sum(e.hot_zone_count for e in self.evidences)
        late_bonus = sum(2.0 for e in self.evidences if e.column_index in (1, 2))
        row_bonus = 1.5 * max(0, len(self.cross_rows) - 1)
        sec_bonus = 2.0 * max(0, len(self.cross_sections) - 1)
        score = (hot_hits * 3.0) + (hot_star_sum * 0.5) + late_bonus + row_bonus + sec_bonus
        self.score = score
        return self.score

def _column_index_from_draw_key(draw_key: str, total_draws: int) -> int:
    """
    Map Draw1..Draw7 to column index (7..1) if total_draws==7.
    When arrays are shortened (6..1 etc.), keep a linear mapping.
    """
    assert draw_key.startswith("Draw")
    n = int(draw_key[4:])
    # Set1 Draw1 contains 7 draws (col7..col1). Draw2 contains 6, etc.
    # We'll map uniformly: leftmost draw in the list is older → higher column index.
    # So column_index = remaining_count - i
    # But since we only have the key, the caller must supply total_draws.
    idx_from_left = n  # not accurate without context; caller passes better info.
    # We don’t rely on this mapping for scoring; we will override with exact position if available.
    return max(1, min(7, 8 - n))

def harvest_triads_from_json(state_json: Dict) -> Dict[str, TriadCandidate]:
    """
    Walk the provided AAT9 JSON tree (exact schema you shared) and collect triad candidates
    from starred boxes across R2/R4/R6/R8, Set1..Set3, Draw1..Draw7.
    Returns dict keyed by triad canonical string (first-appearance order) -> TriadCandidate.
    """
    out: Dict[str, TriadCandidate] = {}
    sections = state_json.get("sections", {})
    for section_name, section in sections.items():
        sets = section.get("sets", {})
        for set_id, set_node in sets.items():
            draws = set_node.get("draws", {})
            # Compute total draws (#cols) using Draw1->draw_data length
            # If unavailable, default to 7.
            set_total = 7
            if "Draw1" in draws:
                set_total = len(draws["Draw1"].get("draw_data", [])) or 7
            for draw_key, draw_node in draws.items():
                pattern_variations = draw_node.get("pattern_variations", {})
                metadata = draw_node.get("metadata", {})
                is_hot = bool(metadata.get("is_hot_zone", False))
                hz_count = int(metadata.get("hot_zone_count", 0))
                # Column index estimate: rightmost (most recent) ~ 1
                col_index = max(1, min(7, set_total - (int(draw_key[4:]) - 1)))
                for row in ("R2", "R4", "R6", "R8"):
                    arr = pattern_variations.get(row, [])
                    for s in arr:
                        tri = infer_core_triad_from_string(s)
                        if tri is None:
                            continue
                        # Triad canonical string (keep first-appearance order)
                        canonical = "".join(tri)
                        cand = out.get(canonical)
                        if not cand:
                            cand = TriadCandidate(
                                triad=Triad(*tri),
                                evidences=[],
                                cross_rows=set(),
                                cross_sections=set(),
                            )
                            out[canonical] = cand
                        cand.evidences.append(
                            BoxEvidence(
                                section=section_name,
                                set_id=set_id,
                                draw_id=draw_key,
                                row=row,
                                column_index=col_index,
                                is_hot_zone=is_hot,
                                hot_zone_count=hz_count,
                            )
                        )
                        cand.cross_rows.add(row)
                        cand.cross_sections.add(section_name)
    # Compute scores deterministically
    for cand in out.values():
        cand.compute_score()
    return out

@dataclass
class PayoutModel:
    # Store/Online payout scaffolds (USD-like units)
    store_boxed: int = 100
    store_straight: int = 500
    online_boxed: int = 160
    online_straight: int = 900

    def eval_cost_payout(
        self,
        combos: Sequence[str],
        play_type: str = "online_straight",
        ticket_value: int = 1,
        rounds: int = 1,
        progression: Optional[List[int]] = None,
    ) -> Dict[str, float]:
        """
        Compute simple envelope: total_cost over N rounds, and the payout if a hit occurs in any round.
        - progression: list of ticket multipliers per round (e.g., [1,1,1,1,1]).
        """
        payout_table = {
            "store_boxed": self.store_boxed,
            "store_straight": self.store_straight,
            "online_boxed": self.online_boxed,
            "online_straight": self.online_straight,
        }
        if play_type not in payout_table:
            raise ValueError(f"Unknown play_type: {play_type}")
        if progression is None:
            progression = [1] * rounds
        elif len(progression) != rounds:
            raise ValueError("progression length must equal rounds")

        base_payout = payout_table[play_type]
        total_cost = 0
        for r in range(rounds):
            total_cost += len(combos) * ticket_value * progression[r]
        # If a single straight/boxed hits in any round, expected payout for that round:
        # Here we give the *best* case envelope (max round multiplier).
        max_mult = max(progression) if progression else 1
        payout = base_payout * ticket_value * max_mult
        return {
            "total_cost": float(total_cost),
            "payout_if_hit": float(payout),
            "net_if_hit": float(payout - total_cost),
        }

@dataclass
class Recommendation:
    triad: str
    score: float
    modes: Dict[str, List[str]]  # "literals", "vtrac_straight", "vtrac_boxed", "pair_mirror_12"
    meta: Dict[str, int]         # counts, doubles, triples, etc.

def build_recommendations(cands: Dict[str, TriadCandidate], top_k: int = 50) -> List[Recommendation]:
    ranked = sorted(cands.values(), key=lambda c: (-c.score, c.triad.as_tuple()))
    out: List[Recommendation] = []
    for cand in ranked[:top_k]:
        tri = cand.triad
        literals = tri.literals()
        v8 = tri.vtrac_straight_8()
        v8_box = tri.vtrac_boxed()
        pm12 = tri.pair_mirror_12()
        meta = {
            "evidence_count": len(cand.evidences),
            "hot_hits": sum(1 for e in cand.evidences if e.is_hot_zone),
            "hot_star_sum": sum(e.hot_zone_count for e in cand.evidences),
            "late_cols": sum(1 for e in cand.evidences if e.column_index in (1, 2)),
            "row_diversity": len(cand.cross_rows),
            "section_diversity": len(cand.cross_sections),
            "literals": len(literals),
            "v8": len(v8),
            "v8_boxed": len(v8_box),
            "pair_mirror_12": len(pm12),
            "doubles_in_literals": sum(1 for n in literals if is_double(n)),
            "triples_in_literals": sum(1 for n in literals if is_triple(n)),
        }
        out.append(Recommendation(
            triad="".join(tri.as_tuple()),
            score=round(cand.score, 3),
            modes={
                "literals": literals,
                "vtrac_straight": v8,
                "vtrac_boxed": v8_box,
                "pair_mirror_12": pm12,
            },
            meta=meta,
        ))
    return out

def ensure_dir(path: str) -> None:
    os.makedirs(path, exist_ok=True)

def write_outputs(state_name: str, out_dir: str, recs: List[Recommendation], meta: Dict) -> None:
    base = os.path.join(out_dir, "data", "outputs", "analysis", "hotzones", state_name)
    ensure_dir(base)

    # per_item as jsonl (one recommendation per line for easy streaming)
    per_item_path = os.path.join(base, f"{state_name}_hotzones_per_item.jsonl")
    with open(per_item_path, "w", encoding="utf-8") as f:
        for r in recs:
            f.write(json.dumps({
                "triad": r.triad,
                "score": r.score,
                "modes": r.modes,
                "meta": r.meta,
            }, ensure_ascii=False) + "\n")

    # top candidates (json)
    top_json_path = os.path.join(base, f"{state_name}_hotzones_top_candidates.json")
    with open(top_json_path, "w", encoding="utf-8") as f:
        json.dump({
            "state": state_name,
            "count": len(recs),
            "recommendations": [r.__dict__ for r in recs],
        }, f, ensure_ascii=False, indent=2)

    # meta.json (config, timestamps, version)
    meta_path = os.path.join(base, f"{state_name}_hotzones_meta.json")
    with open(meta_path, "w", encoding="utf-8") as f:
        json.dump(meta, f, ensure_ascii=False, indent=2)

    # CSV mirror (light)
    csv_path = os.path.join(base, f"{state_name}_hotzones_top_candidates.csv")
    with open(csv_path, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["rank", "triad", "score", "literals", "vtrac_straight", "vtrac_boxed", "pair_mirror_12",
                         "evidence_count", "hot_hits", "hot_star_sum", "late_cols", "row_diversity", "section_diversity"])
        for i, r in enumerate(recs, start=1):
            writer.writerow([
                i, r.triad, r.score,
                " ".join(r.modes["literals"]),
                " ".join(r.modes["vtrac_straight"]),
                " ".join(r.modes["vtrac_boxed"]),
                " ".join(r.modes["pair_mirror_12"]),
                r.meta["evidence_count"], r.meta["hot_hits"], r.meta["hot_star_sum"],
                r.meta["late_cols"], r.meta["row_diversity"], r.meta["section_diversity"],
            ])

def main() -> None:
    ap = argparse.ArgumentParser(
        description="AAT9 Hot Zones — Part 7: Combination & Wagering Engine"
    )
    ap.add_argument("--json", required=True,
                    help="Path to AAT9 JSON string-table environment (sections→sets→draws…) for a single state.")
    ap.add_argument("--out-dir", default=".",
                    help="Project root (we write into data/outputs/analysis/hotzones/<STATE>/)")
    ap.add_argument("--state", default=None,
                    help="Override state_name in JSON (if needed).")
    ap.add_argument("--top-k", type=int, default=50,
                    help="Number of triad families to emit.")
    ap.add_argument("--preview-roi", action="store_true",
                    help="Print ROI envelopes for the top-5 recommendations.")
    ap.add_argument("--log-level", default="INFO",
                    choices=["DEBUG", "INFO", "WARNING", "ERROR"])
    args = ap.parse_args()
    logging.basicConfig(level=getattr(logging, args.log_level))

    with open(args.json, "r", encoding="utf-8") as f:
        data = json.load(f)

    state_name = args.state or data.get("state_name", "UnknownState")
    if not state_name or state_name == "UnknownState":
        raise SystemExit("state_name missing — provide --state override or include in JSON.")

    logging.info("Harvesting triads from starred boxes …")
    cands = harvest_triads_from_json(data)
    if not cands:
        logging.warning("No triads harvested; check JSON structure or hot-zone metadata.")

    logging.info("Building recommendations …")
    recs = build_recommendations(cands, top_k=args.top_k)

    meta = {
        "contract": "hotzones.part7.v1",
        "state": state_name,
        "source": os.path.abspath(args.json),
        "counts": {
            "triad_candidates": len(cands),
            "top_emitted": len(recs),
        },
        "notes": "Outputs are lean Analyzer artifacts for Hot Zones.",
    }

    write_outputs(state_name, args.out_dir, recs, meta)
    logging.info("Wrote Hot Zones outputs under data/outputs/analysis/hotzones/%s", state_name)

    if args.preview_roi and recs:
        pm = PayoutModel()
        print("\nROI envelopes (top-5; online_straight, ticket=$1, 3 rounds [1,1,2]):")
        for i, r in enumerate(recs[:5], start=1):

            env = pm.eval_cost_payout(v8, play_type="online_straight", ticket_value=1, rounds=3, progression=[1,1,2])
            print(f"#{i:02d} triad={r.triad} score={r.score:.2f}  "
                  f"v8={len(v8)}  cost={env['total_cost']:.0f}  "
                  f"payout_if_hit={env['payout_if_hit']:.0f}  net_if_hit={env['net_if_hit']:.0f}")

if __name__ == "__main__":
    main()
