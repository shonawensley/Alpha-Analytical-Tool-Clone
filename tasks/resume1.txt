def _normalize_set_value(value: object) -> Optional[int]:
    if value is None: return None

    return int(digits) if digits else None

def _has_progression(values: list[Optional[int]]) -> bool:
    seq = sorted({v for v in values if v is not None})
    if len(seq) < 2:
        return False
    # reward consecutive runs like 3→2→1 (we only need a run length >=2)
    run = 1
    for i in range(1, len(seq)):
        if seq[i] == seq[i-1] + 1 or seq[i] == seq[i-1] - 1:
            run += 1
            if run >= 2:
                return True
        else:
            run = 1
    return False
