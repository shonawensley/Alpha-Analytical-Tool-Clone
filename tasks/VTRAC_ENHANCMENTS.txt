double_hits (or doubles_hit_count if that’s your field—code handles both)

Keep your telemetry: Combined × 1.15, Midday/Evening × 1.00.

Scoring rules (precise)

Base score (per section):

score = w_overlap   * overlap
      + w_stable    * stable_weighted_sum(stable_columns)
      + w_consensus * (consensus_col1 + consensus_col2)   # capped if no stability
      + w_tokenEcho * cross_section_echo
      + w_hot       * hot_norm
      + w_super     * superhot_norm
      + w_mask      * mask_drop_norm
      + w_mirror    * mirror_norm
      + w_doubles   * doubles_norm

score *= (1.15 if section == "Combined" else 1.00)


Consensus Rescue (strict):
If overlap == 0 and consensus_col1 or consensus_col2 and len(stable_columns) ≥ 1 ⇒ multiply the base score by 0.35 and flag weak_positive_rescue. (This keeps “signal‑present” cases visible but never dominant.)

Tiers (for quick reading):

overlap ≥ 4 → Tier A

overlap = 3 → Tier B+

overlap = 2 → Tier B

overlap = 1 → Tier C

overlap = 0 → Tier Z (only surfaced via rescue rule)

Token family promotion (ledger):

Build a ledger across all input JSONs: for every token (e.g., V4x2_5x2) track the set of states seen and the number of sections it appears in per state.

Apply a small bump when:

token appears in ≥ 3 states, or

token appears in ≥ 2 sections within the same state run.

Use this only to order the recommended_tokens[] list (doesn’t swamp the section score).

Straights blending (never primary):
Use available top_straights in each section purely as a tie‑breaker and for display; don’t demote a section just because “straight overlap = 0” in a narrow validation window.

#!/usr/bin/env python3
"""
V-TRAC Enhanced Validator → Compact Scorer/Exporter
- Reads *your existing* enhanced JSONs (no schema change)
- Computes per-section confidence scores with strict consensus rescue
- Builds a token-family ledger to lightly promote recurring families
- Exports a compact CSV + JSON per state×section

USAGE:
  python vtrac_score_and_export.py /path/to/folder/with_jsons

Outputs:
  /path/to/folder/vtrac_compact_report.csv
  /path/to/folder/vtrac_compact_report.json
"""
import sys, os, glob, json, re
from collections import defaultdict, Counter

WEIGHTS = {
    "overlap": 3.0,
    "stable": 1.6,
    "consensus": 1.0,
    "token_echo": 1.2,
    "hot": 0.25,
    "super": 0.35,
    "mask": 0.8,
    "mirror": 0.5,
    "doubles": 0.3,
    "combined_prior": 1.15,
    "rescue_multiplier": 0.35,
}

COL_W = {7:0.5, 6:0.7, 5:0.9, 4:1.1, 3:1.6, 2:2.0, 1:2.4}

def as_int_list(cols):
    out = []
    for c in cols or []:
        try: out.append(int(str(c).strip()))
        except: pass
    return out

def norm01(value, lo, hi):
    if hi <= lo: return 0.0
    x = (value - lo) / float(hi - lo)
    return max(0.0, min(1.0, x))

def parse_date_from_name(name):
    # ..._20251023_020648.json → 2025-10-23
    m = re.search(r'_(20\d{2})(\d{2})(\d{2})_', name)
    if not m: return ""
    y, mo, d = m.groups()
    return f"{y}-{mo}-{d}"

def load_jsons(root):
    return sorted(glob.glob(os.path.join(root, "*_vtrac_enhanced_*.json")))

def get_section_names(j):
    # Be robust to keys; prefer canonical set
    keys = list((j.get("sections") or {}).keys())
    out = []
    for k in ("Combined","Midday","Evening"):
        if k in keys: out.append(k)
    # append any others (if present)
    for k in keys:
        if k not in out: out.append(k)
    return out

def tokens_for_section(j, s):
    sec = (j.get("sections") or {}).get(s, {})
    sig = (sec.get("signals") or {})
    anl = (sec.get("analyzer_signatures") or {})
    winners_tokens = sig.get("top_vtrac_box_signatures") or []
    analyzer_tokens = (anl.get("primary") or [])
    return winners_tokens, analyzer_tokens

def straight_list_for_section(j, s):
    sec = (j.get("sections") or {}).get(s, {})
    mets = (sec.get("analyzer_metrics") or {}).get("primary") or {}
    # Supports either list of strings or list of dicts with score
    ts = mets.get("top_straights") or []
    if ts and isinstance(ts[0], dict):
        ts = [d.get("straight") for d in sorted(ts, key=lambda d: d.get("score", 0), reverse=True)]
    return ts[:3]

def metrics_for_section(j, s):
    sec = (j.get("sections") or {}).get(s, {})
    sig = (sec.get("signals") or {})
    mets = (sec.get("analyzer_metrics") or {}).get("primary") or {}
    hot = sig.get("hot")
    superhot = sig.get("superhot")
    stable_cols = as_int_list(sig.get("stable_columns") or [])
    c1 = bool(sig.get("consensus_col1"))
    c2 = bool(sig.get("consensus_col2"))

    mask_drop = mets.get("mask_drop_count")
    mirror_supported = mets.get("mirror_supported")
    if isinstance(mirror_supported, (int, float)):
        mirror_supported = mirror_supported > 0

    doubles = mets.get("double_hits")
    if doubles is None: doubles = mets.get("doubles_hit_count", 0)

    # Prefer section_summaries for counts if present
    summ = (j.get("section_summaries") or {}).get(s, {})
    hot_count = summ.get("hot_count")
    superhot_count = summ.get("superhot_count")

    return {
        "hot": hot, "superhot": superhot,
        "hot_count": hot_count, "superhot_count": superhot_count,
        "stable_cols": stable_cols,
        "consensus_col1": c1, "consensus_col2": c2,
        "mask_drop": mask_drop if mask_drop is not None else 0,
        "mirror": bool(mirror_supported),
        "doubles": doubles if doubles is not None else 0,
    }

def stable_score(stable_cols):
    return sum(COL_W.get(c, 0.0) for c in stable_cols)

def section_prior(section_name):
    return WEIGHTS["combined_prior"] if section_name == "Combined" else 1.0

def compute_cross_section_echo(section_tokens_map):
    # tokens_map: section -> set(tokens)
    echo = {}
    sections = list(section_tokens_map.keys())
    for s in sections:
        mine = section_tokens_map[s]
        others = set().union(*[section_tokens_map[o] for o in sections if o != s]) if len(sections) > 1 else set()
        echo[s] = len(mine & others)
    return echo

def tier_from_overlap(overlap):
    if overlap >= 4: return "A"
    if overlap == 3: return "B+"
    if overlap == 2: return "B"
    if overlap == 1: return "C"
    return "Z"

def build_global_token_ledger(all_docs):
    # Count unique states and per-state section repetition
    token_states = defaultdict(set)
    token_sections_by_state = defaultdict(lambda: defaultdict(int))
    for doc in all_docs:
        state = doc["__state__"]
        sections = doc["__sections__"]
        for s, winners_tokens in sections["winners_tokens"].items():
            for t in winners_tokens:
                token_states[t].add(state)
                token_sections_by_state[state][t] += 1
    return token_states, token_sections_by_state

def main(root):
    paths = load_jsons(root)
    if not paths:
        print("No *_vtrac_enhanced_*.json files found.")
        sys.exit(1)

    # First pass: load and stage for ledger
    staged = []
    for p in paths:
        with open(p, "r") as f:
            j = json.load(f)
        state = (j.get("state") or j.get("jurisdiction") or os.path.basename(p).split("_vtrac_enhanced_")[0])
        date = parse_date_from_name(os.path.basename(p))
        sect_names = get_section_names(j)

        winners_tokens_by_s = {}
        for s in sect_names:
            wt, _ = tokens_for_section(j, s)
            winners_tokens_by_s[s] = set(wt)

        staged.append({
            "__path__": p,
            "__state__": state,
            "__date__": date,
            "__sect_names__": sect_names,
            "__sections__": { "winners_tokens": winners_tokens_by_s },
            "doc": j,
        })

    token_states, token_sections_by_state = build_global_token_ledger(staged)

    rows = []
    for item in staged:
        p, j, state, date, sect_names = item["__path__"], item["doc"], item["__state__"], item["__date__"], item["__sect_names__"]

        # For normalization within this state, gather metric ranges
        raw_mask = []
        raw_doubles = []
        # prepare tokens per section for cross-section echo
        tokens_map = {}

        section_cache = {}
        for s in sect_names:
            wt, at = tokens_for_section(j, s)
            tokens_map[s] = set(wt)
            m = metrics_for_section(j, s)
            section_cache[s] = {"wt": wt, "at": at, "m": m}
            raw_mask.append(m["mask_drop"])
            raw_doubles.append(m["doubles"])

        mask_min, mask_max = (min(raw_mask), max(raw_mask)) if raw_mask else (0,0)
        dbl_min,  dbl_max  = (min(raw_doubles), max(raw_doubles)) if raw_doubles else (0,0)
        echo = compute_cross_section_echo(tokens_map)

        for s in sect_names:
            wt = section_cache[s]["wt"]
            at = section_cache[s]["at"]
            m  = section_cache[s]["m"]

            overlap = len(set(wt) & set(at))
            st_cols = m["stable_cols"]
            st_score = stable_score(st_cols)
            consensus_count = int(m["consensus_col1"]) + int(m["consensus_col2"])

            # hot/superhot normalization
            hot_count = m["hot_count"] if m["hot_count"] is not None else (m["hot"] if isinstance(m["hot"], (int,float)) else 0)
            superhot_count = m["superhot_count"] if m["superhot_count"] is not None else (m["superhot"] if isinstance(m["superhot"], (int,float)) else 0)

            # clamp to reasonable ranges for norming
            hot_norm = min(1.0, (hot_count or 0) / 8.0)
            super_norm = min(1.0, (superhot_count or 0) / 12.0)

            mask_norm = norm01(m["mask_drop"], mask_min, mask_max)
            mirror_norm = 1.0 if m["mirror"] else 0.0
            doubles_norm = norm01(m["doubles"], dbl_min, dbl_max)

            # Base score
            score = (
                WEIGHTS["overlap"]   * overlap
              + WEIGHTS["stable"]    * st_score
              + WEIGHTS["consensus"] * (consensus_count if st_cols else 0)  # cap if no stability
              + WEIGHTS["token_echo"]* echo.get(s, 0)
              + WEIGHTS["hot"]       * hot_norm
              + WEIGHTS["super"]     * super_norm
              + WEIGHTS["mask"]      * mask_norm
              + WEIGHTS["mirror"]    * mirror_norm
              + WEIGHTS["doubles"]   * doubles_norm
            )
            score *= section_prior(s)

            flags = []
            # Consensus Rescue
            if overlap == 0 and consensus_count >= 1 and len(st_cols) >= 1:
                score *= WEIGHTS["rescue_multiplier"]
                flags.append("weak_positive_rescue")

            # Tier (based on overlap)
            tier = tier_from_overlap(overlap)
            if overlap == 0: flags.append("zero_overlap")

            # Token recommendations with family promotion
            tok_scores = []
            for t in wt:
                base = 1.0
                base += 0.5 if t in at else 0.0
                # local echo (other sections in same state)
                local_rep = token_sections_by_state[state].get(t, 0)
                base += 0.2 * max(0, local_rep - 1)
                # global: seen in ≥3 states
                if len(token_states[t]) >= 3:
                    base += 0.3
                tok_scores.append((round(base,3), t))
            tok_scores.sort(reverse=True)
            recommended = [t for _, t in tok_scores][:3]

            # Straights (as tie-break info only)
            straights = straight_list_for_section(j, s)

            rows.append({
                "date": date,
                "state": state,
                "section": s,
                "overlap": overlap,
                "stable_cols_count": len(st_cols),
                "stable_cols": st_cols,
                "consensus_col1": m["consensus_col1"],
                "consensus_col2": m["consensus_col2"],
                "cross_section_echo": echo.get(s, 0),
                "hot_count": hot_count,
                "superhot_count": superhot_count,
                "mask_drop": m["mask_drop"],
                "mirror_supported": m["mirror"],
                "double_hits": m["doubles"],
                "confidence_score": round(score, 3),
                "tier": tier,
                "flags": flags,
                "top_tokens": wt,
                "recommended_tokens": recommended,
                "top_straights": straights,
                "source": os.path.basename(p),
            })

    # Write outputs
    out_json = os.path.join(root, "vtrac_compact_report.json")
    out_csv  = os.path.join(root, "vtrac_compact_report.csv")

    with open(out_json, "w") as f:
        json.dump(rows, f, indent=2)

    # Minimal CSV writer (no pandas)
    headers = ["date","state","section","overlap","stable_cols_count","stable_cols",
               "consensus_col1","consensus_col2","cross_section_echo",
               "hot_count","superhot_count","mask_drop","mirror_supported","double_hits",
               "confidence_score","tier","flags","top_tokens","recommended_tokens","top_straights","source"]
    def csv_escape(x):
        s = x if isinstance(x, str) else json.dumps(x, separators=(",",":"))
        if "," in s or '"' in s or "\n" in s:
            s = '"' + s.replace('"', '""') + '"'
        return s
    with open(out_csv, "w") as f:
        f.write(",".join(headers) + "\n")
        for r in rows:
            line = ",".join(csv_escape(r[h]) for h in headers)
            f.write(line + "\n")

    print(f"Wrote {len(rows)} rows")
    print(f"- JSON: {out_json}")
    print(f"- CSV : {out_csv}")

if __name__ == "__main__":
    root = sys.argv[1] if len(sys.argv) > 1 else "."
    main(root)
