from __future__ import annotations
from pathlib import Path
from typing import Iterable, Optional

import pandas as pd

from alpha_analytical import stable
from alpha_analytical.stable.post_pass_families import derive_vtrac_index_for_canonical
from alpha_analytical.vtrac import get_vtrac_index


_WINNER_KEYS = ("Canonical", "Winner", "winner", "Combo", "combo")


def _digits_only(s: str) -> str:
    return "".join(ch for ch in s if ch.isdigit())


def _winner_value(row: pd.Series) -> str:
    for key in _WINNER_KEYS:
        if key in row:
            v = row[key]
            if isinstance(v, str) and v.strip():
                return v.strip()
    return ""


def _canon_3v(value: str) -> str:
    # winners should already be 3-digit strings; still normalize defensively
    return stable.canon(_digits_only(str(value or "")))


def _family_id_from_canon(canon_3v: str) -> Optional[int]:
    if not canon_3v:
        return None
    try:
        return derive_vtrac_index_for_canonical(canon_3v, get_vtrac_index)
    except Exception:
        return None


def _load_csv(path: Optional[str | Path]) -> pd.DataFrame:
    if not path:
        return pd.DataFrame()
    p = Path(path)
    if not p.exists():
        return pd.DataFrame()
    try:
        return pd.read_csv(p)
    except Exception:
        return pd.DataFrame()


def attach_stable_evidence(
    winners: pd.DataFrame,
    *,
    families_path: Optional[str | Path] = None,
    families_df: Optional[pd.DataFrame] = None,
    scores_path: Optional[str | Path] = None,
    scores_df: Optional[pd.DataFrame] = None,
) -> pd.DataFrame:
    """
    Enrich a winners DataFrame with Stable evidence keyed by family_id.

    - Winners -> stable_winner / stable_canonical / family_id
    - Families evidence -> family_score/rank + flags, joined on family_id
    - Rows evidence -> top row per family (prefer section='Combined'), joined on family_id
    """
    winners_df = winners.copy()
    if winners_df.empty:
        return winners_df

    # Normalize winners and compute family_id
    winners_df["stable_winner"] = winners_df.apply(_winner_value, axis=1)
    winners_df["stable_canonical"] = winners_df["stable_winner"].map(_canon_3v)
    winners_df["family_id"] = winners_df["stable_canonical"].map(_family_id_from_canon)

    # Load artifacts
    fams = families_df.copy() if isinstance(families_df, pd.DataFrame) else _load_csv(families_path)
    scores = scores_df.copy() if isinstance(scores_df, pd.DataFrame) else _load_csv(scores_path)
    if not isinstance(fams, pd.DataFrame):
        fams = pd.DataFrame()
    if not isinstance(scores, pd.DataFrame):
        scores = pd.DataFrame()

    # ---- Family evidence (rank, score parts, flags)
    family_evidence = pd.DataFrame()
    if not fams.empty and {"family_id", "family_score"}.issubset(fams.columns):
        fam = fams.dropna(subset=["family_id", "family_score"]).copy()
        fam["family_id"] = fam["family_id"].astype("Int64")
        fam.sort_values("family_score", ascending=False, inplace=True, ignore_index=True)
        fam["family_rank"] = fam["family_score"].rank(method="dense", ascending=False).astype(int)

        keep_cols: Iterable[str] = [
            "family_id", "family_score", "family_rank",
            "section_count", "progression_flag", "last_remaining_3v",
            "any_doubles_support", "hot_density",
            "fam_cov", "fam_hpr", "fam_perm", "fam_repeat",
            "fam_cons", "fam_hot", "fam_straight2", "fam_straight3",
            "fam_doubles", "fam_section_bonus", "fam_progression_bonus", "fam_last_remaining_bonus",
        ]
        available = [c for c in keep_cols if c in fam.columns]
        family_evidence = (
            fam.loc[:, available]
               .drop_duplicates(subset=["family_id"], keep="first")
               .reset_index(drop=True)
        )

    # ---- Row evidence (best score per family; prefer Combined)
    row_evidence = pd.DataFrame()
    if not scores.empty and {"family_id", "score"}.issubset(scores.columns):
        sc = scores.copy()
        # prioritize Combined, then any section by score
        sc["__sec_priority"] = (sc["section"].astype(str) != "Combined").astype(int) if "section" in sc.columns else 1
        sc.sort_values(["family_id", "__sec_priority", "score"], ascending=[True, True, False], inplace=True)
        row_keep: Iterable[str] = [
            "family_id",             # join key
            "score", "type", "rows", "why",
            "score_cov", "score_hpr", "score_perm", "score_repeat",
            "score_straight", "score_single", "score_cons", "score_hot",
            "score_mirror", "score_dom", "score_len", "score_hidden",
            "mirror", "straight2", "straight3", "single_left",
            "cons_full", "cons_3v", "dom_last", "dom_pair", "hidden3v",
            "section"
        ]
        available = [c for c in row_keep if c in sc.columns]
        row_evidence = (
            sc.loc[:, available]
              .drop_duplicates(subset=["family_id"], keep="first")
              .rename(columns={
                  "score": "row_score",
                  "type": "row_type",
                  "rows": "row_rows",
                  "why": "row_why",
              })
        )

    # ---- Merge: winners + family_evidence + row_evidence
    enriched = winners_df.merge(family_evidence, on="family_id", how="left")
    enriched = enriched.merge(row_evidence, on="family_id", how="left")

    # Simple status to avoid NaNs propagating to UI
    ok_mask = enriched["row_score"].notna()
    enriched["evidence_status"] = ok_mask.map({True: "ok", False: "no_row_for_family"})

    # Put helper columns at end
    helper_cols = ["stable_winner", "stable_canonical"]
    base_cols = [c for c in enriched.columns if c not in helper_cols]
    enriched = enriched[base_cols + helper_cols]
    return enriched

Why this solves extended digits (your question about “31166”, “007788”, etc.)

Extended strings are great for analysis, but they’re not reliable join keys. Multiple 3‑value families can be embedded in a long substring, and leading zeros complicate plain string matches.

Stable already organizes patterns by V‑TRAC family. By deriving family_id from the winner (which is always 3‑value), and tagging spotlight rows with that same family_id, the join becomes unambiguous and robust—even when the spotlight shows extended substrings.

You keep the extended evidence in the CSV (for diagnostics), but downstream enrichment joins on family_id. Result: no NaNs, and extended cluster insight is preserved.
