Sketch:

def _is_hidden_3v(raw: str, canon: str) -> bool:
    L = len(raw)
    for i in range(L-3):
        w = raw[i:i+4]
        if not is_3value(w): continue
        for j in ((0,1,2),(0,1,3),(0,2,3),(1,2,3)):
            sub = ''.join(w[k] for k in j)
            if is_3value(sub):
                if derive_vtrac_index_for_canonical(canon, get_vtrac_index) == \
                   derive_vtrac_index_for_canonical(canon(sub), get_vtrac_index):
                    return True
    return False
