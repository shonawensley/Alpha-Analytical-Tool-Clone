Compact reference implementation (drop‑in guard + ranking core)

This snippet is self‑contained and shows the exact extraction, gating, scoring hooks, and tie‑break sort. Integrate where you build candidates and write winners.

# hot_zones_guard.py
from dataclasses import dataclass, field
from typing import List, Dict, Tuple, Iterable

VT_MIRROR = {"0":"5","1":"6","2":"7","3":"8","4":"9","5":"0","6":"1","7":"2","8":"3","9":"4"}

def canonical_triad(s: str) -> str:
    """Return canonical sorted triad string, e.g. '494' -> '449'."""
    assert len(s) == 3 and s.isdigit()
    return ''.join(sorted(s))

def vt_mirror_triad(s: str) -> str:
    """Mirror each digit via VT map then canonicalize."""
    return canonical_triad(''.join(VT_MIRROR[d] for d in s))

def guard_triads_from_set1_col(draw_data_value: str) -> List[str]:
    """
    Input: the literal 3-digit value observed in Combined ▸ Set1 ▸ Draw1 at col1/col2.
    Output: [canonical, optional canonical(VT-mirror)] distinct triads.
    """
    base = canonical_triad(draw_data_value)
    mirr = vt_mirror_triad(draw_data_value)
    return [base] if mirr == base else [base, mirr]

def vt_lane_present(vt_triad: str, lane_evidence: Dict[str, bool]) -> bool:
    """
    lane_evidence is precomputed from your analyzer (e.g., vt_straight, vt_only_lane).
    This keeps guard tied to real VT presence in Combined Set1 C1/C2.
    """
    return bool(lane_evidence.get(vt_triad, False))

@dataclass
class EvidenceRow:
    triad: str
    vt_triad: str
    score: float
    reasons: List[str] = field(default_factory=list)
    guard_injected: bool = False
    literal_hit: bool = False  # true if the literal Set1 digits appear
    col1: bool = False
    funnel_precol1: bool = False

@dataclass
class TopAggregate:
    triad: str
    vt_triad: str
    score_max: float
    score_mean: float
    guard_hits: int
    literal_hits: int
    tags: List[str]

def aggregate_and_rank(rows: Iterable[EvidenceRow]) -> List[TopAggregate]:
    """
    Group by (triad, vt_triad), compute score_max/mean and guard/literal counts,
    then apply deterministic tie-break sorting.
    """
    from collections import defaultdict
    acc = defaultdict(list)
    for r in rows:
        acc[(r.triad, r.vt_triad)].append(r)

    tops: List[TopAggregate] = []
    for (tri, vt), grp in acc.items():
        score_max = max(g.score for g in grp)
        score_mean = sum(g.score for g in grp) / len(grp)
        guard_hits = sum(1 for g in grp if g.guard_injected)
        literal_hits = sum(1 for g in grp if g.literal_hit)
        tags = sorted({t for g in grp for t in g.reasons})
        tops.append(TopAggregate(tri, vt, score_max, score_mean, guard_hits, literal_hits, tags))

    tops.sort(key=lambda t: (-t.score_max, -t.guard_hits, -t.literal_hits, -t.score_mean))
    return tops

def apply_guard_and_score(
    combined_set1_cols: Dict[int, str],  # {1: '494', 2: '858'} for example
    hot_zone_count: int,
    max_star_threshold: int,
    vt_lane_flags: Dict[str, bool],  # vt_triad -> bool (Combined Set1 C1/C2 presence)
    base_rows: List[EvidenceRow],
    w_guard_bonus: float = 1.5,
) -> List[EvidenceRow]:
    """
    Build minimal guard triads from col1/col2 under strict activation, gate by VT lane presence,
    add bonus + reason tags, and return augmented evidence rows (still score-driven).
    """
    rows = list(base_rows)
    if hot_zone_count >= max_star_threshold:
        for col_idx in (1, 2):
            if col_idx not in combined_set1_cols:
                continue
            val = combined_set1_cols[col_idx]
            if not (val and len(val) == 3 and val.isdigit()):
                continue
            for tri in guard_triads_from_set1_col(val):
                vt = canonical_triad(''.join(VT_MIRROR[d] for d in tri))  # vt family key
                if not vt_lane_present(vt, vt_lane_flags):
                    continue  # respect VT-lane gating in Combined
                # Guard evidence row with modest, deterministic bonus
                rows.append(EvidenceRow(
                    triad=tri,
                    vt_triad=vt,
                    score=w_guard_bonus,
                    reasons=["guard_set1"] + (["col1"] if col_idx == 1 else ["funnel_precol1"]),
                    guard_injected=True,
                    literal_hit=True,  # literal Set1 digits by construction
                    col1=(col_idx == 1),
                    funnel_precol1=(col_idx == 2),
                ))
    return rows
