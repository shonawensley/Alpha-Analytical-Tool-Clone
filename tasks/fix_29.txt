1) Create the real builder: src/reporting/winner_report_full.py
# src/reporting/winner_report_full.py
from __future__ import annotations
from pathlib import Path
from typing import Iterable, Tuple, Dict, Set
import pandas as pd

# ---- Public entrypoint -------------------------------------------------------

def build_and_save_full_report(
    state4: str,
    winner_3d: str,
    draw: str,   # "Midday" or "Evening"
    winner_date: str,  # "YYYY-MM-DD" string; used in filename + banner
    tables_root: Path | None = None,
    out_root: Path | None = None,
) -> Path:
    """
    Render the Analyzer-style HTML (3 panes: Midday/Evening/Combined)
    for a single winner input. Returns the output HTML path.
    """
    state4 = str(state4)
    winner = normalize_token(winner_3d)

    repo = Path(__file__).resolve().parents[2]  # repo root: .../Alpha-Analytical-Tool
    tables_root = tables_root or repo / "data" / "outputs" / "tables" / state4
    out_root    = out_root    or repo / "data" / "outputs" / "analysis" / "winners" / state4

    # 1) Load the three pipeline tables (string-safe)
    paths = {
        "Midday":    tables_root / f"{state4}_Midday_combined.csv",
        "Evening":   tables_root / f"{state4}_Evening_combined.csv",
        "Combined":  tables_root / f"{state4}_Combined_combined.csv",
    }
    for name, p in paths.items():
        if not p.exists():
            raise FileNotFoundError(f"Missing table for {name}: {p}")

    tables = {k: load_table_csv(v) for k, v in paths.items()}

    # 2) Resolve V-TRAC index + its full "purple" set and the "green straights"
    from modules.vtrac_reference import get_vtrac_index, get_index_set, get_index_straights
    index = get_vtrac_index(winner)
    purple_set: Set[str]  = set(map(normalize_token, get_index_set(index)))
    green_set: Set[str]   = set(map(normalize_token, get_index_straights(index)))

    # 3) Build HTML (3 panes; purple + green overlay)
    html = render_three_pane_html(
        state4=state4, winner=winner, draw=draw, date_str=winner_date,
        tables=tables, purple=purple_set, greens=green_set
    )

    # 4) Save
    out_root.mkdir(parents=True, exist_ok=True)
    out_path = out_root / f"{state4}_{draw}_{winner_date}_winner_{winner}_analyzer.html"
    out_path.write_text(html, encoding="utf-8")
    return out_path

# ---- Helpers -----------------------------------------------------------------

VISIBLE_COLS = ["7","6","5","4","3","2","1"]  # table value columns

def load_table_csv(path: Path) -> pd.DataFrame:
    df = pd.read_csv(path, dtype=str, keep_default_na=False, na_filter=False)
    # normalize value columns once; keep star markers for display but match on stripped tokens
    for c in VISIBLE_COLS:
        if c in df.columns:
            df[c] = df[c].astype(str)
    return df

def normalize_token(s: str) -> str:
    return str(s).replace("*", "").replace("’", "").replace("'", "").strip()

def style_cell(val: str, purple: Set[str], greens: Set[str]) -> str:
    norm = normalize_token(val)
    if norm in greens:
        return "background-color:#26A269; color:white; font-weight:600;"  # green overlay
    if norm in purple:
        return "background-color:#6D49A3; color:white;"
    return ""

def render_three_pane_html(
    state4: str, winner: str, draw: str, date_str: str,
    tables: Dict[str,pd.DataFrame], purple: Set[str], greens: Set[str]
) -> str:
    parts = []
    banner = (f"<h2>Winners Logger – {state4} – {draw} – {date_str} – "
              f"Winner: <code>{winner}</code> – V‑TRAC index overlay</h2>")
    parts.append(banner)
    for pane in ("Midday", "Evening", "Combined"):
        df = tables[pane].copy()
        # apply style to the 7 visible columns only
        styler = df.style.applymap(lambda v: style_cell(v, purple, greens), subset=VISIBLE_COLS)
        parts.append(f"<h3>{pane}</h3>" + styler.to_html())
    return "\n<hr/>\n".join(parts)
