# alpha_analytical/digit_reduction/analyzer_v2/winners_overlay.py
from __future__ import annotations
from dataclasses import dataclass
from pathlib import Path
from typing import Dict, Any, Iterable, List, Tuple, Optional
import json, re, itertools, datetime as dt

from . import io  # uses your analyzer_v2 io helpers (training_dir_for_state, etc.)

# ---- V-TRAC family helpers (local, no external imports) ----------------------

_MIRROR = {"0":"5","1":"6","2":"7","3":"8","4":"9","5":"0","6":"1","7":"2","8":"3","9":"4"}

def _digits(s: str) -> List[str]:
    return [c for c in s if c.isdigit()]

def _canon3(s: str) -> str:
    ds = [c for c in _digits(s)][:3]
    return "".join(sorted(ds)) if len(ds) == 3 else ""

def _vtrac_local_key(s: str) -> Tuple[int, int, int]:
    """
    Local mirror-family key, stable and order-free.
    Map each digit d to min(d, mirror(d)) in 0..4, sort ascending → (a,b,c).
    """
    ds = _digits(s)
    if len(ds) < 3:
        return (-1, -1, -1)
    tri = ds[:3]
    fam = []
    for ch in tri:
        d = int(ch)
        m = int(_MIRROR[ch])
        fam.append(min(d, m))     # 0..4
    fam.sort()
    return tuple(fam)  # e.g., (1,2,3)

def _vtrac_local_index(s: str) -> int:
    """
    Deterministic 0..124 index for the local family key (base-5 encoding of sorted mirror digits).
    NOTE: This is a local Analyzer index (does not depend on external canonical APIs).
    """
    a, b, c = _vtrac_local_key(s)
    if a < 0:
        return -1
    return a * 25 + b * 5 + c

# ---- Winner spec -------------------------------------------------------------

@dataclass
class WinnerSpec:
    combo: str                   # "237"
    variant: str = "Combined"    # "Combined" | "Midday" | "Evening"
    when: Optional[str] = None   # free text or ISO date for labeling

    @property
    def canon(self) -> str:
        return _canon3(self.combo)

    @property
    def canon_permutations(self) -> List[str]:
        if len(self.canon) != 3: 
            return []
        return list({"".join(p) for p in itertools.permutations(self.canon)})

    @property
    def vtrac_idx(self) -> int:
        return _vtrac_local_index(self.canon)

# ---- Training+HTML loaders ---------------------------------------------------

def _stacked_html_path(state: str, analysis_root: Path) -> Path:
    # Original stacked report produced by the reducer
    return analysis_root / "digit_reduction" / state / f"{state}_digit_reduction_report_stacked.html"

def _training_json_latest(state: str, analysis_root: Path) -> Optional[Path]:
    """
    Best-effort: pick the most recent training JSON the reducer wrote:
    data/outputs/analysis/digit_reduction/<STATE>/training/*logs.json
    """
    tdir = analysis_root / "digit_reduction" / state / "training"
    if not tdir.exists():
        return None
    candidates = sorted(tdir.glob("*_digit_reduction_logs.json"), key=lambda p: p.stat().st_mtime, reverse=True)
    return candidates[0] if candidates else None

# ---- JSON map builder --------------------------------------------------------

def _iter_items_from_training(training_json: Path) -> Iterable[Dict[str, Any]]:
    data = json.loads(training_json.read_text(encoding="utf-8"))
    # Expecting {'items':[...]} from your reducer logs; fall back if needed.
    items = data.get("items", data if isinstance(data, list) else [])
    for it in items:
        yield it

def _get(it: Dict[str, Any], key: str, default=None):
    return it.get(key, default)

def _first_match_step(steps: List[Dict[str, Any]], perms: List[str]) -> Tuple[int, Optional[str]]:
    """
    Return earliest step index (int) and value string where value is a permutation match.
    """
    for st in steps:
        val = str(st.get("value", ""))
        if len(val) == 3 and val in perms:
            return int(st.get("step", 0)), val
    return -1, None

def _first_vtrac_match_step(steps: List[Dict[str, Any]], vidx: int) -> Tuple[int, Optional[str]]:
    for st in steps:
        val = str(st.get("value", ""))
        if len(val) == 3 and _vtrac_local_index(val) == vidx:
            return int(st.get("step", 0)), val
    return -1, None

def build_winner_map(state: str, winner: WinnerSpec, *, analysis_root: Path) -> Dict[str, Any]:
    """
    Scan reducer training logs and build a machine-readable map of winner/vtrac hits.
    """
    tj = _training_json_latest(state, analysis_root)
    results: List[Dict[str, Any]] = []
    if tj is None or not tj.exists():
        return {"state": state, "winner": winner.combo, "variant": winner.variant,
                "vtrac_local_index": winner.vtrac_idx, "items": results, "error": "training_json_missing"}

    for it in _iter_items_from_training(tj):
        section = (_get(it, "section", "") or _get(it, "area_section", "")).strip()  # 'Midday'|'Evening'|'Combined'
        if winner.variant and section and section != winner.variant:
            continue

        steps: List[Dict[str, Any]] = it.get("steps", [])
        final = it.get("final", {}) or {}
        final_val = str(final.get("value", ""))

        # exact-permutation match
        s_idx, s_val = _first_match_step(steps, winner.canon_permutations)
        f_exact = (final_val in winner.canon_permutations)

        # vtrac-family match
        v_idx, v_val = _first_vtrac_match_step(steps, winner.vtrac_idx)
        f_vtrac = (_vtrac_local_index(final_val) == winner.vtrac_idx)

        if s_idx >= 0 or v_idx >= 0 or f_exact or f_vtrac:
            results.append({
                "area": _get(it, "area", ""),
                "section": section,
                "set": _get(it, "set", ""),
                "draw": _get(it, "draw", ""),
                "col": _get(it, "col", ""),
                "method": _get(it, "method", ""),
                "mode": _get(it, "mode", ""),
                "earliest_exact_step": s_idx,
                "earliest_exact_value": s_val,
                "final_exact_match": bool(f_exact),
                "earliest_vtrac_step": v_idx,
                "earliest_vtrac_value": v_val,
                "final_vtrac_match": bool(f_vtrac),
                "final_value": final_val,
                "final_vtrac_local_index": _vtrac_local_index(final_val),
            })

    return {
        "state": state,
        "winner": winner.combo,
        "winner_canon": winner.canon,
        "variant": winner.variant,
        "vtrac_local_index": winner.vtrac_idx,
        "generated_at": dt.datetime.utcnow().isoformat(timespec="seconds") + "Z",
        "items": results,
    }

# ---- Stacked HTML annotator (non-destructive) --------------------------------

_CSS = """
/* Analyzer V2 winner overlay */
.anv2-banner {font-family: system-ui, sans-serif; font-size:14px; margin:10px 0; padding:8px 10px; border:1px solid #ddd; background:#f8f9fb;}
.anv2-chip {display:inline-block; padding:2px 6px; margin:0 4px; border-radius:4px; font-weight:600;}
.anv2-chip.win {background:#ffe493; border:1px solid #d24a00; color:#5b3300;}
.anv2-chip.vtrac {background:#b4f0ff; border:1px solid #00749c; color:#003a50;}
.anv2-win {background:#ffe493; border:2px solid #d24a00; border-radius:3px; padding:0 3px;}
.anv2-vt  {background:#b4f0ff; border:2px solid #00749c; border-radius:3px; padding:0 3px;}
"""

def _highlight_numbers(html: str, perms: List[str], vtrac_set: set[str]) -> str:
    """
    Wrap 3-digit tokens that match winner permutations or share the winner's local V-TRAC index.
    We only touch isolated 3-digit tokens (non-digit boundaries) to avoid longer Orig numbers.
    """
    token_re = re.compile(r'(?<!\d)(\d{3})(?!\d)')

    def repl(m: re.Match) -> str:
        tok = m.group(1)
        if tok in perms:
            return f"<span class='anv2-win'>{tok}</span>"
        if tok in vtrac_set:
            return f"<span class='anv2-vt'>{tok}</span>"
        return tok

    return token_re.sub(repl, html)

def _collect_all_triplets(html: str) -> List[str]:
    return re.findall(r'(?<!\d)(\d{3})(?!\d)', html)

def annotate_stacked_html(state: str, winner: WinnerSpec, *, analysis_root: Path) -> Optional[Path]:
    src = _stacked_html_path(state, analysis_root)
    if not src.exists():
        return None

    html = src.read_text(encoding="utf-8")

    # Build vtrac-equivalent token set from the tokens already present in the report
    tokens = _collect_all_triplets(html)
    vt_idx = winner.vtrac_idx
    vtrac_equivs = {t for t in tokens if _vtrac_local_index(t) == vt_idx}

    annotated = _highlight_numbers(html, winner.canon_permutations, vtrac_equivs)

    banner = (
        f"<div class='anv2-banner'>"
        f"<strong>Analyzer V2 · Winners overlay</strong> — "
        f"Winner <span class='anv2-chip win'>{winner.combo}</span> "
        f"(canon: {winner.canon}) · Variant: {winner.variant} · "
        f"V‑TRAC family <span class='anv2-chip vtrac'>#{vt_idx}</span>"
        f"{' · ' + winner.when if winner.when else ''}"
        f"</div>"
    )
    head_inject = f"<style>{_CSS}</style>\n{banner}"

    # inject <style> + banner after <body> or at top if missing
    annotated = re.sub(r'(<body[^>]*>)', r'\1' + head_inject, annotated, flags=re.IGNORECASE)
    if annotated == html:  # no <body>; prepend
        annotated = head_inject + annotated

    outdir = analysis_root / "digit_reduction" / state / "analyzer_v2" / "winners"
    outdir.mkdir(parents=True, exist_ok=True)
    stamp = dt.datetime.now().strftime("%Y%m%d")
    out = outdir / f"{stamp}_{winner.variant}_winner_overlay.html"
    out.write_text(annotated, encoding="utf-8")
    return out

# ---- Orchestrator ------------------------------------------------------------

def run_winner_overlay(
    state: str,
    winner_combo: str,
    *,
    variant: str = "Combined",
    analysis_root: Path | str = Path("data/outputs/analysis"),
    when: Optional[str] = None,
) -> Dict[str, Any]:
    """
    One-shot overlay:
      - builds JSON winner map from training logs
      - produces a CSV flat table of hits
      - writes an annotated copy of the stacked HTML with highlights
    """
    analysis_root = Path(analysis_root)
    spec = WinnerSpec(winner_combo, variant=variant, when=when)

    # JSON map
    wmap = build_winner_map(state, spec, analysis_root=analysis_root)
    outdir = analysis_root / "digit_reduction" / state / "analyzer_v2" / "winners"
    outdir.mkdir(parents=True, exist_ok=True)
    stamp = dt.datetime.now().strftime("%Y%m%d")
    jpath = outdir / f"{stamp}_{variant}_winner_map.json"
    jpath.write_text(json.dumps(wmap, ensure_ascii=False, indent=2), encoding="utf-8")

    # CSV flat table (quick QA)
    import csv
    cpath = outdir / f"{stamp}_{variant}_winner_hits.csv"
    with cpath.open("w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        header = ["area","section","set","draw","col","method","mode",
                  "earliest_exact_step","earliest_exact_value","final_exact_match",
                  "earliest_vtrac_step","earliest_vtrac_value","final_vtrac_match",
                  "final_value","final_vtrac_local_index"]
        writer.writerow(header)
        for r in wmap.get("items", []):
            writer.writerow([r.get(k,"") for k in header])

    # HTML overlay (if the stacked HTML exists)
    hpath = annotate_stacked_html(state, spec, analysis_root=analysis_root)

    return {
        "state": state,
        "variant": variant,
        "winner": spec.combo,
        "winner_canon": spec.canon,
        "vtrac_local_index": spec.vtrac_idx,
        "json": str(jpath),
        "csv": str(cpath),
        "html": (str(hpath) if hpath else None),
        "n_hits": len(wmap.get("items", [])),
    }
