Cell hook: wherever each <td> is emitted, add:

from alpha_analytical.digit_reduction.long_string_windows import get_long_string_boxes

# build a fast lookup set at render-time
_boxes = get_long_string_boxes(table_kind)  # "midday"/"evening"/"combined"
LS = {(b.table, s, d, rt, c) for b in _boxes
      for s in b.sets for d in b.draws for rt in b.row_types for c in b.cols}

def _is_ls_box(table_kind, set_no, draw_no, row_type, col_index) -> bool:
    return (table_kind, set_no, draw_no, row_type, col_index) in LS

# when building each cell...
classes = []
if _is_ls_box(table_kind, set_no, draw_no, row_type, col_index):
    classes += ["ls-box","ls-box-edge"]   # background + edge
# retain existing winner/family/straight class logic (text color stays on top)
td = f'<td class="{" ".join(classes)}">{cell_html}</td>'
