COMBO_SINGLE_LATE = 334
COMBO_SINGLE_VERY_LATE = 501
COMBO_DOUBLE_LATE = 667
COMBO_DOUBLE_VERY_LATE = 1000

def is_double_combo(boxed: str) -> bool:
    # boxed is sorted 'aab' or 'abb' or 'abc'
    a, b, c = boxed[0], boxed[1], boxed[2]
    return (a == b != c) or (a != b == c) or (a == c != b)  # exactly two same

def classify_gap(gap: int, boxed: str) -> str | None:
    if is_double_combo(boxed):
        if gap >= COMBO_DOUBLE_VERY_LATE: return "red"
        if gap >= COMBO_DOUBLE_LATE:      return "blue"
    else:  # singles / triples
        if gap >= COMBO_SINGLE_VERY_LATE: return "red"
        if gap >= COMBO_SINGLE_LATE:      return "blue"
    return None

Drop‑in helper (Codex can paste this)

from collections import defaultdict
from typing import Dict, List, Tuple, Optional

MIRROR_CLASS = {0:0,5:0, 1:1,6:1, 2:2,7:2, 3:3,8:3, 4:4,9:4}
VARIANT_BADGE = {"combined": "C", "midday": "M", "evening": "E"}

def canon_boxed(s: str) -> str:
    return "".join(sorted((s or "").strip().zfill(3)))

def combo_type(s: str) -> str:
    c = canon_boxed(s)
    return "triple" if c[0]==c[1]==c[2] else ("double" if c[0]==c[1] or c[1]==c[2] else "single")

def draws_since(draws: List[str], combo: str) -> int:
    tgt = canon_boxed(combo)
    for idx, d in enumerate(draws):
        if canon_boxed(d) == tgt:
            return idx
    return len(draws)

def classify(ds: int, combo: str) -> Optional[str]:
    t = combo_type(combo)
    if t == "double":
        return "R" if ds >= 1000 else ("B" if ds >= 667 else None)
    if t == "single":
        return "R" if ds >= 501 else ("B" if ds >= 334 else None)
    # triples uncommon; treat like doubles for overdue signaling
    return "R" if ds >= 1000 else ("B" if ds >= 667 else None)

def mirror_class(d: int) -> int: return MIRROR_CLASS[d]

def family_key_for_double(combo: str) -> Tuple[int,int]:
    """Key for V‑TRAC double family (mirror_class(repeated), mirror_class(singleton))."""
    c = canon_boxed(combo)             # e.g., '225' or '044'
    if not (c[0]==c[1] or c[1]==c[2]): # only doubles
        raise ValueError("family_key_for_double expects a double")
    repeated = int(c[1])               # middle is always one of the repeated digits in sorted form
    singleton = int(c[0] if c[0]!=c[1] else c[2])
    return (mirror_class(repeated), mirror_class(singleton))

def build_due_double_families(
    draws_by_variant: Dict[str, List[str]],  # {'combined': [...], 'midday': [...], 'evening': [...]}
    combos: List[str]                        # all 90 boxed doubles, or precomputed set
) -> List[Dict]:
    # 1) per-variant gaps and classes
    gaps: Dict[str, Dict[str, int]] = {}
    cls:  Dict[str, Dict[str, Optional[str]]] = {}
    for vkey, draws in draws_by_variant.items():
        gaps[vkey] = {c: draws_since(draws, c) for c in combos}
        cls[vkey]  = {c: classify(gaps[vkey][c], c) for c in combos}

    # 2) keep only combos that are R/B in at least one variant
    qualifying = [c for c in combos if any(cls[v][c] in ("R","B") for v in ["combined","midday","evening"])]

    # 3) group into families
    fam_bins: Dict[Tuple[int,int], List[str]] = defaultdict(list)
    for c in qualifying:
        fam_bins[family_key_for_double(c)].append(c)

    # 4) score families and build display tokens
    out = []
    for fkey, members in fam_bins.items():
        tokens = []
        score = 0
        best_gap = None
        for c in sorted(set(members)):
            for vkey in ("combined","midday","evening"):
                badge = VARIANT_BADGE[vkey]
                k = cls[vkey][c]
                if k in ("R","B"):
                    tokens.append(f"{k}{c}{badge}")
                    score += 2 if k=="R" else 1
                    g = gaps[vkey][c]
                    best_gap = g if best_gap is None else min(best_gap, g)
        # order tokens: R before B, then by larger gap, then by text
        tokens.sort(key=lambda t: (0 if t[0]=="R" else 1, -int(gaps[[k for k,v in VARIANT_BADGE.items() if VARIANT_BADGE[k]==t[-1]][0]][t[1:4]]), t))
        out.append({"family": fkey, "tokens": " ".join(tokens), "score": score, "best_gap": best_gap or 0})

    # 5) rank families by score (ties: lowest best_gap)
    out.sort(key=lambda r: (-r["score"], r["best_gap"]))
    return out[:5]  # top 5 families
